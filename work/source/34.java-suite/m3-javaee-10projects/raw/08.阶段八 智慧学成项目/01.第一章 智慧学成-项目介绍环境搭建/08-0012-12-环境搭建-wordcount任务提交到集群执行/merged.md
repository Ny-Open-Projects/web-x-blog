  ```scala
  val counts = sc.textFile("https://xc-online-bakup:9000/word.txt")
  .flatMap(line => line.split(" "))
  .map(word => (word, 1))
  .reduceByKey(_ + _)
  .sortBy(_._2, ascending = false)
  counts.collect().foreach(println(_))
  ```
