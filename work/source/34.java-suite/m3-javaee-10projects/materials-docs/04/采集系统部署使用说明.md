    var arr, reg = new RegExp("(^| )" + name + "=([^;]*)(;|$)");

function generateUUID() {
    var d = new Date().getTime();
    return 'xxxxxxxxxxxxxxxxyxxxxxxxxxxxxxxx'.replace(/[xy]/g, function (c) {
        var r = (d + Math.random() * 16) % 16 | 0;
        d = Math.floor(d / 16);
        return (c == 'x' ? r : (r & 0x3 | 0x8)).toString(16);
    });
}
// 获取唯⼀标识
function getSid() {
    var sid = '';
    if (getCookie("sid")) {
        sid = getCookie("sid")
    } else {
        sid = generateUUID();
        setCookie("sid", sid);
    }
    return sid;
}

    public Logger getLogger(OuLogType logType) {
        switch (logType) {
            case KAFKA:
                KafkaAppender kafkaAppender = 
createKafkaAppender(KAFKA.getCode(), logConfigService.getKafkaTopic(), 
logConfigService.getKafkaHost(), logConfigService.getKafkaPattern());
                configAppender(kafkaAppender, KAFKA.getCode());
                return LogManager.getLogger(KAFKA.getCode());
            case FLUME:
                final FlumeAppender flumeAppender = createFlumeAppender();
                flumeAppender.start();
                configAppender(flumeAppender, FLUME.getCode());
                return LogManager.getLogger(FLUME.getCode());
            default:
                return LogManager.getLogger("Console");
        }

    private void configAppender(Appender appender, String loggerName) {
        configuration.addAppender(appender);
        AppenderRef ref = AppenderRef.createAppenderRef(loggerName, null, null);
        AppenderRef[] refs = new AppenderRef[]{ref};
        LoggerConfig loggerConfig = LoggerConfig.createLogger(false,
                Level.ALL, loggerName, "true", refs, null, configuration, null);
        loggerConfig.addAppender(appender, null, null);
        configuration.addLogger(loggerName, loggerConfig);
        loggerContext.updateLoggers();
    }
    /**
     * 创建 FlumeAppender
     */
    private FlumeAppender createFlumeAppender() {
        if (this.FLUME_APPENDER != null) {
            return this.FLUME_APPENDER;
        }
        this.FLUME_APPENDER = configuration.getAppender(FLUME.getCode());
        return this.FLUME_APPENDER;
    }
    /**
     * 创建 KafkaAppender
     *
     * @param loggerName ⽇志名
     * @param topic      kafka topic
     * @param host       kafka host
     * @param pattern    kafka ⽇志格式
     * @return
     */
    private KafkaAppender createKafkaAppender(String loggerName, String topic, 
String host, String pattern) {
        // 如果已在存直接返回
        if (this.KAFKA_APPENDER != null) {
            return this.KAFKA_APPENDER;
        }
        final PatternLayout layout = PatternLayout.newBuilder()
                .withCharset(Charset.forName("UTF-8"))
                .withConfiguration(configuration)
                .withPattern(pattern)
                .build();
        Property property = Property.createProperty("bootstrap.servers", host);
        this.KAFKA_APPENDER = KafkaAppender.newBuilder()
                .setConfiguration(configuration)
                .setName(loggerName)
                .setLayout(layout)
                .setTopic(topic)
                .setPropertyArray(new Property[]{property})
                .build();
        this.KAFKA_APPENDER.start();
        return this.KAFKA_APPENDER;
    }
