        ```java
        public class QuotaServiceImpl extends ServiceImpl<QuotaEntity, Long> implements QuotaService {
            @Override
            public IPage<QuotaEntity> queryPage(Long page, Long pageSize, String name) {
                Page<QuotaEntity> pageResult = new Page<>(page, pageSize);
                LambdaQueryWrapper<QuotaEntity> wrapper = new LambdaQueryWrapper<>();
                if (!Strings.isNullOrEmpty(name)) {
                    wrapper.like(QuotaEntity::getName, name);
                }
                return this.page(pageResult, wrapper);
            }

            @Override
            public List<String> getAllSubject() {
                QueryWrapper<QuotaEntity> wrapper = new QueryWrapper<>();
                wrapper.lambda().select(QuotaEntity::getSubject);
                return this.list(wrapper).stream().map(q -> q.getSubject()).collect(Collectors.toList());
            }
        }
        ```

        ```java
        public class QuotaController {

            @PostMapping
            public boolean create(@RequestBody QuotaVO vo) {
                try {
                    QuotaEntity quotaEntity = new QuotaEntity();
                    BeanUtils.copyProperties(vo, quotaEntity);
                    return quotaService.save(quotaEntity);
                } catch (DuplicateKeyException e) {
                    throw new BusinessException("已存在名称");
                }
            }

            @GetMapping
            public Page<QuotaEntity> queryPage(@RequestParam(value = "page", required = false, defaultValue = "1") Long page,

                return quotaService.queryPage(page, pageSize, null);
        ```

        ```java
            @Autowired
            private QuotaService quotaService;

            @Override
            public void connectionLost(Throwable throwable) {
                System.out.println("连接断开");
                emqClient.connect();
                quotaService.getAllSubject().forEach(s -> {
                    try {
                        emqClient.subscribe(topicName + "/" + s);
                    } catch (MqttException e) {
                        e.printStackTrace();
                    }
                });
            }

            @Override
            public void messageArrived(String topic, MqttMessage mqttMessage) throws Exception {
                String payload = new String(mqttMessage.getPayload());
                System.out.println("收到消息：" + payload);
            }
        ```
