@Override
@Transactional(rollbackFor = {Exception.class}, noRollbackFor = {LogicException.class})
public boolean createTask(TaskViewModel taskViewModel) throws LogicException {
    // 检查任务是否已存在
    checkCreateTask(taskViewModel.getInnerCode(), taskViewModel.getProductType());
    if (hasTask(taskViewModel.getInnerCode(), taskViewModel.getProductType())) {
        throw new LogicException("该机器有未完成的同类工单");
    }
    
    // 创建任务实体
    TaskEntity taskEntity = new TaskEntity();
    taskEntity.setTaskCode(GenerateTaskCode());
    taskEntity.setProperties(taskViewModel);
    taskEntity.setStatus(VMSYSTEM.TASK_STATUS_CREATED);
    taskEntity.setAddr(VMService.getVMInfo(taskViewModel.getInnerCode()).getAddr());
    this.save(taskEntity);
    
    // 根据产品类型设置任务类型
    if (taskEntity.getProductType() != null) {
        VMSystem.TASK_TYPE_SUPPLY;
    }
    
    return true;
}
