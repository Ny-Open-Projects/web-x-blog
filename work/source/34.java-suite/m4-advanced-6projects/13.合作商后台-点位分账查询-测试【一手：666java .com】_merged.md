    ```java
    @Override
    public Page<OrderCollectEntity> search(Long pageIndex, Long pageSize, Integer partnerId, String nodeName, String start, String end) {
        // 构建查询条件
        LambdaQueryWrapper<OrderCollectEntity> qw = new LambdaQueryWrapper<>();
        qw.eq(OrderCollectEntity::getPartnerId, partnerId);
        qw.eq(OrderCollectEntity::getNodeName, nodeName);
        qw.between(OrderCollectEntity::getDate, start, end);
        qw.orderByDesc(OrderCollectEntity::getDate);
        qw.last("limit " + pageSize);
        // 调用服务层方法
        return orderCollectService.list(qw);
    }
    ```
