    ```json
              "fieldName": "bytes"
            }
          ],
          "granularitySpec": {
            "type": "uniform",
            "queryGranularity": "minute",
            "segmentsGranularity": "2018-01-01T00:00:00Z",
            "rollup": true
          }
        },
        "ioConfig": {
          "type": "index",
          "firehose": {
            "type": "index",
            "spec": "rollup-data.json"
          }
        }
      }
    }
    ```

    *   **关键参数说明**:
        *   `dataSource`: 指定数据源名称。
        *   `parser.type`: 指定数据解析器类型为 `string`。
        *   `parseSpec.format`: 指定数据格式为 `json`。
        *   `timestampSpec.column`: 指定时间戳字段名为 `timestamp`。
        *   `dimensionsSpec.dimensions`: 定义维度字段，如 `srcIP` 和 `dstIP`。
        *   `metricsSpec`: 定义度量指标，如 `count` 和 `bytes`。
        *   `granularitySpec.type`: 定义粒度类型为 `uniform`。
        *   `granularitySpec.queryGranularity`: 定义查询粒度为 `minute`。
        *   `granularitySpec.segmentsGranularity`: 定义段粒度。
        *   `granularitySpec.rollup`: 设置为 `true` 表示启用数据聚合（Rollup）。
        *   `ioConfig.firehose.spec`: 指定数据文件路径为 `rollup-data.json`。
