    ```java
    @GetMapping("/list")
    @SysLog("查询商品信息")
    public R<List<GoodsInfo>> list(GoodsInfoPageDTO data) {
        LqWrapper<GoodsInfo> wrapper = Wraps.lqQ();
        wrapper.like(GoodsInfo::getName, data.getName());
        wrapper.like(GoodsInfo::getCode, data.getCode());
        wrapper.eq(GoodsInfo::getBarCode, data.getBarCode());
        wrapper.geHeader(GoodsInfo::getCreateTime, data.getStartCreateTime());
        wrapper.leFooter(GoodsInfo::getCreateTime, data.getEndCreateTime());
        wrapper.orderByDesc(GoodsInfo::getCreateTime);
        return success(goodsInfoService.list(wrapper));
    }
    ```
