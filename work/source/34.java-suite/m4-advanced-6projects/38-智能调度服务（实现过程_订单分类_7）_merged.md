            String address = receiverFullAddress(order);

                exceptionHappend(msg: "收件人地址不能为空");
            }
            // 调用百度地图工具类，根据地址获取对应的经纬度坐标

            String[] originArray = location.split(regex: ",");
            location = originArray[1] + "," + originArray[0];
            String url = "http://api.map.baidu.com/reverse_geocoding/v3/?ak=" + AK + "&output=json&coordtype=wgs84ll&location=" + location + "&ak=" + AK;
            String json = loadJSONUrl(url);
            System.out.println(json);
            if (json != null && !"{}".equals(json)) {
                Map map = JSON.parseObject(json, Map.class);
                if ("0".equals(map.getOrDefault("status", "0").toString())) {
                    Map childMap = map.get("result");
                    Map areaMap = (Map) childMap.get("addressComponent");
