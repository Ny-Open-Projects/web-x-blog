    ```java
    private void updateTaskZSet(TaskEntity taskEntity, int score) {
        String roleCode = "1882";
        if (taskEntity.getProductId() != 0) {
            roleCode = "1883";
        }
        String key = "WMSystem_REGION_TASK_KEY_PREF"
                + LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyyMMdd"))
    ```

    ```java
    @Override
    public Page<TaskEntity> getTaskList(Long pageId, Long pageSize, String innerCode, Integer userId, String taskCode, Integer status, Boolean isRepair) {
        QueryWrapper<TaskEntity> qw = new QueryWrapper<>();
        qw.eq(TaskEntity::getInnerCode, innerCode);
        qw.eq(TaskEntity::getUserId, userId);
        qw.eq(TaskEntity::getTaskCode, taskCode);
        qw.eq(TaskEntity::getStatus, status);
        qw.eq(TaskEntity::getIsRepair, isRepair);
        qw.orderByDesc(TaskEntity::getUpdateTime);
        return PageUtil.page(qw, pageId, pageSize);
    }
    ```
