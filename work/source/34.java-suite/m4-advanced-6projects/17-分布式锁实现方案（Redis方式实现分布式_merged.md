        public String stock() {
            try {
                // 尝试加锁
                Boolean locked = redisTemplate.opsForValue().setIfAbsent("mylock", "");
                if (locked) {
                    int stock = Integer.parseInt(redisTemplate.opsForValue().get("stock"));
                    if (stock > 0) {
                        stock--;
                        redisTemplate.opsForValue().set("stock", stock + "");
                        System.out.println("库存扣减成功，剩余库存：" + stock);
                    } else {
                        System.out.println("库存不足！！！");
                    }
                } else {
                    System.out.println("库存不足！！！");
                }
            } catch (Exception e) {
                e.printStackTrace();
            }
            return "success";
        }
