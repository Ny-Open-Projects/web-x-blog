                continue;

            if (redisTemplate.boundValueOps(key).get() == null) {
                httpUtil.httpPost(quotaDTO.getWebhook(), quotaDTO);
                redisTemplate.boundValueOps(key).set(quotaDTO.getStringValue(), quotaDTO.getCycle(), TimeUnit.MINUTES);
            }
        }

                                 @RequestParam(value = "tag", required = false, defaultValue = "") String tag) {
            return deviceService.queryPage(page, pageSize, tag, null);
        }

        @PostMapping("/clientAction")
        public void clientAction(@RequestParam Map<String, String> param) {
            String deviceId = param.get("clientID");
            if (param.get("action").equals("client_connected")) {
                deviceService.updateOnline(deviceId, online: true);
            }
            if (param.get("action").equals("client_disconnected")) {
                deviceService.updateOnline(deviceId, online: false);
            }
        }
