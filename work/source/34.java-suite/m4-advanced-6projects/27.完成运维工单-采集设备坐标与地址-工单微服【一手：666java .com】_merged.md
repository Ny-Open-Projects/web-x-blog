    ```java
    // 重载完成工单方法，支持经纬度参数
    @Override
    public boolean completeTask(Long id, Double lat, Double lon, String addr) {
        TaskEntity taskEntity = this.getTaskById(id);
        if (taskEntity.getTaskStatus() == VMSysten.TASK_STATUS_FINISH || taskEntity.getTaskStatus() == VMSysten.TASK_STATUS_CANCEL) {
            throw new LogicException("工单已经结束");
        }
        taskEntity.setTaskStatus(VMSysten.TASK_STATUS_FINISH);
        this.updateById(taskEntity);

        // 构建协议对象并设置经纬度
        TaskCompleteContract taskCompleteContract = new TaskCompleteContract();
        taskCompleteContract.setInnerCode(taskEntity.getInnerCode());
        taskCompleteContract.setProductTypeCode(taskEntity.getProductTypeCode());
        taskCompleteContract.setLat(lat);
        taskCompleteContract.setLon(lon);
        taskCompleteContract.setAddr(addr);

        // 发送消息到MQTT
        try {
            mqttProducer.send(TopicConfig.COMPLETED_TASK_TOPIC, 2, taskCompleteContract);
        } catch (Exception e) {
            log.error("发送工单完成协议出错");
            throw new LogicException("发送工单完成协议出错");
        }
        return true;
    }

    // 查询所有工单状态
    @Override
    public List<TaskStatusTypeEntity> getAllStatus() {
        QueryWrapper<TaskStatusTypeEntity> qw = new QueryWrapper<>();
        qw.lambda().eq(TaskStatusTypeEntity::getTaskStatusId, VMSysten.TASK_STATUS_CREATE);
        return statusTypeService.list(qw);
    }
    ```

    **3. `TaskController.java` (控制器层)**

    ```java
    // 新增完成工单接口，接收经纬度
    @GetMapping("/complete")
    public boolean complete(@PathVariable("taskId") Long taskId,
                            @RequestParam Double lat,
                            @RequestParam Double lon,
                            String addr) {
        return taskService.completeTask(taskId, lat, lon, addr);
    }

    // 新增取消工单接口
    @PostMapping("/cancel")
    public boolean cancel(@PathVariable("taskId") String taskId,
                          @RequestBody CancelTaskViewModel cancelVm) {
        Long id = Long.valueOf(taskId);
        TaskEntity task = taskService.getTaskById(id);
        if (task.getAssignUserId().intValue() != 0) {
            throw new LogicException("操作非法");
        }
        return taskService.cancelTask(id, cancelVm);
    }

    // 新增查询所有工单状态接口
    @GetMapping("/allTaskStatus")
    public List<TaskStatusTypeEntity> getAllStatus() {
        return taskService.getAllStatus();
    }
    ```
