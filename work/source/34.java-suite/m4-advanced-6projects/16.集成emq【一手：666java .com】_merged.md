        @Component
        public class EmqMsgCallback implements MqttCallbackExtended {
            @Autowired
            private EmqClient emqClient;
            @Autowired
            private EmqConfig config;

            @Override
            public void connectionLost(Throwable throwable) {
                log.info("emq connect lost", throwable);
                try {
                    TimeUnit.SECONDS.sleep(18);
                    emqClient.connect(callback);
                    emqClient.subscribe(config.getSubscribeTopic(), callback);
                } catch (Exception e) {
                    log.info("订阅失败", e);
                }
            }

            @Override
            public void messageArrived(String topic, MqttMessage mqttMessage) throws Exception {
                CompletableFuture.runAsync(() -> {
                    System.out.println(topic);
                });
                String msgType = new String(mqttMessage.getPayload());
                log.info("***********msgType {} payload {}***********", msgType, mqttMessage.getPayload());
            }

            @Override
            public void deliveryComplete(IMqttDeliveryToken iMqttDeliveryToken) {
            }

            @Override

                    return;

            }
