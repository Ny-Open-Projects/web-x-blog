        ```java
        // 远程加载 Maven 仓库中的 jar 包
        String url = "http://localhost:8080/kie-drools-wb/maven2wb/com/mspace_1/myproject_1/1.0.0/myproject_1-1.0.0.jar";

        // 通过 Resource 资源对象加载 jar 包
        UrlResource resource = (UrlResource) kieServices.getResources().newUrlResource(url);
        resource.setUsername("kie");
        resource.setPassword("kie");
        resource.setBasicAuthentication("enabled");

        // 将资源转换为输入流
        InputStream inputStream = resource.getInputStream();

        // 创建仓库对象
        KieRepository repository = kieServices.getRepository();

        // 通过输入流读取 Maven 仓库中的 jar 包数据，包装成 KieModule 模块添加到仓库中
        KieModule kieModule = repository.addKieModule(kieServices.getResources().newInputStreamResource(inputStream));

        // 基于 KieModule 模块创建容器对象，从容器中可以获取 session 会话
        KieContainer kieContainer = kieServices.newKieContainer(kieModule.getReleaseId());
        KieSession session = kieContainer.newKieSession();

        Person person = new Person();
        person.setAge(10);
        session.insert(person);
        session.fireAllRules();
        session.dispose();
        ```
