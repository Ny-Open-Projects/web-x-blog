```go
type MsgReadTransferClient interface {
    Push(msg *mq.MsgMarkRead) error
}

type msgReadTransferClient struct {
    pusher *mq.Pusher
}

func NewMsgReadTransferClient(addr []string, topic string, opts ...mq.PushOption) MsgReadTransferClient {
    pusher := mq.NewPusher(addr, topic, opts...)
    return &msgReadTransferClient{pusher: pusher}
}

func (c *msgReadTransferClient) Push(msg *mq.MsgMarkRead) error {
    body, err := json.Marshal(msg)
    if err != nil {
        return err
    }
    return c.pusher.Push(string(body))
}
```

还有一个小地方需要处理：WebSocket 中还要对消息 ID 做发送，增加对消息 ID 的推送，消息通过 single 方法推送给对应的连接：

```go
func single(srv *websocket.Server, data *ws.Push, recvId string) error {
    rconn := srv.GetConn(recvId)
    if rconn == nil {
        return nil
    }
    return srv.Send(websocket.NewMessage(data.SendId, ws.Chat{
        ConversationId: data.ConversationId,
        ChatType:       data.ChatType,
        SendTime:       data.SendTime,
        Msg: ws.Msg{
            MsgId:     data.MsgId,
            MType:     data.MType,
            Content:   data.Content,
            ReadRecords: data.ReadRecords,
        },
    }), rconn)
}
```

重启服务，重新发送消息，可以看到消息里已经有消息 ID 了。消息 ID 有了之后，接下来做消息的已读记录处理：把这个消息 ID 发送过去，代表当前这个用户已读了。先看其他人会不会收到，收到了，对应消息记录的已读记录的处理结果已经到了，只是还没有标记出已读记录。

再在里面增加已读的记录，重新连接再发送。看代码发现发送不成功，要先在这里注释一下，因为这里会判断 ID 不为空就重新设置。重启一下，只要重启消息队列内的消费者就可以了。重新发送新的消息，发送之后他们都会接收到。对这个消息重新进行已读发送，这次就会有已读的记录了。但记录的是 Base64 的值，直接展示不太好看，需要知道哪些用户是已读。为此增加一个 API 查询聊天记录的已读与未读。API 已经写好并启动，服务也启动起来了。把刚刚已读的聊天消息 ID 复制到查询聊天记录中，发送查询。

第一次请求是权限不足，给它增加一个请求头。请求之后输出结果：上面是已读的，下面是未读的。已读的有两个，一个是发送者，一个是本次请求的用户；未读的是剩下的。再让另一个用户标记已读后查询，可以看到已读多了一个，说明已读未读功能确实没有问题了。
