## AIC-529：笔试：手写沙箱执行器

**答**：实现 `createSandbox({ timeoutMs, maxMemoryMB, allowedPaths })`。方法：`execute(cmd, args)` → `{ code, stdout, stderr, timedOut }`。规则：1) **超时保护**：timeoutMs 超时 → kill + 返回 timedOut=true；2) **内存限制**：NODE_OPTIONS=--max-old-space-size=maxMemoryMB；3) **命令白名单**：只允许白名单命令（node/npx/npm/tsc——防止 rm -rf 等）；4) **执行审计**：每次执行的 cmd/args/code/duration 记录（审计链）；5) **并发限制**：最多同时 N 个沙箱进程（资源保护 RBL-690）。返回 `{ execute, stats }`。边界：白名单外的命令（拒绝）；超时（kill + timedOut）；并发超限（排队）。

```javascript
function createSandbox({ timeoutMs = 30000, maxMemoryMB = 512, allowedCmds = ['node', 'npx', 'npm', 'tsc'],
  maxConcurrent = 3, spawnFn } = {}) {
  const audit = []; let active = 0; const queue = [];
  function _doExecute(cmd, args) {
    return new Promise((resolve) => {
      const start = Date.now();
      const child = spawnFn ? spawnFn(cmd, args) : mockSpawn(cmd, args);
      let stdout = '', stderr = '';
      let timedOut = false;
      const timer = setTimeout(() => { timedOut = true; child.kill?.(); }, timeoutMs);
      child.on('data', d => stdout += d);
      child.on('error', d => stderr += d);
      child.on('close', (code) => {
        clearTimeout(timer); active--;
        const duration = Date.now() - start;
        audit.push({ cmd, args, code, duration, timedOut });
        pump();
        resolve({ code: code ?? -1, stdout, stderr, timedOut,
          duration, cmd: `${cmd} ${args.join(' ')}` });
      });
    });
  }
  function mockSpawn(cmd, args) {
    const events = { listeners: {} };
    const obj = {
      on(ev, cb) { (events.listeners[ev] ??= []).push(cb); },
      stdout: { on: (ev, cb) => { if (ev === 'data') setTimeout(() => cb(`output of ${cmd}`), 1); } },
      stderr: { on: (ev, cb) => {} },
      kill: () => { events.listeners.close?.forEach(cb => cb(-1)); },
    };
    setTimeout(() => events.listeners.close?.forEach(cb => cb(0)), 5);
    return obj;
  }
  function pump() {
    while (active < maxConcurrent && queue.length) {
      const { cmd, args, resolve } = queue.shift();
      active++;
      _doExecute(cmd, args).then(resolve);
    }
  }
  return {
    async execute(cmd, args = []) {
      if (!allowedCmds.includes(cmd))
        return { code: -1, stdout: '', stderr: `Command not allowed: ${cmd}`, timedOut: false, rejected: true };
      if (active >= maxConcurrent) {
        return new Promise(resolve => { queue.push({ cmd, args, resolve }); pump(); });
      }
      active++;
      return _doExecute(cmd, args);
    },
    stats: () => ({ active, queued: queue.length, audit: [...audit] }),
  };
}

// 自测
const sandbox = createSandbox({ timeoutMs: 100, maxConcurrent: 2,
  spawnFn: (cmd, args) => {
    const obj = { on(ev, cb) { if (ev === 'close') setTimeout(() => cb(0), 5); },
      stdout: { on: (ev, cb) => { if (ev === 'data') setTimeout(() => cb(`ok`), 1); } },
      stderr: { on: () => {} }, kill: () => {} };
    return obj;
  }});
// 白名单内
let r = await sandbox.execute('node', ['test.js']);
console.log(r.code === 0 && !r.timedOut);
// 白名单外拒绝
r = await sandbox.execute('rm', ['-rf', '/']);
console.log(r.rejected === true && r.stderr.includes('not allowed'));
// 审计
console.log(sandbox.stats().audit.length === 2);         // 两次执行记录
// 超时
const timeoutSandbox = createSandbox({ timeoutMs: 10,
  spawnFn: () => { const obj = { on(ev, cb) { /* 永不 close */ },
    stdout: { on: () => {} }, stderr: { on: () => {} }, kill: () => {} }; return obj; }});
r = await timeoutSandbox.execute('node', ['long.js']);
console.log(r.timedOut === true);                        // 超时检测
// 拒绝的不计入审计
console.log(sandbox.stats().audit.every(a => !a.cmd.includes('rm')));
```

**评分点**：
- 白名单命令（防止 rm -rf / curl 等危险命令——Agent 代码不可信）
- 超时 kill（防死循环/资源耗尽——资源保护 RBL-690）；并发限制
- 审计链（每次执行的 cmd/args/duration——安全审计 RBL-528）；拒绝不计入审计（区分拒绝与执行）
- 能说清「Agent 沙箱的必要性：AI 生成的代码不可信（可能含 rm -rf）——沙箱是 Agent 自主执行的最后一道防线」属于加分
