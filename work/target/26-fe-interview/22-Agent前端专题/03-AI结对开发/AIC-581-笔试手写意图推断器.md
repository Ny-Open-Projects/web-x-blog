## AIC-581：笔试：手写意图推断

**答**：实现 `inferIntent(diff)`。diff: `{ newFiles: string[], modifiedFiles: string[], newImports: string[], changedPatterns: { conditionals, errorHandling, memoHooks, extractedFunctions } }`. 规则（按优先级）：1) bugfix：conditionals ≥ 3 && errorHandling ≥ 2 && newFiles 空；2) performance：memoHooks ≥ 2（不受 newFiles 限制——性能优化可能加新文件）；3) refactor：extractedFunctions ≥ 2 && newImports 空；4) feature：newFiles > 0 || newImports > 0；5) 默认：'modification'（confidence 50）。返回 `{ intent, confidence, reviewFocus }`。边界：全部为 0（默认 modification）；conditionals 和 memoHooks 同时高（先匹配 bugfix——优先级）；newFiles > 0 且 conditionals 高（feature 优先于 bugfix——有新文件 = 新功能而非修 bug？——不：优先级顺序是 bugfix > performance > refactor > feature——条件判断优先）。

```javascript
function inferIntent(diff) {
  const { newFiles = [], newImports = [],
    changedPatterns: cp = { conditionals: 0, errorHandling: 0, memoHooks: 0, extractedFunctions: 0 } } = diff;
  // 优先级：bugfix > performance > refactor > feature > default
  if (cp.conditionals >= 3 && cp.errorHandling >= 2 && newFiles.length === 0)
    return { intent: 'bugfix', confidence: 80, reviewFocus: '边界条件与错误处理' };
  if (cp.memoHooks >= 2)
    return { intent: 'performance', confidence: 75, reviewFocus: '性能优化正确性' };
  if (cp.extractedFunctions >= 2 && newImports.length === 0)
    return { intent: 'refactor', confidence: 70, reviewFocus: '行为不变性验证' };
  if (newFiles.length > 0 || newImports.length > 0)
    return { intent: 'feature', confidence: 75, reviewFocus: '新功能的边界与安全' };
  return { intent: 'modification', confidence: 50, reviewFocus: '变更影响面' };
}

// 自测
// bugfix：高 conditionals + errorHandling + 无新文件
let r = inferIntent({ newFiles: [], modifiedFiles: ['fix.ts'],
  newImports: [], changedPatterns: { conditionals: 5, errorHandling: 3, memoHooks: 0, extractedFunctions: 0 } });
console.log(r.intent === 'bugfix' && r.confidence === 80);
console.log(r.reviewFocus.includes('边界'));
// performance：高 memoHooks
r = inferIntent({ newFiles: [], changedPatterns: { conditionals: 1, errorHandling: 0, memoHooks: 3, extractedFunctions: 0 } });
console.log(r.intent === 'performance');
// refactor：extractedFunctions 高
r = inferIntent({ newFiles: [], newImports: [], changedPatterns: { conditionals: 0, errorHandling: 0, memoHooks: 0, extractedFunctions: 3 } });
console.log(r.intent === 'refactor');
// feature：有新文件
r = inferIntent({ newFiles: ['new-component.tsx'], changedPatterns: { conditionals: 0, errorHandling: 0, memoHooks: 0, extractedFunctions: 0 } });
console.log(r.intent === 'feature');
// 优先级：bugfix 条件满足但也有 memoHooks
r = inferIntent({ newFiles: [], changedPatterns: { conditionals: 5, errorHandling: 3, memoHooks: 3, extractedFunctions: 0 } });
console.log(r.intent === 'bugfix');                     // bugfix 优先
// 默认
r = inferIntent({ changedPatterns: { conditionals: 0, errorHandling: 0, memoHooks: 0, extractedFunctions: 0 } });
console.log(r.intent === 'modification' && r.confidence === 50);
```

**评分点**：
- 五种意图 + 优先级顺序（bugfix > performance > refactor > feature > modification——按审查紧急度排序）
- 优先级的语义（bugfix 最紧急（修 bug 比加功能紧急）——条件判断优先级映射审查紧急度）；默认 modification
- 边界（全零默认/条件重叠优先级/无 newFiles 的 bugfix 限制）
- 能说清「意图推断决定审查策略：bugfix 审边界、feature 审完整性、refactor 验行为不变——先推断意图再定审查重点」属于加分
