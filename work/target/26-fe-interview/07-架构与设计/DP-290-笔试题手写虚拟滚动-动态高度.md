## DP-290：笔试题：手写虚拟滚动（动态高度版）？
**题目**：实现动态高度虚拟列表：1) 每项渲染后测量真实高度入缓存；2) 未测量项用估算值，前缀和偏移计算；3) 缓存更新后重算可视区（防跳动）；4) scrollToIndex 精确定位（估算 + 校正）。
**评分点**：
- 前缀和数组维护正确（增量更新而非全量重建）。
- 测量时机（ResizeObserver 或渲染后 rAF 读取 offsetHeight）。
- 总高修正与滚动位置补偿（测量差异 delta 应用后 scrollTop 调整）。
- scrollToIndex 二分定位 + 测量校正。
```javascript
function createVirtualList({ container, count, estimate = 44 }) {
  const heights = new Array(count).fill(estimate);
  const offsets = [0];
  function buildOffsets() { for (let i = 0; i < count; i++) offsets[i + 1] = offsets[i] + heights[i]; }  // 2) 前缀和
  buildOffsets();
  function locate(idx) {           // 二分查偏移
    let lo = 0, hi = offsets.length - 1;
    while (lo < hi) { const mid = (lo + hi) >> 1; offsets[mid + 1] <= idx ? lo = mid + 1 : hi = mid; }
    return lo;
  }
  function render() {
    const st = container.scrollTop, vh = container.clientHeight;
    const start = Math.max(0, locate(st) - 5);
    const end = Math.min(count, locate(st + vh) + 5);
    return { start, end, top: offsets[start], total: offsets[count] };
  }
  function measure(idx, h) {       // 1) 测量入缓存
    const delta = h - heights[idx];
    heights[idx] = h;
    for (let i = idx + 1; i <= count; i++) offsets[i] += delta;   // 3) 增量重算
    const deltaScroll = st(idx) - offsets[idx];  // 位置补偿（视口内首项测量后修正）
  }
  function scrollToIndex(idx) {    // 4) 估算 + 校正
    container.scrollTop = offsets[idx];
    requestAnimationFrame(() => { const item = container.querySelector(`[data-idx="${idx}"]`);
      if (item) { const diff = item.offsetTop - offsets[idx]; if (diff) container.scrollTop += diff; } });
  }
  return { render, measure, scrollToIndex };
}
```
