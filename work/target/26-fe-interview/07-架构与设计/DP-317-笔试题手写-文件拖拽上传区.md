## DP-317：笔试题：手写文件拖拽上传区（类型校验/进度/粘贴）？
**答**：实现 DropZone：1) dragover 高亮/dragleave 去亮（防抖动：enter/leave 计数）；2) 文件类型与大小校验（accept/maxSize），错误提示；3) 上传进度（XMLHttpRequest 可监听）；4) 支持粘贴上传（paste 事件拿 clipboardData.files）。
**评分点**：
- enter/leave 计数防抖正确（子元素触发 leave 不清高亮）。
- 校验先行（不发请求就报错）。
- XHR upload.onprogress 正确（fetch 无上传进度）。
- paste 支持完整。
```javascript
function setupDropZone(el, { accept = [], maxSize = 10 << 20, onUploaded }) {
  let depth = 0;                                            // 1) 计数防抖
  el.addEventListener('dragenter', e => { e.preventDefault(); if (++depth === 1) el.classList.add('dragging'); });
  el.addEventListener('dragleave', () => { if (--depth === 0) el.classList.remove('dragging'); });
  el.addEventListener('dragover', e => e.preventDefault());
  el.addEventListener('drop', e => { e.preventDefault(); depth = 0; el.classList.remove('dragging');
    handleFiles([...e.dataTransfer.files]); });
  document.addEventListener('paste', e => handleFiles([...e.clipboardData.files]));   // 4)
  function handleFiles(files) {
    for (const f of files) {
      if (accept.length && !accept.some(ext => f.name.endsWith(ext))) { alert(`${f.name} 类型不支持`); continue; }  // 2)
      if (f.size > maxSize) { alert(`${f.name} 超过大小限制`); continue; }
      upload(f);
    }
  }
  function upload(file) {
    const xhr = new XMLHttpRequest();
    const bar = createProgress(file.name);
    xhr.upload.onprogress = e => bar.set(e.loaded / e.total);          // 3)
    xhr.onload = () => xhr.status < 300 ? onUploaded(JSON.parse(xhr.responseText)) : bar.fail();
    xhr.open('POST', '/upload');
    const fd = new FormData(); fd.append('file', file);
    xhr.send(fd);
  }
}
```
