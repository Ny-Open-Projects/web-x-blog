## DP-347：笔试题：手写数字滚动动画（rAF+easeOut）？
**题目**：实现 animateNumber(el, from, to, { duration, format })：1) rAF 驱动帧率无关；2) easeOutCubic 缓动；3) 中途更新目标值（从当前值继续，不跳变）；4) 页面隐藏暂停恢复正确。
**评分点**：
- rAF + 起始时间戳正确（performance.now）。
- 缓动公式正确。
- 中途更新从当前插值继续。
- 卸载取消 rAF。
```javascript
function animateNumber(el, to, { duration = 1200, format = n => Math.round(n).toLocaleString() } = {}) {
  let raf = 0;
  let current = parseFloat(el.dataset.value ?? '0');
  function play(from, target) {
    cancelAnimationFrame(raf);
    const start = performance.now();
    const ease = t => 1 - Math.pow(1 - t, 3);                  // 2) easeOutCubic
    const step = (now) => {
      const p = Math.min(1, (now - start) / duration);
      current = from + (target - from) * ease(p);
      el.textContent = format(current);
      if (p < 1) raf = requestAnimationFrame(step);
      else el.dataset.value = String(target);
    };
    raf = requestAnimationFrame(step);
  }
  play(current, to);                                           // 3) 从当前值起
  return { update: (newTo) => play(current, newTo),            // 中途更新
    cancel: () => cancelAnimationFrame(raf) };                 // 4)
}
```
