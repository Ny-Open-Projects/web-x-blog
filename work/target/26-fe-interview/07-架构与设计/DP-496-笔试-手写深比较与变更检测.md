## DP-496：笔试：手写深比较

**答**：实现 deepEqual：① 基本类型 === + NaN 相等（Object.is）；② 对象/数组递归（key 集合一致 + 值深比）；③ 循环引用防御（WeakMap 记录已比较对——栈溢出防护）；④ 性能优化：引用相同短路（=== 命中直接 true——大多数比较到此结束）；⑤ 变更路径输出：deepDiff(a, b) 返回变更路径清单（'list[2].status'——精确 patch 的基础）。写出实现。

```javascript
function deepEqual(a, b, seen = new WeakMap()) {
  if (Object.is(a, b)) return true;                    // 引用相同/NaN 相等 — 快速路径
  if (typeof a !== 'object' || typeof b !== 'object' || !a || !b) return false;
  if (seen.get(a) === b) return true;                  // 循环引用
  seen.set(a, b);
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  const ka = Object.keys(a), kb = Object.keys(b);
  if (ka.length !== kb.length) return false;
  return ka.every((k) => deepEqual(a[k], b[k], seen));
}

function deepDiff(a, b, path = '', out = []) {
  if (Object.is(a, b)) return out;
  if (typeof a !== 'object' || typeof b !== 'object' || !a || !b)
    return out.concat({ path, from: a, to: b });
  const keys = new Set([...Object.keys(a ?? {}), ...Object.keys(b ?? {})]);
  for (const k of keys) {
    const p = path ? (Array.isArray(a) ? `${path}[${k}]` : `${path}.${k}`) : String(k);
    deepDiff(a?.[k], b?.[k], p, out);
  }
  return out;
}
// deepDiff({list:[{status:'a'}]}, {list:[{status:'b'}]}) → [{path:'list[0].status', from:'a', to:'b'}]
```

**评分点**：
- 引用相同短路（性能关键——大多数比较到此结束）
- 循环引用的 WeakMap 防御
- 数组与对象的 key 集合比较
- 变更路径输出（精确 patch 的基础）
