## DP-500：笔试：手写权限内核

**答**：实现权限内核 `createAuth`：① RBAC：角色 → 权限点展开（admin: ['order:*']）；② ABAC：资源属性条件（resource.ownerId === userId 或 has perm:any）；③ 通配符匹配（order:* 前缀）；④ deny 优先（显式拒绝集合 > 允许集合）；⑤ 判定缓存（has 结果缓存 + 权限变更清缓存）；⑥ 变更订阅（响应式权限——UI 自动更新）。写出实现。

```javascript
export function createAuth({ roles = {}, initialRoles = [], deny = [], userId }) {
  let userRoles = new Set(initialRoles);
  let denySet = new Set(deny);
  let allowSet = new Set();
  const cache = new Map(); const subs = new Set();
  function rebuild() {
    allowSet = new Set();
    for (const r of userRoles) for (const p of roles[r] ?? []) allowSet.add(p);
    cache.clear();                                     // 变更清缓存
  }
  rebuild();
  function match(perm) {
    if (denySet.has(perm)) return false;               // deny 优先
    if (allowSet.has(perm)) return true;
    return [...allowSet].some((p) => p.endsWith('*') && perm.startsWith(p.slice(0, -1)));
  }
  function has(perm, resource) {
    const key = `${perm}:${resource?.id ?? ''}`;
    if (!cache.has(key)) {
      const base = match(perm);
      const abac = resource ? base && (resource.ownerId === userId || match(`${perm}:any`)) : base;
      cache.set(key, abac);
    }
    return cache.get(key);
  }
  return { has, hasAny: (l) => l.some((p) => has(p)), hasAll: (l) => l.every((p) => has(p)),
    setRoles(list) { userRoles = new Set(list); rebuild(); subs.forEach((f) => f()); },
    sub(fn) { subs.add(fn); return () => subs.delete(fn); } };
}
```

**评分点**：
- deny 优先语义（显式拒绝覆盖允许）
- ABAC 的资源属性条件（ownerId 判定）
- 通配符 + 判定缓存（变更清缓存——性能与正确性平衡）
- 变更订阅（响应式权限——UI 自动更新）
