## ARC-403：笔试题：手写配置驱动的灰度路由
**题目**：实现 route(user, rules)：规则形如 { match: { region?: string, percent?: number, whitelist?: string[] }, to: 'v2' }；按顺序命中第一条规则；percent 用稳定哈希分桶（同一用户结果恒定）；无命中返回 default。
**评分点**：
- 规则顺序优先匹配，先白名单后百分比。
- 稳定哈希（非 Math.random）保证同用户同桶。
- percent 语义正确（0~100，hash % 100 < percent）。
- 字段缺省即不限该维度。
```javascript
function hashUser(uid) {
  let h = 0;
  for (const ch of uid) h = (h * 31 + ch.charCodeAt(0)) >>> 0;   // 稳定哈希
  return h;
}
function route(user, rules, fallback = 'default') {
  for (const r of rules) {
    const { match: m, to } = r;
    if (m.region && user.region !== m.region) continue;          // 条件不符跳过
    if (m.whitelist?.length && !m.whitelist.includes(user.id)) continue;
    if (m.percent != null && hashUser(user.id) % 100 >= m.percent) continue;
    return to;                                                    // 第一条命中即返回
  }
  return fallback;
}
const rules = [
  { match: { whitelist: ['u100'] }, to: 'v2' },                   // 名单优先
  { match: { region: 'cn', percent: 20 }, to: 'v2' },             // cn 20% 放量
  { match: { region: 'cn' }, to: 'v1' }                           // 其余 cn 保持 v1
];
console.log(route({ id: 'u42', region: 'cn' }, rules));           // 稳定结果
```
