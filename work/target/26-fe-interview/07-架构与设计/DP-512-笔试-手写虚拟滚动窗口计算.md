## DP-512：笔试：手写虚拟窗口

**答**：实现虚拟列表的窗口计算：① 定高版：start/end/offset 的 O(1) 计算；② 不定高版：累计高度数组 + 二分查找定位（offsets 有序）；③ 二分查找：找「第一个 start > scrollTop 的项」（上边界）；④ 高度校准：实测后 offsets 数组的增量更新（diff 传播到后续项）；⑤ scrollToIndex 的精确计算。写出核心实现。

```javascript
function createVirtualizer({ count, estimateSize, overscan = 5 }) {
  const offsets = [];                                 // 累计高度
  const sizes = new Map();                            // 实测缓存
  function rebuild() {
    offsets.length = 0;
    let acc = 0;
    for (let i = 0; i < count; i++) { offsets.push(acc); acc += sizes.get(i) ?? estimateSize(i); }
    offsets.push(acc);                                // 总高哨兵
  }
  rebuild();
  function binarySearch(top) {                        // 找第一个 start > top 的项
    let lo = 0, hi = offsets.length - 1;
    while (lo < hi) { const mid = (lo + hi) >> 1; offsets[mid] <= top ? lo = mid + 1 : hi = mid; }
    return lo - 1 < 0 ? 0 : lo - 1;
  }
  function measure(index, real) {
    const old = sizes.get(index) ?? estimateSize(index);
    if (Math.abs(old - real) < 1) return;             // 阈值防抖
    sizes.set(index, real);
    const diff = real - old;
    for (let i = index + 1; i <= count; i++) offsets[i] += diff;   // 增量传播
  }
  function getRange(scrollTop, viewH) {
    const start = Math.max(0, binarySearch(scrollTop) - overscan);
    const end = Math.min(count, binarySearch(scrollTop + viewH) + overscan);
    return { start, end, offsetY: offsets[start], total: offsets[count] };
  }
  return { getRange, measure, scrollToIndex: (i) => offsets[i] };
}
```

**评分点**：
- 累计数组 + 二分查找（不定高 O(log n)）
- 高度校准的增量传播（diff 只更新后续——不整表重算）
- 阈值防抖（1px 内不触发重算）
- overscan 缓冲 + 总高哨兵
