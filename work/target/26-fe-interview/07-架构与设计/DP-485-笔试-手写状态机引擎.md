## DP-485：笔试：手写状态机引擎

**答**：实现状态机引擎 `createMachine`：① 迁移表驱动：{ state: { event: { target, guard, action } } }；② send(event, payload)：guard 校验 → action 执行 → 状态迁移 → 订阅通知；③ can(event)：UI 驱动（当前状态允许的事件集合——按钮渲染）；④ 历史与回退：迁移日志栈 + back()（恢复上状态 + context 快照）；⑤ context 管理：action 返回新 context（不可变更新）；⑥ 非法迁移告警（连续非法事件计数——配置错误信号）。写出实现。

```javascript
export function createMachine({ states, initial, context = {} }) {
  let current = initial;
  let ctx = { ...context };
  const subs = new Set();
  const history = [];
  let illegal = 0;

  function send(event, payload = {}) {
    const t = states[current]?.on?.[event];
    if (!t) {
      if (++illegal > 10) console.warn('[fsm] 非法迁移过多, 检查配置');
      return { ok: false, reason: 'illegal' };
    }
    if (t.guard && !t.guard(ctx, payload)) return { ok: false, reason: 'guard' };
    states[current]?.exit?.(ctx);
    history.push({ from: current, event, ctx: { ...ctx } });
    ctx = t.action?.({ ...ctx }, payload) ?? ctx;     // 不可变 context
    current = t.target;
    states[current]?.enter?.(ctx, payload);
    subs.forEach((f) => f({ state: current, ctx, event }));
    return { ok: true, state: current, ctx };
  }
  function back() {
    const h = history.pop();
    if (!h) return { ok: false };
    current = h.from; ctx = h.ctx;                    // 快照恢复
    subs.forEach((f) => f({ state: current, ctx, event: '__back__' }));
    return { ok: true };
  }
  return { send, back, can: (e) => !!states[current]?.on?.[e],
    get state() { return current; }, get ctx() { return ctx; },
    sub(fn) { subs.add(fn); return () => subs.delete(fn); } };
}
// UI: <button disabled={!fsm.can('pay')} onClick={() => fsm.send('pay', { amount })}>
```

**评分点**：
- guard/action/target 的完整迁移语义
- can() 驱动 UI（非法状态不可能出现）
- back() 的 context 快照恢复
- 非法迁移计数告警（配置错误信号）
