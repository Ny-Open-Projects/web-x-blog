## DP-308：笔试题：手写监控错误聚合看板数据层？
**题目**：实现 ErrorAggregator：1) ingest(errors) 批量入库；2) 指纹聚合（stack 前 3 帧 + 消息参数归一化）；3) 时间窗口查询（hourly rate：count / pv）；4) 突增检测（同环比 3 倍 + 最小样本）。
**评分点**：
- 指纹归一化正确（数字/URL 参数替换）。
- 聚合结构支持窗口查询（bucket by hour）。
- 突增检测含最小样本与环比基准。
- 输出可用于告警规则消费。
```javascript
class ErrorAggregator {
  #buckets = new Map();        // 'fp|hourKey' → count
  ingest(errors) { for (const e of errors) { const fp = fingerprint(e);
    const key = `${fp}|${hourKey(e.ts)}`;
    this.#buckets.set(key, (this.#buckets.get(key) ?? 0) + 1); } }
  rate(fp, from, to, pvs) {    // 3) 窗口查询
    let count = 0;
    for (let h = from; h < to; h += 3600_000) count += this.#buckets.get(`${fp}|${hourKey(h)}`) ?? 0;
    return { count, rate: count / pvs };
  }
  spike(fp, now, pvs) {        // 4) 突增
    const cur = this.rate(fp, now - 3600_000, now, pvs.current);
    const base = this.rate(fp, now - 7 * 3600_000, now - 3600_000, pvs.base);   // 同环比基准
    return { spike: cur.count >= 20 && cur.rate > base.rate * 3, cur: cur.count, base: base.count };
  }
}
function fingerprint(e) {
  const stack = (e.stack ?? '').split('\n').slice(0, 3).join('|');
  return `${stack}::${(e.message ?? '').replace(/\d+/g, 'N').replace(/https?:\/\/\S+/g, 'URL')}`;
}
```
