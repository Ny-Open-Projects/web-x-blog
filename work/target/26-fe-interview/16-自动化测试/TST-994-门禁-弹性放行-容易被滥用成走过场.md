## TST-994：门禁「弹性放行」容易被滥用成走过场，怎么把每次豁免都管得住？
**答**：
- 结论：豁免必须「有理由、有审批、有期限、有回填」——每次 bypass 记录 reason/approver/expiresAt/补偿测试任务，到期自动回收权限。
- 要点：豁免按模块/信号粒度而非全局；豁免次数超阈值触发复盘；审计日志不可篡改。
- 价值：既保留紧急通道，又让「谁放的水」可追溯。
- 追问钩子：某团队一个月用了 15 次豁免，你怎么用数据判断是流程合理还是门禁设计有问题？

```javascript
function grantBypass({ module, signal, reason, approver, hours = 24 }) {
  if (!reason || !approver) throw new Error('豁免必须提供理由与审批人');
  return {
    id: `BP_${Date.now()}`, module, signal, reason, approver,
    expiresAt: new Date(Date.now() + hours * 3600e3).toISOString(),
    compensationTask: `补测-${module}-${signal}`, // 到期必须补测
    active: true,
  };
}
function activeBypasses(bypasses) {
  return bypasses.filter(b => b.active && new Date(b.expiresAt) > new Date());
}
function abuseAlert(bypasses, threshold = 10) {
  const byModule = {};
  for (const b of bypasses) (byModule[b.module] ||= 0)++;
  return Object.entries(byModule).filter(([, n]) => n > threshold);
}
const bp = grantBypass({ module: 'pay', signal: 'visual', reason: '大促临时改图', approver: 'lead' });
console.log(bp.compensationTask); // '补测-pay-visual'
```
