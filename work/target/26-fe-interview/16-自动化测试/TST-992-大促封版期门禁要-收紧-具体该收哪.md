## TST-992：大促封版期门禁要「收紧」，具体该收哪些、松哪些？
**答**：
- 结论：收紧的是「放行门槛」与「审批层级」——核心路径必须全绿、覆盖率增量不得下降、任何豁免需双人审批；松的是「非核心页面的视觉/低优用例」可转建议级。
- 节奏：封版前 3 天冻结门禁配置，禁止临时放宽；封版期只允许 hotfix 通道进入。
- 兜底：保留一条紧急通道但强制事后 24h 内补测与复盘。
- 追问钩子：封版当天 P0 bug 修复需要改 3 个核心文件但只来得及跑单测，门禁怎么放又不失控？

```javascript
const PROMO_POLICY = {
  tightened: [
    '核心路径 E2E 必须 100% 通过',
    '增量覆盖率不得低于基线',
    '任何豁免需 tech-lead + QA lead 双人审批',
  ],
  relaxed: ['非核心页视觉回归转建议级', '低优 flaky 用例隔离'],
  emergency: { channel: 'hotfix', requirePostTestHours: 24, requireRetro: true },
};
function promoGate({ isCore, e2ePass, diffCoverageOK, approvers }) {
  if (isCore && !e2ePass) return { pass: false, reason: '核心路径 E2E 未全绿' };
  if (isCore && !diffCoverageOK) return { pass: false, reason: '增量覆盖率不足' };
  if (!isCore) return { pass: true, level: 'suggestion' };
  return { pass: approvers.length >= 2, reason: '需双人审批' };
}
console.log(promoGate({ isCore: true, e2ePass: true, diffCoverageOK: true, approvers: ['a'] }));
// { pass:false, reason:'需双人审批' }
```
