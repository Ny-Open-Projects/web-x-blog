### 精讲 1 · [HW-038] 手写 Vue reactive 核心

**为什么考这个**：响应式是 Vue 的立身之本，也是区分「用过 Vue」和「读过 Vue」的分水岭。30 行能跑的 reactive，比十分钟「依赖收集」的口头描述有说服力。原卡给的是 Vue2 `Object.defineProperty` 版本，这里给 Vue3 `Proxy` 版本——两者的差异恰好是最高频的追问，所以两份都要能写。

**标准实现**：

```javascript
// Vue3 响应式核心：reactive / track / trigger / effect
let activeEffect = null;            // 当前正在执行的副作用
const effectStack = [];             // 嵌套 effect 栈：内层跑完恢复外层
const targetMap = new WeakMap();    // target -> Map<key, Set<effect>>

function reactive(target) {
  if (target === null || typeof target !== 'object') return target;
  return new Proxy(target, {
    get(obj, key, receiver) {
      const res = Reflect.get(obj, key, receiver);
      track(obj, key);                                      // 读取时收集依赖
      return typeof res === 'object' ? reactive(res) : res; // 惰性深层代理：用到才包
    },
    set(obj, key, value, receiver) {
      const old = obj[key];
      const ok = Reflect.set(obj, key, value, receiver);
      if (!Object.is(old, value)) trigger(obj, key);         // 值真变了才触发
      return ok;
    },
    deleteProperty(obj, key) {
      const had = key in obj;
      const ok = Reflect.deleteProperty(obj, key);
      if (had) trigger(obj, key);                            // 删除也能触发，Proxy 独有
      return ok;
    },
  });
}

function track(target, key) {
  if (!activeEffect) return;            // 没有副作用在读，不收集
  let depsMap = targetMap.get(target);
  if (!depsMap) targetMap.set(target, (depsMap = new Map()));
  let dep = depsMap.get(key);
  if (!dep) depsMap.set(key, (dep = new Set()));
  dep.add(activeEffect);                // Set 去重：同一 effect 只记一次
  activeEffect.deps.push(dep);          // 反向记录，便于执行前清理旧依赖
}

function trigger(target, key) {
  const dep = targetMap.get(target)?.get(key);
  if (!dep) return;
  [...dep].forEach((run) => run());     // 拷贝：effect 内可能新增依赖，直接遍历会死循环
}

function effect(fn) {
  const runner = () => {
    cleanup(runner);                    // 先清旧依赖，避免条件分支残留导致多余触发
    activeEffect = runner;
    effectStack.push(runner);
    try { return fn(); }
    finally {
      effectStack.pop();
      activeEffect = effectStack[effectStack.length - 1] || null;
    }
  };
  runner.deps = [];
  runner();
  return runner;
}

function cleanup(runner) {
  runner.deps.forEach((dep) => dep.delete(runner));
  runner.deps.length = 0;
}

// ================= 验证 =================
const state = reactive({ count: 0, nested: { msg: 'a' }, list: [] });
let countRuns = 0;
let nestedRuns = 0;
effect(() => { countRuns++; state.count; });      // 只依赖 count
effect(() => { nestedRuns++; state.nested.msg; }); // 依赖嵌套字段

state.count++;
state.count++;
console.log('count effect 执行次数(含首次):', countRuns);   // 1 + 2 = 3
state.nested.msg = 'b';
console.log('nested effect 执行次数(含首次):', nestedRuns); // 2：首次 + 一次深层修改
state.added = 1;                                  // 新增属性也触发
state.list.push(9);                               // 数组索引 + length 都触发
console.log('状态:', state.nested.msg, state.added, state.list.length);
```

五个函数的职责必须一句话说清：`reactive` 包代理、`track` 收集、`trigger` 派发、`effect` 注册副作用、`cleanup` 清旧依赖。跑一遍输出是「count 3 次 / nested 2 次」，说明深层字段和数组都被代理到了。

**必须覆盖的边界**：
- 嵌套对象要**惰性代理**：在 `get` 里才 `reactive`，而不是初始化时递归遍历一遍，否则大对象首屏白付成本。
- 数组：`push` 会同时触发索引 set 和 `length` set，`trigger` 必须容忍同一 key 被通知两次，别做成「一次操作一次渲染」的强假设。
- 副作用去重：用 `Set` 存 effect；同时 `track` 时反向记录 `runner.deps`，下次执行前 `cleanup`，否则条件分支切换后旧依赖仍然会触发。
- 嵌套 effect：`effectStack` 保存外层 runner，内层结束后必须恢复 `activeEffect`，不然内层读到的属性会错误地收集到内层 effect 上。
- 值未变不触发：用 `Object.is` 比较，而不是 `!==`（`NaN` 自比较为 false）。
- 删除属性：`deleteProperty` 陷阱要单独写，这正是 `defineProperty` 做不到的事。

**追问链**：
1. **Proxy 和 defineProperty 的本质差别？** → 前者拦截整个对象，新增属性、删除属性、数组索引与 `length`、`in`、`ownKeys` 都能拦；后者只能劫持初始化时已存在的属性，Vue2 因此要用 `$set` 补丁。
2. **为什么必须用 `Reflect`？** → 保证 getter/setter 里的 `this` 指向代理对象（receiver 透传），否则访问器属性会把 `this` 逃逸回原始对象，导致依赖收集错位；同时 `set`/`deleteProperty` 统一返回布尔值，严格模式下失败语义一致。
3. **为什么依赖表是三层结构？** → `WeakMap<target, Map<key, Set<effect>>>`：WeakMap 让 target 失去引用后可被 GC；按 key 精确触发，改 A 不会惊动只读 B 的 effect；Set 做去重。
4. **Proxy 的代价是什么？** → 不支持 IE 且无法 polyfill；每次 get 多一层 trap 调用，热路径上比原生访问慢；所以 Vue3 对 `ref` 的原始值做了浅层优化。

### 精讲 2 · [HW-040] 手写 React useState

**为什么考这个**：useState 是 hooks 体系的第一块砖，写出来就等于把「为什么 hooks 不能写在 if 里」讲清楚了。面试官要看到的不是 API 复刻，而是**顺序对齐**与**更新调度**这两个机制。

**标准实现**：

```javascript
// 简版 React：hooks 存在数组里，按调用顺序用自增下标对齐
let hookIndex = 0;
const hooks = [];
let scheduled = false;
let App = null;
let renders = 0;

function useState(initial) {
  const i = hookIndex++;
  // 只在第一次渲染时初始化；传函数则惰性求值（React 同款语义）
  if (!(i in hooks)) hooks[i] = typeof initial === 'function' ? initial() : initial;
  const setState = (next) => {
    const value = typeof next === 'function' ? next(hooks[i]) : next; // 函数式更新
    if (Object.is(value, hooks[i])) return;  // 同值 bail-out：不触发渲染
    hooks[i] = value;
    schedule();
  };
  return [hooks[i], setState];
}

function schedule() {
  if (scheduled) return;                     // 同一轮多次 setState 合并成一次渲染
  scheduled = true;
  queueMicrotask(() => { scheduled = false; render(); });
}

function render() {
  hookIndex = 0;                             // 每次渲染下标归零，重新按顺序对齐
  renders++;
  App();
}

function mount(component) {
  App = component;
  render();
}

// ================= 验证 =================
const log = [];
let setCount = null;

function Counter() {
  const [n, setN] = useState(0);
  const [label] = useState(() => 'counter');   // 工厂函数只在首次执行
  setCount = setN;
  log.push(`${label}:${n}`);
}

const flush = () => new Promise((r) => setTimeout(r, 0)); // 等微任务跑完

(async () => {
  mount(Counter);
  console.log('首屏:', log, '渲染次数:', renders);

  setCount(1);
  setCount(2);                                 // 同一轮两次 setState → 只渲染一次
  await flush();
  console.log('批量更新后:', log, '渲染次数:', renders);

  setCount((n) => n + 1);                      // 函数式更新基于最新值
  await flush();
  console.log('函数式更新后:', log, '渲染次数:', renders);

  setCount(3);                                 // 与当前值相同，不触发渲染
  await flush();
  console.log('同值更新后 渲染次数:', renders);
})();
```

运行结果值得逐行念给面试官听：首次渲染 1 次；同一轮调两次 `setCount` 只触发 1 次渲染（渲染次数从 1 变 2，值直接是 2）；函数式更新基于最新值算出 3；`setCount(3)` 与当前值相同则完全跳过。四个数字就是四个考点。

**必须覆盖的边界**：
- **按调用顺序对齐**：状态存在模块级数组里，靠 `hookIndex++` 定位。条件分支里调用 hooks 会让下标错位，读到的就是别人的 state——这是「Hooks 规则」的物理原因，必须亲手指出。
- **函数式更新**：`setState(v => v + 1)` 要基于 `hooks[i]` 的最新值计算，而不是闭包里的旧值，否则连点两次只加一次。
- **批量更新**：同一轮多次 `setState` 用微任务合并成一次渲染，否则每次 set 都同步重渲染会白白多跑几帧。
- **同值 bail-out**：`Object.is` 相等就不调度，这是 `setState(0)` 不会死循环的原因。
- **惰性初始化**：`useState(() => heavy())` 只在首次执行工厂函数，后续渲染不能重复调用。
- **渲染期重置下标**：`hookIndex = 0` 必须在组件函数执行前完成。

**追问链**：
1. **为什么 hooks 不能写在条件语句里？** → 状态靠调用顺序与数组下标绑定，少调一次后面的 hooks 全体前移，读到错误状态；React 内部用链表实现，原理相同。
2. **setState 是同步还是异步？** → 在 React 18 的自动批处理下是异步的（事件、Promise、setTimeout 都会被批处理），同一轮多次更新合并；`flushSync` 可强制同步。
3. **闭包陷阱怎么解释？** → 每次渲染的函数组件都有独立作用域，事件处理器捕获的是当次渲染的 state 快照；要拿最新值用函数式更新或 ref。
4. **和 useReducer 的关系？** → useState 可看作只有一个 action 的 useReducer；状态转移逻辑复杂、需要多字段联动时上 reducer。

### 精讲 3 · [HW-042] 手写 useEffect 简版

**为什么考这个**：useEffect 是"副作用"与"渲染"边界所在，考察点是**执行时机**（渲染后）、**依赖语义**（三态）、**清理顺序**（先旧后新）三件事，几乎每一条都能对应到线上 bug。

**标准实现**：

```javascript
// useEffect 简版：渲染期只登记，提交后统一执行（cleanup 先于新 effect）
let hookIndex = 0;
let slots = [];
let pending = [];

function useEffect(fn, deps) {
  const i = hookIndex++;
  const prev = slots[i];
  const changed =
    !prev ||                                   // 首次挂载必跑
    deps === undefined ||                      // 不传 deps：每次渲染都跑
    deps.length !== prev.deps.length ||
    deps.some((d, k) => !Object.is(d, prev.deps[k])); // 逐项浅比较
  pending.push({ i, fn, deps, changed });
}

function commit() {
  const jobs = pending;
  pending = [];
  for (const { i, fn, deps, changed } of jobs) {
    if (!changed) continue;                    // 依赖没变，跳过整轮
    slots[i]?.cleanup?.();                     // 关键顺序：旧 cleanup → 新 effect
    const cleanup = fn();
    slots[i] = { deps, cleanup: typeof cleanup === 'function' ? cleanup : undefined };
  }
}

function render(component) {
  hookIndex = 0;                               // 每次渲染下标归零
  pending = [];
  component();
  commit();
}

function unmount() {
  slots.forEach((s) => s?.cleanup?.());
  slots = [];
}

// ================= 验证 =================
const logs = [];
let props = { id: 'A', n: 0 };
let forced = 0;

function Profile() {
  forced++;                                    // 标记组件函数真的重新执行了
  useEffect(() => {
    logs.push('mount');                        // [] ：只挂载跑一次
    return () => logs.push('unmount');         // 卸载时才清理
  }, []);
  useEffect(() => {
    const n = props.n;                         // 捕获当次渲染的值（闭包快照）
    logs.push(`fetch:${n}`);                  // [n]：依赖变化才重跑
    return () => logs.push(`cancel:${n}`);
  }, [props.n]);
  useEffect(() => {
    logs.push('always');                       // 无 deps：每次都跑
  });
}

render(Profile);
props = { ...props, n: 0 };
render(Profile);                               // n 没变：只跑 always
props = { ...props, n: 1 };
render(Profile);                               // n 变了：先 cancel:0 再 fetch:1
unmount();
console.log('组件执行次数:', forced);
console.log(logs);
```

关键在 `commit` 那十行：所有 effect 先登记再统一执行，`slots[i]?.cleanup?.()` 一定在新 effect 之前。日志输出 `mount, fetch:0, always` → 第二次渲染只有 `always` → 第三次是 `cancel:0, fetch:1, always` → 卸载时 `unmount, cancel:1`，顺序一目了然。

**必须覆盖的边界**：
- **deps 三种语义**：`undefined` 每次渲染后都跑；`[]` 只挂载跑一次；`[a, b]` 依赖变化才跑。三者写错一个就是死循环或数据不更新。
- **cleanup 顺序**：先执行上一轮 cleanup，再执行新 effect；卸载时按登记顺序统一清理。顺序反了会出现「旧定时器没清就开新的」。
- **浅比较**：逐项 `Object.is`，长度不同直接判定变化。这也是「deps 里放对象会每次重跑」的原因。
- **异步提交**：渲染期不能有副作用，必须等 DOM 提交后再执行（浏览器里是绘制后，用 `queueMicrotask`/`setTimeout` 模拟）。
- **闭包快照**：effect 里读到的 props 是当次渲染的值，cleanup 里若引用外层变量，捕获的也是当次值——所以示例里先 `const n = props.n`。
- **执行次数与渲染次数不是一回事**：依赖没变时组件函数照跑，但 effect 跳过。

**追问链**：
1. **useEffect 和 useLayoutEffect 的区别？** → 前者在绘制后异步执行，不阻塞首屏；后者在 DOM 变更后、绘制前同步执行，用于测量布局、避免闪烁，代价是阻塞渲染。
2. **为什么依赖数组少写会出 bug？** → effect 捕获的是旧闭包，读到过期 state；ESLint 的 exhaustive-deps 规则就是防这个。
3. **effect 里的请求怎么取消？** → cleanup 里 `controller.abort()`，或置 `mounted` 标志位，见 HW-138 / HW-046 的竞态处理。
4. **StrictMode 下为什么执行两次？** → React 18 开发环境故意 mount→unmount→mount，暴露「cleanup 写不干净」的副作用；生产环境不会。

### 精讲 4 · [HW-043] 手写同层简单 diff

**为什么考这个**：diff 是虚拟 DOM 的心脏。面试官并不指望你写 React 的完整算法，但要看你能不能讲清**为什么能降到 O(n)**，以及 key 到底解决了什么问题。

**标准实现**：

```javascript
// 同层 diff：节点 { type, key, props, children }，输出补丁列表
function diff(oldNode, newNode, patches = [], path = 'root') {
  if (!oldNode) { patches.push({ path, op: 'CREATE', node: newNode }); return patches; }
  if (!newNode) { patches.push({ path, op: 'REMOVE' }); return patches; }
  if (oldNode.type !== newNode.type) {          // 类型不同：整棵替换，不再往下比
    patches.push({ path, op: 'REPLACE', node: newNode });
    return patches;
  }
  diffProps(oldNode, newNode, patches, path);
  diffChildren(oldNode.children || [], newNode.children || [], patches, path);
  return patches;
}

function diffProps(oldNode, newNode, patches, path) {
  const set = {};
  const a = oldNode.props || {};
  const b = newNode.props || {};
  for (const k of Object.keys(b)) if (a[k] !== b[k]) set[k] = b[k];   // 新增 / 修改
  const remove = Object.keys(a).filter((k) => !(k in b));             // 删除
  if (Object.keys(set).length) patches.push({ path, op: 'PROPS', set });
  if (remove.length) patches.push({ path, op: 'PROPS', remove });
}

function diffChildren(oldList, newList, patches, path) {
  // key 优先；没有 key 就用下标兜底（等价于 React 的 index key，重排必然低效）
  const oldMap = new Map(oldList.map((c, i) => [c.key ?? `i:${i}`, { node: c, index: i }]));
  newList.forEach((child, i) => {
    const k = child.key ?? `i:${i}`;
    const hit = oldMap.get(k);
    const childPath = `${path}.children[${i}]`;
    if (!hit) { patches.push({ path: childPath, op: 'CREATE', node: child }); return; }
    oldMap.delete(k);                            // 标记已复用
    if (hit.index !== i) patches.push({ path: childPath, op: 'MOVE', from: hit.index });
    diff(hit.node, child, patches, childPath);   // 同 key 的节点继续向下比
  });
  for (const { index } of oldMap.values()) {     // 没被复用的：删除
    patches.push({ path: `${path}.children[${index}]`, op: 'REMOVE' });
  }
}

// ================= 验证 =================
const el = (type, key, props = {}, children = []) => ({ type, key, props, children });
const oldTree = el('ul', 'root', { class: 'list' }, [
  el('li', 'a', { text: 'A' }),
  el('li', 'b', { text: 'B' }),
  el('li', 'c', { text: 'C' }),
]);
const newTree = el('ul', 'root', { class: 'list active' }, [
  el('li', 'c', { text: 'C2' }),                  // 位置 0：从 2 移过来且文本变了
  el('li', 'a', { text: 'A' }),
  el('li', 'b', { text: 'B' }),
  el('li', 'd', { text: 'D' }),                   // 新增
]);
console.log(JSON.stringify(diff(oldTree, newTree), null, 2));
```

输出里能看到三件事：根节点 `class` 变化产生 PROPS 补丁；`c` 从下标 2 移到 0 产生 MOVE 且文本变化继续向下 diff；`d` 是新节点产生 CREATE。每个补丁都带 path，方便在真实 DOM 上定位。

**必须覆盖的边界**：
- **同层比较**：只比同一层级，跨层移动视为「删除 + 新建」整棵子树，这是把复杂度从 O(n³) 降到 O(n) 的前提。
- **type 不同直接 REPLACE**：不再深入，否则白算一整棵子树。
- **props 三类变化**：新增/修改（set）、删除（remove）要分开，删不掉的 props（比如事件监听）是内存泄漏源头。
- **key 复用**：有 key 时按 key 配对，无 key 时退化为下标配对；用下标做 key 时列表头部插入会「全量更新」，这就是「不要用 index 做 key」的实证。
- **补丁顺序**：先 props 后 children，先父后子；CREATE 的节点不需要再 diff 子树。
- **纯函数约定**：diff 不碰 DOM，只产出补丁，渲染层再应用——这一层分离是测试与 SSR 的基础。

**追问链**：
1. **为什么需要 key？** → key 是节点的稳定身份。没有 key 只能按位置猜，头部插入会让所有节点都「内容变了」；有 key 才能识别「同一个节点挪了位置」。
2. **MOVE 是怎么算的？** → 这一版是朴素的「位置变了就 MOVE」，最坏产生 n 次移动；工程实现会用最长递增子序列（LIS）找出「不用动」的节点，其余才移动。
3. **Vue 的双端 diff 好在哪？** → 头头、尾尾、头尾、尾头四指针同时逼近，常见场景（头部插入、尾部追加）只需 O(1) 次移动；React 的 Fiber 用单链表 + `oldIndex` 比较。
4. **diff 结果怎么落地？** → 补丁按 depth 排序后批量应用，减少重排；真实框架还会合并同一帧的补丁。

### 精讲 5 · [HW-105] 错误边界 + chunk 加载失败重试

**为什么考这个**：这是唯一一道「线上真的救过命」的题。发版后旧 hash 的 chunk 404 是 CDN 部署的经典事故，只会写 try/catch 的人答不出「为什么必须整页刷新」。

**标准实现**：

```jsx
import React from 'react';

const CHUNK_ERROR = /Failed to fetch dynamically imported module|Loading chunk \d+ failed|Importing a module script failed/i;
const RELOAD_KEY = 'chunk-reload-at';

const report = (payload) => {
  navigator.sendBeacon?.('/log', JSON.stringify(payload));   // 上报：不阻塞渲染
};

export class ErrorBoundary extends React.Component {
  state = { error: null, retries: 0 };

  static getDerivedStateFromError(error) {
    return { error };                                  // 渲染期抛错 → 立刻切降级 UI
  }

  componentDidCatch(error, info) {
    report({ type: 'render-error', msg: String(error), stack: info.componentStack });
    if (CHUNK_ERROR.test(String(error))) this.reloadOnce();   // chunk 404：整页刷新拿新 hash
  }

  reloadOnce() {
    const last = Number(sessionStorage.getItem(RELOAD_KEY) || 0);
    if (Date.now() - last < 10_000) return;             // 10s 内只刷一次，避免死循环
    sessionStorage.setItem(RELOAD_KEY, String(Date.now()));
    location.reload();
  }

  retry = () => {
    if (this.state.retries >= 2) return;                // 手动重试封顶
    this.setState((s) => ({ error: null, retries: s.retries + 1 }));  // 清错误 → 子树重新挂载
  };

  render() {
    const { error } = this.state;
    if (!error) return this.props.children;
    return this.props.fallback
      ? this.props.fallback({ error, retry: this.retry })
      : React.createElement('div', null, '页面加载失败', React.createElement('button', { onClick: this.retry }, '重试'));
  }
}

// 懒加载重试：网络抖动 2 次退避重试，仍失败再交给 ErrorBoundary
export function lazyWithRetry(factory, attempts = 2) {
  return React.lazy(async () => {
    for (let i = 0; ; i += 1) {
      try { return await factory(); }
      catch (err) {
        if (i >= attempts) throw err;
        await new Promise((r) => setTimeout(r, 300 * 2 ** i));
      }
    }
  });
}
```

三层防线要在回答时按顺序点出来：ErrorBoundary 兜渲染期错误 → `lazyWithRetry` 兜网络抖动 → chunk 404 整页重载拿新 hash。

**必须覆盖的边界**：
- **Error Boundary 抓不到什么**：事件处理器、异步回调、SSR、自身抛出的错误。这三类要交给 `window.onerror` / `unhandledrejection`。
- **必须用 class**：`getDerivedStateFromError` 与 `componentDidCatch` 没有 hooks 版本，用函数组件答直接扣分。
- **防循环重载**：用 sessionStorage 打时间戳（示例 10 秒内只刷一次），否则 CDN 还没同步完就无限刷新，用户看到白屏闪烁。
- **区分错误类型**：只有 chunk/import 失败才重载，业务渲染错误重载没用，反而丢掉用户输入。
- **重试要封顶**：手动重试 2 次、自动退避 2 次，之后展示降级 UI 与反馈入口。
- **降级 UI 要有出口**：`fallback` 接收 `retry`，否则用户只能刷新页面。

**追问链**：
1. **chunk 失败后为什么不能只重新 import？** → 模块加载失败会被浏览器缓存为失败状态，同一 URL 再 import 仍走缓存；且旧页面里的 chunk 映射已失效，必须整页拿新 HTML。
2. **`React.lazy` 的重试为什么要放在 `lazy` 的工厂里？** → 因为失败发生在 import 阶段，包在组件外层 catch 不到；工厂函数内重试才能真正重发请求。
3. **Error Boundary 能不能嵌套？** → 能，且应该按路由/区块分层，颗粒度越细，崩一块不至于整页白屏。
4. **生产环境怎么定位？** → `componentDidCatch` 上报 `error.stack` + `componentStack` + 版本号，配合 sourcemap 还原。

### 精讲 6 · [HW-138] useQuery 最小实现（缓存 / 竞态 / 重试 / 失效 / 卸载中止）

**为什么考这个**：这是手写题的「综合大题」。单独考防抖、考竞态都不难，难的是把**缓存共享、在途去重、失效重取、卸载中止**放进同一份状态里还不打架。能把这道题讲成状态机，基本就够 P6 的深度。

**标准实现**：

```javascript
import { useCallback, useEffect, useRef, useState } from 'react';

// 缓存层是框架无关的：Map<key, entry>，entry 用同一个对象做「可变节点」
const cache = new Map();

function getEntry(key) {
  let entry = cache.get(key);
  if (!entry) {
    entry = { data: undefined, error: null, ts: 0, promise: null, controller: null, subs: new Set() };
    cache.set(key, entry);
  }
  return entry;
}

export function invalidate(key) {          // 失效：把 ts 归零 + 通知所有订阅组件重取
  const entry = cache.get(key);
  if (!entry) return;
  entry.ts = 0;
  entry.subs.forEach((notify) => notify());
}

export function useQuery(key, fetcher, { staleTime = 30_000, retries = 2 } = {}) {
  const [, force] = useState(0);
  const alive = useRef(true);
  const entry = getEntry(key);

  const run = useCallback(() => {
    const e = getEntry(key);
    if (e.promise) return e.promise;                    // 在途共享：并发调用只发一次
    const controller = new AbortController();
    e.controller = controller;
    e.promise = (async () => {
      for (let attempt = 0; ; attempt += 1) {
        try {
          const data = await fetcher({ key, signal: controller.signal });
          Object.assign(e, { data, error: null, ts: Date.now(), promise: null });  // 同一对象上更新
          e.subs.forEach((n) => n());
          return data;
        } catch (err) {
          if (err.name === 'AbortError') return undefined;   // 主动中止不算错误
          if (attempt >= retries) {
            Object.assign(e, { error: err, promise: null });
            e.subs.forEach((n) => n());
            throw err;
          }
          await new Promise((r) => setTimeout(r, 300 * 2 ** attempt)); // 指数退避重试
        }
      }
    })().finally(() => { e.promise = null; });
    return e.promise;
  }, [key, fetcher, retries]);

  useEffect(() => {
    alive.current = true;
    const notify = () => { if (alive.current) force((v) => v + 1); };
    entry.subs.add(notify);
    if (Date.now() - entry.ts > staleTime) run().catch(() => {});  // 新鲜则直接用缓存
    return () => {                                     // 卸载：退订 + 中止在途请求
      alive.current = false;
      entry.subs.delete(notify);
      entry.controller?.abort();
    };
  }, [entry, run, staleTime]);

  return {
    data: entry.data,
    error: entry.error,
    isLoading: entry.data === undefined && !entry.error,
    isFetching: Boolean(entry.promise),
    refetch: run,
  };
}
```

设计上刻意做了两层分离：`cache` 是框架无关的模块级单例，`useQuery` 只是把它接到 React 生命周期上。六个语义逐条对应：`ts + staleTime` 是缓存新鲜度；`e.promise` 是在途共享（天然去重）；`AbortController` 是取消；`attempt` 循环是退避重试；`subs` 是失效通知；`alive` 是卸载保护。

**必须覆盖的边界**：
- **同一 key 并发挂载只发一次请求**：第二个组件直接拿到 `e.promise`，两个组件共享同一份数据，这是 React Query 与 `useEffect + fetch` 最大的体验差异。
- **竞态**：key 变化时旧请求必须 abort，否则旧响应晚到会覆盖新 key 的 data。`controller` 挂在 entry 上，cleanup 里能精确中止。
- **AbortError 不算失败**：中止要走 `return` 而不是进重试，否则用户切页面会触发一串无意义重试。
- **入口对象身份稳定**：用 `Object.assign` 原地更新而不是 `cache.set(key, {...})`，否则 effect 捕获的旧引用读到永远过期的值。
- **失效重取**：`invalidate` 把 `ts` 归零并通知订阅者，组件不卸载也能拿到新数据。
- **状态机完整**：`isLoading`（首次无数据）、`isFetching`（在途）、`data`、`error` 四态不能混，loading 与 fetching 混用会导致列表刷新时整页骨架屏闪烁。

**追问链**：
1. **为什么不用 useEffect 里直接 fetch？** → 没有缓存与去重，两个组件各发一次；切回页面必重新转圈；卸载后 setState 报警告；没有统一重试与失效入口。
2. **缓存什么时候清？** → 需要 `gcTime`（默认 5 分钟）+ 引用计数：最后一个订阅者卸载后启动定时器，超时删除 entry；否则 SPA 常驻会一直吃内存。
3. **重试为什么要退避？** → 服务端 5xx 或网络抖动时，立即重试只会加剧拥塞；指数退避 + 抖动（jitter）避免重试风暴。
4. **和 SWR 的差别？** → SWR 的 `mutate` 更偏乐观更新，React Query 的 `queryClient` 提供了更细的失效粒度（按 key 前缀批量失效）；核心机制一致：stale-while-revalidate。

### HW-037 虚拟列表核心（定高）

**评分点**：三个量 + buffer + 总高度。定高才好算，不定高要先测量再二分查找。

```javascript
function getVisibleRange(scrollTop, viewportH, total, itemH, buffer = 3) {
  const start = Math.max(0, Math.floor(scrollTop / itemH) - buffer);
  const count = Math.ceil(viewportH / itemH) + buffer * 2;   // 上下各留 buffer 条防白屏
  return {
    start,
    end: Math.min(total, start + count),
    offset: start * itemH,          // 只渲染可见区，用 transform 平移
    totalHeight: total * itemH,     // 撑出真实滚动条
  };
}
```

**追问**：为什么用 transform 而不是改 `padding-top`？→ transform 只触发合成，不触发布局；改 padding 会让每帧都 layout。定高场景可以直接算；不定高需要 `ResizeObserver` 测量 + 前缀和二分。

### HW-039 Vue v-model 双向绑定（简版）

**评分点**：编译（找 `v-model` 与 `{{}}`）+ 数据劫持（定义 getter/setter）+ 更新函数表。一句话总结：**v-model = `:value` + `@input` 的语法糖**。

```javascript
class MiniVue {
  constructor({ el, data }) {
    this.root = document.querySelector(el);
    this.data = data;
    this.updaters = new Map();                 // key -> Set<updater>
    Object.keys(data).forEach((k) => this._reactive(k, data[k]));
    this._compile(this.root);
  }
  _reactive(key, value) {
    const set = new Set();
    this.updaters.set(key, set);
    Object.defineProperty(this.data, key, {
      get: () => value,
      set: (next) => { value = next; set.forEach((fn) => fn(next)); },   // 数据 → 视图
    });
  }
  _bind(key, updater) {
    this.updaters.get(key).add(updater);
    updater(this.data[key]);                   // 初始化视图
  }
  _compile(node) {
    node.childNodes.forEach((child) => {
      if (child.nodeType === 3) {              // 文本节点：{{ msg }}
        const m = child.textContent.match(/\{\{\s*(\w+)\s*\}\}/);
        if (m) this._bind(m[1], (v) => { child.textContent = v; });
      } else if (child.nodeType === 1) {
        if (child.hasAttribute('v-model')) {
          const key = child.getAttribute('v-model');
          this._bind(key, (v) => { child.value = v; });          // 数据 → 视图
          child.addEventListener('input', (e) => { this.data[key] = e.target.value; }); // 视图 → 数据
        }
        this._compile(child);
      }
    });
  }
}
```

**追问**：Vue3 的 v-model 为什么改了？→ `value`/`input` 改成 `modelValue`/`update:modelValue`，一个组件可挂多个 v-model；本质仍是 props 下行 + 事件上行，**单向数据流 + 语法糖**。

### HW-041 useMemo / useCallback

**评分点**：useMemo 缓存值，useCallback 缓存函数引用，`useCallback(fn, deps) === useMemo(() => fn, deps)`。依赖浅比较，`deps` 省略等于不缓存。

```javascript
let hookIndex = 0;
const slots = [];
const sameDeps = (a, b) =>
  a === b || (a && b && a.length === b.length && a.every((v, i) => Object.is(v, b[i])));

function useMemo(factory, deps) {
  const i = hookIndex++;
  const prev = slots[i];
  if (!prev || !sameDeps(prev.deps, deps)) slots[i] = { value: factory(), deps };
  return slots[i].value;
}
function useCallback(fn, deps) {
  return useMemo(() => fn, deps);              // 把函数当值缓存
}
```

**追问**：useMemo 能当性能优化银弹吗？→ 不能。它自身有比较与存储成本，且不保证不被丢弃（React 文档明确说语义上是性能提示，不是缓存保证）；真正的优化顺序是减少渲染范围 → 拆分组件 → 再考虑 memo。

### HW-044 受控组件表单验证 hook

**评分点**：`values` 与 `errors` 分开存；change 只校验当前字段，submit 全量校验；`handleSubmit` 要 `preventDefault`，不通过就聚焦第一个错误字段。

```javascript
function validateAll(values, validators) {
  const errors = {};
  for (const [name, rule] of Object.entries(validators)) {
    const msg = rule(values[name], values);   // 返回文案即错误，undefined 表示通过
    if (msg) errors[name] = msg;
  }
  return errors;
}
// hook 里：handleChange(name, value) 只跑 validators[name]；
// handleSubmit 跑 validateAll，Object.keys(errors).length 决定是否提交
```

**追问**：受控与非受控怎么选？→ 受控强一致、可即时校验、便于联动，代价是每次输入都重渲染；非受控（`defaultValue` + ref）适合大表单与文件输入，校验时机放到 blur/submit。antd Form 内部是「受控 + 字段级订阅」，只重渲染变化的字段。

### HW-045 useDebounce

**评分点**：返回的是**延迟后的值**，不是防抖函数。精髓在 cleanup：值再变就清掉上一轮定时器，天然只剩最后一轮生效。

```javascript
// useState / useEffect 来自 react
function useDebounce(value, delay = 300) {
  const [debounced, setDebounced] = useState(value);
  useEffect(() => {
    const timer = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(timer);   // 关键：下一轮进来先取消上一轮
  }, [value, delay]);
  return debounced;
}
```

**追问**：为什么不直接防抖请求函数？→ 搜索场景要防的是「值触发的副作用」，用延迟值更声明式；且多个下游（请求、埋点、空状态）共享同一个节流后的值，不会各自防抖导致节奏不一致。

### HW-046 useRequest 最小版

**评分点**：`loading/data/error` 三态、`run` 返回 Promise、**竞态 id**、卸载后不 setState。

```javascript
// useState / useRef / useEffect / useCallback 来自 react
function useRequest(fn) {
  const [state, setState] = useState({ loading: false, data: null, error: null });
  const latest = useRef(0);
  const mounted = useRef(true);
  useEffect(() => () => { mounted.current = false; }, []);
  const run = useCallback(async (...args) => {
    const id = ++latest.current;             // 只认最新一次
    setState({ loading: true, data: null, error: null });
    try {
      const data = await fn(...args);
      if (id === latest.current && mounted.current) setState({ loading: false, data, error: null });
      return data;
    } catch (error) {
      if (id === latest.current && mounted.current) setState({ loading: false, data: null, error });
      throw error;
    }
  }, [fn]);
  return { ...state, run };
}
```

**追问**：竞态 id 和 AbortController 该用哪个？→ 两者互补：id 保证「不写入过期结果」，abort 保证「不浪费带宽并释放连接」。生产实现两个都要，且 abort 后要吞掉 `AbortError`。

### HW-047 keep-alive 思想

**评分点**：缓存的是**组件实例 + vnode + DOM**，切走不销毁、切回不重建；缓存必须有 LRU 上限，淘汰时才真正 destroy。

```javascript
class KeepAlive {
  constructor(max = 5) { this.max = max; this.cache = new Map(); }
  get(key, create) {
    if (this.cache.has(key)) {
      const inst = this.cache.get(key);
      this.cache.delete(key);       // 命中先删再插 → 移到队尾
      this.cache.set(key, inst);
      inst.activated?.();           // 只触发 activated，不重新 mount
      return inst;
    }
    const inst = create();
    this.cache.set(key, inst);
    if (this.cache.size > this.max) {
      const [oldestKey, oldest] = this.cache.entries().next().value;
      this.cache.delete(oldestKey);
      oldest.destroyed?.();         // 只有被淘汰才真正销毁
    }
    return inst;
  }
}
```

**追问**：React 没有官方 keep-alive 怎么办？→ 三种做法：CSS `display:none` 保活（最简单，但状态在父层）、状态提升到 store、或用 Offscreen/Activity（React 实验特性）。Vue 的 `include/exclude` 就是给缓存加白名单，避免内存无上限增长。

## 常见扣分点

| 扣分点 | 后果 | 正确做法 |
|---|---|---|
| reactive 不处理嵌套 | 深层修改不触发更新 | `get` 里惰性 `reactive` |
| effect 不去重/不清理 | 一次修改触发多次渲染，条件切换后幽灵触发 | `Set` 存 effect + 执行前 `cleanup` |
| useState 忘记重置下标 | 第二次渲染读到别人的 state | 每次渲染前 `hookIndex = 0` |
| useEffect 依赖写漏 | 闭包读旧值，数据不更新 | 依赖写全，或改用函数式更新 |
| diff 跨层比较 | 复杂度退化为 O(n³) | 同层比较，跨层视为删除 + 新建 |
| diff 用 index 当 key | 头部插入导致全量更新 | 用业务唯一 id |
| 表单 change 就全量校验 | 未填字段全红，体验崩 | change 校验当前字段，submit 全量 |
| useRequest 不做竞态保护 | 旧响应覆盖新数据 | 自增 id + `AbortController` |
| keep-alive 无上限 | 内存持续增长 | LRU + `include/exclude` |
| ErrorBoundary 用函数组件 | 根本捕获不到 | class + `getDerivedStateFromError` |
| chunk 失败无限重载 | 用户白屏闪烁 | sessionStorage 时间戳防循环 |

## 现场写题的 5 分钟节奏

1. **先问场景**：数据量级？要不要兼容 IE？是面试演示还是生产可用？——决定你用 Proxy 还是 defineProperty、要不要加 buffer。
2. **先写主流程，再补边界**：先让 `effect` 能触发、`diff` 能出补丁，再补嵌套、key、清理。
3. **注释写机制，不写废话**：`// 先 cleanup 再跑新 effect` 比 `// 执行 effect` 值钱十倍。
4. **主动报边界**：写完立刻说「这里我没处理循环引用 / 这里没做 LIS 优化」，比被问出来强。
5. **留一个可运行验证**：能打印出触发次数、补丁列表，面试官会明显加分。

## 一句话总结

框架原理手写题的评分从来不是「代码长短」，而是三个问题：**依赖收得全不全、更新调度得对不对、异常兜不兜得住**。把这三件事写进注释、讲进追问，30 行就够了。
