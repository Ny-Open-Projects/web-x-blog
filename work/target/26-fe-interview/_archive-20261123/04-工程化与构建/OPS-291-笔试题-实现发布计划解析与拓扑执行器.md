## OPS-291：手写发布计划执行器

**需求与边界**：
- 需求：解析发布计划 YAML（steps+depends+role），校验拓扑可执行、执行时校验「当前计划 hash 与审批快照一致」（计划冻结，SCN-OPS-273）；按拓扑批次执行，失败逆序回滚。
- 边界：① 环检测 fail-fast；② hash 不匹配拒绝执行（偏离重新审批）；③ 每步幂等键（planId+stepId）；④ 回滚失败不中断（state 记录）。

**答**：
```javascript
const yaml = require('js-yaml');
function parsePlan(yamlStr) {
  const plan = yaml.load(yamlStr);
  const ids = plan.steps.map(s => s.app);
  if (new Set(ids).size !== ids.length) throw new Error('步骤重复');
  // 环检测（Kahn 剩余非空即有环）
  const indeg = new Map(plan.steps.map(s => [s.app, (s.depends ?? []).length]));
  let ready = plan.steps.filter(s => !s.depends?.length);
  let processed = 0;
  while (ready.length) {
    processed += ready.length;
    for (const r of ready)
      for (const s of plan.steps)
        if (s.depends?.includes(r.app) && indeg.get(s.app) > 0 && indeg.set(s.app, indeg.get(s.app) - 1).get(s.app) === 0)
          ready.push(s);
    ready = ready.filter(s => indeg.get(s.app) === 0);   // 简化：每轮取零入度
    if (ready.every(s => s.done)) break;
  }
  if (processed < plan.steps.length) throw new Error('CYCLE_DETECTED');
  return plan;
}
async function executePlan(plan, approvedHash) {
  const currentHash = sha256(JSON.stringify(plan.steps));
  if (currentHash !== approvedHash)
    return { rejected: 'PLAN_DRIFTED', note: '计划与审批快照不一致，需重新审批' };   // 冻结校验
  const batches = topoBatches(plan.steps);
  const done = [], state = {};
  for (const batch of batches) {
    const results = await Promise.allSettled(
      batch.map(s => execStep(s, { idempotencyKey: `${plan.id}:${s.app}` })));
    if (results.some(r => r.status === 'rejected')) {
      for (const s of [...done].reverse())
        try { await rollback(s); state[s.app] = 'ROLLED_BACK'; }
        catch (e) { state[s.app] = 'ROLLBACK_FAILED'; }   // 回滚失败不中断
      return { status: 'ROLLED_BACK', state };
    }
    batch.forEach(s => { done.push(s); state[s.app] = 'DONE'; });
  }
  return { status: 'OK', state };
}
```

**评分点**:
- 环检测（Kahn 剩余非空）fail-fast。
- 计划冻结 hash 校验（偏离拒绝执行——SCN-OPS-273 核心）。
- 幂等键 + 批次并行 + 逆序回滚不中断。
- 语义校验（步骤重复）在解析层完成。
