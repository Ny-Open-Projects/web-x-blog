## SCN-698：手写告警风暴限流聚合器

**需求与边界**：
- 需求：滑动窗口（1 分钟）统计告警密度；密度超阈值进入风暴模式：critical 全放行、major 采样 50%、minor/info 采样 10%，其余聚合计数；风暴解除后聚合摘要一次性输出。
- 边界：① 密度判定用滑动窗口（环形数组）非固定窗；② 风暴进出有迟滞（进 100 条/分，出需 < 50 条/分）；③ 每条被限流的告警计入 dropped 计数按来源聚合。

**答**：
```javascript
function createAlarmLimiter({ inRate = 100, outRate = 50, windowMs = 60_000 } = {}) {
  const win = [];                                      // [ts]
  let storm = false, stormCount = 0;
  const dropped = new Map();                           // source -> count
  const summary = [];
  function shouldPass(alarm, now) {
    push(now);
    if (storm) {
      if (density() < outRate) {                       // 解除（迟滞阈值更低）
        storm = false;
        summary.push({ droppedTotal: [...dropped.values()].reduce((s, v) => s + v, 0),
                       droppedBySource: Object.fromEntries(dropped) });
        dropped.clear();
        return true;
      }
      stormCount++;
      if (alarm.level === 'critical') return true;     // 紧急全放行
      const rate = alarm.level === 'major' ? 0.5 : 0.1;
      const pass = Math.random() < rate;
      if (!pass) dropped.set(alarm.source, (dropped.get(alarm.source) || 0) + 1);
      return pass;
    }
    if (density() > inRate) {                          // 进入风暴
      storm = true; stormCount = 0;
      report({ type: 'ALARM_STORM', density: density() });
      return alarm.level === 'critical';
    }
    return true;
  }
  function push(now) {
    win.push(now);
    while (win[0] < now - windowMs) win.shift();       // 滑动窗口清理
  }
  const density = () => win.length;
  return { shouldPass, get storm() { return storm; }, get summary() { return summary; } };
}
```

**评分点**:
- 滑动窗口密度统计（逐条 shift 清理）。
- 分级放行（critical 全放/major 50%/其余 10%）。
- 风暴进出迟滞（100 进 / 50 出，防抖动）。
- 限流告警按来源聚合计数 + 解除时摘要输出。
