## SCN-598：手写验证链路超时降级调度器

**需求与边界**：
- 需求：每个验证器有独立超时；超时/失败自动降级到下一级；降级期间对失败验证器定期探活（指数退避），恢复后回到首选链路。
- 边界：① 探活不阻塞当前验证；② 恢复判断需要连续 2 次成功；③ 调度器可查询当前生效链路。

**答**：
```javascript
function createDegradationScheduler(chain, { probeBase = 10_000 } = {}) {
  let active = chain, probes = 0, probeTimer = null;
  function degrade(from, reason) {
    active = chain.filter(v => v !== from).concat();   // 移除故障验证器
    scheduleProbe(from);
    return { active, degraded: true, from, reason };
  }
  function scheduleProbe(name) {                       // 指数退避探活
    const delay = probeBase * 2 ** Math.min(probes++, 5);
    probeTimer = setTimeout(async () => {
      const ok = await probe(name).catch(() => false);
      if (ok) recoveries[name] = (recoveries[name] || 0) + 1;
      else recoveries[name] = 0;
      if (recoveries[name] >= 2) {                     // 连续 2 次成功才恢复
        active = chain;                                // 回到首选链路
        probes = 0; delete recoveries[name];
        return;
      }
      scheduleProbe(name);
    }, delay);
  }
  const recoveries = {};
  async function run(scene) {
    for (const name of active) {
      try {
        const r = await withTimeout(registry.get(name).run(scene), registry.get(name).timeout);
        if (r.pass) return r;
        degrade(name, 'user-fail-hard');
      } catch (e) { degrade(name, e.message); }
    }
    return { pass: false, reason: 'EXHAUSTED' };
  }
  function probe(name) { return registry.get(name).ping(); }
  return { run, get active() { return active; } };
}
const withTimeout = (p, ms) => Promise.race([
  p, new Promise((_, rej) => setTimeout(() => rej(new Error('TIMEOUT')), ms)),
]);
```

**评分点**:
- 按级独立超时 + 失败自动降级。
- 指数退避探活 + 连续 2 次成功才恢复（防抖动）。
- 探活异步不阻塞当前验证。
- active 链路可查询（可观测）。
