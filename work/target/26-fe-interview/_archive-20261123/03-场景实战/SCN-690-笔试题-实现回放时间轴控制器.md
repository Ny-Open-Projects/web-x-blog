## SCN-690：手写回放时间轴控制器

**需求与边界**：
- 需求：时间范围分片（5 分钟/片）；play 按倍速推进；seek 二分定位分片并应用快照+增量；临近片尾预载下一片；play/pause/seek 状态一致。
- 边界：① seek 打断播放后从新位置继续；② 预载失败不影响播放（退化为即时加载）；③ 快进到达终点自动停止。

**答**：
```javascript
function createTimeline({ start, end, sliceMs = 300_000, loader }) {
  const slices = [];
  for (let t = start; t < end; t += sliceMs)
    slices.push({ start: t, end: Math.min(t + sliceMs, end), data: null, loading: false });
  let cursor = start, speed = 1, playing = false, timer = null;
  async function ensureSlice(ts) {
    const s = slices[bisect(slices, x => x.start <= ts)];
    if (!s.data && !s.loading) {
      s.loading = true;
      s.data = await loader(s).catch(() => null);      // 失败返回 null
      s.loading = false;
    }
    return s;
  }
  function bisect(arr, pred) {
    let lo = 0, hi = arr.length - 1;
    while (lo < hi) { const m = (lo + hi + 1) >> 1; pred(arr[m]) ? lo = m : hi = m - 1; }
    return lo;
  }
  async function play() {
    if (playing) return;
    playing = true;
    await ensureSlice(cursor);
    tick();
  }
  function tick() {
    if (!playing) return;
    timer = setTimeout(async () => {
      cursor += sliceMs / 10 * speed;                  // 每步推进 slice/10
      if (cursor >= end) { pause(); cursor = end; onEnd?.(); return; }
      const s = await ensureSlice(cursor);
      applyUpto(s, cursor);
      if (cursor > s.end - sliceMs / 5) ensureSlice(cursor + sliceMs);   // 预载下一片
      tick();
    }, sliceMs / 10);
  }
  async function seek(ts) {
    pause();
    cursor = Math.max(start, Math.min(end, ts));
    const s = await ensureSlice(cursor);
    applySnapshot(s); applyUpto(s, cursor);            // 快照 + 增量
  }
  function pause() { playing = false; clearTimeout(timer); }
  function setSpeed(v) { speed = v; }
  return { play, pause, seek, setSpeed, get cursor() { return cursor; } };
}
```

**评分点**:
- 分片二分定位 + 快照直达（不全量重放）。
- 倍速推进用绝对 slice 换算（不依赖 setInterval 精度）。
- 预载下一片（失败不阻塞播放）。
- seek/pause/play 状态机一致，终点自动停。
