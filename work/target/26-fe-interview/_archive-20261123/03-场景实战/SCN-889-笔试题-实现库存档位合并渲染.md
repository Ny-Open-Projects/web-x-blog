## SCN-889：手写库存档位渲染器

**需求与边界**：
- 需求：库存推送 500ms 合并窗口渲染最新档位；档位由真实库存映射（充足/紧张/仅剩少量/售罄）；档位下降时短高亮（500ms）不闪烁；售罄触发全局锁定回调一次。
- 边界：① 展示档位永远 ≤ 真实库存语义（不多报）；② 合并窗口内最新覆盖旧值（只渲染最新）；③ 售罄后停止订阅。

**答**：
```javascript
function createStockRenderer({ window = 500 } = {}) {
  let latest = null, timer = null, soldOut = false, highlightTimer = null;
  const LEVELS = [[0, 'SOLD_OUT'], [10, 'LOW'], [0.3, 'MID'], [Infinity, 'OK']];
  function levelOf(remain, total) {
    for (const [th, lv] of LEVELS)
      if (lv === 'MID' || lv === 'OK') { if (remain < total * th) return lv; }
      else if (remain <= th) return lv;
    return 'OK';
  }
  function onPush(evt, total, { onSoldOut, unsubscribe }) {
    if (soldOut) return;
    latest = evt;                                      // 只留最新
    timer ??= setTimeout(() => {
      timer = null;
      const prevLevel = currentLevel;
      currentLevel = levelOf(latest.remain, total);
      render(currentLevel);
      if (currentLevel < prevLevel) {                  // 档位下降：短高亮
        el.classList.add('stock-drop');
        highlightTimer = setTimeout(() => el.classList.remove('stock-drop'), 500);
      }
      if (currentLevel === 'SOLD_OUT') {               // 售罄终态
        soldOut = true;
        clearTimeout(highlightTimer);
        unsubscribe();                                 // 停止订阅
        onSoldOut();                                   // 全局锁定回调（一次性）
      }
    }, window);
  }
  let currentLevel = 'OK';
  return { onPush, get level() { return currentLevel; } };
}
```

**评分点**:
- 合并窗口只渲染最新（高频推送不打架）。
- 档位映射语义单调（档位 ≤ 真实库存，不多报）。
- 下降短高亮 500ms（紧迫感不闪烁）。
- 售罄终态一次性：停订阅 + 全局锁定回调 + 清高亮 timer。
