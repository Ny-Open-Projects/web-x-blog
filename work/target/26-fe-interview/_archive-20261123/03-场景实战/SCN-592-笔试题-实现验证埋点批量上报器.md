## SCN-592：手写验证埋点批量上报器

**需求与边界**：
- 需求：队列满 N 条或定时 T 批量上报；失败落 IndexedDB 补偿重放；每条带递增序号；页面卸载用 sendBeacon 强制刷出。
- 边界：① 重放时保持序号顺序；② 补偿队列上限（超过丢弃最旧并计数）；③ 上报去重（服务端按 sid+seq）；④ flush 并发保护。

**答**：
```javascript
function createReporter({ batch = 20, every = 5000, maxPersist = 500 } = {}) {
  const queue = []; let seq = 0, flushing = false;
  function push(event, { urgent = false } = {}) {
    queue.push({ ...event, seq: ++seq, sid: SID });
    if (urgent || queue.length >= batch) flush();
  }
  async function flush() {
    if (flushing || !queue.length) return;
    flushing = true;
    const data = queue.splice(0, batch);
    try {
      const res = await fetch('/collect', {
        method: 'POST', body: JSON.stringify(data), keepalive: true,
      });
      if (!res.ok) throw new Error(res.status);
    } catch {
      await persist(data);                            // 失败落库补偿
    } finally { flushing = false; }
  }
  async function persist(data) {
    const db = await openDB('risk_log');
    const tx = db.transaction('events', 'readwrite');
    for (const e of data) tx.store.put({ seq: e.seq, event: e });
    // 上限控制：超 maxPersist 丢弃最旧
    const cnt = await db.count('events', 'events');
    if (cnt > maxPersist) await db.delete('events', 'events', IDBKeyRange.upperBound(cnt - maxPersist));
  }
  async function replay() {                           // 启动时按序重放
    const db = await openDB('risk_log');
    const all = await db.getAllFromIndex('events', 'by_seq');   // 按序号有序
    for (let i = 0; i < all.length; i += batch)
      await fetch('/collect', { method: 'POST', body: JSON.stringify(all.slice(i, i + batch)) });
    await db.clear('events');
  }
  window.addEventListener('pagehide', () => {
    if (queue.length) navigator.sendBeacon('/collect', JSON.stringify(queue));
  });
  setInterval(flush, every);
  return { push, flush, replay };
}
```

**评分点**:
- 双触发批量（数量/时间）+ flushing 并发保护。
- 失败落 IndexedDB + 启动按 seq 有序重放。
- pagehide 时 sendBeacon 保卸载不丢。
- 补偿队列上限丢弃最旧（内存/存储守恒）。
