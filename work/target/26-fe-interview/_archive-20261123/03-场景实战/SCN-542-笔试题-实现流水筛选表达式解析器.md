## SCN-542：手写流水筛选表达式解析器

**需求与边界**：
- 语法：`类型:转账`、`金额>=100`、`状态:失败 OR 状态:关闭`、自由关键词；支持 AND（空格）、OR、括号。
- 输出：解析为查询 DSL（服务端执行）；非法语法给出错误位置。
- 边界：① 值含空格用引号；② 未知字段报错；③ OR 优先级低于 AND；④ 空表达式返回全部。

**答**：
```javascript
const FIELDS = new Set(['类型', '金额', '状态', '渠道']);
function tokenize(input) {
  const tokens = []; let i = 0;
  while (i < input.length) {
    const c = input[i];
    if (c === ' ') { i++; continue; }
    if (c === '(' || c === ')') { tokens.push({ t: c }); i++; continue; }
    if (c === '"') {                                   // 引号值
      const j = input.indexOf('"', i + 1);
      if (j < 0) throw new Error(`未闭合引号 @${i}`);
      tokens.push({ t: 'word', v: input.slice(i + 1, j) }); i = j + 1; continue;
    }
    let j = i;
    while (j < input.length && !"() ".includes(input[j])) j++;
    tokens.push({ t: 'word', v: input.slice(i, j) }); i = j;
  }
  return tokens;
}
function parseExpr(tokens, pos = 0) {                    // OR 层
  let left = parseAnd(tokens, pos); let { node, pos: p } = left; pos = p;
  while (pos < tokens.length && tokens[pos].v === 'OR') {
    const right = parseAnd(tokens, pos + 1);
    node = { op: 'or', l: node, r: right.node }; pos = right.pos;
  }
  return { node, pos };
}
function parseAnd(tokens, pos) {                         // AND 层（空格/隐式）
  let node = parseAtom(tokens, pos); pos = node.pos;
  while (pos < tokens.length && tokens[pos].t === 'word' && tokens[pos].v !== 'OR') {
    const right = parseAtom(tokens, pos);
    node = { node: { op: 'and', l: node.node, r: right.node }, pos: right.pos }; pos = node.pos;
  }
  return { node, pos };
}
function parseAtom(tokens, pos) {
  const tk = tokens[pos];
  if (!tk) return { node: { op: 'match-all' }, pos };
  if (tk.t === '(') { const r = parseExpr(tokens, pos + 1); return { node: r.node, pos: r.pos + 1 }; }
  const m = tk.v.match(/^(.+?)(>=|<=|>|<|:)(.+)$/);
  if (m) {
    if (!FIELDS.has(m[1])) throw new Error(`未知字段 ${m[1]}`);
    return { node: { op: 'cond', field: m[1], cmp: m[2], value: m[3] }, pos: pos + 1 };
  }
  return { node: { op: 'keyword', value: tk.v }, pos: pos + 1 };
}
// parse('类型:转账 金额>=100 (状态:失败 OR 状态:关闭)') =>
// { op:'and', l:{op:'and', l:{cond 类型:转账}, r:{cond 金额>=100}}, r:{op:'or', ...} }
```

**评分点**:
- 三层递归下降（OR → AND → 原子）体现优先级。
- 引号值、括号嵌套、未知字段报错带位置。
- 输出 DSL 而非直接过滤（服务端执行的意识）。
- 空表达式 match-all 兜底。
