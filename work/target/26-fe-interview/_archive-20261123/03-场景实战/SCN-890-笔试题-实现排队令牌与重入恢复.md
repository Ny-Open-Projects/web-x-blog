## SCN-890：手写排队令牌管理

**需求与边界**：
- 需求：进入排队领取 token（服务端原子）；页面关闭前保存 token（重入恢复位置）；token 定期续期（服务端校验活性）；放行（position=0）携带 token 跳购买页。
- 边界：① token 只存 sessionStorage（标签页生命周期，不跨标签共享防并发）；② 续期失败 2 次提示退出重排；③ 放行 token 一次性（跳转后失效防重复进入）。

**答**：
```javascript
function createQueueToken(activityId) {
  const KEY = `q_${activityId}`;
  let token = sessionStorage.getItem(KEY), renewTimer = null, failures = 0;
  async function join() {
    if (token) return resume();                        // 重入恢复
    const res = await api.post('/queue/join', { activityId });
    token = res.token;
    sessionStorage.setItem(KEY, token);                // 离开可恢复
    startRenew();
    return { position: res.position };
  }
  async function resume() {                            // 重入：服务端确认位置仍在
    const res = await api.post('/queue/resume', { activityId, token }).catch(() => null);
    if (!res?.valid) { clear(); return join(); }       // token 失效重排
    startRenew();
    return { position: res.position };
  }
  function startRenew() {
    clearInterval(renewTimer);
    renewTimer = setInterval(async () => {
      const ok = await api.post('/queue/renew', { token }).catch(() => false);
      if (!ok && ++failures >= 2) {
        clear();
        toast('排队已超时，请重新排队');
      }
    }, RENEW_EVERY);
  }
  function admit() {                                   // 放行：一次性 token
    clearInterval(renewTimer);
    sessionStorage.removeItem(KEY);
    location.href = `/buy?activityId=${activityId}&token=${token}`;
    token = null;
  }
  function clear() { sessionStorage.removeItem(KEY); token = null; clearInterval(renewTimer); }
  return { join, admit, clear, get token() { return token; } };
}
```

**评分点**:
- sessionStorage 生命周期选择（不跨标签防并发排队）。
- 重入 resume 校验（token 失效静默重排）。
- 续期活性检测 + 失败 2 次退出提示。
- 放行一次性（消费即清，防重复进入购买页）。
