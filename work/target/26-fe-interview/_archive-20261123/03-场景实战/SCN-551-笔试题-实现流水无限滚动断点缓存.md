## SCN-551：手写无限滚动列表的数据层（游标 + 会话缓存 + 新流水提示）

**需求与边界**：
- 需求：游标分页无限加载；离开页面保存游标与已加载数（sessionStorage 加密摘要），回来恢复；加载期间有新流水到达时不打断，显示「N 笔新流水」。
- 边界：① 恢复时游标仍有效（服务端校验，过期则重置）；② 触底节流 + in-flight 去重；③ 恢复数据校验完整性（哈希）。

**答**：
```javascript
function createInfiniteFlows(api, { pageSize = 20 } = {}) {
  let list = [], cursor = null, loading = false;
  let bufferCount = 0, newestTs = 0;
  async function loadMore() {
    if (loading) return;                              // in-flight 去重
    loading = true;
    try {
      const res = await api.get('/flows', { params: { cursor, size: pageSize } });
      list = list.concat(res.list);
      cursor = res.next;
      newestTs = Math.max(newestTs, ...res.list.map(r => r.ts));
      return list;
    } finally { loading = false; }
  }
  function onNewFlows(events) {                       // WS 新流水
    const fresh = events.filter(e => e.ts > newestTs);
    if (fresh.length && !atTop()) bufferCount += fresh.length;   // 非顶部进缓冲
    else list.unshift(...fresh);
  }
  const atTop = () => true;                           // 由视图层注入滚动判断
  function checkpoint() {                             // 离开时存断点
    sessionStorage.setItem('flows_ck', JSON.stringify({
      cursor, newestTs,
      digest: hash(JSON.stringify(list.map(r => r.id))),        // 完整性摘要
    }));
  }
  async function restore() {
    const ck = JSON.parse(sessionStorage.getItem('flows_ck') || 'null');
    if (!ck) return loadMore();
    const probe = await api.get('/flows/validate', { params: { cursor: ck.cursor } });
    if (!probe.valid) { sessionStorage.removeItem('flows_ck'); return loadMore(); }  // 游标过期重置
    cursor = ck.cursor; newestTs = ck.newestTs;
    return { restored: true, bufferCount: 0 };
  }
  return { loadMore, onNewFlows, checkpoint, restore, get buffer() { return bufferCount; } };
}
```

**评分点**:
- 游标恢复 + 服务端有效性探测（过期重置而非报错）。
- 新流水缓冲不打断浏览（atTop 才直插）。
- in-flight 去重与 finally 复位。
- 断点摘要（digest）体现数据完整性意识。
