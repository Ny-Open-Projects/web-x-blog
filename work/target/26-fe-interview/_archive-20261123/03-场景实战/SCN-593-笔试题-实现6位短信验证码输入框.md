## SCN-593：手写 6 位验证码输入框逻辑

**需求与边界**：
- 需求：6 个分格输入，输满自动提交；支持粘贴（一次填 6 位拆分）；退格回退上一格；方向键导航；只剩未填格聚焦。
- 边界：① 只允许数字；② 粘贴含非数字过滤；③ 中途删除后焦点保持；④ 自动提交防重复（in-flight 保护）。

**答**：
```javascript
function createOtpInput({ length = 6, onSubmit }) {
  const digits = new Array(length).fill('');
  let pos = 0, submitting = false;
  function render() { return digits.join(' '); }
  function nextFocus() {
    const empty = digits.findIndex(d => d === '');
    pos = empty === -1 ? length - 1 : empty;
    return pos;
  }
  function input(ch, at = pos) {
    ch = String(ch).replace(/\D/g, '');              // 只留数字
    if (!ch) return false;
    for (const c of ch) {                            // 粘贴多字符逐格填充
      if (at >= length) break;
      digits[at++] = c;
    }
    pos = Math.min(at, length - 1);
    if (digits.every(d => d !== '')) autoSubmit();   // 输满自动提交
    return true;
  }
  async function autoSubmit() {
    if (submitting) return;                          // 防重复提交
    submitting = true;
    try { await onSubmit(digits.join('')); }
    finally { submitting = false; }
  }
  function backspace() {
    if (digits[pos] === '' && pos > 0) pos--;        // 当前空则回退一格
    digits[pos] = '';
    return true;
  }
  function paste(text) {
    const clean = text.replace(/\D/g, '').slice(0, length);
    if (!clean) return false;
    for (let i = 0; i < clean.length; i++) digits[i] = clean[i];
    pos = Math.min(clean.length, length - 1);
    if (clean.length === length) autoSubmit();
    return true;
  }
  return { input, backspace, paste, nextFocus, get value() { return digits.join(''); }, render };
}
```

**评分点**:
- 输满自动提交 + in-flight 防重复。
- 粘贴拆分逐格填充与非数字过滤。
- 退格智能回退（当前空则回上一格）。
- 焦点始终落在首个空格。
