## SCN-638：手写审批条件路由求值器

**需求与边界**：
- 条件语法：`amount > 10000 && dept == '研发' || urgency == 'P0'`；分支按顺序求值，首个命中的生效，else 兜底。
- 要求：① 词法+递归下降解析（AND/OR/比较/括号）；② 求值返回命中分支及条件路径；③ dry-run 模式返回「每个分支的求值结果」供预演展示；④ 未知变量报错。

**答**：
```javascript
function tokenize(s) {
  return s.match(/\s*(&&|\|\||==|!=|>=|<=|>|<|\(|\)|[\w.\u4e00-\u9fa5]+|'[^']*')\s*/g)
    .map(t => t.trim()).filter(Boolean);
}
function parseOr(tokens) {
  let node = parseAnd(tokens);
  while (tokens[0] === '||') node = { op: '||', l: node, r: parseAnd(tokens.slice(1)) };
  return node;
}
function parseAnd(tokens) {
  let node = parseAtom(tokens);
  while (tokens[0] === '&&') node = { op: '&&', l: node, r: parseAtom(tokens.slice(1)) };
  return node;
}
function parseAtom(tokens) {
  if (tokens[0] === '(') { const n = parseOr(tokens.slice(1)); return n; }  // 简化：括号由外层消费
  const [field, cmp, ...rest] = tokens;
  const val = rest[0].replace(/'/g, '');
  return { op: 'cmp', field, cmp, val, rest: rest.slice(1) };
}
function evalNode(node, data, trace) {               // trace 记录预演路径
  if (node.op === '&&' || node.op === '||') {
    const l = evalNode(node.l, data, trace), r = evalNode(node.r, data, trace);
    return node.op === '&&' ? (l.v && r.v) : (l.v || r.v);
  }
  if (!(node.field in data)) throw new Error(`未知变量 ${node.field}`);
  const v = data[node.field], t = node.val;
  const num = Number(v);
  const cmpV = node.cmp === '>' ? (isNaN(num) ? v > t : num > Number(t))
    : node.cmp === '>=' ? num >= Number(t)
    : node.cmp === '<' ? num < Number(t)
    : node.cmp === '<=' ? num <= Number(t)
    : node.cmp === '==' ? String(v) === t
    : String(v) !== t;
  trace.push({ expr: `${node.field} ${node.cmp} ${node.val}`, result: cmpV });
  return cmpV;
}
function route(branches, data, { dryRun = false } = {}) {
  for (const b of branches) {
    const trace = [];
    const hit = b.cond === 'else' ? true : evalNode(b.cond, data, trace).v;
    if (dryRun) b.trace = trace;                     // 预演：每个分支的条件求值明细
    if (hit) return { branch: b, trace };
  }
  throw new Error('NO_BRANCH_MATCHED');              // 无 else 兜底时必须报错
}
```

**评分点**:
- 递归下降（OR→AND→原子）优先级正确。
- dry-run 返回每分支求值 trace（预演可解释）。
- 未知变量 fail-fast；无 else 未命中报错（不放行）。
- 数字/字符串比较类型处理。
