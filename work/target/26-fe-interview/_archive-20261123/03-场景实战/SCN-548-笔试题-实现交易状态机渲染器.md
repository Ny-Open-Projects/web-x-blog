## SCN-548：手写交易状态机渲染器

**需求与边界**：
- 需求：定义迁移表（合法状态流转）、按状态渲染文案/颜色/动作、非法迁移拒绝并告警、未知状态落入兜底态。
- 边界：① version 单调性；② PROCESSING 超时转 STUCK 态（不判失败）；③ 动作执行前校验当前态合法。

**答**：
```javascript
const TXFSM = {
  transitions: {                                   // 合法迁移白名单
    INIT:       ['PROCESSING', 'CLOSED'],
    PROCESSING: ['SUCCESS', 'FAIL', 'STUCK'],
    STUCK:      ['SUCCESS', 'FAIL'],
    SUCCESS:    ['REFUND'],
    FAIL:       ['PROCESSING'],                    // 重试
    REFUND:     [], CLOSED: [],
  },
  view: {
    INIT: { text: '已创建', cls: 'gray' }, PROCESSING: { text: '处理中', cls: 'blue' },
    STUCK: { text: '处理时间较长，以最终通知为准', cls: 'blue' },
    SUCCESS: { text: '成功', cls: 'green' }, FAIL: { text: '失败', cls: 'red' },
    REFUND: { text: '已退款', cls: 'green' }, CLOSED: { text: '已关闭', cls: 'gray' },
  },
  actions: { SUCCESS: ['receipt', 'refund'], FAIL: ['retry'], STUCK: ['contact'] },
};
function createTxFSM(tx) {
  let cur = tx.status, ver = tx.version;
  function transition(to, incomingVersion) {
    if (incomingVersion <= ver) return cur;                       // 旧事件忽略
    ver = incomingVersion;
    if (!(TXFSM.transitions[cur] || []).includes(to)) {
      report({ type: 'ILLEGAL_TRANSITION', from: cur, to });      // 非法迁移告警不崩溃
      return cur;
    }
    cur = to;
    return cur;
  }
  function render() {
    const v = TXFSM.view[cur] || TXFSM.view.PROCESSING;           // 未知态兜底
    return { ...v, actions: TXFSM.actions[cur] || [],
             can: (a) => (TXFSM.actions[cur] || []).includes(a) };
  }
  return { transition, render, get status() { return cur; } };
}
```

**评分点**:
- 迁移白名单 + 非法迁移告警而非抛错崩溃。
- version 单调守卫防乱序覆盖。
- STUCK 兜底态（处理中不判失败——资损意识）。
- 未知状态 fallback 与动作白名单 can()。
