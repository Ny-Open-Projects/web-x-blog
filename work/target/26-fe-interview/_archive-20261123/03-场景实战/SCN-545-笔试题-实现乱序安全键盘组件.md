## SCN-545：手写乱序安全键盘逻辑

**需求与边界**：
- 需求：数字 0-9 随机布局（每次打开重新洗牌）、最大长度截断、退格、确认回调；防复制/粘贴/拖拽。
- 边界：① 可注入 seed 保证测试确定性；② 连点去抖（同键 80ms 内重复点击忽略）；③ 输入内容只存缓冲，DOM 上不显示明文。

**答**：
```javascript
function createKeypad({ seed = Math.random(), maxLength = 6 } = {}) {
  const rnd = mulberry32(Math.floor(seed * 2 ** 31));     // 可注入的伪随机
  const keys = shuffle([...Array(10).keys()], rnd);
  let buf = '', lastKey = null, lastTs = 0;
  function shuffle(arr, rnd) {
    for (let i = arr.length - 1; i > 0; i--) {
      const j = Math.floor(rnd() * (i + 1));
      [arr[i], arr[j]] = [arr[j], arr[i]];
    }
    return arr;
  }
  return {
    layout: keys,
    press(k, now = Date.now()) {
      if (k === lastKey && now - lastTs < 80) return null;   // 连点去抖
      lastKey = k; lastTs = now;
      if (k === 'del') { buf = buf.slice(0, -1); return buf; }
      if (!Number.isInteger(k) || buf.length >= maxLength) return null;
      buf += k;
      return buf;
    },
    confirm() {
      if (buf.length < maxLength) throw new Error('INCOMPLETE');
      const out = buf; buf = '';                             // 确认后立即清缓冲
      return out;                                            // 由调用方加密上传
    },
    get length() { return buf.length; },
  };
}
function mulberry32(a) {
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}
// DOM 层：按键显示 • 而非数字明文（可配置显示数字但 DOM 不存 value 属性）；
// onPaste/onCopy/onDrop 全部 preventDefault。
```

**评分点**:
- 可注入 seed 的洗牌（测试确定性 + 生产随机双模式）。
- 连点去抖与 maxLength 截断边界。
- 确认后清空缓冲（明文最小存活）。
- 按键事件防复制/粘贴/拖拽的意识。
