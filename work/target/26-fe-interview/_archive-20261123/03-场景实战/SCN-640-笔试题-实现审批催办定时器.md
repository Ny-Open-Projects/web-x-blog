## SCN-640：手写 SLA 工作日历倒计时

**需求与边界**：
- 输入：到达时间戳、SLA 工时（小时）、工作日历（工作日 9-18 点 + 节假日表）；输出 deadline 时间戳与剩余工作时长。
- 边界：① SLA 按「工作小时」累计（跨天/跨周末顺延）；② 剩余时间每分钟可查（绝对时间计算不漂移）；③ deadline 落在非工作时段顺延到下个工作时段起点。

**答**：
```javascript
function workDeadline(arriveTs, slaHours, cal) {     // cal: { workStart: 9, workEnd: 18, holidays: Set }
  let remainMs = slaHours * 3600_000, cursor = new Date(arriveTs);
  while (remainMs > 0) {
    if (isWorkday(cursor, cal)) {
      const dayStart = new Date(cursor); dayStart.setHours(cal.workStart, 0, 0, 0);
      const dayEnd = new Date(cursor); dayEnd.setHours(cal.workEnd, 0, 0, 0);
      const from = cursor > dayStart ? cursor : dayStart;      // 取「当前位置」与时段起点较大者
      if (from < dayEnd) {
        const span = Math.min(dayEnd - from, remainMs);
        remainMs -= span;
        if (remainMs <= 0) return from.getTime() + span;       // 在本时段内完成
      }
    }
    cursor = nextMorning(cursor);                    // 跳到下一天工作开始（周末自动跳过）
  }
  return cursor.getTime();
}
function isWorkday(d, cal) {
  const day = d.getDay();
  return day !== 0 && day !== 6 && !cal.holidays.has(d.toISOString().slice(0, 10));
}
function nextMorning(d) {
  const n = new Date(d); n.setDate(n.getDate() + 1); n.setHours(9, 0, 0, 0);
  return isWorkday(n, cal) ? n : nextMorning(n);     // 递归跳过周末与节假日
}
function remainWorkMs(deadline, cal) {               // 剩余工作时长（不含非工作时段）
  const now = Date.now();
  if (now >= deadline) return 0;
  let total = 0, cursor = new Date(now);
  while (cursor.getTime() < deadline) {
    if (isWorkday(cursor, cal)) {
      const s = new Date(cursor); s.setHours(9, 0, 0, 0);
      const e = new Date(cursor); e.setHours(18, 0, 0, 0);
      const from = Math.max(cursor.getTime(), s.getTime());
      total += Math.max(0, Math.min(e.getTime(), deadline) - from);
    }
    cursor = nextMorning(cursor);
  }
  return total;
}
```

**评分点**:
- 工作小时累计（跨天/周末顺延）而非自然时间。
- deadline 落在非工作时段顺延到时段起点。
- 剩余时长按工作日历计算（超时判定口径一致）。
- 节假日 Set 查询 O(1)，递归跳日有终止条件。
