## SCN-595：手写行为特征聚合器

**需求与边界**：
- 需求：实时接收行为事件（move/click/key），只存统计量不存原始事件（内存 O(1)）；输出特征：移动速率、方向熵、点击间隔变异系数、按键节奏均值/方差。
- 边界：① 方向熵用 8 象限直方图计算；② 样本数 < 阈值输出 insufficient；③ 数值防御（NaN/Infinity 过滤）。

**答**：
```javascript
function createBehaviorAggregator({ minMoves = 20 } = {}) {
  let moves = 0, totalDist = 0, lastT = 0, lastPos = null;
  const quads = new Array(8).fill(0);                // 方向 8 象限直方图
  const clickGaps = [], keyGaps = [];
  let lastClick = 0, lastKey = 0;
  const ok = v => Number.isFinite(v);
  function onMove(e) {
    if (lastPos && e.t > lastT) {
      const dx = e.x - lastPos.x, dy = e.y - lastPos.y;
      const d = Math.hypot(dx, dy);
      if (!ok(d)) return;                            // NaN/Infinity 防御
      totalDist += d;
      const quad = Math.floor(((Math.atan2(dy, dx) + Math.PI) / (2 * Math.PI)) * 8) % 8;
      quads[quad]++;
    }
    moves++; lastT = e.t; lastPos = { x: e.x, y: e.y };   // 只存最近点，O(1) 内存
  }
  const stats = (arr) => {
    if (arr.length < 2) return null;
    const mean = arr.reduce((s, v) => s + v, 0) / arr.length;
    const varr = arr.reduce((s, v) => s + (v - mean) ** 2, 0) / (arr.length - 1);
    return { mean, std: Math.sqrt(varr), cv: mean ? Math.sqrt(varr) / mean : 0 };
  };
  function entropy() {                               // 方向熵
    const total = quads.reduce((s, v) => s + v, 0);
    if (!total) return 0;
    return -quads.reduce((s, q) => s + (q ? (q / total) * Math.log2(q / total) : 0), 0);
  }
  return {
    onMove,
    onClick(t) { if (lastClick) clickGaps.push(t - lastClick); lastClick = t; },
    onKey(t)  { if (lastKey) keyGaps.push(t - lastKey); lastKey = t; },
    features() {
      if (moves < minMoves) return { insufficient: true };
      return {
        moveRate: totalDist / Math.max(1, moves),
        turnEntropy: entropy(),                       // 真人熵高（方向多样）
        clickGap: stats(clickGaps),
        keyRhythm: stats(keyGaps),
      };
    },
  };
}
```

**评分点**:
- 内存 O(1)（只存统计量与最近点）——原始终迹不落地。
- 方向熵 8 象限（真人高熵、机器低熵）。
- stats 带样本下限与 cv（变异系数）。
- NaN/Infinity 数值防御。
