## SCN-689：手写告警脉冲聚合渲染

**需求与边界**：
- 需求：告警触发点位脉冲动画（scale 正弦波动）；同点位多告警聚合为一个标记（数字角标 = 数量，颜色取最高级）；紧急告警优先于一般告警的视觉；脉冲数量上限（最多 10 个同时）。
- 边界：① 聚合计数异步累加；② 脉冲在标签层（DOM）实现避免 3D 层重绘；③ 清理：告警销毁后脉冲移除且不泄漏。

**答**：
```javascript
function createAlarmPulse(container) {
  const byPos = new Map();                             // posKey -> { maxLevel, count, el }
  const active = [];
  const SEV = { info: 0, minor: 1, major: 2, critical: 3 };
  function onAlarm(alarm) {
    const key = `${alarm.deviceGroup}`;
    const cur = byPos.get(key) ?? { maxLevel: 'info', count: 0, el: null };
    cur.count++;
    if (SEV[alarm.level] > SEV[cur.maxLevel]) cur.maxLevel = alarm.level;   // 最高级着色
    if (!cur.el) { cur.el = makePulseEl(container); active.push(cur); trimActive(); }
    cur.el.querySelector('.count').textContent = cur.count;
    cur.el.className = `pulse level-${cur.maxLevel}`;
    byPos.set(key, cur);
  }
  function trimActive() {                              // 脉冲上限：只留最严重 10 个
    if (active.length > 10) {
      active.sort((a, b) => SEV[b.maxLevel] - SEV[a.maxLevel]);
      const removed = active.splice(10);
      removed.forEach(c => { c.el.remove(); byPos.delete(c.key); });   // 释放
    }
  }
  function onResolved(alarm) {
    const cur = byPos.get(alarm.deviceGroup);
    if (!cur) return;
    if (--cur.count <= 0) {                            // 全部销警才移除
      cur.el.remove();
      active.splice(active.indexOf(cur), 1);
      byPos.delete(alarm.deviceGroup);
    } else cur.el.querySelector('.count').textContent = cur.count;
  }
  // 脉冲动画：CSS animation（scale+opacity 正弦近似），标签层不触发 3D 重绘
  return { onAlarm, onResolved };
}
function makePulseEl(container) {
  const el = document.createElement('div');
  el.className = 'pulse';
  el.innerHTML = '<span class="count"></span>';
  container.appendChild(el);
  return el;
}
```

**评分点**:
- 同点位聚合（计数+最高级着色）。
- 脉冲上限裁剪（保留最严重），销毁时清理不泄漏。
- CSS 动画在标签层（不触发 3D 重绘）。
- count 递减销警语义（部分销警保留标记）。
