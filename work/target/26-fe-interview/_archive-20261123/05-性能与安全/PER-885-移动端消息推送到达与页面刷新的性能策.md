## PER-885：移动端消息推送到达与页面刷新的性能策略：推送到达时的精准刷新与节流

**答**：
- 推送刷新的坑：每条推送都触发刷新 → 消息洪泛时页面狂刷（列表跳动、请求风暴）；无脑全量刷新浪费流量，用户看不到变化。
- 精准刷新策略：推送带数据版本/变更摘要（哪些实体变了），页面按需局部更新（只刷变化的项）；洪泛时合帧（100ms 窗口内多条推送合并为一次刷新）。
- 可见性判断：页面不可见时只更新角标不刷新列表；回到前台（visibilitychange）时用当前数据版本拉增量 diff（不是全量）。
- 度量：刷新次数/分钟、增量刷新命中率、因刷新导致的列表跳动次数（用户滚动中禁止刷新，结束后再刷）。

```javascript
function pushRefreshManager({ windowMs = 100 } = {}) {
  let pending = null, timer = null;
  function onPush(msg) {                                       // msg: {entity, version, digest}
    if (!document.visibilityState.startsWith('visible')) {     // 后台只更新角标
      updateBadge(); return;
    }
    (pending ??= new Map()).set(msg.entity, msg.version);      // 洪泛合并
    if (!timer) timer = setTimeout(flush, windowMs);
  }
  async function flush() {
    timer = null;
    const entities = [...pending.keys()]; pending = null;
    if (userScrolling()) { queueAfterScroll(flush); return; }  // 滚动中延后刷新
    const diff = await fetchDelta(entities);                   // 拉增量（带版本）
    patchList(diff);                                           // 局部更新，不重排整列
    report('push-refresh', { entities: entities.length, mode: 'delta' });
  }
  return { onPush };
}
```
