## PER-517：手写一个并行性能测试调度器：把 100 个页面的跑分任务分发到多个浏览器实例并发执行并汇总

**答**：
- 核心是**并发池模型**：起 N 个浏览器实例，任务队列分发，单实例串行跑避免互相抢 CPU 导致数据污染。
- 要点：并发数不是越大越好——同机并发会互相干扰精度，跑分任务并发一般不超过 CPU 核数；失败任务要有重试与最终失败清单。
- 边界：汇总要等全部 settle（Promise.allSettled），不能让单页失败炸掉整批。
- 评分点：worker 池实现与错误隔离。

```javascript

const puppeteer = require('puppeteer');

async function pool(items, workerCount, fn) {
  const results = new Array(items.length);
  let i = 0;
  await Promise.all(Array.from({ length: workerCount }, async () => {
    const browser = await puppeteer.launch({ headless: 'new' }); // 每个并发独占一个浏览器
    while (i < items.length) {
      const idx = i++;
      try {
        results[idx] = { ok: true, ...(await fn(browser, items[idx])) };
      } catch (e) {
        results[idx] = { ok: false, url: items[idx], err: e.message };
      }
    }
    await browser.close();
  }));
  return results;
}

const med = a => a.filter(Number.isFinite).sort((x, y) => x - y)[a.length >> 1] ?? -1;
async function perfOne(browser, path) {
  const page = await browser.newPage();
  await page.goto(`https://example.com${path}`, { waitUntil: 'load', timeout: 30000 });
  const lcp = await page.evaluate(() => new Promise(res => {
    new PerformanceObserver(l => res(l.getEntries().at(-1)?.startTime ?? -1))
      .observe({ type: 'largest-contentful-paint', buffered: true });
    setTimeout(() => res(-1), 10000);
  }));
  await page.close();
  return { path, lcp };
}

(async () => {
  const pages = Array.from({ length: 100 }, (_, i) => `/page/${i}`);
  const rs = await pool(pages, 4, perfOne);        // 并发=核数，保证测量精度
  console.log('失败页:', rs.filter(r => !r.ok));
  console.log('全站 LCP 中位数:', med(rs.map(r => r.lcp)));
})();
```
