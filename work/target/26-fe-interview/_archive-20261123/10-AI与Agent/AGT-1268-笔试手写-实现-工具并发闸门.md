## 1268：笔试手写：实现 工具并发闸门

**需求**：实现 makeGate(limit)：返回 acquire(key) => Promise<release>。同一 key（工具名）最多 limit 个并发执行，超出的按 FIFO 排队；release() 后唤醒队首。不同 key 互不影响。提供 gate.stats() 输出各 key 的 active/queued。

```js
function makeGate(limit) {
  const state = new Map(); // key -> { active, queue }
  const st = key => {
    if (!state.has(key)) state.set(key, { active: 0, queue: [] });
    return state.get(key);
  };
  const pump = key => {
    const s = st(key);
    while (s.active < limit && s.queue.length) {
      s.active++;
      s.queue.shift()(); // 取出等待中的 resolve
    }
  };
  return {
    acquire(key) {
      const s = st(key);
      return new Promise(resolve => {
        const tryRun = () => {
          if (s.active < limit) { s.active++; resolve(release); }
          else s.queue.push(tryRun);
        };
        const release = () => { s.active--; pump(key); };
        tryRun();
      });
    },
    stats: () => Object.fromEntries(
      [...state].map(([k, s]) => [k, { active: s.active, queued: s.queue.length }])
    ),
  };
}
```

**评分点**：
- per-key 状态隔离（限流按工具而非全局）
- 队列存 resolve 而非 promise（唤醒即取）
- release 后 pump 补位（FIFO 公平）
- 能讨论：排队超时与取消（等待者支持 abort）是生产必备
