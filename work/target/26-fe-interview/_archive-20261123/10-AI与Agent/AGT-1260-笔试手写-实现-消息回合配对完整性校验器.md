## 1260：笔试手写：实现 消息回合配对完整性校验器

**需求**：实现 validateTurns(messages)：messages 为 OpenAI 风格 { role: 'system'|'user'|'assistant'|'tool', tool_calls?, tool_call_id? }[]。校验：每个 tool 消息必须对应前文 assistant 的某个 tool_calls id；assistant 的每个 tool_call 必须有后续 tool 回复；role 序列不允许 tool 开头。返回 { valid, broken: [{ index, reason }] }。

```js
function validateTurns(messages) {
  const broken = [];
  const openCalls = new Map(); // tool_call_id -> index
  messages.forEach((m, i) => {
    if (m.role === 'tool') {
      if (!m.tool_call_id) broken.push({ index: i, reason: 'tool 消息缺 tool_call_id' });
      else if (!openCalls.has(m.tool_call_id))
        broken.push({ index: i, reason: '无对应 assistant tool_call' });
      else openCalls.delete(m.tool_call_id); // 已回复，闭合
    }
    if (m.role === 'assistant' && m.tool_calls) {
      for (const c of m.tool_calls) {
        if (openCalls.has(c.id)) broken.push({ index: i, reason: `tool_call ${c.id} 未回复又发起` });
        openCalls.set(c.id, i);
      }
    }
  });
  openCalls.forEach((idx, id) => broken.push({ index: idx, reason: `tool_call ${id} 未收到回复` }));
  if (messages[0]?.role === 'tool') broken.push({ index: 0, reason: 'tool 消息不能开头' });
  return { valid: broken.length === 0, broken };
}
```

**评分点**：
- tool_call_id 的开合配对用 Map 记账（set/delete）
- 三类破损：缺 id、无配对、未回复
- 重复发起同一未回复 call 也算坏
- 能讨论：多轮 tool 循环中该配对是调用 API 的硬前置
