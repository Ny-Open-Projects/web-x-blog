## RBL-943：笔试：手写幻觉依赖防线（lockfile + provenance）

**需求**：实现 `guardAIDeps(files, lockfile, { declaredDeps })`。files: `{ name, code, provenance?: 'ai'|'human' }[]`。规则：1) 提取 import 的顶层包名（from 'x' / require('x') / import 'x'；相对路径 ./ ../ 跳过；@scope/pkg 取 scope+名）；2) 包不在 lockfile → blocker 'hallucinated'（无论 provenance）；3) provenance='ai' 且包不在 declaredDeps（package.json 声明）→ blocker 'undeclared-ai-dep'（AI 引入依赖必须显式 PR）；4) provenance='human' 未声明 → warn；5) 返回 `{ blockers, warnings }` 去重。边界：无 import；provenance 缺省按 human。

```javascript
const topLevel = (spec) => spec.startsWith('@')
  ? spec.split('/').slice(0, 2).join('/')
  : spec.split('/')[0];

function guardAIDeps(files, lockfile = {}, { declaredDeps = [] } = {}) {
  const blockers = [], warnings = [];
  const seen = new Set();
  const declared = new Set(declaredDeps);
  for (const f of files ?? []) {
    const prov = f.provenance ?? 'human';
    for (const m of (f.code ?? '').matchAll(/(?:from|require\()\s*['"]([^.'"][^'"]*)['"]|import\s+['"]([^.'"][^'"]*)['"]/g)) {
      const spec = (m[1] ?? m[2]).trim();
      const pkg = topLevel(spec);
      const key = `${f.name}:${pkg}`;
      if (seen.has(key)) continue;
      seen.add(key);
      if (!(pkg in lockfile)) {
        blockers.push({ type: 'hallucinated', pkg, file: f.name,
          detail: `${pkg} 不在 lockfile——疑似幻觉依赖（slopsquatting 风险）` });
        continue;
      }
      if (prov === 'ai' && !declared.has(pkg)) {
        blockers.push({ type: 'undeclared-ai-dep', pkg, file: f.name,
          detail: 'AI 代码引入依赖但无显式声明 PR' });
      } else if (prov === 'human' && !declared.has(pkg)) {
        warnings.push({ type: 'undeclared', pkg, file: f.name });
      }
    }
  }
  return { blockers, warnings };
}

// 自测
const lockfile = { lodash: { v: '4' }, '@scope/ui': { v: '1' } };
const declaredDeps = ['lodash'];
const files = [
  { name: 'a.js', code: "import _ from 'lodash';\nimport x from '@scope/ui/Button';", provenance: 'human' },
  { name: 'b.js', code: "import y from 'lodash-utils-deep';", provenance: 'ai' },    // 幻觉
  { name: 'c.js', code: "import _ from 'moment';", provenance: 'ai' },               // 在 lockfile 但未声明
  { name: 'd.js', code: "import _ from 'moment';", provenance: 'human' },            // 未声明 warn
  { name: 'e.js', code: "import { x } from './local';" },                            // 相对跳过
];
let r = guardAIDeps(files, lockfile, { declaredDeps });
console.log(r.blockers.some(b => b.type === 'hallucinated' && b.pkg === 'lodash-utils-deep'));
console.log(r.blockers.some(b => b.type === 'undeclared-ai-dep' && b.pkg === 'moment'));
console.log(r.warnings.some(w => w.type === 'undeclared' && w.pkg === 'moment'));
console.log(r.blockers.length === 2);

// 豁免：AI + 已声明
r = guardAIDeps([{ name: 'f.js', code: "import _ from 'lodash';", provenance: 'ai' }],
  lockfile, { declaredDeps: ['lodash'] });
console.log(r.blockers.length === 0 && r.warnings.length === 0);

// 相对路径跳过 + 子路径归一
r = guardAIDeps([{ name: 'g.js', code: "import { B } from '@scope/ui/Extra';" }],
  lockfile, { declaredDeps: [] });
console.log(r.warnings[0].pkg === '@scope/ui');          // scope 归一
console.log(guardAIDeps([], {}, {}).blockers.length === 0);
```

**评分点**：
- 双防线：幻觉（不在 lockfile 一律阻断——slopsquatting 向量）与 AI 未声明（provenance 强化）；scope 包归一
- human 未声明降级为 warn（分级治理）；相对路径跳过
- 去重（同文件同包）；边界
- 能说清「AI 代码与人写代码同一标准 + provenance 强化门禁——幻觉依赖是 AI 时代新供应链向量」属于加分
