## RBL-544：手写确定性构建器（排序组装 + 双次 diff 校验）
**需求**：543 落地——实现 `buildDeterministic(modules)`：1) 并行转换（乱序完成）；2) **组装前按模块 id 全排序**（消除完成顺序影响）；3) 路径相对化（sourceMap sources 去绝对路径）；4) `verifyDeterminism()`：双次构建 byte 比对；5) 附用例：并行乱序仍产出确定性产物。

```javascript
import { createHash } from 'crypto'
const md5 = (s) => createHash('md5').update(s).digest('hex')

async function buildDeterministic(modules, { transformParallel = true } = {}) {
  // 1) 并行转换（完成顺序不确定）
  const results = transformParallel
    ? await Promise.all(modules.map(async (m) => ({ id: m.id, code: await Promise.resolve(transform(m)) })))
    : modules.map((m) => ({ id: m.id, code: transform(m) }))

  // 2) 组装前全排序（按模块 id）——消除完成顺序影响
  results.sort((a, b) => a.id.localeCompare(b.id))

  // 3) 路径相对化 + 统一分隔符
  function relativize(code) {
    return code.replaceAll('/Users/dev/project', '.').replaceAll('\\', '/')
  }
  const chunks = results.map((r) => ({ id: r.id, code: relativize(r.code) }))

  // 4) 产物 hash（确定性产物 = 确定性 hash）
  const hash = md5(chunks.map((c) => `${c.id}:${c.code}`).join('\n'))
  return { chunks, hash }
}

function transform(m) {
  // 模拟带破坏源的转换：随机延迟（乱序完成）+ 绝对路径
  return new Promise((r) => setTimeout(
    () => r(`/* /Users/dev/project/${m.id}.map */ code(${m.id})`),
    Math.random() * 10))
}

async function verifyDeterminism(modules) {
  const b1 = await buildDeterministic(modules)
  const b2 = await buildDeterministic(modules)
  const same = b1.hash === b2.hash
  return { deterministic: same, hash: b1.hash }
}

// 用例
const modules = [{ id: 'b.js' }, { id: 'a.js' }, { id: 'c.js' }]
const v = await verifyDeterminism(modules)
console.log(v.deterministic)                       // true：并行乱序仍确定
const b = await buildDeterministic(modules)
console.log(b.chunks.map((c) => c.id))             // ['a.js','b.js','c.js']：组装有序
console.log(b.chunks[0].code.includes('./a.js.map'))   // true：路径相对化
```
**评分点**：
- 并行转换 + 排序组装（计算并行、组装有序——性能与确定性兼得）
- 路径相对化（绝对路径破坏源修复）
- 双次构建 byte-diff 校验（verifyDeterminism）
- 产物 hash 的确定性传递（远程缓存前提）
