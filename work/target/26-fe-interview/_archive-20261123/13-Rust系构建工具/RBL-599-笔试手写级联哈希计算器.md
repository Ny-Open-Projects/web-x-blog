## RBL-599：笔试：手写级联哈希计算器（chunk 内容 + 引用链传播）

**需求**：实现 `computeHashes(chunks)`。chunks: `{ id, code, imports: string[] }[]`（imports 是依赖 chunk id）。要求：1) 依赖者的 hash = hash(自身 code + 所有依赖的 hash 按字典序拼接)；2) 循环引用的 chunk 组成 SCC，SCC 内 hash = hash(组内全部 code 拼接)；3) hash 用提供的 `h(str)->string`（题外实现可自写 djb2）取前 8 位 hex；4) 无依赖 chunk 直接 hash(code)。相同输入必须产出相同结果。返回 `Map<id, hash>`。

```javascript
const h = (s) => { let x = 5381; for (let i = 0; i < s.length; i++) x = ((x << 5) + x + s.charCodeAt(i)) >>> 0; return x.toString(16).padStart(8, '0').slice(0, 8); };

function computeHashes(chunks) {
  const byId = new Map(chunks.map(c => [c.id, c]));
  const hashes = new Map();
  // SCC（简化：题目数据规模小，直接 DFS 环检测分组）
  const sccOf = new Map();
  const comps = findSCCs(chunks, byId);                 // 复用 tarjan 思路
  for (const comp of comps) {
    const key = comp.slice().sort().join(',');
    const seed = comp.slice().sort().map(id => byId.get(id).code).join('');
    const selfHash = h(seed);
    for (const id of comp) { sccOf.set(id, key); hashes.set(id, null); compMap.set(key, { comp, selfHash }); }
  }
  const compMap = new Map();
  for (const comp of comps) {
    const key = comp.slice().sort().join(',');
    if (!compMap.has(key)) compMap.set(key, { comp, selfHash: h(comp.slice().sort().map(id => byId.get(id).code).join('')) });
  }
  // 按依赖序求值（SCC 图上拓扑）
  const resolve = (id, visiting = new Set()) => {
    const key = sccOf.get(id);
    if (hashes.get(id)) return hashes.get(id);
    if (visiting.has(key)) return compMap.get(key).selfHash;      // 环内：用组内自身 hash
    visiting.add(key);
    const { comp, selfHash } = compMap.get(key);
    const depHashes = [];
    for (const cid of comp)
      for (const dep of byId.get(cid).imports)
        if (sccOf.get(dep) !== key) depHashes.push(resolve(dep, visiting));
    const v = h(selfHash + depHashes.sort().join(','));
    for (const cid of comp) hashes.set(cid, v);                   // SCC 共享同一 hash
    return v;
  };
  chunks.forEach(c => resolve(c.id));
  return hashes;
}
function findSCCs(chunks, byId) { /* 复用 RBL-593 tarjan */ return tarjan(chunks.map(c => c.id), new Map(chunks.map(c => [c.id, c.imports]))); }

// 自测：A -> B -> C 链；改 C 使 A/B 级联失效；环 X<->Y 共享 hash
const base = [
  { id: 'A', code: 'a', imports: ['B'] },
  { id: 'B', code: 'b', imports: ['C'] },
  { id: 'C', code: 'c', imports: [] },
  { id: 'X', code: 'x', imports: ['Y'] },
  { id: 'Y', code: 'y', imports: ['X'] },
];
const h1 = computeHashes(base);
const modified = base.map(c => c.id === 'C' ? { ...c, code: 'c2' } : c);
const h2 = computeHashes(modified);
// 断言：C 变 -> B 变 -> A 变（级联）；X/Y hash 相同（SCC 共享）
console.log(h1.get('C') !== h2.get('C'), h1.get('B') !== h2.get('B'), h1.get('A') !== h2.get('A'));
console.log(h1.get('X') === h1.get('Y'));
console.log(computeHashes(base).get('A') === h1.get('A'));       // 确定性
```

**评分点**：
- 级联正确：hash 输入 = 自身内容 + 依赖 hash（字典序拼接），依赖变则依赖者变
- 循环引用处理：SCC 内共享同一个 hash（基于组内全部 code），不死循环
- 确定性验证（同输入同输出）与 8 位 hex 截断符合规格
- 能说清「runtime 引用改写依赖此 hash，故必须先算被引用者」属于加分
