## RBL-514：手写反向依赖索引与失效传播（传递性失效 + 副作用保守）
**需求**：实现 `createDepGraph()`：1) `addModule(id, { imports, sideEffect })`；2) `invalidate(changedId)`：沿反向依赖递归失效（传递性），**副作用模块的下游保守全失效**（即使 import 的是具名导出）；3) 返回失效集合（含原因链）；4) 附用例：传递性失效与副作用保守扩散。

```javascript
function createDepGraph() {
  const modules = new Map()      // id -> { imports, sideEffect }
  const reverse = new Map()      // targetId -> Set<importerId>
  const listeners = new Set()

  function addModule(id, { imports = [], sideEffect = false }) {
    modules.set(id, { imports, sideEffect })
    for (const t of imports) {
      (reverse.get(t) ?? reverse.set(t, new Set()).get(t)).add(id)
    }
    // 重算依赖更新：移除旧边（真实实现 diff）
    return this
  }

  function invalidate(changedId, { visited = new Map() } = {}) {
    // visited: id -> reason chain
    if (visited.has(changedId)) return visited
    visited.set(changedId, ['root-change'])
    const m = modules.get(changedId)
    // 副作用模块：保守策略——其下游全部失效（即使具名导入）
    const importers = reverse.get(changedId) ?? new Set()
    for (const imp of importers) {
      const impMod = modules.get(imp)
      const chain = [`changed:${changedId}${m?.sideEffect ? '(side-effect→保守)' : ''}`]
      if (!visited.has(imp)) {
        visited.set(imp, chain)
        // 传递性失效：importer 的下游递归
        invalidate(imp, { visited })
      }
    }
    return visited
  }

  return { addModule, invalidate, reverse: () => Object.fromEntries([...reverse].map(([k, v]) => [k, [...v]])) }
}

// 用例
const g = createDepGraph()
g.addModule('a.js', { imports: [] })
g.addModule('b.js', { imports: ['a.js'] })                       // b import a
g.addModule('c.js', { imports: ['b.js'] })                       // c import b
g.addModule('polyfill.js', { imports: [], sideEffect: true })    // 副作用模块
g.addModule('d.js', { imports: ['polyfill.js'] })                // d 依赖副作用模块
g.addModule('e.js', { imports: ['d.js'] })

// 传递性失效：a 变 → b、c 连锁
const inv1 = g.invalidate('a.js')
console.log([...inv1.keys()])        // ['a.js', 'b.js', 'c.js']

// 副作用保守：polyfill 变 → d、e 全失效
const inv2 = g.invalidate('polyfill.js')
console.log([...inv2.keys()])        // ['polyfill.js', 'd.js', 'e.js']
console.log(inv2.get('e.js')[0])     // 'changed:polyfill.js(side-effect→保守)'
```
**评分点**：
- 反向依赖索引（importers Map——失效传播的基础结构）
- 传递性失效递归（visited Map 防环 + 原因链记录）
- 副作用模块保守扩散（宁可多失效不可漏——语义标注）
- 新依赖边注册（reverse Map 维护）
