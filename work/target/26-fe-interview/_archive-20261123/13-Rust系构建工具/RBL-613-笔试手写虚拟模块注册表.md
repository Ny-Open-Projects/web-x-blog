## RBL-613：笔试：手写虚拟模块注册表（id 编码 + 参数快照 + 失效回调）

**需求**：实现 `createVirtualRegistry()`：`register(name, loader)` 注册虚拟模块（loader(params) -> string）；`resolve(id)` 返回虚拟模块句柄或 null——id 形如 `virtual:name?k=v&k2=v2`；`load(id)` 返回 `{ code, snapshot }`，snapshot 是 params 的规范化字符串（键排序）。`diff(id, newParams)` 比较新参数与上次 load 的快照，变化则返回 true 并触发该模块注册的 `onInvalidate` 回调（若有）。要求：参数值做 URI 编码防注入（含 `?`/`&`/`/`）；同名重复注册抛错；未注册名返回 null；diff 未 load 过的模块以空快照为基准。

```javascript
function createVirtualRegistry() {
  const mods = new Map();                                // name -> { loader, onInvalidate }
  const lastSnap = new Map();                            // id -> snapshot
  const encode = (o) => Object.keys(o).sort()
    .map(k => `${encodeURIComponent(k)}=${encodeURIComponent(o[k])}`).join('&');

  return {
    register(name, loader, onInvalidate) {
      if (mods.has(name)) throw new Error(`重复注册: ${name}`);
      mods.set(name, { loader, onInvalidate });
    },
    parse(id) {
      const m = /^virtual:([^?]+)(?:\?(.*))?$/.exec(id);
      if (!m || !mods.has(m[1])) return null;
      const params = Object.fromEntries(new URLSearchParams(m[2] || ''));
      return { name: m[1], params };
    },
    async load(id) {
      const p = this.parse(id); if (!p) return null;
      const code = await mods.get(p.name).loader(p.params);
      const snapshot = encode(p.params);
      lastSnap.set(id, snapshot);
      return { code, snapshot };
    },
    diff(id, newParams) {
      const ns = encode(newParams), os = lastSnap.get(id) ?? '';
      if (ns === os) return false;
      lastSnap.set(id, ns);
      const p = this.parse(id);
      if (p && !p.name.startsWith('@')) {                // 已注册且非内部 -> 触发失效
        // parse 只对注册名有效，onInvalidate 在注册表中
      }
      const name = id.match(/^virtual:([^?]+)/)?.[1];
      const mod = name && mods.get(name);
      if (mod?.onInvalidate) mod.onInvalidate(id);
      return true;
    },
  };
}

// 自测
const reg = createVirtualRegistry();
let invalidated = [];
reg.register('routes', async (p) => `export default ${JSON.stringify(p)}`,
  (id) => invalidated.push(id));

console.log((await reg.load('virtual:routes?lang=zh&mode=a')).code.includes('lang'));
console.log(reg.parse('virtual:routes?lang=zh').name === 'routes');
console.log(reg.parse('virtual:nope') === null);
console.log(reg.diff('virtual:routes?lang=zh', { lang: 'zh', mode: 'a' }) === false); // 快照一致（键序无关）
console.log(reg.diff('virtual:routes?lang=zh', { lang: 'en' }) === true);
console.log(invalidated.length === 1 && invalidated[0] === 'virtual:routes?lang=zh');

// 参数注入防护
console.log((await reg.load('virtual:routes?path=/a?b&x=1')).snapshot.includes('%2F') ||
            (await reg.load('virtual:routes?path=a%2Fb')).snapshot.includes('%2F'));
let dup; try { reg.register('routes', async () => ''); } catch (e) { dup = e; }
console.log(/重复注册/.test(dup.message));
```

**评分点**：
- id 解析与未注册名返回 null 正确；重复注册抛错
- 快照规范化（键排序 + URI 编码）使 diff 与键序无关且防分隔符注入
- diff 语义正确：变化才触发 onInvalidate，未 load 过以空快照为基准
- 能说明「忘记 invalidate 的静默失效是虚拟模块典型 bug」属于加分
