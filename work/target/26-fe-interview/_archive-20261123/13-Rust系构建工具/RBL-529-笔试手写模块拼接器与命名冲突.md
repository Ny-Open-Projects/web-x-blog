## RBL-529：手写模块拼接器（执行序排列 + 命名冲突重命名）
**需求**：实现 `flattenModules(modules)`：1) `modules: Map<id, { code: string, topLevelVars: string[], imports: string[], hoistable: bool }>`；2) 执行序：导入提升——按导入图深度优先后序排列；不可 hoist 模块标记 isolated；3) 命名冲突：跨模块同名顶层变量重命名（`name__modId`），生成 renaming map；4) 输出拼接产物 + renaming map；5) 附用例：执行序与冲突重命名。

```javascript
function flattenModules(modules) {
  const order = []
  const visited = new Set()
  const isolated = new Set()

  // 1) 执行序：导入图深度优先后序（导入提升语义）
  function visit(id) {
    if (visited.has(id)) return
    visited.add(id)
    const m = modules.get(id)
    if (!m.hoistable) { isolated.add(id); return }       // 不可 hoist：独立作用域
    for (const dep of m.imports) visit(dep)
    order.push(id)                                        // 后序：依赖先执行
  }
  for (const id of modules.keys()) visit(id)

  // 2) 命名冲突重命名
  const renaming = new Map()
  const seenNames = new Map()   // name -> 首个模块 id
  for (const id of order) {
    const m = modules.get(id)
    for (const v of m.topLevelVars) {
      if (!seenNames.has(v)) seenNames.set(v, id)
      else {
        renaming.set(`${id}:${v}`, `${v}__${id.replace(/\W/g, '_')}`)   // 冲突：重命名
      }
    }
  }
  // 重命名应用到代码（真实实现走 AST 重命名；此处文本替换示意）
  function applyRenaming(id, code) {
    let out = code
    for (const [key, newName] of renaming) {
      const [mid, name] = key.split(':')
      if (mid === id) out = out.replace(new RegExp(`\\b${name}\\b`, 'g'), newName)
    }
    return out
  }

  // 3) 拼接
  const chunks = order.map((id) => {
    const m = modules.get(id)
    return { id, code: applyRenaming(id, m.code), renamed: [...renaming.keys()].filter((k) => k.startsWith(`${id}:`)) }
  })
  return { chunks, isolated: [...isolated], renaming: Object.fromEntries(renaming) }
}

// 用例
const modules = new Map([
  ['utils', { code: 'const helper = () => 1;', topLevelVars: ['helper'], imports: [], hoistable: true }],
  ['app', { code: 'const helper = () => 2; console.log(helper);', topLevelVars: ['helper'], imports: ['utils'], hoistable: true }],
  ['legacy', { code: 'eval("x"); const x = 1;', topLevelVars: ['x'], imports: [], hoistable: false }],   // eval 不可 hoist
])
const r = flattenModules(modules)
console.log(r.chunks.map((c) => c.id))         // ['utils', 'app']：utils 先执行（后序）
console.log(r.renaming)                        // { 'app:helper': 'helper__app' }：冲突重命名
console.log(r.isolated)                        // ['legacy']：eval 退回独立作用域
```
**评分点**：
- 导入提升执行序（DFS 后序——依赖先执行）
- 命名冲突检测与重命名（renaming map + 首模块优先）
- 不可 hoist 判定（eval/动态 import 退独立作用域）
- 重命名对代码的作用（文本替换示意，真实走 AST）
