## RBL-584：笔试：手写原子产物发布器（临时目录 + manifest 校验 + 回滚）

**需求**：实现 `class ArtifactPublisher`，用注入的 `fs` 适配器（方法均为 async：`writeFile/rm/rename/readFile/exists/readdir`，内存实现自备）。`publish(artifacts, { name })`：1) 写入 `.tmp-{name}` 临时目录；2) 写 `.tmp-{name}/manifest.json`（含 artifacts 文件名与 size 列表）；3) 校验：readdir 临时目录所得文件集与 manifest 一致（多文件/缺文件都算失败）；4) 校验通过则 rename 为 `{name}`（若目标已存在先备份为 `.bak-{name}`）；5) 任一步失败清理临时目录并恢复备份（回滚）。返回 `{ ok, version }` 或抛错。

```javascript
class ArtifactPublisher {
  constructor(fs) { this.fs = fs; }
  async publish(artifacts, { name }) {
    const tmp = `.tmp-${name}`, dest = name, bak = `.bak-${name}`;
    const manifest = { files: artifacts.map(a => ({ name: a.name, size: a.content.length })) };
    try {
      await this.fs.rm(tmp, { force: true });
      for (const a of artifacts) await this.fs.writeFile(`${tmp}/${a.name}`, a.content);
      await this.fs.writeFile(`${tmp}/manifest.json`, JSON.stringify(manifest));
      await this.#verify(tmp, manifest);
      // 备份旧版
      const hadOld = await this.fs.exists(dest);
      if (hadOld) await this.fs.rename(dest, bak);
      try {
        await this.fs.rename(tmp, dest);
      } catch (e) {                       // 切换失败：回滚
        if (hadOld) await this.fs.rename(bak, dest);
        throw e;
      }
      if (hadOld) await this.fs.rm(bak, { force: true }).catch(() => {});
      return { ok: true, version: manifest };
    } catch (e) {
      await this.fs.rm(tmp, { force: true }).catch(() => {});
      throw e;
    }
  }
  async #verify(dir, manifest) {
    const actual = (await this.fs.readdir(dir)).sort();
    const expect = [...manifest.files.map(f => f.name), 'manifest.json'].sort();
    if (actual.length !== expect.length || actual.some((f, i) => f !== expect[i]))
      throw new Error(`产物校验失败: 缺失=${expect.filter(f => !actual.includes(f))} 多余=${actual.filter(f => !expect.includes(f))}`);
  }
}

// 内存 fs 自测
function memFs() {
  const store = new Map();
  const p = (op) => async (...a) => { store.set(op + ':' + a[0], a); };
  return {
    store,
    writeFile: async (n, c) => { store.set(n, c); },
    rm: async (n) => { for (const k of [...store.keys()]) if (k.startsWith(n)) store.delete(k); },
    rename: async (a, b) => { for (const k of [...store.keys()]) if (k.startsWith(a)) { store.set(b + k.slice(a.length), store.get(k)); store.delete(k); } },
    exists: async (n) => store.has(n) || [...store.keys()].some(k => k.startsWith(n + '/')),
    readdir: async (n) => [...new Set([...store.keys()].filter(k => k.startsWith(n + '/')).map(k => k.slice(n.length + 1).split('/')[0]))],
  };
}

const f = memFs();
const pub = new ArtifactPublisher(f);
await pub.publish([{ name: 'a.js', content: 'xx' }], { name: 'dist' });
// 断言：f.store.has('dist/a.js')，含 dist/manifest.json
// 二次发布切换成功后 .bak-dist 已清理
await pub.publish([{ name: 'b.js', content: 'yy' }], { name: 'dist' });
console.log([...f.store.keys()]);    // dist/b.js, dist/manifest.json，无 .bak-dist
```

**评分点**：
- 临时目录写完才校验，校验以 readdir 与 manifest 文件集比对（多/缺文件双向检测）
- rename 切换前备份旧版，切换失败回滚备份——目标目录永不处于半成品态
- 失败路径清理临时目录（不残留垃圾）；manifest 是完整性标志，最后写入
- 能说清「rename 同分区原子」与 CDN 场景半成品事故的关联属于加分
