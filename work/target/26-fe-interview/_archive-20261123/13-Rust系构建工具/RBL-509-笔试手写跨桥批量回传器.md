## RBL-509：手写跨桥批量回传器（攒批 + 变更瘦身 + 预算上限）
**需求**：模拟 napi 桥——实现 `createBridgeBatcher()`：1) `push(moduleId, content, changed)`：攒批——未变更模块只传引用标记；2) `flush({ budgetBytes })`：预算内尽量多模块一次回传（超预算的分批），返回批次数与字节数；3) `stats()`：对比「逐个回传」节省的跨桥次数；4) 附用例：瘦身与预算分批。

```javascript
function createBridgeBatcher({ budgetBytes = 1024 } = {}) {
  const queue = new Map()       // moduleId -> { content, changed }
  let naiveCount = 0            // 假设逐个回传的次数
  let flushedBatches = 0
  let flushedBytes = 0
  const listeners = new Set()

  function push(moduleId, content, changed = true) {
    queue.set(moduleId, { content, changed })
    naiveCount++
  }

  function flush({ budgetBytes: budget = budgetBytes } = {}) {
    const batches = []
    let cur = [], curBytes = 2 /* 包头 */
    // 变更瘦身：未变更只传引用标记（~8 字节）
    const entries = [...queue.entries()].map(([id, m]) => ({
      id,
      payload: m.changed ? m.content : `ref:${id}`,
      changed: m.changed,
    }))
    for (const e of entries) {
      const bytes = Buffer ? Buffer.byteLength(e.payload) : e.payload.length
      if (curBytes + bytes + 8 > budget && cur.length) {
        batches.push(cur)                     // 超预算：当前批封口
        cur = []; curBytes = 2
      }
      cur.push(e)
      curBytes += bytes + 8
    }
    if (cur.length) batches.push(cur)
    flushedBatches += batches.length
    flushedBytes += curBytes
    queue.clear()
    batches.forEach((b) => listeners.forEach((fn) => fn(b)))   // 每批一次跨桥
    return { batches: batches.length, bytes: curBytes, saved: naiveCount - batches.length }
  }

  return {
    push, flush,
    stats: () => ({ naiveCount, flushedBatches, flushedBytes, saved: naiveCount - flushedBatches }),
    subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
  }
}

// 用例
const b = createBridgeBatcher({ budgetBytes: 100 })
const big = 'x'.repeat(60)
b.push('m1', big)                  // 60B
b.push('m2', 'small1', true)       // 6B
b.push('m3', 'small2', true)
b.push('m4', big, false)           // 未变更：瘦身 ref:m4
b.push('m5', big, false)
const r = b.flush({ budgetBytes: 100 })
console.log(r.batches)             // ≥2：预算分批（60B+6B+6B=80 满批，ref 批第二）
console.log(b.stats().saved > 0)   // true：5 次逐个 → 2 批
```
**评分点**：
- 变更瘦身（未变更传 ref 标记 ~8B）
- 预算分批（超预算封口当前批）
- stats 的节省口径（naive vs batched 次数）
- flush 清队列（幂等）
