## RBL-518：手写持久任务图运行器（指纹缓存 + 拓扑调度 + 按需执行）
**需求**：517 落地——实现 `createTaskGraph()`：1) `task(id, deps, fn)`：注册任务；2) `run(id)`：按需执行——依赖递归求值（拓扑序）+ **指纹缓存**（输入指纹与持久缓存一致 → 直接返回缓存输出，跳过执行）；3) 指纹 = 任务自身输入版本 + 全部依赖输出 hash（级联失效自动成立）；4) 并行：无依赖任务并发求值；5) 附用例：缓存命中与级联失效。

```javascript
function createTaskGraph({ store = new Map() } = {}) {
  const tasks = new Map()     // id -> { deps, fn, version }
  const running = new Map()   // 并发去重：同任务同 Promise
  let runs = 0

  function task(id, deps, fn) {
    tasks.set(id, { deps, fn, version: 0 })
    return this
  }

  function fingerprint(id) {
    const t = tasks.get(id)
    // 指纹 = 自身版本 + 依赖指纹（级联）
    const depFp = t.deps.map((d) => fingerprint(d)).join('|')
    return `${t.version}:${depFp}`
  }

  async function run(id) {
    // 并发去重：同任务进行中直接复用 Promise
    if (running.has(id)) return running.get(id)
    const p = (async () => {
      const t = tasks.get(id)
      const fp = fingerprint(id)
      const cached = store.get(id)
      if (cached && cached.fp === fp) {
        return cached.output                      // 指纹命中：跳过执行（持久 memoization）
      }
      // 依赖先求值（拓扑序由递归保证）
      const depOutputs = {}
      await Promise.all(t.deps.map(async (d) => { depOutputs[d] = await run(d) }))
      const output = await t.fn(depOutputs)       // 执行
      runs++
      store.set(id, { fp, output })               // 持久缓存
      return output
    })()
    running.set(id, p)
    const result = await p
    running.delete(id)
    return result
  }

  function invalidate(id) { tasks.get(id).version++ }   // 变更：版本 bump → 指纹变 → 级联失效

  return { task, run, invalidate, stats: () => ({ runs }), store }
}

// 用例
const tg = createTaskGraph()
tg.task('src', [], async () => 'src-code')
tg.task('transform', ['src'], async ({ src }) => `transformed(${src})`)
tg.task('bundle', ['transform'], async ({ transform }) => `bundle-of(${transform})`)

console.log(await tg.run('bundle'))       // bundle-of(transformed(src-code))
console.log(tg.stats().runs)              // 3：三个任务各执行一次

// 全量重跑：指纹全部命中 → 零执行
await tg.run('bundle')
console.log(tg.stats().runs)              // 3：缓存命中跳过

// src 变更 → 级联失效（transform/bundle 指纹变 → 重算，src 本身也重算）
tg.invalidate('src')
console.log(await tg.run('bundle'))       // bundle-of(transformed(src-code))
console.log(tg.stats().runs)              // 6：src/transform/bundle 级联重算
```
**评分点**：
- 指纹级联（自身版本 + 依赖指纹——失效自动传播）
- 并发去重（running Map 复用进行中 Promise）
- 拓扑序由递归求值保证 + 无依赖并发（Promise.all）
- 持久缓存接口（store 可换磁盘实现）
