## RBL-680：笔试：手写遥测数据脱敏器（白名单裁剪 + 分桶 + fail-open）

**需求**：实现 `sanitize(event, whitelist, buckets)`。whitelist: `{ field: 'raw'|'bucket'|'hash'|'platform' }[]`。规则：1) 只保留 whitelist 声明的字段；2) 'bucket' 处理：durationMs 按 buckets 边界（升序数组）分桶为 `<50ms`、`50-200ms` 等（左闭右开，最后一段含右端）；3) 'platform'：归到大类（darwin/linux/windows/other，基于前缀匹配）；4) 'hash'：简单 hash 字符串（djb2 hex）；5) 白名单外的字段一律丢弃；6) 'bucket' 但值为非数字时字段丢弃（不猜测）。返回脱敏后对象；event 为 null 返回 {}。

```javascript
const djb2 = (s) => { let h = 5381; for (const c of s) h = ((h << 5) + h + c.charCodeAt(0)) >>> 0; return h.toString(16); };

function sanitize(event, whitelist, buckets) {
  if (!event) return {};
  const out = {};
  for (const { field, mode } of whitelist) {
    const v = event[field];
    if (v === undefined) continue;
    switch (mode) {
      case 'raw': out[field] = v; break;
      case 'hash': out[field] = djb2(String(v)); break;
      case 'platform': out[field] = platformBucket(String(v)); break;
      case 'bucket':
        if (typeof v !== 'number' || !Number.isFinite(v)) continue;   // 非数字丢弃
        out[field] = bucketOf(v, buckets); break;
      default: continue;                                             // 未知模式丢弃
    }
  }
  return out;
}
const platformBucket = (p) =>
  p.startsWith('darwin') ? 'darwin' : p.startsWith('linux') ? 'linux'
  : p.startsWith('win') ? 'windows' : 'other';
function bucketOf(v, buckets) {
  const sorted = [...buckets].sort((a, b) => a - b);
  let prev = 0;
  for (const b of sorted) {
    if (v < b) return `${prev === 0 ? '<' : prev + '-'}${b}${prev === 0 ? '' : 'ms'}`.replace(/ms?$/, prev === 0 ? 'ms' : 'ms');
    prev = b;
  }
  return `>=${sorted[sorted.length - 1]}ms`;
}

// 自测
const buckets = [50, 200];
const wl = [
  { field: 'flag', mode: 'raw' },
  { field: 'durationMs', mode: 'bucket' },
  { field: 'platform', mode: 'platform' },
  { field: 'repoName', mode: 'hash' },
];
let r = sanitize(
  { flag: 'aggressiveShake', durationMs: 120, platform: 'linux-x64',
    repoName: 'internal-app', secretPath: '/Users/x/.env', password: 'hunter2' }, wl, buckets);
console.log(JSON.stringify(r));
// flag 原值；durationMs -> '50-200ms'；platform -> 'linux'；repoName -> hash；敏感字段丢弃

console.log(sanitize({ durationMs: 30 }, wl, buckets).durationMs === '<50ms');
console.log(sanitize({ durationMs: 200 }, wl, buckets).durationMs === '>=200ms');   // 右端含
console.log(sanitize({ durationMs: 'fast' }, wl, buckets).durationMs === undefined);
console.log(sanitize(null, wl, buckets));
console.log(Object.keys(sanitize({ flag: 'x', unknown: 1 }, wl, buckets)).join(','));  // flag
```

**评分点**：
- 白名单裁剪严格（白名单外字段一律丢弃，含敏感值验证）；未知 mode 也丢弃
- 分桶边界正确（左闭右开、最后段含右端、非数字丢弃不猜测）
- platform 大类归并与 hash 匿名化实现
- 能说清「遥测失败 fail-open 不阻塞构建」与「最小数据是开发者工具合规底线」属于加分
