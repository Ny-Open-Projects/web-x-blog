## RBL-533：手写常量折叠与死代码消除（L1+L2 压缩管线）
**需求**：实现 `minimizeSimple(ast)`（对简化 AST）：1) 常量折叠：二值运算两侧均为字面量 → 折叠；2) 死代码：恒假分支删除（`if (false)`）、未使用变量删除（作用域内无引用）；3) 字面量内联：`const a = 5` 且仅右值引用 → 替换后删声明；4) 多遍迭代直到不再变化（fixpoint）；5) 附用例：折叠/删除/内联/fixpoint。

```javascript
function minimizeSimple(ast) {
  // ast: { type: 'Program', body: [nodes] }
  // node: { type: 'Expr', value } | { type: 'VarDecl', name, init, refs } | { type: 'If', test, then, else }
  let changed = true
  const usedNames = collectUses(ast)

  function fold(node) {
    if (node.type === 'Expr' && typeof node.value === 'object' && node.value.op) {
      const { op, a, b } = node.value
      if (typeof a === 'number' && typeof b === 'number') {
        node.value = { folded: op === '+' ? a + b : op === '*' ? a * b : a - b }
        return true
      }
    }
    return false
  }

  function simplify(node) {
    let c = false
    if (node.type === 'If' && node.test === false) {
      node.type = 'Dead'                       // 恒假分支：标记死代码
      return true
    }
    if (node.type === 'Expr') c = fold(node) || c
    return c
  }

  function collectUses(n) {
    // 简化：VarDecl 的 refs 计入使用
    const used = new Set()
    function walk(node) {
      if (node.type === 'VarDecl') node.refs?.forEach((r) => used.add(r))
      if (node.type === 'Expr' && node.value?.uses) node.value.uses.forEach((u) => used.add(u))
    }
    walk(n)
    return used
  }

  // fixpoint 迭代
  let iterations = 0
  while (changed && iterations < 10) {
    changed = false
    iterations++
    const uses = collectUses(ast)
    const nextBody = []
    for (const node of ast.body) {
      changed = simplify(node) || changed
      if (node.type === 'Dead') continue                       // 删除死分支
      if (node.type === 'VarDecl') {
        // 字面量内联：init 为数字且 refs 全部可替换
        if (typeof node.init === 'number' && (node.refs ?? []).length > 0) {
          // 用 uses 判断：所有引用可内联（简版：直接内联并删声明）
          inlineMap.set(node.name, node.init)
          changed = true
          continue                                             // 删声明
        }
        if (!(uses.has(node.name))) continue                   // 未使用变量：删除
      }
      nextBody.push(node)
    }
    ast.body = nextBody
  }

  return { ast, iterations }
}

const inlineMap = new Map()

// 用例
const ast = {
  type: 'Program',
  body: [
    { type: 'VarDecl', name: 'x', init: 5, refs: ['y'] },            // 字面量 → 内联
    { type: 'Expr', value: { op: '+', a: 1, b: 2, uses: [] } },       // 折叠为 3
    { type: 'If', test: false, then: 'x' },                          // 死分支删除
    { type: 'VarDecl', name: 'unused', init: 9, refs: [] },          // 未使用删除
    { type: 'Expr', value: { op: '-', a: { folded: 3 }, b: 1, uses: [] } },   // 依赖折叠结果再折叠
  ],
}
const r = minimizeSimple(ast)
console.log(r.iterations >= 2)     // true：fixpoint 多遍（折叠产生新可折叠）
console.log(r.ast.body)
// [ { type: 'Expr', value: { folded: 2 } } ] —— x 内联/死分支删/unused 删/3-1=2
console.log([...inlineMap.entries()])   // [['x', 5]]：内联映射
```
**评分点**：
- 常量折叠（两侧字面量 → 折叠值）
- 死代码删除（恒假分支/未使用变量）
- 字面量内联（init 数字 → inlineMap + 删声明）
- fixpoint 多遍迭代（L1 产物成为 L2 输入——级联优化）
