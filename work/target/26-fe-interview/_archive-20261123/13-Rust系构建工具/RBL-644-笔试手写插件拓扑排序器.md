## RBL-644：笔试：手写插件拓扑排序器（enforce 分段 + before/after 依赖）

**需求**：实现 `orderPlugins(plugins, { core })`。插件： `{ name, enforce?: 'pre'|'post', before?: string[], after?: string[] }`。core 为内置核心插件数组（固定在 pre 之后 normal 之前）。排序规则：1) enforce 分段 pre/core/normal/post；2) 段内：`after` 声明排到目标之后、`before` 声明排到目标之前（同段内生效；跨段声明忽略）；3) 未声明间保持注册序；4) 循环依赖抛错（含环路径）；5) before/after 指向不存在的插件忽略。输出扁平数组。

```javascript
function orderPlugins(plugins, { core = [] } = {}) {
  const segOf = (p) => p.enforce === 'pre' ? 0 : p.enforce === 'post' ? 2 : 1;
  const groups = [[], [...core], [], []];
  for (const p of plugins) groups[segOf(p)].push(p);

  const topo = (list) => {
    const byName = new Map(list.map(p => [p.name, p]));
    const names = new Set(byName.keys());
    const done = new Set(), visiting = new Set(), out = [];
    const visit = (p, path) => {
      if (done.has(p.name)) return;
      if (visiting.has(p.name))
        throw new Error('插件顺序环: ' + [...path.slice(path.indexOf(p.name)), p.name].join(' -> '));
      visiting.add(p.name); path.push(p.name);
      // after: 先访问依赖目标
      for (const t of p.after || []) if (names.has(t)) visit(byName.get(t), path);
      out.push(p);
      // before: 目标应排在自己后 —— 等价于目标 after 自己
      // 实现：先把自己加入 out 前检查「谁 before 我」？简化为反向边处理：
      // 这里用统一边：before[X] 表示我必须在 X 前
      visiting.delete(p.name); path.pop(); done.add(p.name);
    };
    const visitAll = (p) => {
      if (done.has(p.name)) return;
      if (visiting.has(p.name)) throw new Error(`插件顺序环: ${p.name}`);
      visiting.add(p.name);
      for (const t of p.after || []) if (names.has(t) && !done.has(t)) visitAll(byName.get(t));
      out.push(p); done.add(p.name); visiting.delete(p.name);
    };
    // before 边展开：把「A before B」转成「B after A」
    for (const p of list) {
      const beforeTargets = (p.before || []).filter(t => names.has(t) && byName.get(t));
      for (const t of beforeTargets) {
        const target = byName.get(t);
        target.after = target.after || [];
        if (!target.after.includes(p.name)) target.after.push(p.name);
      }
    }
    list.forEach(visitAll);
    return out;
  };

  return [...groups[0], ...topo(groups[1]), ...topo(groups[2]), ...topo(groups[3])];
}

// 自测
const P = (name, extra = {}) => ({ name, ...extra });
// 1) enforce 分段
let out = orderPlugins([
  P('normal1'), P('post1', { enforce: 'post' }), P('pre1', { enforce: 'pre' }),
], { core: [P('core1')] });
console.log(out.map(p => p.name).join(','));             // pre1,core1,normal1,post1

// 2) after 声明
out = orderPlugins([
  P('b'), P('a', { after: ['b'] }),                      // a 必须在 b 后
]);
console.log(out.map(p => p.name).join(','));             // b,a

// 3) before 声明
out = orderPlugins([
  P('later'), P('early', { before: ['later'] }),
]);
console.log(out.map(p => p.name).join(','));             // early,later

// 4) 环
let err; try { orderPlugins([
  P('x', { after: ['y'] }), P('y', { after: ['x'] }),
]); } catch (e) { err = e; }
console.log(/环/.test(err.message));

// 5) 指向不存在插件忽略
out = orderPlugins([P('only', { after: ['ghost'], before: ['ghost2'] })]);
console.log(out.length === 1);

// 6) core 段不受 before/after 影响
out = orderPlugins([P('u1', { before: ['core1'] })], { core: [P('core1')] });
console.log(out.map(p => p.name).join(','));             // core1,u1
```

**评分点**：
- enforce 三段 + core 固定段正确；段内拓扑基于 after 边，before 展开为反向 after
- 环检测含路径；跨段声明忽略、幽灵目标忽略
- 未声明插件保持注册序（拓扑仅在声明间生效，避免过度重排）
- 能说清「顺序契约从注册序到显式声明的可维护性收益」属于加分
