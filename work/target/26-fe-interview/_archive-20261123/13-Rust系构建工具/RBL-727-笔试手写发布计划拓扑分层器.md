## RBL-727：笔试：手写发布计划拓扑分层器（依赖分层 + 语义跳过 + 部分失败）

**需求**：实现 `releasePlan(affected, graph)`。affected: 需发布的包名集合；graph: `Record<name, { deps: string[], semanticChanged: boolean }>`. 规则：1) affected 中 semanticChanged=false 的包跳过（内部实现变化免发布）但**其依赖者不受影响照常发布**；2) 分层：依赖全部不在 affected（或已在前层）的包为同一层——层内可并发；3) 返回 `layers: string[][]`（拓扑序）；4) 环内包合并到同层（标 warning）。

```javascript
function releasePlan(affected, graph) {
  const todo = [...affected].filter(p => graph[p].semanticChanged !== false);
  const layers = []; const published = new Set(); const warnings = [];
  const visiting = new Set();

  function ready(p) {
    return graph[p].deps.every(d => !affected.has(d) || published.has(d) || graph[d]?.semanticChanged === false);
  }
  function topo(p, stack) {
    if (published.has(p) || !todo.includes(p)) return;
    if (visiting.has(p)) { warnings.push(`环: ${[...stack, p].join(' -> ')}`); return; }
    visiting.add(p);
    for (const d of graph[p].deps) topo(d, [...stack, p]);
    visiting.delete(p);
  }

  while (todo.some(p => !published.has(p))) {
    const layer = todo.filter(p => !published.has(p) && ready(p));
    if (!layer.length) {
      // 环兜底：把剩余的放同层
      const rest = todo.filter(p => !published.has(p));
      warnings.push(`依赖环兜底同层发布: ${rest.join(',')}`);
      layers.push(rest);
      rest.forEach(p => published.add(p));
      break;
    }
    layers.push([...layer]);
    layer.forEach(p => published.add(p));
  }
  return { layers, warnings };
}

// 自测
const graph = {
  ui: { deps: [], semanticChanged: true },
  hooks: { deps: ['ui'], semanticChanged: true },
  utils: { deps: [], semanticChanged: false },           // 语义不变：免发布
  web: { deps: ['ui', 'hooks', 'utils'], semanticChanged: true },
};
const affected = new Set(['ui', 'hooks', 'utils', 'web']);
const { layers, warnings } = releasePlan(affected, graph);
console.log(layers[0].join(','));                        // ui（utils 语义不变跳过）
console.log(layers[1].join(','));                        // hooks
console.log(layers[2].join(','));                        // web
console.log(warnings.length === 0);
console.log(!layers.flat().includes('utils'));

// 全跳过
const { layers: l2 } = releasePlan(new Set(['utils']), graph);
console.log(l2.length === 0);

// 环兜底
const g2 = { a: { deps: ['b'], semanticChanged: true }, b: { deps: ['a'], semanticChanged: true } };
const r3 = releasePlan(new Set(['a', 'b']), g2);
console.log(r3.warnings.length >= 1 && r3.layers.length >= 1);
```

**评分点**：
- 语义跳过正确（semanticChanged=false 不发布、不影响依赖者）
- 分层就绪判定正确（依赖已发布或不受影响才进当前层）；拓扑序保证
- 环兜底同层 + warning；全跳过返回空 layers
- 能说清「语义级 affected 让『免发布』成为一等决策」属于加分
