## RBL-883：笔试：手写第三方脚本加载器（consent + idle + 超时）

**需求**：实现 `createTpLoader(configs, { consentGranted, loadScript, idle })`。configs: `{ id, src, consent?, idle?: boolean, timeout }[]`。规则：1) consent 声明的脚本：consentGranted 返回 false → 返回 { status: 'blocked' }（不加载）；2) idle=true：用注入 idle（模拟 requestIdleCallback）延迟加载；3) loadScript 注入：`(src) => Promise`（可超时失败）；加载超时（timeout ms 内未完成）→ { status: 'timeout' } 且**不重试不冒泡**；4) 加载去重（同 id 并发只加载一次，共享结果）；5) 返回 `{ load(id) -> Promise<{ status: 'loaded'|'blocked'|'timeout'|'failed' }>, loaded() }`。边界：未知 id 抛错；consent 网络失败按未同意处理。

```javascript
function createTpLoader(configs, { consentGranted = async () => true, loadScript, idle = (fn) => fn() } = {}) {
  const byId = new Map(configs.map(c => [c.id, c]));
  const inflight = new Map();
  function load(id) {
    const cfg = byId.get(id);
    if (!cfg) throw new Error(`未知第三方: ${id}`);
    if (inflight.has(id)) return inflight.get(id);
    const p = (async () => {
      if (cfg.consent) {
        try { if (!(await consentGranted(cfg.consent))) return { status: 'blocked' }; }
        catch { return { status: 'blocked' }; }          // consent 异常按未同意（保守）
      }
      const run = () => Promise.race([
        loadScript(cfg.src).then(() => ({ status: 'loaded' })),
        new Promise(res => setTimeout(() => res({ status: 'timeout' }), cfg.timeout)),
      ]);
      return cfg.idle ? idle(run) : run();
    })();
    inflight.set(id, p);
    p.finally(() => inflight.delete(id));
    return p;
  }
  return { load,
    loaded: () => [...inflight.keys()] };
}

// 自测
const loadedScripts = [];
const loader = createTpLoader([
  { id: 'analytics', src: '/tp/a.js', consent: 'analytics', timeout: 100 },
  { id: 'ads', src: '/tp/b.js', consent: 'ads', idle: true, timeout: 50 },
  { id: 'chat', src: '/tp/c.js', timeout: 80 },
], {
  consentGranted: async (cat) => cat !== 'ads',          // ads 未同意
  loadScript: async (src) => {
    if (src.includes('c.js')) await sleep(200);          // chat 超时
    loadedScripts.push(src);
  },
  idle: (fn) => fn(),                                    // 立即（测试）
});
// 并发去重
const [a1, a2] = await Promise.all([loader.load('analytics'), loader.load('analytics')]);
console.log(a1 === a2 && a1.status === 'loaded' && loadedScripts.length === 1);

const ads = await loader.load('ads');
console.log(ads.status === 'blocked' && loadedScripts.every(s => !s.includes('b.js')));

const chat = await loader.load('chat');
console.log(chat.status === 'timeout');                  // 超时且不重试
console.log(loader.loaded().join(',') === 'chat');       // inflight 清理（analytics 已完成移除）

let e; try { loader.load('ghost'); } catch (x) { e = x; }
console.log(/未知/.test(e.message));
```

**评分点**：
- consent 门控（异常按未同意的保守默认——RBL-827 哲学）；去重共享（inflight 合并）
- idle 注入与超时竞速（timeout 结果不重试不冒泡——失败隔离）；inflight 清理
- 未知 id 抛错；边界
- 能说清「第三方加载器统一入口让性能税显性化，独立预算是责任主体分离」属于加分
