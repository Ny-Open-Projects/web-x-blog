## RBL-638：笔试：手写远程缓存完整性校验与权限分级判定

**需求**：实现两个函数。`verifyBundle(files, manifest)`：files 为 `{ name, content }[]`（content 是字符串，用注入 hash(content) 模拟），manifest: `{ hashes: Record<name, string> }`。校验：1) manifest 中每个文件都存在且 hash 匹配；2) files 中不在 manifest 里的「多余文件」也报错（防夹带）；3) manifest.json 本身不算产物文件。全部通过返回 true，否则抛出含首个问题文件名的错误。`canWrite(env)`：env: `{ branch, isPR, token: 'read'|'write'|null }`——主分支 CI 且 write token 可写；PR runner 只读（返回 false 并给 reason 'pr-readonly'）；无 token false；本地开发（branch 为 null 且非 CI）write token 时允许但标记 risky（返回 { ok: true, risky: true }）。

```javascript
function verifyBundle(files, manifest, { hash }) {
  const byName = new Map(files.map(f => [f.name, f.content]));
  const known = new Set(Object.keys(manifest.hashes));
  for (const [name, h] of Object.entries(manifest.hashes)) {
    if (!byName.has(name)) throw new Error(`缺失产物: ${name}`);
    if (hash(byName.get(name)) !== h) throw new Error(`哈希不匹配: ${name}`);
  }
  for (const f of files)
    if (!known.has(f.name) && f.name !== 'manifest.json')
      throw new Error(`多余文件: ${f.name}（不在 manifest 中）`);
  return true;
}

function canWrite(env) {
  const { branch, isPR, token } = env;
  if (!token) return { ok: false, reason: 'no-token' };
  if (isPR) return { ok: false, reason: 'pr-readonly' };
  if (token === 'write' && (branch === 'main' || branch === null)) {
    return branch === null ? { ok: true, risky: true } : { ok: true };
  }
  return { ok: false, reason: 'branch-not-allowed' };
}

// 自测
const h = (s) => 'h_' + s;
const files = [
  { name: 'index.js', content: 'ABC' },
  { name: 'index.js.map', content: 'DEF' },
];
const manifest = { hashes: { 'index.js': h('ABC'), 'index.js.map': h('DEF') } };
console.log(verifyBundle(files, manifest, { hash: h }) === true);

// 篡改检测
let e1; try { verifyBundle([{ name: 'index.js', content: 'XX' }], manifest, { hash: h }); } catch (e) { e1 = e; }
console.log(/哈希不匹配.*index\.js/.test(e1.message));

// 多余文件
let e2; try { verifyBundle([...files, { name: 'evil.sh', content: 'x' }],
  manifest, { hash: h }); } catch (e) { e2 = e; }
console.log(/多余文件.*evil\.sh/.test(e2.message));

// 缺失
let e3; try { verifyBundle([files[0]], manifest, { hash: h }); } catch (e) { e3 = e; }
console.log(/缺失产物.*index\.js\.map/.test(e3.message));

// 权限
console.log(canWrite({ branch: 'main', isPR: false, token: 'write' }).ok === true);
console.log(canWrite({ branch: 'feat', isPR: true, token: 'write' }).reason === 'pr-readonly');
console.log(canWrite({ branch: null, isPR: false, token: 'write' }).risky === true);
console.log(canWrite({ branch: 'main', isPR: false, token: null }).reason === 'no-token');
console.log(canWrite({ branch: 'dev', isPR: false, token: 'read' }).ok === false);
```

**评分点**：
- 校验三向：缺失/哈希不匹配/多余文件，错误信息含首个问题文件名（可诊断）
- manifest.json 自身豁免产物校验（边界处理）
- 权限分级正确：PR 只读（防投毒核心）、本地 write 标记 risky、分支白名单
- 能说清「缓存产物是可执行代码，签名校验不可省」属于加分
