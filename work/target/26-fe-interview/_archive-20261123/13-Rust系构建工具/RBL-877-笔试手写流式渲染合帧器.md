## RBL-877：笔试：手写流式渲染合帧器（rAF 批处理 + 尾段渲染）

**需求**：实现 `createStreamingRenderer({ render, renderTail })`。`append(text)`：累积 token；每帧最多渲染一次：完整内容传给 render（首帧/闭合段落变化时）且未闭合尾段传给 renderTail。`flush()`：立即渲染待处理内容（停止生成时调用）。`stop()`：abort 标记——后续 append 忽略，flush 已有内容并触发 onStop 回调。规则：1) rAF 合帧（多次 append 一帧一次渲染）；2) 渲染参数：`{ full, tail }`（full 为完整文本，tail 为最后一个未闭合段落——以 '\n\n' 分割）；3) dispose 后 append 忽略；4) 渲染调用计数可通过 opts 计数验证。返回对象方法齐全。

```javascript
function createStreamingRenderer({ render, renderTail, onStop, raf = requestAnimationFrame, caf = cancelAnimationFrame } = {}) {
  let buf = ''; let full = ''; let rafId = null; let stopped = false; let disposed = false;
  function paint() {
    rafId = null;
    if (disposed) return;
    const blocks = buf.split('\n\n');
    const tail = blocks[blocks.length - 1] ?? '';
    render?.(buf);
    renderTail?.(tail);
  }
  return {
    append(text) {
      if (stopped || disposed) return;
      buf += text;
      if (rafId == null) rafId = raf(() => paint());     // 合帧
    },
    flush() {
      if (rafId != null) { caf(rafId); rafId = null; }
      if (buf && !disposed) paint();
    },
    stop() {
      stopped = true;
      this.flush();
      onStop?.(buf);                                     // 半截消息保留
    },
    dispose() {
      disposed = true;
      if (rafId != null) { caf(rafId); rafId = null; }
    },
    get content() { return buf; },
  };
}

// 自测：合成 rAF（手动触发帧）
let frameCb = null;
const raf = (cb) => { frameCb = cb; return 1; };
const caf = () => { frameCb = null; };
const paints = [], tails = [];
const r = createStreamingRenderer({
  render: (full) => paints.push(full),
  renderTail: (tail) => tails.push(tail),
  raf, caf,
});

r.append('hello ');
r.append('world');                                      // 同帧两次 append
console.log(paints.length === 0);                       // 未渲染（等帧）
frameCb();                                              // 触发帧
console.log(paints.length === 1 && paints[0] === 'hello world');
console.log(tails[0] === 'hello world');

r.append('!\n\nnew block ');                            // 新段落（尾段变化）
frameCb();
console.log(tails[tails.length - 1] === 'new block ');
console.log(paints[paints.length - 1] === 'hello world!\n\nnew block ');

// stop：flush + 后续忽略
r.stop();
console.log(paints.length === 3);                       // stop 立即 flush
r.append('ignored');
frameCb();
console.log(paints.length === 3 && r.content === 'hello world!\n\nnew block ');

r.dispose();
r.append('more');
console.log(paints.length === 3);
```

**评分点**：
- rAF 合帧正确（pending 累积、一帧一渲染）；flush 立即渲染（stop 前调用）
- 尾段提取（\n\n 分割最后块）传 renderTail；stop 后 append 忽略且触发 onStop（半截保留）
- dispose 清理 rAF 与状态；合成 rAF 注入使测试确定性
- 能说清「流式渲染性能本质是批处理，与 HMR 合帧/进度合帧同构」属于加分
