## RBL-660：笔试：手写路径安全校验器（穿越检测 + glob 锁定 + define 白名单）

**需求**：实现三个校验函数。1) `insideRoot(resolved, root)`：resolved 相对 root 不越界（`../` 或绝对路径在根外）返回 true；root 自身返回 true；等价路径（`/a/b/../c`）先规范化。2) `validateGlob(globPattern, base)`：base 之后的模式部分不得含 `..` 段与绝对路径特征（`/` 开头），否则抛错（防 glob 目录逃逸）。3) `validateDefine(define)`：key 为 `process.env`（整对象）抛错；`process.env.X` 形式通过；其他 key（如 `__APP_VERSION__`）通过。返回值为布尔/抛错要明确。

```javascript
const normalize = (p) => p.split('/').reduce((acc, seg) => {
  if (seg === '..') acc.pop();                           // 空栈时 '..' 越界
  else if (seg && seg !== '.') acc.push(seg);
  return acc;
}, []);

function insideRoot(resolved, root) {
  const r = normalize(resolved), rt = normalize(root);
  if (r.length < rt.length) return false;                // '..' 弹空后比根浅
  for (let i = 0; i < rt.length; i++)
    if (r[i] !== rt[i]) return false;
  return true;
}

function validateGlob(pattern, base) {
  if (!insideRoot(base, '/root')) throw new Error(`base 越界: ${base}`);
  const rel = pattern.startsWith(base) ? pattern.slice(base.length) : pattern;
  const segs = rel.split('/').filter(Boolean);
  if (segs.includes('..')) throw new Error(`glob 模式越界: ${pattern}`);
  if (pattern.startsWith('/')) throw new Error(`glob 模式为绝对路径: ${pattern}`);
  return true;
}

function validateDefine(define) {
  for (const key of Object.keys(define)) {
    if (key === 'process.env')
      throw new Error('禁止整对象注入 process.env——逐个声明所需变量');
    if (/^process\.env\.[A-Za-z_$][\w$]*$/.test(key)) continue;
    if (key === '__proto__' || key === 'constructor') throw new Error(`危险 key: ${key}`);
  }
  return true;
}

// 自测
console.log(insideRoot('/root/src/a.js', '/root') === true);
console.log(insideRoot('/root', '/root') === true);
console.log(insideRoot('/root/src/../src/x.js', '/root') === true);    // 规范化
console.log(insideRoot('/root/../secrets.env', '/root') === false);
console.log(insideRoot('/other/x.js', '/root') === false);

let e1; try { validateGlob('/root/../etc/*', '/root'); } catch (x) { e1 = x; }
console.log(/越界/.test(e1.message));
console.log(validateGlob('/root/src/*.js', '/root') === true);
let e2; try { validateGlob('/*', '/root'); } catch (x) { e2 = x; }
console.log(/绝对路径/.test(e2.message));

console.log(validateDefine({ 'process.env.NODE_ENV': 'production', __APP_VERSION__: '1' }) === true);
let e3; try { validateDefine({ 'process.env': { SECRET: 1 } }); } catch (x) { e3 = x; }
console.log(/禁止整对象/.test(e3.message));
let e4; try { validateDefine({ __proto__: 'x' }); } catch (x) { e4 = x; }
console.log(/危险/.test(e4.message));
```

**评分点**：
- 穿越检测基于段级规范化（不是字符串 includes，'..foo' 不误判）；根自身/规范化等价路径正确
- glob 校验三防：`..` 段、绝对路径、base 自身越界
- define 白名单：整对象拒绝、单变量/自定义全局通过，__proto__/constructor 显式拒绝
- 能说清「.env 意外打包与 define 整对象注入是密钥泄漏两大源」属于加分
