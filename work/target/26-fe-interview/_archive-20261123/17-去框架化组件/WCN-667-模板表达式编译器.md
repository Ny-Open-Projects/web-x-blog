## WCN-667：笔试：实现模板表达式编译器（词法分析 + 递归下降解析 + 生成取值函数）

**答**：
- 需求：实现 parseExpr(src) 解析简单表达式（标识符、成员访问 a.b.c、字符串、数字、+ 运算），返回 AST；再实现 evaluate(ast, ctx) 从上下文取值。
- 思路：tokenizer 切 token；递归下降解析 primary → member → additive；AST 节点 type 区分 ident/member/lit/binary。
- 边界：括号优先级；上下文缺字段返回 undefined 而非抛错；非法 token 抛带位置的错误。

```javascript
function tokenize(src) {
  const tokens = []; let i = 0;
  while (i < src.length) {
    const c = src[i];
    if (/\s/.test(c)) { i++; continue; }
    if (/[A-Za-z_$]/.test(c)) {
      let s = i; while (/[\w$]/.test(src[i] || '')) i++;
      tokens.push({ t: 'id', v: src.slice(s, i) }); continue;
    }
    if (/\d/.test(c)) {
      let s = i; while (/\d/.test(src[i] || '')) i++;
      tokens.push({ t: 'num', v: Number(src.slice(s, i)) }); continue;
    }
    if (c === '.') { tokens.push({ t: '.' }); i++; continue; }
    if (c === '+') { tokens.push({ t: '+' }); i++; continue; }
    throw new Error(`非法字符 ${c} @ ${i}`);
  }
  return tokens;
}
function parseExpr(src) {
  const ts = tokenize(src); let p = 0;
  function primary() {
    const t = ts[p++];
    if (t.t === 'num') return { type: 'lit', v: t.v };
    if (t.t === 'id') {
      let node = { type: 'ident', name: t.v };
      while (ts[p]?.t === '.') { p++; node = { type: 'member', obj: node, prop: ts[p++].v }; }
      return node;
    }
    throw new Error('期望标识符或数字');
  }
  function expr() {
    let left = primary();
    while (ts[p]?.t === '+') { p++; left = { type: 'binary', op: '+', left, right: primary() }; }
    return left;
  }
  return expr();
}
function evaluate(ast, ctx) {
  switch (ast.type) {
    case 'lit': return ast.v;
    case 'ident': return ctx[ast.name];
    case 'member': return evaluate(ast.obj, ctx)?.[ast.prop];
    case 'binary': return evaluate(ast.left, ctx) + evaluate(ast.right, ctx);
  }
}
const ast = parseExpr('user.address.city');
console.log(evaluate(ast, { user: { address: { city: '深圳' } } })); // 深圳
```

**评分点**：
- tokenizer 正确切分标识符/数字/点/加号，非法字符报错带位置。
- 递归下降解析支持成员访问链与 + 运算（优先级正确）。
- evaluate 对上下文缺字段安全返回 undefined。
- 生成可复用的 AST 结构（便于后续编译为代码）。
