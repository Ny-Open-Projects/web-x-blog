## ARC-234：权限系统的性能优化：权限快照过大怎么办？
**答**：
- 大快照的两个代价：传输体积（几万权限码的 SaaS）与判定开销（每次渲染遍历数组）。
- 体积优化：**按域分片**——只下发当前应用涉及的权限域（进入中台某子系统才拉那一片）；位图编码（权限码全平台有序号，快照即 bitset，几千码压缩到几百字节）；字段级缓存 + 版本号 diff 增量。
- 判定优化：快照进 Set/位图，O(1) 查询；路由守卫在路由注册期预解析 perm 谓词，渲染期零字符串拼接。
- 度量：快照传输 P90、首屏权限就绪时间（blocking 快照是首屏串行点，尽量异步 + 骨架）。

```typescript
// 位图权限快照：位数组 + O(1) 判定
const ALL_PERMS = registerTable.codes;                    // 全平台有序权限码表（带版本）
const idx = new Map(ALL_PERMS.map((c, i) => [c, i]));
function toBitmap(perms: string[]): Uint8Array {
  const bm = new Uint8Array(Math.ceil(ALL_PERMS.length / 8));
  for (const p of perms) bm[idx.get(p)! >> 3] |= 1 << (idx.get(p)! & 7);
  return bm;
}
function has(bm: Uint8Array, code: string) {
  const i = idx.get(code); return i != null && (bm[i >> 3] & (1 << (i & 7))) !== 0;
}
// 增量同步：版本号落后 → 服务端只返回变更的码列表 → 客户端 patch 位图
```
