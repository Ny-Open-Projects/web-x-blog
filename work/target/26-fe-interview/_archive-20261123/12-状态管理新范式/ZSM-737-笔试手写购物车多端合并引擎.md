## ZSM-737：手写购物车多端合并引擎（版本裁决 + pending 保留 + 合并语义）
**需求**：实现 `createCartMerge()`：1) `applyRemote({ version, items })`：版本单调检查；合并语义——同 SKU 取 max(qty)、远端没有而本地有的条目**若非 pending 则视为被其他端删除**（提示并移除）、pending 条目保留；2) `addItem(sku, qty)`：乐观 +pending；`confirm(sku, serverItem)`：确认替换；3) `derived()`：小计/角标派生；4) 事件通知带 changes（added/removed/qtyChanged）；5) 附用例：三端冲突合并的最终态验证。

```javascript
function createCartMerge() {
  const items = new Map()       // sku -> { qty, price, pending }
  let version = 0
  const listeners = new Set()

  function addItem(sku, qty = 1, price = null) {
    const cur = items.get(sku)
    items.set(sku, { qty: (cur?.qty ?? 0) + qty, price: price ?? cur?.price ?? null, pending: true })
    notify({ added: [sku] })
  }

  function confirm(sku, serverItem) {
    items.set(sku, { ...serverItem, pending: false })    // 确认替换
    notify({ confirmed: [sku] })
  }

  function applyRemote({ version: v, items: remote }) {
    if (v <= version) return { applied: false, reason: 'stale' }   // 旧版本丢弃
    const changes = { added: [], removed: [], qtyChanged: [] }
    const next = new Map()
    const remoteSkus = new Set(remote.map((i) => i.sku))

    for (const r of remote) {
      const local = items.get(r.sku)
      if (!local) {
        next.set(r.sku, { ...r, pending: false })        // 他端新增
        changes.added.push(r.sku)
      } else if (local.pending) {
        // 本地在途：合并语义 max(qty)
        const qty = Math.max(local.qty, r.qty)
        next.set(r.sku, { ...r, qty, pending: true })
        if (qty !== r.qty) changes.qtyChanged.push(r.sku)
      } else if (r.qty !== local.qty) {
        next.set(r.sku, { ...r, pending: false })        // 远端权威
        changes.qtyChanged.push(r.sku)
      } else {
        next.set(r.sku, { ...r, pending: false })
      }
    }
    // 远端没有且非 pending：被其他端删除
    for (const [sku, it] of items) {
      if (!remoteSkus.has(sku) && !it.pending) {
        changes.removed.push(sku)                        // 不进 next：删除胜
      } else if (!remoteSkus.has(sku) && it.pending) {
        next.set(sku, it)                                // pending 保留
      }
    }
    items.clear()
    for (const [k, v] of next) items.set(k, v)
    version = v
    if (changes.removed.length) changes.notice = `购物车已在其他设备更新`
    notify(changes)
    return { applied: true, changes }
  }

  function derived() {
    let subtotal = 0, count = 0
    for (const it of items.values()) {
      subtotal += (it.price ?? 0) * it.qty
      count += it.qty
    }
    return { subtotal, count, pendingCount: [...items.values()].filter((i) => i.pending).length }
  }

  function notify(changes = {}) {
    listeners.forEach((fn) => fn({ items: [...items.entries()], version, ...derived(), changes }))
  }

  return { addItem, confirm, applyRemote, derived, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)), snapshot: () => [...items.entries()] }
}

// 用例：三端冲突
const cart = createCartMerge()
cart.addItem('a', 2, 1000)          // 本端：a=2 pending
cart.confirm('a', { sku: 'a', qty: 2, price: 1000 })
cart.addItem('b', 1, 500)           // b pending

const res = cart.applyRemote({
  version: 5,
  items: [
    { sku: 'a', qty: 5, price: 1000 },   // 他端 a=5 → max(2,5)=5? 不对：a 非 pending → 远端权威 5
    { sku: 'c', qty: 1, price: 300 },    // 他端新增 c
    // b 未出现：pending → 保留
  ],
})
console.log(res.applied, res.changes)
// applied=true, { added: ['c'], qtyChanged: ['a'], removed: [], pendingCount: 1 }
const d = cart.derived()
console.log(d.count)                 // 5+1(b pending)+1(c)=7
console.log(d.pendingCount)          // 1（b 在途）
```
**评分点**：
- 版本单调裁决（旧推送丢弃返回 stale）
- 合并语义三分支（pending 取 max / 非 pending 远端权威 / 远端缺省即删除）
- pending 保留不被推送吞掉（本地在途不丢）
- changes 事件含 added/removed/qtyChanged + 用户提示字段
