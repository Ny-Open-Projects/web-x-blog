## ZSM-197：手写 useDraftAutosave（草稿自动保存：脏检查 + 防抖 + 冲突保护）
**需求**：实现 `useDraftAutosave(docId, initial, { save, delay = 1000 })`：1) 内容变化即「脏」，防抖 delay 后自动 save（只在脏时发）；2) save 成功后基线更新（不再脏）；3) 保存中用户继续编辑：不阻塞输入，保存完成后若内容又变了继续下轮；4) 冲突保护：save 带 baseVersion，服务端发现版本落后（409）时停止自动保存并暴露 conflict，由用户手动解决；5) 提供 { draft, setDraft, status, conflict, saveNow, reset }。

```javascript
export function useDraftAutosave(docId, initial, { save, delay = 1000 }) {
  const [draft, setDraftRaw] = useState(initial)
  const [meta, setMeta] = useState({ status: 'clean', conflict: null })
  const baseline = useRef(initial)      // 已保存基线
  const version = useRef(0)
  const timer = useRef(), alive = useRef(true)
  useEffect(() => () => { alive.current = false }, [])

  const setDraft = useCallback((updater) => {
    setDraftRaw(updater)
    setMeta(m => m.conflict ? m : { ...m, status: 'dirty' })   // 冲突期暂停自动保存
    clearTimeout(timer.current)
    timer.current = setTimeout(() => doSave(), delay)          // 防抖自动保存
  }, [delay])

  async function doSave() {
    const content = draftRef.current
    if (deepEq(content, baseline.current)) {                   // 脏检查：没变不保存
      setMeta(m => ({ ...m, status: 'clean' }))
      return
    }
    setMeta(m => ({ ...m, status: 'saving' }))
    try {
      const res = await save(docId, content, version.current)  // 带版本号
      if (!alive.current) return
      baseline.current = content                              // 基线推进
      version.current = res.version
      setMeta(m => ({ ...m, status: 'clean', conflict: null }))
    } catch (e) {
      if (e.status === 409) {                                  // 版本冲突
        setMeta(m => ({ ...m, status: 'conflict', conflict: e.server }))
        return                                                  // 停止自动保存
      }
      setMeta(m => ({ ...m, status: 'error' }))
    }
  }

  const reset = useCallback(() => {
    clearTimeout(timer.current)
    setDraftRaw(baseline.current)                              // 回到已保存基线
    setMeta({ status: 'clean', conflict: null })
  }, [])

  return { draft, setDraft, ...meta, saveNow: doSave, reset }
}

// 评分点：
// 1. 脏检查基线（baseline）而不是「有输入就保存」
// 2. 保存中继续编辑：防抖续期，保存完成后自动开启下一轮
// 3. 版本号乐观锁 + 409 停止自动保存暴露 conflict
// 4. saveNow 手动直存（提交前调用）
// 5. reset 回基线，冲突后用户「放弃我的修改」路径
```
