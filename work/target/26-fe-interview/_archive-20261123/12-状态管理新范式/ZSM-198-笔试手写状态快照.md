## ZSM-198：手写状态回放器（action log 录制与重放：时间旅行调试核心）
**需求**：实现 `createRecorder(store)`：1) 录制：包装 store.dispatch，把每个 action 按序存入 log（含时间戳与执行后的状态快照可选项）；2) 重放 `replay(log, upto)`：从 initial 开始按序 dispatch 到第 upto 条，返回最终状态；3) 快进优化：若 log 中含「状态快照」条目，从最近快照开始重放，跳过其前的 action；4) 撤销 `undo()`：回退到上一个 action 前的状态（要求 store 的 reducer 纯函数）；5) 检测非纯 reducer：重放两次结果不同抛出错误。核心考点：纯函数前提 + 快照跳跃重放。

```javascript
export function createRecorder({ reducer, initial }) {
  const log = []

  function dispatch(action) {
    const state = reducer(currentState(), action)
    log.push({ ...action, __ts: Date.now(), __state: state })   // 附快照
    return state
  }
  let cache = null
  function currentState() {
    if (cache) return cache
    cache = log.reduce((s, a) => reducer(s, a), initial)
    return cache
  }

  function replay(upto = log.length) {
    const first = log.slice(0, upto).reverse().find(a => a.__state != null)
    if (!first) return log.slice(0, upto).reduce((s, a) => reducer(s, a), initial)
    // 快进：从最近快照之后开始重放
    const idx = log.indexOf(first) + 1
    let s = first.__state
    for (let i = idx; i < upto; i++) s = reducer(s, log[i])
    return s
  }

  function undo() {
    if (!log.length) return currentState()
    log.pop()
    cache = null                       // 失效缓存，从头重算（纯函数保证一致）
    return currentState()
  }

  function verifyPure(times = 2) {     // 纯函数自检：两次重放结果必须一致
    const a = replay(), b = replay()
    if (JSON.stringify(a) !== JSON.stringify(b))
      throw new Error('reducer is impure: replay mismatch')
    return true
  }
  return { dispatch, replay, undo, verifyPure, log: () => [...log] }
}

// 评分点：
// 1. log 结构：action + 时间戳 + 状态快照
// 2. replay 从最近快照快进，跳过前缀 action
// 3. undo = pop + 重算（依赖 reducer 纯函数前提）
// 4. verifyPure 两次重放对比检测副作用 reducer
// 5. cache 失效时机正确（pop 后清缓存）
```
