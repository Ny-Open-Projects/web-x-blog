## ZSM-864：手写播放器队列引擎（指针锁 + 随机历史栈 + ended 竞态）
**需求**：863 落地——实现 `createPlayerQueue()`：1) `next({ reason })`：切到下一首——reason 'ended'（自动）与 'manual'（人工）**指针偏移锁**防双跳（ended 与 manual 并发只跳一次）；2) `shuffle()`：随机模式——乱序指针但保留历史栈（`history` 记录播放顺序，prev() 按历史回溯而非顺序）；3) `insertNext(trackId)`：插播（当前指针 +1 插入）；4) `removeCurrent()`：移除当前曲自动跳下一首；5) 附用例：ended+manual 并发只跳一次、随机回溯。

```javascript
function createPlayerQueue(tracks) {
  const order = [...tracks]
  let pointer = 0
  let mode = 'sequence'          // sequence | shuffle
  const history = []             // 播放历史栈（随机回溯用）
  let pointerLock = false        // 指针偏移锁：ended 与 manual 并发防双跳
  const listeners = new Set()

  function current() { return order[pointer] }

  function advance({ reason }) {
    if (pointerLock) return { skipped: false, reason: 'LOCKED' }   // 已在推进：幂等
    pointerLock = true
    try {
      history.push(order[pointer])
      if (mode === 'shuffle') {
        // 随机：从剩余未播放中随机（历史栈保证可回溯）
        const candidates = order.map((_, i) => i).filter((i) => i !== pointer)
        pointer = candidates[Math.floor(Math.random() * candidates.length)]
      } else {
        pointer = (pointer + 1) % order.length
      }
      listeners.forEach((fn) => fn({ current: current(), reason }))
      return { skipped: true, current: current() }
    } finally {
      setTimeout(() => { pointerLock = false }, 0)   // 锁释放：下一帧解锁（同 tick 幂等）
    }
  }

  function next({ reason = 'manual' } = {}) {
    return advance({ reason })
  }

  // ended 与 manual 并发：锁保证只跳一次
  function onEnded() { return advance({ reason: 'ended' }) }

  function prev() {
    const last = history.pop()
    if (last != null) {
      pointer = order.indexOf(last)
      listeners.forEach((fn) => fn({ current: current(), reason: 'prev' }))
      return current()
    }
    pointer = (pointer - 1 + order.length) % order.length    // 无历史：顺序回退
    return current()
  }

  function shuffle() {
    mode = mode === 'shuffle' ? 'sequence' : 'shuffle'
    if (mode === 'shuffle') history.length = 0     // 开随机：重置历史
    listeners.forEach((fn) => fn({ mode, current: current() }))
    return mode
  }

  function insertNext(trackId) {
    order.splice(pointer + 1, 0, trackId)          // 插播：当前 +1
    listeners.forEach((fn) => fn({ inserted: trackId, next: current() }))
  }

  function removeCurrent() {
    order.splice(pointer, 1)
    if (pointer >= order.length) pointer = 0       // 尾部移除：回到头
    listeners.forEach((fn) => fn({ removed: true, current: current() }))
    return current()
  }

  return {
    next, onEnded, prev, shuffle, insertNext, removeCurrent,
    current, get mode() { return mode },
    subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
  }
}

// 用例：ended + manual 并发只跳一次
const q = createPlayerQueue(['a', 'b', 'c', 'd'])
console.log(q.current())                          // 'a'

// 并发：歌曲自然结束 + 用户点下一首（同 tick）
const r1 = q.onEnded()                            // 锁住：跳到 b
const r2 = q.next({ reason: 'manual' })           // LOCKED：不跳
console.log(r1.skipped, r2.skipped)               // true false
console.log(q.current())                          // 'b'（只跳一次）

// 随机模式历史回溯
q.shuffle()                                       // 'shuffle'
const before = q.current()
q.next({ reason: 'manual' })
console.log(q.prev() === before)                  // true：历史栈回溯

// 插播与删除
q.insertNext('x')
q.next({ reason: 'manual' })
console.log(q.current())                          // 'x'
console.log(q.removeCurrent())                    // 下一首自动接续
```
**评分点**：
- 指针偏移锁（ended 与 manual 并发只跳一次——微任务级幂等）
- 随机历史栈（prev 按历史回溯而非顺序回退）
- insertNext 的指针 +1 插入语义
- removeCurrent 的边界（尾部移除回头部）
