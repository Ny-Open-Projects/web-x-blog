## ZSM-804：手写综合迷你状态库（store+selector+persist+devtools 一体）
**需求**：30 分钟白板题——实现 `createMiniZustand(initial, { persist, devtools })`：1) `use(selector, equalityFn)`：useSyncExternalStore 接线 + 结果比较闸门；2) `setState/getState/subscribe` 基础三件；3) persist：set 后节流写 localStorage（partialize 支持），启动时 hydrate + 版本 migrate；4) devtools：set 时 console.group 打印 diff（简化版时间线）；5) 附用例验证订阅与持久化。

```javascript
function createMiniZustand(initial, { persist, devtools } = {}) {
  let state = initial
  const subs = new Set()
  const listeners = new Set()   // devtools

  function getState() { return state }

  function setState(updater, actionName) {
    const prev = state
    state = typeof updater === 'function' ? updater(prev) : updater
    // 通知
    subs.forEach((fn) => fn(state, prev))
    // devtools：diff 时间线
    if (devtools) {
      const keys = Object.keys(prev).filter((k) => !Object.is(prev[k], state[k]))
      console.group(`%c setState:${actionName ?? 'anon'}`, 'color:#888')
      keys.forEach((k) => console.log(`${k}:`, prev[k], '→', state[k]))
      console.groupEnd()
    }
    // persist：节流写
    if (persist) schedulePersist()
  }

  let persistTimer = null
  function schedulePersist() {
    clearTimeout(persistTimer)
    persistTimer = setTimeout(() => {
      const pick = persist.partialize?.(state) ?? state
      try { localStorage.setItem(persist.name, JSON.stringify({ v: persist.version ?? 1, state: pick })) }
      catch (e) { console.warn('[persist] write failed', e) }
    }, 300)
  }

  // hydrate：版本迁移
  if (persist) {
    try {
      const raw = localStorage.getItem(persist.name)
      if (raw) {
        const { v, state: saved } = JSON.parse(raw)
        const migrated = v !== (persist.version ?? 1) && persist.migrate
          ? persist.migrate(saved, v) : saved
        state = { ...initial, ...migrated }   // merge：持久值覆盖初始值
      }
    } catch (e) { console.warn('[persist] hydrate failed, using initial', e) }
  }

  // hook：useSyncExternalStore + 闸门
  function use(selector = (s) => s, equalityFn = Object.is) {
    let last
    const getSnapshot = () => {
      const next = selector(state)
      if (last !== undefined && equalityFn(last.out, next)) return last.out   // 闸门
      last = { out: next }
      return next
    }
    return React.useSyncExternalStore(
      (fn) => { subs.add(fn); return () => subs.delete(fn) },
      getSnapshot,
      () => selector(initial),
    )
  }

  return { use, getState, setState, subscribe: (fn) => (subs.add(fn), () => subs.delete(fn)) }
}

// 用例
const store = createMiniZustand(
  { count: 0, user: { name: 'a' }, token: 'secret' },
  {
    persist: { name: 'app', version: 2, partialize: (s) => ({ count: s.count }), migrate: (s, v) => s },
    devtools: true,
  },
)
store.setState((s) => ({ count: s.count + 1 }), 'inc')
// devtools 输出: setState:inc  count: 0 → 1
store.setState((s) => ({ count: s.count + 1 }), 'inc')
// persist 节流后 localStorage: { v: 2, state: { count: 2 } }（token 被 partialize 排除）
```
**评分点**：
- useSyncExternalStore 接线 + 双闸门（equalityFn 比较返回旧引用）
- persist 节流 + partialize 白名单 + 版本 migrate + 失败容错
- devtools 的 diff 时间线（keys 过滤同值）
- SSR 侧 getServerSnapshot 用 initial（快照稳定）
