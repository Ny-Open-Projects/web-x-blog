## ZSM-862：手写地图视口查询协调器（bbox 去重 + zoom 粒度切换 + 帧避让）
**需求**：861 落地——实现 `createMapDataCoordinator()`：1) `onViewport(bbox, zoom)`：视口指纹去重（量化 bbox 后比对——相近视口不重复请求）+ debounce 300ms + 前序 abort；2) zoom 粒度切换：zoom < 12 请求聚合数据（cluster）、≥ 12 请求明细（detail）——粒度切换时清空旧粒度缓存；3) 帧避让：相机动画期间（`isAnimating` 标记）查询挂起，动画结束 idle 执行；4) 附用例：指纹去重与粒度切换清缓存。

```javascript
function createMapDataCoordinator({ api, debounceMs = 300 } = {}) {
  let lastFingerprint = null
  let debounce = null
  let controller = null
  let animating = false
  const pendingCalls = []
  const caches = { cluster: new Map(), detail: new Map() }
  let currentGranularity = null
  const listeners = new Set()

  function fingerprint(bbox, zoom) {
    // 量化 bbox（0.01° 网格）：相近视口归并
    const q = (v) => Math.round(v * 100) / 100
    return `${q(bbox.minLat)},${q(bbox.minLng)},${q(bbox.maxLat)},${q(bbox.maxLng)}@${Math.round(zoom)}`
  }

  function onViewport(bbox, zoom) {
    const fp = fingerprint(bbox, zoom)
    if (fp === lastFingerprint) return                // 指纹去重
    lastFingerprint = fp
    clearTimeout(debounce)
    debounce = setTimeout(() => {
      if (animating) { pendingCalls.push({ bbox, zoom }); return }   // 帧避让：动画期挂起
      execute(bbox, zoom)
    }, debounceMs)
  }

  function setAnimating(v) {
    animating = v
    if (!v && pendingCalls.length) {
      const { bbox, zoom } = pendingCalls.shift()
      execute(bbox, zoom)                             // 动画结束：idle 执行挂起查询
    }
  }

  function execute(bbox, zoom) {
    const granularity = zoom < 12 ? 'cluster' : 'detail'
    // 粒度切换：清空旧粒度缓存
    if (currentGranularity && currentGranularity !== granularity) {
      caches[currentGranularity].clear()
    }
    currentGranularity = granularity
    const cache = caches[granularity]
    const key = fingerprint(bbox, zoom)
    if (cache.has(key)) return cache.get(key)         // 同粒度缓存命中

    controller?.abort()                               // 前序 abort（647）
    controller = new AbortController()
    api.queryPOI(bbox, granularity, { signal: controller.signal })
      .then((data) => { cache.set(key, data); listeners.forEach((fn) => fn({ granularity, data })) })
      .catch((e) => { if (e.name !== 'AbortError') console.warn('[map-query]', e) })
  }

  return {
    onViewport, setAnimating,
    get cacheSize() { return caches.cluster.size + caches.detail.size },
    subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
  }
}

// 用例
const queries = []
const coord = createMapDataCoordinator({
  api: { queryPOI: async (bbox, g, { signal }) => { queries.push({ bbox, g }); await new Promise((r) => setTimeout(r, 5)); return [{ id: 1, g }] } },
})
// 相近视口：指纹归并
coord.onViewport({ minLat: 39.901, minLng: 116.401, maxLat: 39.902, maxLng: 116.402 }, 11)
coord.onViewport({ minLat: 39.9012, minLng: 116.4012, maxLat: 39.9021, maxLng: 116.4021 }, 11)   // 量化后同指纹
await new Promise((r) => setTimeout(r, 350))
console.log(queries.length)                     // 1：去重

// 粒度切换：zoom 13 → detail
coord.onViewport({ minLat: 39.90, minLng: 116.40, maxLat: 39.92, maxLng: 116.42 }, 13)
await new Promise((r) => setTimeout(r, 350))
console.log(queries.at(-1).g)                   // 'detail'
console.log(coord.cacheSize)                    // cluster 缓存被清：只剩 detail

// 帧避让
coord.setAnimating(true)
coord.onViewport({ minLat: 40, minLng: 117, maxLat: 40.02, maxLng: 117.02 }, 13)
await new Promise((r) => setTimeout(r, 350))
console.log(queries.length)                     // 仍 2：动画期挂起
coord.setAnimating(false)
await new Promise((r) => setTimeout(r, 10))
console.log(queries.length)                     // 3：动画结束执行
```
**评分点**：
- 视口指纹量化去重（0.01° 网格归并相近视口）
- 粒度切换清缓存（cluster/detail 数据模型不同）
- 帧避让（动画期挂起 + 结束后 idle 执行）
- 前序 abort（647 竞态防御复用）
