## ZSM-178：手写 store 切片工厂（slice 工厂 + 命名空间 + 依赖注入）
**需求**：实现 `createSlice({ name, initial, actions, selectors })` 与 `composeSlices(slices)`：1) 每个切片的状态挂在 `state[name]` 命名空间下；2) actions 内的 set 只写本切片命名空间；3) actions 可声明 `deps` 引用其他切片的 actions（跨切片调用，如「下单后清购物车」）；4) selectors 接收全局 state 并默认记忆化（参数不变返回上次结果）。禁止循环依赖。

```javascript
export function createSlice({ name, initial, actions, selectors = {}, deps = [] }) {
  return { name, initial, actions, selectors, deps }
}

export function composeSlices(slices) {
  const state = {}; const actions = {}; const selectors = {}
  slices.forEach(s => { state[s.name] = s.initial })

  const set = (name, patch) => {
    state[name] = { ...state[name],
      ...(typeof patch === 'function' ? patch(state[name]) : patch) }
  }
  slices.forEach(s => {
    const sliceDeps = Object.fromEntries(s.deps.map(d => [d, actions[d]]))
    if (s.deps.includes(s.name)) throw new Error('circular slice dep')
    for (const [key, fn] of Object.entries(s.actions))
      actions[`${s.name}/${key}`] = (...args) => fn(set.bind(null, s.name),
                                                   () => state[s.name], sliceDeps)(...args)
    for (const [key, fn] of Object.entries(s.selectors))
      selectors[`${s.name}/${key}`] = memoize((st) => fn(st))
  })
  return { getState: () => state, actions, selectors }
}

function memoize(fn) {
  let lastInput, lastOutput
  return (input) => lastInput === input ? lastOutput
    : (lastInput = input, lastOutput = fn(input))
}
```
