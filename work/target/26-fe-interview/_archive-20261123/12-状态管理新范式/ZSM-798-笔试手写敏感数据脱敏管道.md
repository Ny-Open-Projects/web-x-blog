## ZSM-798：手写敏感数据脱敏管道（清单驱动 + 递归 mask + 原型污染拦截）
**需求**：实现 `createSanitizer(rules)`：1) `sanitize(obj, { level })`：递归遍历——字段名命中敏感清单按级别处理：'log' 级 mask（`pii: 手机号 138****5678`）、'export' 级直接剔除 pii/financial 字段、'debug' 级透传；2) 数组/嵌套对象递归、Map/Set 支持；3) `safeAssign(target, input)`：原型污染键拦截；4) 循环引用防护（WeakSet）；5) 附用例：递归脱敏 + 导出级剔除 + 污染拦截。

```javascript
function maskValue(kind, v) {
  const s = String(v ?? '')
  if (kind === 'phone') return s.replace(/^(\d{3})\d+(\d{4})$/, '$1****$2')
  if (kind === 'token') return s.slice(0, 4) + '***'
  return s.replace(/./g, '*')
}

function createSanitizer(rules, { cycleGuard = new WeakSet() } = {}) {
  // rules: { store: { field: 'phone'|'pii'|'financial'|'internal' } }
  function sanitize(obj, { level = 'log', store } = {}) {
    if (obj == null || typeof obj !== 'object') return obj
    if (cycleGuard.has(obj)) return '[Circular]'
    cycleGuard.add(obj)

    const fieldRules = (store && rules[store]) ?? {}
    const out = Array.isArray(obj) ? [] : {}

    const entries = obj instanceof Map ? [...obj.entries()] : obj instanceof Set ? [...obj].map((v, i) => [i, v]) : Object.entries(obj)

    for (const [k, v] of entries) {
      if (['__proto__', 'constructor', 'prototype'].includes(String(k))) continue   // 污染拦截
      const rule = fieldRules[k]
      // export 级：pii/financial 直接剔除
      if (level === 'export' && (rule === 'pii' || rule === 'financial')) continue
      if (rule && level === 'log') {
        out[k] = rule === 'phone' || rule === 'token' ? maskValue(rule, v)
               : rule === 'financial' ? '¥***' : maskValue('pii', v)
        continue
      }
      if (v instanceof Map) { out[k] = new Map([...v].map(([mk, mv]) => [mk, sanitize(mv, { level, store })])); continue }
      if (v instanceof Set) { out[k] = new Set([...v].map((sv) => sanitize(sv, { level, store }))); continue }
      out[k] = (v && typeof v === 'object') ? sanitize(v, { level, store }) : v
    }
    cycleGuard.delete(obj)     // 允许重复 sanitize 非循环结构
    return out
  }

  function safeAssign(target, input) {
    const clean = {}
    for (const [k, v] of Object.entries(input)) {
      if (['__proto__', 'constructor', 'prototype'].includes(k)) continue
      clean[k] = v
    }
    target.setState?.(clean) ?? Object.assign(target, clean)
    return clean
  }

  return { sanitize, safeAssign }
}

// 用例
const rules = {
  user: { phone: 'phone', idCard: 'pii', balance: 'financial', token: 'token' },
}
const san = createSanitizer(rules)

const state = {
  name: '张三',
  phone: '13812345678',
  profile: { idCard: '110101199001011234', address: { city: '北京' } },
  orders: [{ id: 1, amount: 999 }],
  meta: new Map([['token', 'abcd1234efgh']]),
}

// log 级：mask
console.log(san.sanitize(state, { level: 'log', store: 'user' }).phone)         // '138****5678'
console.log(san.sanitize(state, { level: 'log', store: 'user' }).profile.idCard)   // '************'（全 mask）
console.log(san.sanitize(state, { level: 'log', store: 'user' }).meta.get('token'))   // 'abcd***'

// export 级：pii/financial 剔除
const exported = san.sanitize(state, { level: 'export', store: 'user' })
console.log('phone' in exported, 'balance' in exported)   // false false

// 污染拦截
const store = { state: {}, setState: (p) => Object.assign(store.state, p) }
san.safeAssign(store, { name: 'ok', __proto__: { evil: true }, constructor: 'x' })
console.log(store.state.evil)     // undefined：污染被拦
```
**评分点**：
- 级别语义（log mask / export 剔除 / debug 透传）
- 递归覆盖（嵌套对象/数组/Map/Set）+ 循环引用防护（WeakSet 进出）
- 原型污染键拦截（safeAssign 与 sanitize 双入口）
- 字段规则分级（phone 格式化 mask / token 前缀保留）
