## ZSM-866：手写画布嵌套树引擎（归一化 CRUD + 撤销 patch + 循环防护）
**需求**：865 落地——实现 `createCanvasTree()`：1) `move(nodeId, newParentId, index)`：移节点——**循环防护**（不能移到自己子孙下）；2) `add/remove`：增删同步 children 索引；3) 撤销：每个操作记录 patch（{nodeId, prevParent, prevIndex}），`undo()` 按栈回退；4) `renderTree()`：从 root 递归输出嵌套结构；5) 附用例：循环防护与撤销链。

```javascript
function createCanvasTree() {
  const nodes = new Map()       // nodeId -> { id, parentId, type, props }
  const children = new Map()    // parentId -> id[]（root 为 null）
  const undoStack = []
  const listeners = new Set()

  function indexOfParent(parentId, nodeId) {
    return (children.get(parentId) ?? []).indexOf(nodeId)
  }

  function add(node, parentId = null, index = null) {
    nodes.set(node.id, { ...node, parentId })
    const list = children.get(parentId) ?? children.set(parentId, []).get(parentId)
    index == null ? list.push(node.id) : list.splice(index, 0, node.id)
    undoStack.push({ op: 'add', nodeId: node.id, parentId, index: list.indexOf(node.id) })
    notify()
    return node.id
  }

  function remove(nodeId) {
    const node = nodes.get(nodeId)
    if (!node) return
    const parentId = node.parentId
    const index = indexOfParent(parentId, nodeId)
    // 递归删除子孙
    const descendants = collectDescendants(nodeId)
    descendants.forEach((d) => nodes.delete(d))
    nodes.delete(nodeId)
    children.set(parentId, (children.get(parentId) ?? []).filter((i) => i !== nodeId))
    children.delete(nodeId)
    undoStack.push({ op: 'remove', node, parentId, index, descendants })
    notify()
  }

  function collectDescendants(nodeId) {
    const out = [nodeId]
    for (const c of children.get(nodeId) ?? []) out.push(...collectDescendants(c))
    return out
  }

  function move(nodeId, newParentId, index = null) {
    const node = nodes.get(nodeId)
    if (!node || !nodes.has(newParentId) && newParentId !== null) throw new Error('invalid move')
    // 循环防护：newParent 不能是自己的子孙
    if (newParentId !== null && collectDescendants(nodeId).includes(newParentId)) {
      throw new Error(`循环引用：${newParentId} 是 ${nodeId} 的子孙`)
    }
    const prevParent = node.parentId
    const prevIndex = indexOfParent(prevParent, nodeId)
    children.set(prevParent, (children.get(prevParent) ?? []).filter((i) => i !== nodeId))
    const list = children.get(newParentId) ?? children.set(newParentId, []).get(newParentId)
    index == null ? list.push(nodeId) : list.splice(index, 0, nodeId)
    node.parentId = newParentId
    undoStack.push({ op: 'move', nodeId, prevParent, prevIndex })
    notify()
  }

  function undo() {
    const op = undoStack.pop()
    if (!op) return null
    switch (op.op) {
      case 'move': {
        const node = nodes.get(op.nodeId)
        children.set(node.parentId, (children.get(node.parentId) ?? []).filter((i) => i !== op.nodeId))
        const list = children.get(op.prevParent) ?? children.set(op.prevParent, []).get(op.prevParent)
        list.splice(Math.min(op.prevIndex, list.length), 0, op.nodeId)
        node.parentId = op.prevParent
        break
      }
      case 'remove': {
        nodes.set(op.node.id, op.node)
        op.descendants.forEach((d) => { if (d !== op.nodeId) nodes.set(d, { ...nodes.get(d), parentId: d }) })  // 简化恢复
        const list = children.get(op.parentId) ?? children.set(op.parentId, []).get(op.parentId)
        list.splice(Math.min(op.index, list.length), 0, op.nodeId)
        break
      }
      case 'add': {
        const node = nodes.get(op.nodeId)
        children.set(node.parentId, (children.get(node.parentId) ?? []).filter((i) => i !== op.nodeId))
        nodes.delete(op.nodeId)
        break
      }
    }
    notify()
    return op
  }

  function renderTree(parentId = null) {
    return (children.get(parentId) ?? []).map((id) => {
      const n = nodes.get(id)
      return { id: n.id, type: n.type, props: n.props, children: renderTree(id) }
    })
  }

  function notify() { listeners.forEach((fn) => fn(renderTree())) }

  return { add, remove, move, undo, renderTree, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)) }
}

// 用例
const tree = createCanvasTree()
tree.add({ id: 'root', type: 'page', props: {} })
tree.add({ id: 'container', type: 'div', props: {} }, 'root')
tree.add({ id: 'btn', type: 'button', props: { text: 'hi' } }, 'container')

// 循环防护：container 移到自己子孙（btn 下）→ 报错
try { tree.move('container', 'btn') } catch (e) { console.log(e.message) }
// 循环引用：btn 是 container 的子孙

tree.move('btn', 'root', 0)                    // 移到 root 顶部
console.log(tree.renderTree('root').map((n) => n.id))   // ['btn', 'container']
tree.undo()
console.log(tree.renderTree('root')[0].children.map((n) => n.id))   // ['btn']：撤销恢复
```
**评分点**：
- 归一化双结构（nodes Map + children 索引）原子 CRUD
- 循环防护（不能移到自己子孙——collectDescendants 检查）
- 撤销 patch 三操作（move/remove/add 各自逆操作）
- renderTree 递归输出（归一化→嵌套的投影）
