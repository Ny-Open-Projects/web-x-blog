## ZSM-789：手写选型六维评分矩阵（加权计算 + 敏感度分析）
**答**：实现 `createSelectionMatrix(dims, options)`：1) `score(option, ratings)`：加权总分；2) `rank()`：按总分排序输出；3) **敏感度分析**：`sensitivity(dim)`——某维度权重变化对排名的影响（对每个 option 计算该维 ±1 分时的排名变化），输出「排名稳定度」；4) `explain(option)`：与最优项的差距分解（每个维度的分差×权重）；5) 附用例：两库排名对权重变化的敏感度。

```javascript
function createSelectionMatrix(dims, options) {
  const W = new Map(dims.map((d) => [d.key, d.weight]))

  function score(option, ratings) {
    let total = 0, wsum = 0
    for (const [k, w] of W) {
      total += w * (ratings[option]?.[k] ?? 0)
      wsum += w
    }
    return +(total / wsum).toFixed(2)
  }

  function rank(allRatings) {
    return [...options]
      .map((o) => ({ option: o, score: score(o, allRatings) }))
      .sort((a, b) => b.score - a.score)
  }

  // 敏感度：dim 权重 ±1 时排名是否变化
  function sensitivity(dim, allRatings) {
    const baseRank = rank(allRatings).map((r) => r.option)
    const results = []
    for (const delta of [-1, 1]) {
      const w2 = new Map(W)
      w2.set(dim, Math.max(0.5, (W.get(dim) ?? 1) + delta))
      const alt = [...options]
        .map((o) => {
          let t = 0, ws = 0
          for (const [k, w] of w2) { t += w * (allRatings[o]?.[k] ?? 0); ws += w }
          return { option: o, score: +(t / ws).toFixed(2) }
        })
        .sort((a, b) => b.score - a.score)
        .map((r) => r.option)
      results.push({ delta, changed: JSON.stringify(alt) !== JSON.stringify(baseRank), ranking: alt })
    }
    return { dim, stable: !results.some((r) => r.changed), details: results }
  }

  // 差距分解：与最优项的分差来源
  function explain(option, allRatings) {
    const best = rank(allRatings)[0].option
    const gaps = []
    for (const [k, w] of W) {
      const diff = (allRatings[best]?.[k] ?? 0) - (allRatings[option]?.[k] ?? 0)
      if (diff !== 0) gaps.push({ dim: k, diff, weightedImpact: +(diff * w).toFixed(2) })
    }
    return { option, vsBest: best, gaps: gaps.sort((a, b) => Math.abs(b.weightedImpact) - Math.abs(a.weightedImpact)) }
  }

  return { score, rank, sensitivity, explain }
}

// 用例
const dims = [
  { key: '粒度', weight: 2 }, { key: 'TS', weight: 1.5 },
  { key: '生态', weight: 1.5 }, { key: '成本', weight: 2 },
]
const opts = ['zustand', 'jotai']
const ratings = {
  zustand: { 粒度: 4, TS: 5, 生态: 4, 成本: 5 },
  jotai: { 粒度: 5, TS: 4, 生态: 3, 成本: 4 },
}
const m = createSelectionMatrix(dims, opts)
console.log(m.rank(ratings))
// [{ option: 'zustand', score: 4.43 }, { option: 'jotai', score: 4.14 }]

const sens = m.sensitivity('粒度', ratings)
console.log(sens.stable, sens.details[0].changed)   // false true：粒度权重变化会翻转排名

console.log(m.explain('jotai', ratings))
// { vsBest: 'zustand', gaps: [{ dim: 'TS', diff: 1, weightedImpact: 1.5 }, ...] }
```
**评分点**：
- 加权评分归一（wsum 归一防总权重影响分数）
- 敏感度分析（权重 ±1 时排名稳定性——决策稳健性检验）
- 差距分解（与最优项逐维分差×权重，改进方向明确）
- 排序输出与 tie 处理（同分按序稳定）
