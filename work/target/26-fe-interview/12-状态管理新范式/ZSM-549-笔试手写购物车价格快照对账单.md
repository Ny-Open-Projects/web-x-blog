## ZSM-549：手写购物车价格快照对账单
**答**：实现 `createCheckout(items)`（items 为含 price 快照的购物车行）：1) `subtotal()`：按快照价小计；2) `applyCoupon(coupon)`：门槛校验（满 X 元）+ 折扣计算（满减/折扣率/上限），折扣**分摊到每个 item**（precise 分摊：余数给最后一项，保证 Σitem.discount = 总折扣）；3) `reconcile(latestPrices)`：对账——快照价 vs 最新价不一致的行打标 drifted 并重算差额提示；4) `total()` 返回应付总额；5) 金额用「分」整数运算防浮点误差。附用例验证分摊余数与对账。

```javascript
function createCheckout(items, { mode = 'global' } = {}) {
  let cart = items.map((it) => ({ ...it, discount: 0, drifted: false, latestPrice: null }))
  let coupon = null

  const subtotal = () => cart.reduce((a, it) => a + it.price * it.qty, 0)   // 单位：分

  function applyCoupon(c) {
    const sub = subtotal()
    if (sub < c.threshold) throw new Error(`未满足门槛 ${c.threshold}`)
    let totalDiscount = c.type === 'percent'
      ? Math.min(Math.floor(sub * c.rate / 100), c.cap ?? Infinity)
      : Math.min(c.amount, sub)
    // 精确分摊：按金额占比，余数给最后一项
    let allocated = 0
    cart = cart.map((it, i) => {
      if (i === cart.length - 1) {
        return { ...it, discount: totalDiscount - allocated }
      }
      const d = Math.floor(totalDiscount * (it.price * it.qty) / sub)
      allocated += d
      return { ...it, discount: d }
    })
    coupon = c
    return totalDiscount
  }

  function reconcile(latestPrices) {
    cart = cart.map((it) => {
      const latest = latestPrices[it.goodsId]
      if (latest != null && latest !== it.price) {
        return { ...it, drifted: true, latestPrice: latest }
      }
      return { ...it, drifted: false, latestPrice: null }
    })
    const drift = cart.filter((it) => it.drifted)
      .reduce((a, it) => a + (it.latestPrice * it.qty - it.price * it.qty), 0)
    return { driftedCount: cart.filter((it) => it.drifted).length, drift }  // 差额：正为需补
  }

  const total = () => subtotal() - cart.reduce((a, it) => a + it.discount, 0)

  return { subtotal, applyCoupon, reconcile, total, cart: () => cart }
}

// 用例（金额单位：分）
const ck = createCheckout([
  { goodsId: 'g1', qty: 1, price: 3333 },   // ¥33.33
  { goodsId: 'g2', qty: 2, price: 1000 },   // ¥20.00
])
console.log(ck.subtotal())                   // 5333
console.log(ck.applyCoupon({ type: 'flat', amount: 1000, threshold: 3000 }))  // 1000
console.log(ck.cart().map((it) => it.discount))  // [624, 376]：Σ=1000，余数给最后一项
console.log(ck.total())                      // 4333
console.log(ck.reconcile({ g1: 4000, g2: 1000 }))  // { driftedCount: 1, drift: 667 }
console.log(ck.cart()[0].drifted)            // true
```
**评分点**：
- 分单位整数运算（浮点金额误差）
- 折扣精确分摊：占比计算 + 余数归最后一项（Σitem=总额恒等式）
- 对账打标不静默改价（drifted 标记 + 差额提示）
- 门槛校验与折扣上限（cap）边界处理
