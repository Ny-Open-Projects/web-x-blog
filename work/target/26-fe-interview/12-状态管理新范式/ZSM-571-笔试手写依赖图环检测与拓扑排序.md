## ZSM-571：手写依赖图环检测与拓扑排序（依赖可视化后端）
**答**：实现 `createDepGraph()`：1) `addEdge(from, to)` 加依赖边（from 依赖 to，如派生 atom 依赖上游）；2) `detectCycle()`：DFS 找所有环，返回具体环路径（如 `a→b→c→a`）而非布尔；3) `topoOrder()`：合法 DAG 的拓扑序（无依赖者先算），有环时抛错并附环路径；4) `levels()`：分层视图——同一层节点可并行求值（依赖层级），可视化用；5) 附用例验证环路径输出与分层。

```javascript
function createDepGraph() {
  const adj = new Map()   // from -> Set<to>（from 依赖 to）

  function addEdge(from, to) {
    (adj.get(from) ??= new Set()).add(to)
  }

  // DFS 找环：返回所有环路径
  function detectCycle() {
    const cycles = []
    const color = new Map()   // 0 白 1 灰(在栈中) 2 黑
    const stack = []
    function dfs(node, path) {
      color.set(node, 1)
      stack.push(node)
      for (const next of adj.get(node) ?? []) {
        if (color.get(next) === 1) {
          const start = stack.indexOf(next)
          cycles.push([...stack.slice(start), next])   // 环路径
        } else if (!color.get(next)) {
          dfs(next, [...path, next])
        }
      }
      stack.pop()
      color.set(node, 2)
    }
    for (const n of adj.keys()) if (!color.get(n)) dfs(n, [n])
    return cycles
  }

  function topoOrder() {
    const cycles = detectCycle()
    if (cycles.length) throw new Error(`存在环: ${cycles[0].join('→')}`)
    const indeg = new Map()
    const nodes = new Set([...adj.keys(), ...[...adj.values()].flatMap((s) => [...s])])
    nodes.forEach((n) => indeg.set(n, 0))
    adj.forEach((tos, from) => tos.forEach((t) => indeg.set(t, (indeg.get(t) ?? 0) + 1)))
    const queue = [...nodes].filter((n) => indeg.get(n) === 0)
    const order = []
    while (queue.length) {
      const n = queue.shift()
      order.push(n)
      for (const t of adj.get(n) ?? []) {
        indeg.set(t, indeg.get(t) - 1)
        if (indeg.get(t) === 0) queue.push(t)
      }
    }
    return order
  }

  function levels() {
    const order = topoOrder()
    const levelOf = new Map()
    for (const n of order) {
      let lvl = 0
      for (const t of adj.get(n) ?? []) lvl = Math.max(lvl, (levelOf.get(t) ?? 0) + 1)
      levelOf.set(n, lvl)
    }
    const out = []
    levelOf.forEach((l, n) => (out[l] ??= []).push(n))
    return out
  }

  return { addEdge, detectCycle, topoOrder, levels }
}

// 用例
const g = createDepGraph()
g.addEdge('total', 'items')      // total 依赖 items
g.addEdge('discount', 'total')
g.addEdge('badge', 'discount')
console.log(g.topoOrder())       // ['badge','items','total','discount'] 或等价序
console.log(g.levels())          // [['items','badge'],['total'],['discount']]：层内可并行

const bad = createDepGraph()
bad.addEdge('a', 'b'); bad.addEdge('b', 'c'); bad.addEdge('c', 'a')
try { bad.topoOrder() } catch (e) { console.log(e.message) }  // 存在环: a→b→c→a
console.log(bad.detectCycle())   // [['a','b','c','a']]
```
**评分点**：
- 环检测输出具体路径而非布尔（三色标记法）
- 拓扑序用 Kahn 入度法（含节点集合补全：只出现在 to 的节点）
- levels 分层 = 最长路径深度（并行求值视图）
- 有环时 topo 抛错且附环路径（错误信息可行动）
