## ZSM-850：手写失误检测器（转写文本的 12 减分行为自动检出）
**答**：把 ZSM-849 落成工具——实现 `detectMistakes(transcript, { type })`：1) 规则化检出可自动识别的失误：营销话术、无数字、无澄清（设计题）、无故障层、术语堆砌（术语密度>阈值且无解释句）、超长；2) 每条失误输出 `{ mistake, evidence, fix, ref }`；3) `deduction()`：按失误类型加权扣分（硬编类最重）；4) 附用例：对一段含多失误的转写检出 5 条。

```javascript
const RULES = [
  { mistake: '营销话术', patterns: ['轻量', '很简单', '一目了然', '完美', '极致'], weight: 1.5, fix: '先问需求再给机制', ref: '802' },
  { mistake: '无数字叙事', test: (t) => !/\d+(\.\d+)?\s*(ms|次|条|%|s)/.test(t), weight: 2, fix: '数字锚点三要素', ref: '815' },
  { mistake: '无澄清(设计题)', onlyType: 'design', test: (t) => !/(半径|规模|冲突|规模多大|多少)/.test(t.slice(0, 200)), weight: 2, fix: '三问先行', ref: '802' },
  { mistake: '无故障层(设计题)', onlyType: 'design', test: (t) => !/(断线|故障|失败|回滚|超时)/.test(t), weight: 1.5, fix: '主动画故障层', ref: '823' },
  { mistake: '术语堆砌', test: (t) => {
      const jargon = (t.match(/CRDT|联邦|幂等|拓扑|DI|SSR|hydration/g) ?? []).length
      const explain = (t.match(/也就是|换句话说|比如说/g) ?? []).length
      return jargon >= 5 && explain === 0
    }, weight: 1.5, fix: '术语落地到具体行为', ref: '803' },
  { mistake: '超长', test: (t) => t.length > 1400, weight: 1, fix: '90s ≈ 350 字×4 控制节奏', ref: '808' },
  { mistake: '无取舍', onlyType: 'design', test: (t) => !/(不选|instead|为什么不|取舍|trade)/.test(t), weight: 1.5, fix: '每方案带「为什么不选 B」', ref: '801' },
]

function detectMistakes(transcript, { type = '原理' } = {}) {
  const findings = []
  for (const rule of RULES) {
    if (rule.onlyType && rule.onlyType !== type) continue
    let hit = false
    if (rule.patterns) {
      const found = rule.patterns.filter((p) => transcript.includes(p))
      hit = found.length > 0
      if (hit) findings.push({ mistake: rule.mistake, evidence: found, weight: rule.weight, fix: rule.fix, ref: rule.ref })
    } else if (rule.test?.(transcript)) {
      findings.push({ mistake: rule.mistake, evidence: ['pattern'], weight: rule.weight, fix: rule.fix, ref: rule.ref })
    }
  }
  const deduction = +findings.reduce((a, f) => a + f.weight, 0).toFixed(1)
  return { findings, deduction, grade: deduction >= 5 ? '高危' : deduction >= 2.5 ? '需改进' : '良好' }
}

// 用例
const bad = `Zustand 这个方案非常轻量完美，用 CRDT hydration DI 拓扑幂等这些技术，性能提升了很多。整体方案是 store 加 selector 加渲染。`
const r = detectMistakes(bad, { type: 'design' })
console.log(r.findings.map((f) => f.mistake))
// ['营销话术', '无数字叙事', '无澄清(设计题)', '无故障层(设计题)', '术语堆砌', '无取舍']
console.log(r.deduction)   // 10.0：5 条权重叠加
console.log(r.grade)       // '高危'
console.log(r.findings[0].fix)   // '先问需求再给机制'（ref: 802）

const good = `我先澄清三个问题：同步半径、规模、冲突语义。方案上选了 WS 而不是 SSE，因为双向需求。断线场景用 lastSeq 追平。实测 INP 从 280ms 降到 140ms。`
console.log(detectMistakes(good, { type: 'design' }).grade)   // '良好'
```
**评分点**：
- 规则化检出（patterns/test 双形态）+ onlyType 按题型过滤
- evidence 证据（命中的话术词）
- 加权扣分与三级 grade（高危/需改进/良好）
- 修正建议带 ref（回链到本库骨架卡）
