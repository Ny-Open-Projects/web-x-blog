## ZSM-858：手写多维表格公式字段引擎（依赖图 + 循环检测 + 视图投影）
**答**：857 落地——实现 `createGridEngine()`：1) schema：字段数组（含 formula 字段引用其他字段）；2) 公式依赖图：构建 + 循环检测（加字段/改公式时重跑）+ 拓扑序求值（写时算：行更新时重算受影响公式列）；3) 视图投影：`project(viewId)`——filter/sort/group 三件套零拷贝派生；4) 附用例：公式列联动重算、循环检测报错、看板分组投影。

```javascript
function createGridEngine(fields, { rows: initRows = [] } = {}) {
  const byKey = new Map(fields.map((f) => [f.key, f]))
  const rows = new Map(initRows.map((r) => [r.id, { ...r }]))
  const listeners = new Set()

  // 公式依赖图：构建 + 环检测
  function buildFormulaGraph() {
    const deps = new Map()
    for (const f of fields) {
      if (f.formula) {
        // 提取公式引用的字段（简版正则 {{field}}）
        deps.set(f.key, [...f.formula.matchAll(/\{\{(\w+)\}\}/g)].map((m) => m[1]))
      }
    }
    // 环检测（DFS）
    const color = new Map(), stack = []
    function dfs(k) {
      color.set(k, 1); stack.push(k)
      for (const d of deps.get(k) ?? []) {
        if (!byKey.has(d)) throw new Error(`公式 ${k} 引用未知字段 {{${d}}}`)
        if (color.get(d) === 1) throw new Error(`公式循环引用: ${[...stack.slice(stack.indexOf(d)), d].join('→')}`)
        if (!color.get(d)) dfs(d)
      }
      stack.pop(); color.set(k, 2)
    }
    for (const k of deps.keys()) if (!color.get(k)) dfs(k)
    // 拓扑序：公式列求值顺序
    const order = []
    const indeg = new Map([...deps.entries()].map(([k, ds]) => [k, ds.filter((d) => deps.has(d)).length]))
    const q = fields.filter((f) => f.formula && indeg.get(f.key) === 0).map((f) => f.key)
    while (q.length) {
      const k = q.shift(); order.push(k)
      for (const [fk, ds] of deps) {
        if (ds.includes(k) && --indeg.get(fk) === 0) q.push(fk)
      }
    }
    return { deps, formulaOrder: order }
  }

  let graph = buildFormulaGraph()

  // 写时算：行更新时重算该行公式列（拓扑序）
  function evalRow(row) {
    const ctx = new Proxy(row, {
      get(t, k) { return t[k] ?? byKey.get(k)?.formula ? t[k] : t[k] },
    })
    for (const key of graph.formulaOrder) {
      const f = byKey.get(key)
      // 公式内 {{field}} 替换为行值求值（demo 用 Function，生产用安全表达式引擎）
      const expr = f.formula.replace(/\{\{(\w+)\}\}/g, (_, k) => `row["${k}"]`)
      try { row[key] = Function('row', `"use strict"; return (${expr})`)(row) }
      catch (e) { row[key] = null }
    }
    return row
  }

  function updateRow(id, patch) {
    const row = rows.get(id)
    if (!row) throw new Error(`row ${id} 不存在`)
    Object.assign(row, patch)
    evalRow(row)                       // 写时算：公式列联动重算
    listeners.forEach((fn) => fn({ type: 'row-updated', id }))
  }

  function addRow(row) {
    rows.set(row.id, evalRow({ ...row }))
    listeners.forEach((fn) => fn({ type: 'row-added', id: row.id }))
  }

  // 视图投影：filter/sort/group 零拷贝
  function project({ filter, sortBy, groupBy } = {}) {
    let out = [...rows.values()]
    if (filter) out = out.filter(filter)
    if (sortBy) out = [...out].sort((a, b) => (a[sortBy] > b[sortBy] ? 1 : -1))
    if (groupBy) {
      const groups = new Map()
      for (const r of out) (groups.get(r[groupBy]) ?? groups.set(r[groupBy], []).get(r[groupBy])).push(r)
      return { grouped: [...groups.entries()] }   // 看板视图：按 select 字段分组
    }
    return { rows: out }
  }

  function notify(e) { listeners.forEach((fn) => fn(e)) }

  return { updateRow, addRow, project, getRow: (id) => rows.get(id), subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)) }
}

// 用例
const fields = [
  { key: 'qty', type: 'number' },
  { key: 'price', type: 'currency' },
  { key: 'total', type: 'formula', formula: '{{qty}} * {{price}}' },
  { key: 'taxed', type: 'formula', formula: '{{total}} * 1.13' },
  { key: 'status', type: 'select', options: ['todo', 'done'] },
]
const g = createGridEngine(fields)
g.addRow({ id: 'r1', qty: 2, price: 100, status: 'todo' })
console.log(g.getRow('r1').total)     // 200：公式列写时算
console.log(g.getRow('r1').taxed)     // 226：公式依赖公式（拓扑序）
g.updateRow('r1', { qty: 3 })
console.log(g.getRow('r1').total, g.getRow('r1').taxed)   // 300 339：联动重算

// 看板投影
console.log(g.project({ groupBy: 'status' }).grouped[0][0])   // 'todo'

// 循环检测
try {
  createGridEngine([
    { key: 'a', type: 'formula', formula: '{{b}} + 1' },
    { key: 'b', type: 'formula', formula: '{{a}} + 1' },
  ])
} catch (e) { console.log(e.message) }   // 公式循环引用: a→b→a
```
**评分点**：
- 公式依赖图（{{field}} 提取 + 未知字段报错 + 循环检测 DFS）
- 拓扑序求值（公式依赖公式的链式正确）
- 写时算（行更新触发该行公式列重算）
- 视图投影零拷贝（filter/sort/groupBy 看板分组）
