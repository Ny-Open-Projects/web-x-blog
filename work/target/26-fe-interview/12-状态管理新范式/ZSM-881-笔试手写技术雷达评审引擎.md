## ZSM-881：手写技术雷达评审引擎（四环归属 + 晋升证据链校验）
**答**：880 落地——实现 `createTechRadar()`：1) `place(tech, ring, evidence)`：技术入环（move 必须带 evidence——无证据拒绝挪环）；2) `move(tech, toRing, evidence)`：环间迁移校验（hold→adopt 必须经过 trial 阶段——不允许跳环）；3) `history(tech)`：迁移历史（证据链完整可审计）；4) `report()`：按环统计 + 无证据变动告警；5) 附用例：跳环拒绝与证据链审计。

```javascript
const RINGS = ['hold', 'assess', 'trial', 'adopt']
const RING_ORDER = { hold: 0, assess: 1, trial: 2, adopt: 3 }

function createTechRadar() {
  const entries = new Map()     // tech -> { ring, history: [{ from, to, evidence, at }] }
  const warnings = []

  function place(tech, ring, evidence) {
    if (!RINGS.includes(ring)) throw new Error(`unknown ring ${ring}`)
    entries.set(tech, { ring, history: [{ from: null, to: ring, evidence, at: Date.now() }] })
    return entries.get(tech)
  }

  function move(tech, toRing, evidence) {
    const e = entries.get(tech)
    if (!e) throw new Error(`unknown tech ${tech}`)
    if (!evidence || evidence.trim().length < 10) {
      warnings.push({ tech, attempt: toRing, reason: '无证据不挪环' })   // 纪律：无证据拒绝
      return { ok: false, reason: 'EVIDENCE_REQUIRED' }
    }
    const fromLevel = RING_ORDER[e.ring]
    const toLevel = RING_ORDER[toRing]
    // 跳环校验：hold→adopt 必须经过 trial（跨 2 环以上拒绝）
    if (Math.abs(toLevel - fromLevel) >= 2) {
      return { ok: false, reason: `${e.ring}→${toRing} 跳环：需经 ${toLevel > fromLevel ? 'trial' : 'assess'} 过渡` }
    }
    e.history.push({ from: e.ring, to: toRing, evidence, at: Date.now() })
    e.ring = toRing
    return { ok: true, ring: toRing }
  }

  function history(tech) { return entries.get(tech)?.history ?? [] }

  function report() {
    const byRing = {}
    for (const [tech, e] of entries) (byRing[e.ring] ??= []).push(tech)
    return { byRing, warnings, evidenceChainIntact: warnings.length === 0 }
  }

  return { place, move, history, report }
}

// 用例
const radar = createTechRadar()
radar.place('zustand', 'adopt', '三年生产验证，团队熟练度高')
radar.place('view-transition', 'assess', 'Chrome 支持，Safari 待观察')
radar.place('xstate', 'hold', '历史遗留：无流程复杂场景')

// 跳环拒绝：hold→adopt
const r1 = radar.move('xstate', 'adopt', '某个项目用了一次感觉不错')
console.log(r1)     // { ok: false, reason: 'hold→adopt 跳环：需经 trial 过渡' }

// 正常晋升：hold→assess（带充分证据）
const r2 = radar.move('xstate', 'trial', '订单向导模块试点 2 周，迁移 5 个流程，测试通过率 100%')
console.log(r2.ok)  // true

// 无证据拒绝
const r3 = radar.move('view-transition', 'trial', '想用')
console.log(r3)     // { ok: false, reason: 'EVIDENCE_REQUIRED' }
console.log(radar.report().warnings.length)   // 1：无证据变动告警
console.log(radar.history('xstate').length)   // 2：证据链完整
```
**评分点**：
- 无证据不挪环（evidence 长度校验 + warnings 告警）
- 跳环校验（跨 2 环拒绝，hold→adopt 必经 trial）
- 迁移历史证据链（可审计）
- report 双输出（环分布 + 链完整性）
