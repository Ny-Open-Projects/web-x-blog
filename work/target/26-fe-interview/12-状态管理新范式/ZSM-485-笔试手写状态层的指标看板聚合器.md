## ZSM-485：手写状态层指标看板聚合器
**答**：实现 `createMetricsAggregator()`：1) `record(name, value)`：写入时序桶（每分钟一个桶，保留 60 分钟）；2) `aggregate(name)`：返回 { count, sum, avg, p50, p95, max }（桶内样本排序计算）；3) `series(name)`：每分钟 avg 序列（画趋势图用）；4) 桶滚动：过期桶自动清除（内存恒定）；5) 多指标命名空间隔离；6) `exportJSON()` 导出全量（看板上报）。

```javascript
export function createMetricsAggregator({ bucketMs = 60_000, buckets = 60 } = {}) {
  const data = new Map()      // name → Map<bucketIdx, number[]>

  function bucketIdx() { return Math.floor(Date.now() / bucketMs) }

  function record(name, value) {
    if (!data.has(name)) data.set(name, new Map())
    const m = data.get(name)
    const idx = bucketIdx()
    if (!m.has(idx)) m.set(idx, [])
    m.get(idx).push(value)
    // 滚动清除过期桶
    for (const k of m.keys()) if (idx - k >= buckets) m.delete(k)
  }

  function percentile(sorted, p) {
    if (!sorted.length) return 0
    return sorted[Math.min(sorted.length - 1, Math.floor(sorted.length * p))]
  }

  function aggregate(name) {
    const all = [...(data.get(name)?.values() ?? [])].flat()
    const sorted = [...all].sort((a, b) => a - b)
    return {
      count: all.length,
      sum: all.reduce((a, b) => a + b, 0),
      avg: all.length ? all.reduce((a, b) => a + b, 0) / all.length : 0,
      p50: percentile(sorted, 0.5), p95: percentile(sorted, 0.95),
      max: sorted.length ? sorted[sorted.length - 1] : 0,
    }
  }

  function series(name) {
    const m = data.get(name)
    if (!m) return []
    const idx = bucketIdx()
    const out = []
    for (let i = buckets - 1; i >= 0; i--) {
      const samples = m.get(idx - i)
      out.push(samples ? samples.reduce((a, b) => a + b, 0) / samples.length : null)   // 空桶 null
    }
    return out
  }

  return { record, aggregate, series,
    exportJSON: () => Object.fromEntries([...data.keys()].map((n) => [n, aggregate(n)])) }
}
```
**评分点**：
- 时序桶结构（分钟桶 + 滚动清除，内存恒定——指标系统的基本盘）
- percentile 索引钳制正确；空桶 null（趋势图断点语义）
- 命名空间隔离 + exportJSON 全量导出
- 桶内 flat 合并跨桶统计的正确性
