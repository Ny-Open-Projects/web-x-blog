## ZSM-749：手写文件版本树（快进判定 + 冲突分叉 + 三选裁决）
**答**：实现 `createFileTree()`：1) `upload({ hash, baseVersion, device })`：三路判定——hash 与已有 committed 节点相同 → merged；baseVersion === head → fast-forward 提交为新 head；否则分叉（pending 节点 + onConflict 回调）；2) `resolve(pendingId, action)`：'keep-mine'（成为新 head）/'keep-theirs'（discarded）/ 'keep-both'（提交但不动 head？——两者并行，head 取其一并记录 both）；3) `history()`：版本树序列化（含分叉标记）；4) 附用例：两设备并发上传检出分叉、resolve 后 head 正确。

```javascript
function createFileTree({ onConflict } = {}) {
  const nodes = new Map()      // id -> { id, parentId, device, hash, status }
  let head = null
  const listeners = new Set()

  function upload({ hash, baseVersion, device }) {
    // 1) 内容相同：自动合并
    for (const n of nodes.values()) {
      if (n.hash === hash && n.status === 'committed') return { merged: true, versionId: n.id }
    }
    // 2) 快进：base 是当前 head（或首次上传）
    if (head === null || baseVersion === head) {
      const id = crypto.randomUUID()
      nodes.set(id, { id, parentId: baseVersion, device, hash, status: 'committed' })
      head = id
      return { committed: true, versionId: id }
    }
    // 3) 分叉：pending 等裁决
    const id = crypto.randomUUID()
    nodes.set(id, { id, parentId: baseVersion, device, hash, status: 'pending' })
    onConflict?.({ pendingId: id, head, base: baseVersion, device })
    return { conflicted: true, pendingId: id }
  }

  function resolve(pendingId, action) {
    const node = nodes.get(pendingId)
    if (!node || node.status !== 'pending') throw new Error('invalid pending node')
    switch (action) {
      case 'keep-mine':
        node.status = 'committed'
        head = pendingId                       // 我的成为新 head
        break
      case 'keep-theirs':
        node.status = 'discarded'              // 丢弃：head 不变
        break
      case 'keep-both':
        node.status = 'committed'
        // 并行保留：head 不变（theirs 仍是权威），记录 both 标记
        node.parallelTo = head
        break
      default:
        throw new Error(`unknown action ${action}`)
    }
    listeners.forEach((fn) => fn(history()))
    return node
  }

  function history() {
    const committed = [...nodes.values()].filter((n) => n.status === 'committed' || n.status === 'pending' || n.status === 'discarded')
    return committed.map((n) => ({
      id: n.id, parentId: n.parentId, device: n.device, status: n.status,
      isHead: n.id === head, forked: n.parentId !== null && n.parentId !== head && nodes.get(n.parentId)?.status === 'committed',
    }))
  }

  return { upload, resolve, history, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)), get head() { return head } }
}

// 用例
let conflicts = []
const tree = createFileTree({ onConflict: (c) => conflicts.push(c) })

const v1 = tree.upload({ hash: 'h1', baseVersion: null, device: 'A' })    // 首传：提交
const v2 = tree.upload({ hash: 'h2', baseVersion: v1.versionId, device: 'A' })   // 快进
console.log(tree.head === v2.versionId)                                   // true

const v3 = tree.upload({ hash: 'h3', baseVersion: v1.versionId, device: 'B' })   // 基于旧版：分叉
console.log(v3.conflicted, conflicts.length)                              // true 1
console.log(tree.head)                                                    // v2：head 未动

tree.resolve(v3.pendingId, 'keep-mine')
console.log(tree.head === v3.pendingId)                                   // true：B 版成为新 head
console.log(tree.history().filter((n) => n.forked).length)                // 1：分叉标记
tree.resolve(v3.pendingId, 'keep-both')                                   // 已非 pending：抛错
try { tree.resolve(v3.pendingId, 'keep-both') } catch (e) { console.log(e.message) }
```
**评分点**：
- 三路判定（hash 去重 / fast-forward / 分叉 pending）
- resolve 三选语义（keep-mine 动 head / keep-theirs 丢弃 / keep-both 并行标记）
- history 含 forked 标记与 isHead（版本树可视化数据）
- 非 pending 节点重复 resolve 抛错（状态机防护）
