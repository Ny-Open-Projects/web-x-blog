## ZSM-444：手写时间旅行跳转器
**答**：实现 `createTimeTravel(store, apply)`：1) `record(patch)` 持续记录动作流（环形 1000 条）；2) 每 100 条自动存检查点快照；3) `seek(actionIndex)`：找到目标动作前最近检查点 → 恢复快照 → 重放其后的动作至目标（重放成本 O(分段) 而非 O(全量)）；4) `seekRatio(ratio)` 按比例跳转；5) seek 后进入「浏览态」：新动作到达时提示分叉并可选「继续录制（从当前点）」；6) 跳转期间暂停实时应用。

```javascript
export function createTimeTravel(store, apply, { checkpointEvery = 100, ring = 1000 } = {}) {
  const actions = []           // { patch }
  const checkpoints = []       // { idx, state }
  let browseIdx = -1           // -1 = 实时态
  let paused = false

  function record(patch) {
    if (paused) return { dropped: true }
    if (browseIdx >= 0) return { diverged: true, hint: 'seek 后新动作分叉，continue() 可从此继续' }
    actions.push({ patch })
    if (actions.length > ring) actions.shift()
    if (actions.length % checkpointEvery === 0)
      checkpoints.push({ idx: actions.length - 1, state: structuredClone(store.getState()) })
    return {}
  }

  function seek(idx) {
    paused = true
    browseIdx = Math.min(idx, actions.length - 1)
    // 最近检查点
    let cp = null
    for (const c of checkpoints) if (c.idx <= browseIdx) cp = c
    if (cp) store.setState(cp.state)
    for (let i = (cp ? cp.idx + 1 : 0); i <= browseIdx; i++) apply(store, actions[i].patch)   // 分段重放
    return { replayed: browseIdx - (cp?.idx ?? -1) }
  }

  function seekRatio(r) { return seek(Math.floor((actions.length - 1) * Math.min(1, Math.max(0, r)))) }
  function live() { paused = false; browseIdx = -1 }           // 回到实时
  function continueHere() { checkpoints.push({ idx: browseIdx, state: structuredClone(store.getState()) }); live() }

  return { record, seek, seekRatio, live, continueHere,
    get browsing() { return browseIdx >= 0 } }
}
```
**评分点**：
- 检查点 + 分段重放（seek 成本 O(检查点间隔) 而非 O(全量)）——核心考点
- 浏览态语义完整（暂停实时 / 分叉提示 / continueHere 以当前点为新基线）
- 快照 structuredClone 防变异污染；环形缓冲防内存膨胀
- seekRatio 边界钳制 [0,1]
