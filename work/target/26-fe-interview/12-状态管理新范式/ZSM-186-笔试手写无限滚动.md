## ZSM-186：手写 useInfiniteScroll（IntersectionObserver 哨兵 + 防重复加载）
**需求**：实现 `useInfiniteScroll({ fetcher, cursor })`：1) 哨兵元素进入视口时触发 fetcher(cursor) 加载下一页并追加；2) 加载中再次进入视口不重复请求；3) fetcher 返回 null/undefined 表示没有下一页，此后不再观察；4) 列表更新后哨兵重新挂载要继续工作；5) 支持 rootMargin 提前量。返回 { items, sentinelRef, loading, hasMore }。

```javascript
export function useInfiniteScroll({ fetcher, rootMargin = '200px' }) {
  const [items, setItems] = useState([])
  const [loading, setLoading] = useState(false)
  const [hasMore, setHasMore] = useState(true)
  const sentinelRef = useRef(null)
  const cursorRef = useRef(undefined)
  const loadingRef = useRef(false)          // 同步守卫：Observer 回调里读

  const loadMore = useCallback(async () => {
    if (loadingRef.current) return          // 进行中：拒绝重入
    loadingRef.current = true
    setLoading(true)
    try {
      const { next, batch } = await fetcher(cursorRef.current)
      if (batch.length) setItems(prev => [...prev, ...batch])
      cursorRef.current = next
      if (next == null) setHasMore(false)   // 终止：解除观察
    } finally {
      loadingRef.current = false
      setLoading(false)
    }
  }, [fetcher])

  useEffect(() => {
    if (!hasMore || !sentinelRef.current) return
    const ob = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting) loadMore()  // 哨兵可见即加载
    }, { rootMargin })                      // 提前 200px 预取
    ob.observe(sentinelRef.current)
    return () => ob.disconnect()            // 哨兵随列表重挂自动重建
  }, [hasMore, loadMore, rootMargin])

  return { items, sentinelRef, loading, hasMore }
}

// 评分点：
// 1. IntersectionObserver 哨兵而非 scroll 事件监听
// 2. loadingRef 同步重入守卫（state 异步更新挡不住并发回调）
// 3. cursor 返回 null 即解除观察、hasMore=false
// 4. rootMargin 提前量实现预取
// 5. 清理 disconnect 防泄漏；列表追加用函数式 setState
```
