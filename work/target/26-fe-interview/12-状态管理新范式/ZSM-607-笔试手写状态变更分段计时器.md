## ZSM-607：手写状态变更分段计时器（状态层/渲染层归因）
**答**：实现 `createLayerProfiler()`：1) `begin(action)`：开始记录一次状态变更；2) `mark(phase, ms)`：登记分段耗时（phase: 'selector' | 'notify' | 'serialize' | 'render'）；3) `end()`：结算该次变更，按「总耗时是否超预算 16ms」标记 slow；4) `aggregate(topN)`：聚合多期记录——按 action 分组输出 {count, slowRate, phaseRank}（每组的分段耗时占比排序），找主导分段；5) 附用例：三次变更数据验证主导分段识别。

```javascript
function createLayerProfiler({ budget = 16 } = {}) {
  const records = []
  let current = null

  function begin(action) {
    current = { action, phases: {}, t0: performance.now() }
  }

  function mark(phase, ms) {
    if (!current) throw new Error('begin() first')
    current.phases[phase] = (current.phases[phase] ?? 0) + ms
  }

  function end() {
    const total = performance.now() - current.t0
    // 相对时长：总耗时减去已登记分段 → 未覆盖部分归入 other
    const known = Object.values(current.phases).reduce((a, b) => a + b, 0)
    if (total > known + 0.5) current.phases.other = +(total - known).toFixed(2)
    current.total = +total.toFixed(2)
    current.slow = current.total > budget
    records.push(current)
    current = null
    return current
  }

  function aggregate(topN = 5) {
    const byAction = new Map()
    for (const r of records) {
      const a = byAction.get(r.action) ?? { count: 0, slow: 0, phases: {} }
      a.count++
      if (r.slow) a.slow++
      for (const [p, ms] of Object.entries(r.phases)) a.phases[p] = (a.phases[p] ?? 0) + ms
      byAction.set(r.action, a)
    }
    return [...byAction.entries()]
      .map(([action, a]) => {
        const total = Object.values(a.phases).reduce((x, y) => x + y, 0)
        // 主导分段：占比排序
        const phaseRank = Object.entries(a.phases)
          .map(([p, ms]) => [p, +(ms / total * 100).toFixed(1)])
          .sort((x, y) => y[1] - x[1])
        return {
          action, count: a.count,
          slowRate: +(a.slow / a.count * 100).toFixed(1),
          totalMs: +total.toFixed(1),
          phaseRank,
        }
      })
      .sort((x, y) => y.totalMs - x.totalMs)
      .slice(0, topN)
  }

  return { begin, mark, end, aggregate }
}

// 用例
const p = createLayerProfiler()
p.begin('cart/addItem')
p.mark('selector', 30)     // O(n) 过滤 selector
p.mark('notify', 2)
p.mark('render', 4)
p.end()

p.begin('cart/addItem')
p.mark('selector', 28)
p.mark('render', 3)
p.end()

p.begin('list/refresh')
p.mark('notify', 3)
p.mark('render', 40)      // 渲染层主导
p.end()

console.log(p.aggregate())
// [
//   { action: 'cart/addItem', count: 2, slowRate: 100, totalMs: 67,
//     phaseRank: [['selector', 86.6], ['render', 10.4], ['notify', 3.0]] },
//   { action: 'list/refresh', count: 1, slowRate: 100, totalMs: 43,
//     phaseRank: [['render', 93], ['notify', 7]] }
// ]
// → cart 慢在 selector，list 慢在渲染：修法完全不同
```
**评分点**：
- 分段登记聚合（selector/notify/render 可扩展 phase）
- 慢变更判定（16ms 帧预算）与 slowRate 统计
- 主导分段占比排序（归因输出可行动）
- 未覆盖时长归 other（不漏账）
