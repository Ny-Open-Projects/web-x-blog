## ZSM-465：手写失效最小化
**答**：实现 `createMinimalInvalidator(qc, deps)`：1) deps：`{ mutation: affectedKeysFn }` 映射——mutation 只失效「结果确实会变的查询」；2) 条件失效：affectedKeysFn(vars, prevData) 返回 `{ keys, skip? }`——如改量未改变总价则跳过 total 查询；3) 失效前快照对比优化：对幂等查询可先 refetch 后对比，无变化不传播失效（回填旧值）；4) 批量 mutation 合并失效；5) report 统计「避免的失效数」（节省的重拉成本）。

```javascript
export function createMinimalInvalidator(qc, deps) {
  let avoided = 0, performed = 0

  async function run(mutation, vars, prevData) {
    const spec = deps[mutation]?.(vars, prevData)
    if (!spec) return
    if (spec.skip) { avoided += spec.keys.length; return }      // 条件跳过
    performed += spec.keys.length
    await Promise.all(spec.keys.map((k) => qc.invalidateQueries({ queryKey: k })))
  }

  // 优化：幂等查询 refetch-then-compare
  async function runWithCompare(mutation, vars, prevData, compareFn) {
    const spec = deps[mutation]?.(vars, prevData)
    if (!spec) return
    for (const key of spec.keys) {
      const before = qc.getQueryData(key)
      await qc.refetchQueries({ queryKey: key })
      const after = qc.getQueryData(key)
      if (compareFn?.(before, after) === false) {
        qc.setQueryData(key, before)                             // 回填旧值
        avoided++; performed--
      }
    }
  }

  return { run, runWithCompare, report: () => ({ performed, avoided,
    ratio: performed + avoided ? avoided / (performed + avoided) : 0 }) }
}
```
**评分点**：
- 条件失效（affectedKeysFn 拿 vars/prevData 判断，改总价无关则 skip）——最小失效核心
- refetch-compare-rollback 优化（幂等查询防误伤，回填旧值）及其适用边界（低频写）
- avoided/performed 可观测（失效最小化的收益量化）
- 批量合并（沿用窗口合并思想，代码可提及）
