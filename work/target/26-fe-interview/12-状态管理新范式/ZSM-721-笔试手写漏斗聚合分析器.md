## ZSM-721：手写漏斗聚合分析器（会话聚合 + 转化率 + 时长分位）
**答**：实现 `createFunnelAnalyzer(steps)`：1) `add(event)`：事件 `{ traceId, name, ts, props? }`；2) `analyze()`：按 traceId 聚合会话，**按顺序匹配漏斗步骤**（必须 step[i] 出现在 step[i-1] 之后才算到达），输出每步到达数/转化率/漏损（上一步到达但未到本步的会话数）；3) 时长指标：相邻步骤间的停留时长 p50/p90；4) `slice(fn)`：按事件属性切片（如 expBucket='B' 的漏斗）；5) 附用例：乱序事件与重复步骤的正确处理。

```javascript
function createFunnelAnalyzer(steps) {
  const events = []     // { traceId, name, ts, props }

  function add(event) { events.push(event) }

  function analyze(filterFn = () => true) {
    // 会话聚合
    const sessions = new Map()
    for (const e of events) {
      if (!filterFn(e)) continue
      ;(sessions.get(e.traceId) ?? sessions.set(e.traceId, []).get(e.traceId)).push(e)
    }
    sessions.forEach((list) => list.sort((a, b) => a.ts - b.ts))   // 时序排序

    const rows = []
    let prevReached = sessions.size
    const prevReachedIds = new Set(sessions.keys())
    let prevTs = new Map([...sessions.keys()].map((id) => [id, sessions.get(id)[0]?.ts ?? 0]))

    for (const step of steps) {
      const reachedIds = new Set()
      const dwell = []
      for (const [id, list] of sessions) {
        if (!prevReachedIds.has(id)) continue        // 上一步未到达：跳过
        const prevT = prevTs.get(id)
        for (const e of list) {
          if (e.name === step && e.ts >= prevT) {    // 顺序匹配：晚于上一步
            reachedIds.add(id)
            dwell.push(e.ts - prevT)
            break
          }
        }
      }
      const sorted = [...dwell].sort((a, b) => a - b)
      const q = (p) => sorted[Math.min(sorted.length - 1, Math.floor(sorted.length * p))]
      rows.push({
        step,
        reached: reachedIds.size,
        rate: prevReached ? +(reachedIds.size / prevReached * 100).toFixed(1) : 0,
        leaked: prevReachedIds.size - reachedIds.size,
        dwellP50: sorted.length ? q(0.5) : null,
        dwellP90: sorted.length ? q(0.9) : null,
      })
      prevReached = reachedIds.size
      prevReachedIds = reachedIds
      prevTs = new Map([...sessions].filter(([id]) => reachedIds.has(id))
        .map(([id, list]) => [id, list.find((e) => e.name === step).ts]))
    }
    return rows
  }

  return { add, analyze }
}

// 用例
const f = createFunnelAnalyzer(['view', 'addCart', 'submit', 'paid'])
const T = 0
;[
  { traceId: 's1', name: 'view', ts: 0 },
  { traceId: 's1', name: 'addCart', ts: 30 },
  { traceId: 's1', name: 'submit', ts: 100 },
  { traceId: 's1', name: 'paid', ts: 200 },
  { traceId: 's2', name: 'view', ts: 0 },
  { traceId: 's2', name: 'addCart', ts: 50 },        // 未 submit
  { traceId: 's3', name: 'paid', ts: 5 },            // 乱序：paid 先于 view 到达但 ts 更早
  { traceId: 's3', name: 'view', ts: 10 },
].forEach((e) => f.add(e))

console.log(f.analyze())
// [
//   { step: 'view', reached: 3, rate: 100, leaked: 0, ... },
//   { step: 'addCart', reached: 2, rate: 66.7, leaked: 1, dwellP50: 40 },
//   { step: 'submit', reached: 1, rate: 50, leaked: 1, ... },
//   { step: 'paid', reached: 1, rate: 100, leaked: 0, dwellP50: 100 },
// ]
// s3 的 paid 在 view 前：不满足顺序 → 不计入 paid 到达
console.log(f.analyze((e) => e.traceId === 's1').length)   // 切片分析
```
**评分点**：
- 顺序匹配语义（step[i] 的 ts ≥ step[i-1]，乱序事件不计入）
- 会话聚合与漏损计算（prevReachedIds 传递）
- 停留时长分位（p50/p90，dwell 从上一步 ts 起算）
- slice 切片分析（A/B 桶对比的入口）
