## ZSM-436：手写状态可视化调试器（树形 diff 面板）
**答**：实现 `createStateInspector(store, mountEl)`：1) 渲染当前 state 为可折叠树（对象/数组/原始值分层）；2) 变更时只重渲染 diff 的节点（path 级更新，不是整树重画）；3) 点击节点可「钉住」该 path，值变化时高亮闪一下；4) 变更历史侧栏（最近 20 条，含 action 与 diff 摘要）；5) 大数组折叠显示条数（渲染前 5 条 + …）；6) destroy 完整清理（退订 + DOM 清空 + 事件解绑）。

```javascript
export function createStateInspector(store, mountEl) {
  let lastState = store.getState()
  const pinned = new Set()
  let hist = []
  const unsub = store.subscribe((s, prev, _patch) => {
    const paths = diffPaths(prev, s)                       // 变更路径列表
    paths.forEach((p) => {
      const node = mountEl.querySelector(`[data-path="${cssEscape(p)}"]`)
      if (node) {
        node.textContent = fmt(getPath(s, p))
        if (pinned.has(p)) node.classList.add('flash')     // 钉住高亮
        setTimeout(() => node.classList.remove('flash'), 600)
      }
    })
    hist = [{ ts: Date.now(), paths }, ...hist].slice(0, 20)
    renderHistory()
    lastState = s
  })

  function renderTree(state, path = '', depth = 0) {
    if (Array.isArray(state) && state.length > 5)
      return `<div data-path="${path}">Array(${state.length}) [${state.slice(0, 5).map(fmt).join(', ')}…]</div>`
    if (state && typeof state === 'object')
      return Object.entries(state).map(([k, v]) =>
        `<div><span class="key">${k}</span>${renderTree(v, path ? `${path}.${k}` : k, depth + 1)}</div>`).join('')
    return `<span data-path="${path}" class="val">${fmt(state)}</span>`
  }

  function destroy() {
    unsub(); mountEl.innerHTML = ''                        // 完整清理
    mountEl.removeEventListener('click', onClick)
  }
  const onClick = (e) => { const p = e.target.closest('[data-path]')?.dataset.path
    if (p) pinned.has(p) ? pinned.delete(p) : pinned.add(p) }
  mountEl.addEventListener('click', onClick)
  mountEl.innerHTML = renderTree(lastState)
  return { destroy, pin: (p) => pinned.add(p) }
}

const fmt = (v) => typeof v === 'string' ? `"${v}"` : String(v)
const cssEscape = (s) => CSS?.escape?.(s) ?? s.replace(/["\\]/g, '\\$&')
```
**评分点**：
- path 级增量更新（querySelector 定位 diff 节点，不整树重画）
- 钉住 + flash 高亮（调试「值何时变」的直观手段）
- 大数组折叠（防 DOM 爆炸）；历史侧栏带 diff 摘要
- destroy 完整清理（退订 + DOM + 事件解绑三件套缺一即泄漏）
