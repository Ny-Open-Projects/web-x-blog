## ZSM-693：手写批量操作结果集归并器（并发限制 + 重试 + 汇总）
**答**：实现 `runBatch(items, task, opts)`：1) 并发限制 N（worker 池模型）；2) 逐条收集结果 `[{ item, status, value?|error? }]`；3) 自动重试：可重试错误（retryable 判定函数）退避重试至多 M 次，重试用同 attempt 计数；4) `onProgress(done, total)` 回调（合并去抖）；5) 中断：`abort()` 停止派发新任务，返回已完成结果集；6) 附用例：3 并发 + 失败重试 + 中断验证。

```javascript
function runBatch(items, task, { concurrency = 5, retryable = () => false, maxRetry = 2, onProgress } = {}) {
  const results = new Array(items.length)
  let done = 0
  let aborted = false
  let nextIndex = 0
  const controllers = new Set()

  const promise = new Promise((resolve) => {
    async function worker() {
      while (!aborted) {
        const i = nextIndex++
        if (i >= items.length) return
        let attempt = 0
        let result
        while (true) {
          const ac = new AbortController()
          controllers.add(ac)
          try {
            const value = await task(items[i], ac.signal, attempt)
            result = { item: items[i], status: 'ok', value }
            break
          } catch (e) {
            if (aborted || ac.signal.aborted) { result = { item: items[i], status: 'aborted' }; break }
            if (retryable(e) && attempt < maxRetry) {
              attempt++
              await new Promise((r) => setTimeout(r, 2 ** attempt * 100))   // 退避
              continue
            }
            result = { item: items[i], status: 'failed', error: e.message, attempts: attempt + 1 }
            break
          } finally {
            controllers.delete(ac)
          }
        }
        results[i] = result
        done++
        onProgress?.(done, items.length)
        if (done >= items.length || (aborted && ![...pending])) resolveDone()
      }
      resolveDone()
    }
    let resolved = false
    function resolveDone() {
      if (resolved) return
      resolved = true
      resolve({
        results,
        summary: {
          ok: results.filter((r) => r?.status === 'ok').length,
          failed: results.filter((r) => r?.status === 'failed').length,
          aborted: results.filter((r) => r?.status === 'aborted' || r === undefined).length,
        },
      })
    }
    const pending = []
    Array.from({ length: Math.min(concurrency, items.length) }, worker)
  })

  return {
    promise,
    abort() {
      aborted = true
      controllers.forEach((c) => c.abort())
    },
  }
}

// 用例
let calls = 0
const batch = runBatch(
  ['a', 'b', 'c', 'd'],
  async (item, signal, attempt) => {
    if (item === 'b' && attempt === 0) { const e = new Error('flaky'); e.status = 500; throw e }   // 首次失败可重试
    await new Promise((r) => setTimeout(r, 10))
    return `${item}-done`
  },
  {
    concurrency: 2,
    retryable: (e) => e.status === 500,
    maxRetry: 2,
    onProgress: (done, total) => console.log(`${done}/${total}`),
  },
)
const { summary, results } = await batch.promise
console.log(summary)        // { ok: 4, failed: 0, aborted: 0 }
console.log(results.map((r) => `${r.item}:${r.status}`))   // ['a:ok','b:ok','c:ok','d:ok']（b 重试后成功）
```
**评分点**：
- worker 池并发模型（nextIndex 原子分发）
- 重试退避 + attempt 计数（结果带 attempts 可观测）
- abort 的信号传播（在途任务 signal abort，未派发停止）
- 进度回调 + 汇总三分桶（ok/failed/aborted）
