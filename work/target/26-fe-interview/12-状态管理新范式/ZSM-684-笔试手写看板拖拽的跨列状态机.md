## ZSM-684：手写看板拖拽的跨列状态引擎（归属 + 序 + 原子移动）
**答**：实现 `createBoard()`：1) 状态 `{ columns: Map<colId, { id, title, order: string[] }>, cardCol: Map<cardId, colId> }`——归属与序分离；2) `move(cardId, toColId, toIndex)`：跨列/同列移动为原子操作（先从源列 splice、再插入目标列），失败（卡片不存在/列不存在）抛错不产生部分变更；3) `rename/addCard/removeCard` 同步维护双结构；4) `snapshot()` 导出列序视图；5) 附用例：同列重排、跨列移动、错误输入不产生部分变更。

```javascript
function createBoard(cols) {
  const columns = new Map()
  const cardCol = new Map()
  for (const c of cols) {
    columns.set(c.id, { ...c })
    c.order.forEach((cardId) => cardCol.set(cardId, c.id))
  }

  function assert(cond, msg) { if (!cond) throw new Error(msg) }

  function move(cardId, toColId, toIndex) {
    const fromColId = cardCol.get(cardId)
    assert(fromColId, `card ${cardId} 不存在`)
    assert(columns.has(toColId), `column ${toColId} 不存在`)
    const from = columns.get(fromColId)
    const to = columns.get(toColId)
    assert(from.order.includes(cardId), '源列不含该卡片（结构不一致）')

    // 原子操作：先取下后插入，两步都成功才提交
    const next = new Map(columns)
    const nextCardCol = new Map(cardCol)
    const fromList = [...from.order]
    fromList.splice(fromList.indexOf(cardId), 1)
    const toList = fromColId === toColId ? fromList : [...to.order]
    if (toIndex == null || toIndex > toList.length) toIndex = toList.length
    toList.splice(Math.max(0, toIndex), 0, cardId)

    next.set(fromColId, { ...from, order: fromList })
    next.set(toColId, { ...to, order: toList })
    if (fromColId !== toColId) {
      nextCardCol.delete(cardId)
      nextCardCol.set(cardId, toColId)
    }
    columns.clear(); columns.set; // 替换
    for (const [k, v] of next) columns.set(k, v)
    for (const [k, v] of nextCardCol) cardCol.set(k, v)
    return { fromColId, toColId }
  }

  function addCard(colId, cardId) {
    assert(columns.has(colId), `column ${colId} 不存在`)
    assert(!cardCol.has(cardId), `card ${cardId} 已存在`)
    columns.get(colId).order.push(cardId)
    cardCol.set(cardId, colId)
  }

  function removeCard(cardId) {
    const colId = cardCol.get(cardId)
    assert(colId, `card ${cardId} 不存在`)
    const col = columns.get(colId)
    columns.set(colId, { ...col, order: col.order.filter((id) => id !== cardId) })
    cardCol.delete(cardId)
  }

  function snapshot() {
    return [...columns.values()].map((c) => ({ id: c.id, title: c.title, cards: c.order }))
  }

  return { move, addCard, removeCard, snapshot }
}

// 用例
const board = createBoard([
  { id: 'todo', title: '待办', order: ['c1', 'c2'] },
  { id: 'doing', title: '进行中', order: ['c3'] },
])
console.log(board.move('c1', 'doing', 0))         // 跨列移动
console.log(board.snapshot())
// [{ id: 'todo', cards: ['c2'] }, { id: 'doing', cards: ['c1', 'c3'] }]
console.log(board.move('c2', 'todo', 1))          // 同列重排（原地）
try { board.move('ghost', 'todo', 0) } catch (e) { console.log(e.message) }   // 卡片不存在
try { board.move('c2', 'nope', 0) } catch (e) { console.log(e.message) }      // 列不存在
console.log(board.snapshot())                     // 未产生部分变更
```
**评分点**：
- 归属与序双结构同步维护（cardCol 与 column.order 一致性）
- 原子移动（先构建 next 再整体提交，失败不产生部分变更）
- 同列/跨列两种路径的 order 处理（同列先删后插）
- assert 前置校验（非法输入 early throw）
