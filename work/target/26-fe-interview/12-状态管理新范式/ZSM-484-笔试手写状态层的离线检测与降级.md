## ZSM-484：手写状态层离线检测与降级
**需求**：实现 `createConnectivityGate()`：1) 三态判定：online / offline / flaky（navigator.onLine 是 true 但请求连续失败）；2) `track(result)`：请求成败喂入滑动窗口（最近 10 次），失败率 > 60% 判 flaky；3) `mode()` 返回当前判定；4) 模式变化事件（驱动 UI 离线横幅与请求策略切换：flaky 时提高超时与重试）；5) 恢复探测：flaky/offline 下周期探测轻量端点（10s），成功即恢复 online；6) 探测退避（连续失败翻倍间隔，上限 60s）。

```javascript
export function createConnectivityGate({ probeUrl = '/healthz', windowSize = 10 } = {}) {
  const results = []
  let mode = 'online'
  let probeTimer = null, backoff = 10_000
  const listeners = new Set()

  function setMode(m) {
    if (mode === m) return
    mode = m
    listeners.forEach((fn) => fn(m))
    scheduleProbe()
  }

  function track(ok) {
    results.push(ok)
    if (results.length > windowSize) results.shift()
    const failRate = results.filter((r) => !r).length / results.length
    setMode(navigator.onLine && failRate <= 0.6 ? 'online' : navigator.onLine ? 'flaky' : 'offline')
  }

  async function probe() {
    try {
      const r = await fetch(probeUrl, { method: 'HEAD', cache: 'no-store' })
      if (r.ok) { backoff = 10_000; setMode('online'); return }
    } catch { /* 继续退避 */ }
    backoff = Math.min(backoff * 2, 60_000)                  // 退避上限
    scheduleProbe()
  }

  function scheduleProbe() {
    clearTimeout(probeTimer)
    if (mode !== 'online') probeTimer = setTimeout(probe, backoff)
  }

  navigator.addEventListener?.('online', () => setMode('online'))
  navigator.addEventListener?.('offline', () => setMode('offline'))

  return {
    track,
    mode: () => mode,
    onChange: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
  }
}
```
**评分点**：
- flaky 态发明正确（navigator.onLine 是 true 但实际不可用——常见盲区）
- 滑动窗口失败率判定 + 模式变化事件驱动策略切换
- 探测退避（指数翻倍上限 60s）防探测风暴
- online/offline 系统事件与主动探测双通道互补
