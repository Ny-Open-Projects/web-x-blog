## ZSM-528：手写乐观更新的对账器
**答**：实现 `createReconciler(store)`：1) `applyOptimistic(opId, mutate)`：立即执行本地变更并挂起待确认操作；2) `confirm(opId, serverState)`：服务端确认后丢弃本地变更、以服务端为准覆盖（服务端可能已合并他人改动）；3) `reject(opId)`：回滚该操作，且**只回滚自己**不影响其后提交的其他乐观变更；4) `pending()` 列出未确认操作；5) 超时自动 reject 可配。附用例验证「回滚不误伤后续变更」。

```javascript
function createReconciler(get, set, notify) {
  const pendingOps = new Map()   // opId -> { prevSlice, opKey, timer }
  const listeners = new Set()
  let version = 0

  function applyOptimistic(opId, opKey, mutate, timeoutMs = 8000) {
    const prev = get()
    mutate(prev)                       // 直接改 store（乐观写）
    const timer = setTimeout(() => reject(opId), timeoutMs)
    pendingOps.set(opId, { prevSlice: prev, opKey, timer })
    version++
    notify(version)
  }

  function confirm(opId, serverState) {
    const op = pendingOps.get(opId)
    if (!op) return
    clearTimeout(op.timer)
    pendingOps.delete(opId)
    set(serverState)                   // 服务端为准：覆盖本地
    version++
    notify(version)
  }

  function reject(opId) {
    const op = pendingOps.get(opId)
    if (!op) return
    clearTimeout(op.timer)
    pendingOps.delete(opId)
    // 只回滚自己：撤销该 op 的字段变更，保留之后其他 op 的变更
    set((cur) => ({
      ...cur,
      [op.opKey]: op.prevSlice[op.opKey],
    }))
    version++
    notify(version)
  }

  return { applyOptimistic, confirm, reject, pending: () => [...pendingOps.keys()] }
}

// 用例
const db = { likes: 0, views: 0 }
const rc = createReconciler(
  () => db,
  (patch) => Object.assign(db, typeof patch === 'function' ? patch(db) : patch),
  () => console.log('render', { ...db })
)

rc.applyOptimistic('op1', 'likes', (s) => { s.likes++ })   // 乐观 +1
rc.applyOptimistic('op2', 'views', (s) => { s.views += 100 })
rc.reject('op1')                                            // 点赞失败
console.log(db)   // { likes: 0, views: 100 }：likes 回滚，views 保留
rc.confirm('op2', { likes: 0, views: 103 })                 // 服务端合并了他人浏览
console.log(db)   // { likes: 0, views: 103 }
```
**评分点**：
- 字段级回滚（按 opKey 只还原自己的切片，不整体回滚）
- confirm 以服务端状态覆盖（对账终局以服务端为准）
- 超时自动 reject 的 timer 清理（确认后 clearTimeout 防泄漏）
- pending 可观测（调试界面能看到未确认操作队列）
