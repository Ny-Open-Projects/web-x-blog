## ZSM-492：手写自动化修复建议器
**需求**：实现 `createFixAdvisor()`：1) 输入诊断信号（如 { type: 'low-cache-hit', key, value }），规则库匹配输出修复建议：`{ cause, fix, code, autoApplicable }`；2) 规则可组合：一条信号可能命中多条规则（按置信度排序）；3) `autoApply`：autoApplicable 且用户授权时自动执行修复（如把 staleTime 从 0 调到 30s），执行前快照可回滚；4) 建议去重（同一 cause 24h 内不重复弹出）；5) `history()` 修复执行记录（成功/失败/回滚）。

```javascript
export function createFixAdvisor(rules) {
  const seen = new Map()       // cause → lastShownAt
  const history = []
  let authorized = false

  function match(signal) {
    const now = Date.now()
    return rules
      .map((r) => r.when(signal) ? { confidence: r.confidence ?? 0, ...r.output(signal) } : null)
      .filter(Boolean)
      .sort((a, b) => b.confidence - a.confidence)
      .filter((s) => {                                        // 24h 去重
        const last = seen.get(s.cause)
        if (last && now - last < 86400_000) return false
        seen.set(s.cause, now)
        return true
      })
  }

  async function autoApply(advice, applyFn) {
    if (!advice.autoApplicable || !authorized) return { applied: false, reason: 'not authorized' }
    const snapshot = applyFn.snapshot?.()
    try {
      await applyFn(advice.code)
      history.push({ cause: advice.cause, at: Date.now(), status: 'applied' })
      return { applied: true, rollback: () => { snapshot && applyFn.restore?.(snapshot);
        history.push({ cause: advice.cause, at: Date.now(), status: 'rolled-back' }) } }
    } catch (e) {
      history.push({ cause: advice.cause, at: Date.now(), status: 'failed' })
      return { applied: false, reason: String(e) }
    }
  }

  return { match, autoApply, setAuthorized: (v) => (authorized = v), history: () => history }
}

// 规则示例
const rules = [{
  when: (s) => s.type === 'low-cache-hit' && s.value < 0.3,
  confidence: 0.8,
  output: () => ({ cause: 'low-cache-hit', fix: '提高 staleTime 或规范化 queryKey',
    code: (api) => api.setStaleTime(30_000), autoApplicable: true }),
}]
```
**评分点**：
- 规则→建议映射 + 置信度排序（多规则组合输出）
- autoApply 安全设计（授权前置 + 快照回滚 + 历史审计）
- 24h 建议去重（不骚扰）
- 规则库与执行器解耦（规则可热更）
