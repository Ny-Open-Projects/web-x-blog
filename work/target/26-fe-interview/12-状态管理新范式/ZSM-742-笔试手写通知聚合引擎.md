## ZSM-742：手写通知聚合引擎（窗口折叠 + 去重 + 已读游标联动）
**答**：实现 `createNotificationHub({ windowMs, max })`：1) `receive(n)`：按 id 去重（重复推送丢弃）；聚合 key = `type:targetId`，窗口内合并 count（取最新内容）；2) `markRead(seq)`：游标只进不退 + 上报回调（去抖）；3) `unread` 派生：seq > cursor 的聚合条目数；4) `route(n, ctx)`：展示决策——urgent 且非 busy → toast；busy/low → 仅列表；5) 环形上限（超 max 淘汰最旧聚合组）；6) 附用例：窗口内折叠计数、重复去重、已读后角标变化。

```javascript
function createNotificationHub({ windowMs = 5 * 60000, max = 50, reportRead } = {}) {
  const groups = new Map()      // dedupeKey -> { count, latest, seq, priority, latestAt }
  const seenIds = new Set()
  let readCursor = 0
  let reportTimer = null
  const listeners = new Set()

  function receive(n, ctx = {}) {
    if (seenIds.has(n.id)) return false                   // 重复推送去重
    seenIds.add(n.id)
    const key = `${n.type}:${n.targetId}`
    const cur = groups.get(key)
    if (cur && Date.now() - cur.latestAt < windowMs) {
      cur.count++
      cur.latest = n
      cur.latestAt = Date.now()
      cur.seq = Math.max(cur.seq, n.seq)
    } else {
      groups.set(key, { count: 1, latest: n, latestAt: Date.now(), seq: n.seq, priority: n.priority })
      // 环形上限：淘汰最旧聚合组
      if (groups.size > max) {
        const oldest = [...groups.entries()].sort((a, b) => a[1].latestAt - b[1].latestAt)[0]
        groups.delete(oldest[0])
      }
    }
    route(n, ctx)
    notify()
    return true
  }

  // 展示决策：busy 不打断
  function route(n, { busy = false } = {}) {
    if (busy) return 'list-only'
    if (n.priority === 'urgent') return 'toast+system'
    if (n.priority === 'normal') return 'toast'
    return 'list-only'
  }

  function markRead(seq) {
    readCursor = Math.max(readCursor, seq)                // 只进不退
    clearTimeout(reportTimer)
    reportTimer = setTimeout(() => reportRead?.(readCursor), 500)   // 上报去抖
    notify()
  }

  function unread() {
    let n = 0
    for (const g of groups.values()) if (g.seq > readCursor) n++
    return n
  }

  function list() {
    return [...groups.values()].sort((a, b) => b.seq - a.seq)
      .map((g) => ({ key: g.latest.type + ':' + g.latest.targetId, count: g.count, title: g.latest.title, seq: g.seq, priority: g.priority, unread: g.seq > readCursor }))
  }

  function notify() { listeners.forEach((fn) => fn({ list: list(), unread: unread() })) }

  return { receive, markRead, unread, list, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)) }
}

// 用例
let reported = null
const hub = createNotificationHub({ reportRead: (s) => (reported = s) })
hub.receive({ id: 'n1', type: 'like', targetId: 'post1', seq: 1, priority: 'normal', title: 'a 赞了你' })
hub.receive({ id: 'n2', type: 'like', targetId: 'post1', seq: 2, priority: 'normal', title: 'b 赞了你' })
hub.receive({ id: 'n3', type: 'like', targetId: 'post1', seq: 3, priority: 'normal', title: 'c 赞了你' })   // 折叠
console.log(hub.list()[0].count)         // 3：窗口内折叠
hub.receive({ id: 'n1', type: 'like', targetId: 'post1', seq: 1, priority: 'normal', title: 'a 赞了你' })   // 重复
console.log(hub.list()[0].count)         // 仍 3
console.log(hub.unread())                // 3（聚合组 seq=3 > cursor 0）
hub.markRead(3)
console.log(hub.unread())                // 0
console.log(reported)                    // 3：上报去抖后
hub.receive({ id: 'n4', type: 'comment', targetId: 'post2', seq: 4, priority: 'low', title: 'x 评论' }, { busy: true })
console.log(hub.list()[0].priority)      // 'low'：busy 时仅进列表
```
**评分点**：
- id 去重 + dedupeKey 窗口折叠（count 累计取最新内容）
- 已读游标只进不退 + 上报去抖
- unread 派生（不存储角标，游标比较）
- busy 展示决策（urgent 也降级为 list-only）
