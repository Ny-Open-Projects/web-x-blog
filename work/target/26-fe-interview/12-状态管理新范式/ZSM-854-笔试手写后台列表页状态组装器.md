## ZSM-854：手写后台列表页状态组装器（URL→Query→选择→批量四层）
**答**：853 的落地——实现 `createAdminListPage()`：1) `parseQuery(search)`：URL 白名单解析（536）+ 默认值；2) `buildQueryKey(q, tenantId)`：含租户前缀与筛选参数的 key；3) `selection`：跨页三态选择（537 复用）+ URL 不存选择（会话态）；4) `batchRun(ids, action)`：并发 5 + 结果集（693 简版）+ 可重试分级；5) 附用例：URL→key 一致性、选择跨页累计、批量部分成功重试。

```javascript
const DEFAULTS = { page: 1, pageSize: 20, sort: 'id_desc', status: 'all' }
const SORTS = new Set(['id_desc', 'id_asc', 'time_desc'])

function createAdminListPage({ api }) {
  // 1) URL 解析：白名单 + 默认值
  function parseQuery(search) {
    const sp = new URLSearchParams(search)
    return {
      page: Math.max(1, parseInt(sp.get('page') ?? '1', 10) || 1),
      pageSize: Math.min(100, parseInt(sp.get('pageSize') ?? '20', 10) || 20),
      sort: SORTS.has(sp.get('sort')) ? sp.get('sort') : DEFAULTS.sort,
      status: ['all', 'paid', 'refunded'].includes(sp.get('status')) ? sp.get('status') : 'all',
      q: (sp.get('q') ?? '').slice(0, 100),
    }
  }

  // 2) queryKey：租户前缀 + 筛选
  function buildQueryKey(q, tenantId) {
    return ['tenant', tenantId, 'orders', 'list', { page: q.page, pageSize: q.pageSize, sort: q.sort, status: q.status, q: q.q }]
  }

  // 3) 跨页三态选择（537 复用）
  const selected = new Map()   // pageId -> Set<rowId>
  let disabled = new Set()
  function toggleRow(pageId, rowId) {
    if (disabled.has(rowId)) return
    const s = selected.get(pageId) ?? new Set()
    s.has(rowId) ? s.delete(rowId) : s.add(rowId)
    if (s.size) selected.set(pageId, s)
    else selected.delete(pageId)
  }
  function selectedCount() { return [...selected.values()].reduce((a, s) => a + s.size, 0) }
  function selectedIds() { return [...selected.values()].flatMap((s) => [...s]) }
  function state3(pageId, pageRowIds, total) {
    const cur = selected.get(pageId) ?? new Set()
    const pageSel = pageRowIds.filter((r) => !disabled.has(r) && cur.has(r)).length
    const selectable = pageRowIds.filter((r) => !disabled.has(r)).length
    const totalSel = selectedCount()
    if (totalSel === 0) return 'none'
    if (pageSel === selectable && totalSel === total - disabled.size) return 'all'
    return 'some'
  }
  function clearSelection() { selected.clear() }

  // 4) 批量操作：并发 5 + 结果集 + 可重试分级（691/693 简版）
  async function batchRun(ids, action, { concurrency = 5 } = {}) {
    const results = new Map()
    let next = 0
    const RETRYABLE = new Set([429, 500, 503, 0])
    async function worker() {
      while (next < ids.length) {
        const id = ids[next++]
        try {
          const v = await api[action](id)
          results.set(id, { status: 'ok', value: v })
        } catch (e) {
          results.set(id, { status: 'failed', reason: e.message, retryable: RETRYABLE.has(e.status ?? 0) })
        }
      }
    }
    await Promise.all(Array.from({ length: Math.min(concurrency, ids.length) }, worker))
    const list = [...results.entries()]
    return {
      summary: {
        ok: list.filter(([, r]) => r.status === 'ok').length,
        failed: list.filter(([, r]) => r.status === 'failed').length,
        retryableIds: list.filter(([, r]) => r.status === 'failed' && r.retryable).map(([id]) => id),
      },
      results,
    }
  }

  return { parseQuery, buildQueryKey, toggleRow, selectedCount, selectedIds, state3, clearSelection, batchRun, disable: (ids) => (disabled = new Set([...disabled, ...ids])) }
}

// 用例
let t = 0
const api = {
  delete: async (id) => { if (id === 'r3') { const e = new Error('busy'); e.status = 500; throw e } return 'ok' },
}
const page = createAdminListPage({ api })
const q = page.parseQuery('?page=2&sort=time_desc&q=abc&evil=1')
console.log(q)                                    // 白名单解析（evil 被丢弃）
console.log(page.buildQueryKey(q, 't1'))          // ['tenant','t1','orders','list', {...}]

page.toggleRow('p1', 'r1'); page.toggleRow('p2', 'r2')   // 跨页累计
console.log(page.selectedCount(), page.state3('p1', ['r1', 'r3'], 4))   // 2 'some'
page.disable(['r3'])
console.log(page.state3('p1', ['r1', 'r3'], 4))   // 仍 'some'（r3 禁选不计）

const { summary } = await page.batchRun(['r1', 'r2', 'r3'], 'delete')
console.log(summary)                              // { ok: 2, failed: 1, retryableIds: ['r3'] }
```
**评分点**：
- URL 白名单解析（evil 参数丢弃 + 默认值兜底）
- queryKey 租户前缀（跨租户缓存隔离 724）
- 跨页选择三态（537 语义复用 + URL 不存选择）
- 批量并发 + 可重试分级（691 语义 + retryableIds 输出）
