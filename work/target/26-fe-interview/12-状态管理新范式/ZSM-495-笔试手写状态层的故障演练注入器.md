## ZSM-495：手写状态层故障演练注入器
**需求**：实现 `createChaosInjector(store)`：1) 注入类型：`inject('slow-set', { ms })`（setState 拖慢）/ `inject('error-set', { rate })`（按概率抛错）/ `inject('leak', { n })`（模拟订阅泄漏）；2) 注入挂载在原 API 包装层，`clear()` 全部卸载恢复原语义；3) 演练断言钩子：注入期间记录「系统行为指标」（如 slow-set 期间 P95、error-set 期间错误率）验证降级/熔断是否生效；4) 生产禁用（环境守卫）；5) 注入事件日志（演练可回溯）。

```javascript
export function createChaosInjector(store, { enabled = false } = {}) {
  if (!enabled && import.meta.env.PROD) return { inject: () => { throw new Error('chaos disabled in prod') }, clear: () => {}, log: () => [] }
  const log = []
  const originals = {}

  function wrap(name, mutate) {
    originals[name] = originals[name] ?? store[name]
    store[name] = mutate(originals[name])
  }

  function inject(type, opts = {}) {
    log.push({ type, opts, at: Date.now() })
    if (type === 'slow-set') {
      wrap('setState', (orig) => (...args) => {
        const start = performance.now()
        const r = orig.apply(store, args)
        const spin = performance.now() + (opts.ms ?? 50) - (performance.now() - start)
        while (performance.now() < spin) {}                   // 拖慢
        return r
      })
    } else if (type === 'error-set') {
      wrap('setState', (orig) => (...args) => {
        if (Math.random() < (opts.rate ?? 0.5)) throw new Error('chaos: injected error')
        return orig.apply(store, args)
      })
    } else if (type === 'leak') {
      wrap('subscribe', (orig) => (fn) => {
        for (let i = 0; i < (opts.n ?? 10); i++) orig.call(store, () => {})   // 泄漏订阅
        return orig(fn)
      })
    }
  }

  function clear() {
    Object.entries(originals).forEach(([k, v]) => (store[k] = v))
    log.push({ type: 'clear', at: Date.now() })
  }

  return { inject, clear, log: () => log }
}
```
**评分点**：
- 三类注入覆盖（延迟/错误/泄漏——状态层三大故障模式）
- 包装层可完全卸载（clear 恢复原语义——演练不留残余）
- 生产环境守卫（chaos 只在测试/预发）
- 注入日志（演练可回溯可复盘）
