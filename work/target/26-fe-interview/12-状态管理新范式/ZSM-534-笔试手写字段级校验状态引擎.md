## ZSM-534：手写字段级校验状态引擎
**答**：实现 `createForm(config)`：1) config 为 `{ field: { rules: [fn], deps: [field] } }`，支持跨字段规则（如「结束时间 > 开始时间」）；2) `setValue(field, value)` 后**异步批量校验**（同一 tick 合并，规则可 async）；3) 字段状态 `{ value, error, touched, validating }`：blur 时 touched=true 且立即校验；4) `validateAll()` 返回是否全部通过且收集错误；5) 竞态处理：连续输入只有最后一次校验结果生效（版本号比对）。附用例。

```javascript
function createForm(config) {
  const fields = new Map()  // field -> { value, error, touched, validating }
  const versions = new Map()
  const listeners = new Set()
  for (const k of Object.keys(config)) fields.set(k, { value: '', error: null, touched: false, validating: false })

  const emit = () => listeners.forEach((fn) => fn(snapshot()))

  async function validateField(field) {
    const st = fields.get(field)
    const ver = (versions.get(field) ?? 0) + 1
    versions.set(field, ver)
    st.validating = true
    emit()
    let error = null
    for (const rule of config[field].rules) {
      const res = await rule(st.value, fieldsProxy())   // 规则可 async
      if (res !== true) { error = typeof res === 'string' ? res : 'invalid'; break }
    }
    if (versions.get(field) !== ver) return   // 竞态：过期结果丢弃
    st.validating = false
    st.error = error
    emit()
  }

  // 跨字段规则通过只读代理读取其他字段
  function fieldsProxy() {
    const o = {}
    for (const [k, v] of fields) o[k] = v.value
    return o
  }

  let scheduled = new Set()
  function setValue(field, value) {
    const st = fields.get(field)
    st.value = value
    scheduled.add(field)
    // 同 tick 合并：依赖字段也一起重校
    queueMicrotask(() => {
      const batch = [...scheduled, ...[...scheduled].flatMap((f) => config[f].deps ?? [])]
      scheduled.clear()
      batch.forEach((f) => validateField(f))
    })
    emit()
  }

  function blur(field) {
    fields.get(field).touched = true
    validateField(field)   // blur 立即校验
  }

  async function validateAll() {
    await Promise.all(Object.keys(config).map((f) => validateField(f)))
    return [...fields.values()].every((f) => !f.error)
  }

  function snapshot() {
    const o = {}
    for (const [k, v] of fields) o[k] = { ...v }
    return o
  }

  return { setValue, blur, validateAll, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)), get: snapshot }
}

// 用例：跨字段规则
const form = createForm({
  start: { rules: [(v) => !!v || '必填'], deps: [] },
  end: {
    deps: ['start'],
    rules: [
      (v, all) => !!v || '必填',
      async (v, all) => new Date(v) > new Date(all.start) || '结束需晚于开始',
    ],
  },
})
form.setValue('start', '2026-01-01')
form.setValue('end', '2025-01-01')     // 触发 end + 依赖重校
setTimeout(() => console.log(form.get().end.error), 10)  // '结束需晚于开始'
```
**评分点**：
- 异步校验竞态版本号比对（过期结果丢弃）
- deps 跨字段依赖联动重校（结束/开始时间类规则）
- 同 tick 批量校验合并（微任务去抖）
- touched/validating/error 三态分离（UI 展示完备）
