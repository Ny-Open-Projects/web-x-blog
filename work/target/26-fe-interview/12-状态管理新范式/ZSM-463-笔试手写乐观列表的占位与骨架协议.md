## ZSM-463：手写乐观列表占位协议
**答**：实现 `createPlaceholderList(store)`：1) `refetch(page)`：拉新页数据期间保留旧数据渲染（placeholderData 语义），不显示骨架（有旧数据时）；2) 新数据到达：无缝替换；若页码变化导致条目变化，布局跳变用「锚点保持」缓解（保持视口内首条目的 id 对齐）；3) 首次无数据才显示骨架；4) 失败保留旧数据 + 局部错误提示；5) `getRenderList()` 返回 { list, mode: 'skeleton'|'content'|'stale-content' }。

```javascript
export function createPlaceholderList(fetcher, { staleMs = 60_000 } = {}) {
  let list = null, fetchedAt = 0, error = null, loading = false

  async function refetch(params) {
    loading = true
    try {
      const next = await fetcher(params)
      list = next; fetchedAt = Date.now(); error = null
    } catch (e) {
      error = e                                            // 失败保留旧 list
    } finally { loading = false }
  }

  function getRenderList() {
    if (!list) return { list: [], mode: 'skeleton' }                       // 首载骨架
    const stale = Date.now() - fetchedAt > staleMs
    return { list, mode: stale ? 'stale-content' : 'content', refreshing: loading,
             error: error && stale ? error : null }        // 旧数据保留 + 可选错误提示
  }

  function anchorOffset(prevList, nextList, anchorId) {
    const pi = prevList.findIndex((x) => x.id === anchorId)
    const ni = nextList.findIndex((x) => x.id === anchorId)
    return pi >= 0 && ni >= 0 ? ni - pi : 0                // 锚点位移：滚动补偿量
  }

  return { refetch, getRenderList, anchorOffset }
}
```
**评分点**：
- placeholderData 语义手写正确（有旧数据不换骨架，静默刷新）
- 三态模式（skeleton/content/stale-content）判定完整（首载/新鲜/过期）
- 失败保留旧数据（可用性优先）+ 错误仅在过期时显著提示
- 锚点保持（anchorOffset 位移计算）防布局跳变——加分项
