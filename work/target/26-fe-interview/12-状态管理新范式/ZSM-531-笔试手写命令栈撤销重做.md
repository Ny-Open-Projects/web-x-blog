## ZSM-531：手写表单设计器的命令栈撤销重做
**答**：实现 `createHistory(initial)`：1) `commit(mutator, label)`：mutator 产出新状态入栈，栈上限可配（超限丢最旧）；2) `undo/redo`：返回对应状态；3) `canUndo/canRedo` 派生；4) **合并策略**：连续同 label 的变更（如连续拖动滑块）在时间窗口（500ms）内合并为一条历史，避免撤销一步挪一格；5) `jumpTo(index)` 支持历史跳转且清空 redo 段。附用例。

```javascript
function createHistory(initial, { limit = 50, mergeMs = 500 } = {}) {
  let past = []        // 历史
  let present = initial
  let future = []      // redo 段
  let lastCommit = null // { label, ts }

  function commit(mutator, label = 'edit') {
    const next = typeof mutator === 'function' ? mutator(present) : mutator
    if (Object.is(next, present)) return
    const now = Date.now()
    // 合并：同 label 且在时间窗内 → 替换 present，不追加历史
    if (lastCommit && lastCommit.label === label && now - lastCommit.ts < mergeMs) {
      present = next
      lastCommit.ts = now
      return
    }
    past.push(present)
    if (past.length > limit) past.shift()   // 超限丢最旧
    present = next
    future = []                              // 新提交清空 redo 段
    lastCommit = { label, ts: now }
  }

  function undo() {
    if (!past.length) return present
    future.unshift(present)
    present = past.pop()
    lastCommit = null        // 手动 undo 打断合并窗口
    return present
  }

  function redo() {
    if (!future.length) return present
    past.push(present)
    present = future.shift()
    lastCommit = null
    return present
  }

  function jumpTo(index) {
    // index 相对 past 的位置：-1 表示回到最初
    const target = past[index] ?? initial
    future = [...past.slice(index + 1), present, ...future]
    past = past.slice(0, index)
    present = target
    lastCommit = null
    return present
  }

  return {
    commit, undo, redo, jumpTo,
    canUndo: () => past.length > 0,
    canRedo: () => future.length > 0,
    get: () => present,
    labels: () => past.map((_, i) => i),
  }
}

// 用例：表单设计器
const h = createHistory({ fields: [], title: '' })
h.commit((s) => ({ ...s, fields: [...s.fields, 'name'] }), 'addField')
h.commit((s) => ({ ...s, fields: [...s.fields, 'phone'] }), 'addField')
h.commit((s) => ({ ...s, title: '报名表' }), 'setTitle')
for (let i = 0; i < 20; i++) h.commit((s) => ({ ...s, drag: i }), 'drag') // 20 次拖动
// 500ms 内的 drag 全部合并 → 一次 undo 回到拖动前
console.log(h.undo().title)   // ''：undo 直接回到 setTitle 前
console.log(h.canRedo())      // true
console.log(h.undo().fields)  // ['name']：addField 两次按 label 分条，逐条回退
```
**评分点**：
- past/present/future 三段栈模型（redo 段在新提交时清空）
- 同 label 时间窗合并（高频微操作不撑爆历史）
- 栈上限淘汰策略（内存可控）
- jumpTo 重构双栈（历史跳转的正确切分）
