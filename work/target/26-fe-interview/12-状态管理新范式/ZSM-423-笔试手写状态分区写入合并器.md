## ZSM-423：手写状态分区写入合并器
**答**：实现 `createPartitionedWriter(store)`：1) `claim(partition, fields[])`：来源声明自己拥有的字段（分区），冲突声明抛错；2) `write(partition, patch)`：只能写自己分区字段，越权抛错；3) 多分区同 tick 写入自动合并为一次 store.setState（一次通知一次渲染）；4) `release(partition)` 释放分区（防泄漏）；5) 同一 partition 重复 claim 覆盖（热更新场景）。

```javascript
export function createPartitionedWriter(store) {
  const parts = new Map()          // partition → Set<fields>
  const pending = new Map()        // partition → patch（tick 内合并）
  let scheduled = false

  function claim(partition, fields) {
    for (const [p, set] of parts) {
      if (p !== partition) for (const f of fields)
        if (set.has(f)) throw new Error(`field ${f} owned by ${p}`)
    }
    parts.set(partition, new Set(fields))          // 重复 claim 覆盖
  }

  function write(partition, patch) {
    const own = parts.get(partition)
    if (!own) throw new Error('not claimed')
    for (const k of Object.keys(patch))
      if (!own.has(k)) throw new Error(`field ${k} not owned`)
    pending.set(partition, { ...pending.get(partition), ...patch })
    if (!scheduled) {
      scheduled = true
      queueMicrotask(() => {
        scheduled = false
        const merged = {}
        pending.forEach((p) => Object.assign(merged, p))
        pending.clear()
        store.setState(merged)                     // 一次通知一次渲染
      })
    }
  }

  function release(partition) { parts.delete(partition) }
  return { claim, write, release }
}
```
**评分点**：
- 分区所有权校验（声明制，越权写抛错防来源互踩）
- 同 tick 合并写入（microtask 聚合，多来源一次通知）
- claim 冲突检测（跨分区字段冲突抛错）；release 防分区泄漏
- pending 按 partition 合并（同分区多次写保留末值语义）
