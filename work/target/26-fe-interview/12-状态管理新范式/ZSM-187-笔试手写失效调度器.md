## ZSM-187：手写失效调度器（同帧去重的批量 invalidate）
**需求**：实现 `createInvalidationScheduler(invalidateFn)`：1) `schedule(keys)` 收集待失效 key，同一微任务周期内多次 schedule 合并去重后统一执行一次 invalidateFn；2) 去重规则：key 完全相等去重，且「前缀已包含」的 key 被更短前缀吸收（`['goods']` 吸收 `['goods','detail','1']`）；3) 执行时按 key 长度升序（先失效父域）；4) 返回的 schedule 支持链式与 Promise 化（`schedule(keys).then(done)`）。适用于 mutation 回调里高频触发失效的场景。

```javascript
export function createInvalidationScheduler(invalidateFn) {
  let scheduled = false
  let pendingSet = []                       // 收集本次周期的 key
  let resolvers = []
  let timer = null

  function absorb(keys) {                   // 前缀吸收：被父域覆盖的子 key 删除
    const sorted = [...pendingSet, ...keys]
      .sort((a, b) => a.length - b.length)
    const kept = []
    for (const k of sorted) {
      const covered = kept.some(p =>
        p.length <= k.length && p.every((v, i) => v === k[i]))
      if (!covered) kept.push(k)
    }
    pendingSet = kept
  }

  function schedule(keys) {
    absorb(Array.isArray(keys) && !Array.isArray(keys[0]) ? [keys] : keys)
    if (!scheduled) {
      scheduled = true
      timer = Promise.resolve().then(() => {          // 微任务级批合并
        const batch = pendingSet.sort((a, b) => a.length - b.length)
        pendingSet = []
        scheduled = false
        batch.forEach(k => invalidateFn(k))           // 父域先失效
        resolvers.forEach(r => r(batch))
        resolvers = []
      })
    }
    return new Promise(r => { resolvers.push(r); return timer })
  }

  return schedule
}

// 1. 微任务批合并：同周期多次 schedule 只执行一次
// 2. 前缀吸收减少冗余失效（父域覆盖子 key）
// 3. 升序排序保证父域先失效
// 4. Promise 化：可 await 完成时机
// 5. 单 key / 多 key 双形态入参兼容
```
