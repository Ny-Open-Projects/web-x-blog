## ZSM-471：手写状态层埋点注入器
**答**：实现 `instrumentActions(store, tracked)`：1) tracked 声明要埋的动作：`{ addToCart: { event: 'cart_add', props: (args, state) => ({...}) } }`；2) 包装动作函数：执行前取前置快照、执行后取后置，成功才上报（失败走 fail 事件）；3) props 工厂拿到 args 与前后状态（可计算 diff 类 props：before/after）；4) 上报节流（同类事件 100ms 合并计数）；5) 埋点异常完全隔离（try-catch，不影响动作本身）；6) `disable()` 全局关闭（性能排查用）。

```javascript
export function instrumentActions(store, tracked, report) {
  let enabled = true
  const merged = new Map()   // event → { count, timer }

  function flush(event) {
    const m = merged.get(event)
    if (!m) return
    report(event, { count: m.count, at: Date.now() })
    merged.delete(event)
  }

  const wrapped = {}
  for (const [name, cfg] of Object.entries(tracked)) {
    const orig = store[name]
    wrapped[name] = (...args) => {
      if (!enabled) return orig(...args)
      const before = structuredClone(store.getState())
      try {
        const r = orig(...args)
        const after = store.getState()
        const m = merged.get(cfg.event) ?? { count: 0 }
        m.count++
        merged.set(cfg.event, m)
        if (!m.timer) m.timer = setTimeout(() => flush(cfg.event), 100)   // 合并上报
        report(cfg.event, { props: cfg.props?.(args, before, after) })
        return r
      } catch (e) {
        report(cfg.event + '_fail', { error: String(e) })                 // 失败事件
        throw e
      }
    }
  }

  return { actions: wrapped, disable: () => { enabled = false; merged.forEach((_, e) => flush(e)) } }
}
```
**评分点**：
- 前后快照驱动的 props 工厂（diff 类埋点数据）
- 成功/失败事件分离；埋点异常隔离（不影响业务动作）
- 同类事件合并上报（100ms 计数，上报量控）
- disable 全局开关（性能排查时关闭埋点开销）
