## ZSM-868：手写白板元素引擎（z 序 LWW + 可视裁剪 + op 压缩）
**答**：867 落地——实现 `createWhiteboard()`：1) `upsert(element)`：元素 upsert 带 op 时间戳（并发 LWW：同 id 后到时间戳胜）+ z 序维护；2) `bringToFront(id)`：z 序置顶（记录 op 广播）；3) `visible(bbox, zoom)`：可视裁剪（元素 bbox 与视口相交）+ zoom 容差压缩点集；4) `compressOp(op, tolerance)`：路径简化（Douglas-Peucker 简版——逐点距离过滤）；5) 附用例：LWW 决胜与压缩减点。

```javascript
function createWhiteboard() {
  const elements = new Map()    // id -> { id, type, points, z, opTs }
  const listeners = new Set()

  // 并发 LWW：同 id 元素按 opTs 决胜
  function upsert(el, { opTs = Date.now() } = {}) {
    const cur = elements.get(el.id)
    if (cur && cur.opTs > opTs) return { applied: false, reason: 'STALE' }   // 旧 op 丢弃
    elements.set(el.id, { ...el, opTs })
    listeners.forEach((fn) => fn({ type: 'upsert', id: el.id }))
    return { applied: true }
  }

  function bringToFront(id) {
    const maxZ = Math.max(-1, ...[...elements.values()].map((e) => e.z))
    const el = elements.get(id)
    if (!el) return
    const opTs = Date.now()
    elements.set(id, { ...el, z: maxZ + 1, opTs })
    listeners.forEach((fn) => fn({ type: 'zorder', id, z: maxZ + 1, opTs }))
  }

  // 可视裁剪：bbox 相交 + zoom 容差压缩
  function elementBBox(el) {
    const xs = el.points.map((p) => p.x), ys = el.points.map((p) => p.y)
    return { minX: Math.min(...xs), maxX: Math.max(...xs), minY: Math.min(...ys), maxY: Math.max(...ys) }
  }
  function intersects(a, b) {
    return !(a.maxX < b.minX || a.minX > b.maxX || a.maxY < b.minY || a.minY > b.maxY)
  }
  function visible(bbox, zoom) {
    const out = []
    for (const el of elements.values()) {
      if (intersects(elementBBox(el), bbox)) {
        out.push({ ...el, points: compressPoints(el.points, toleranceFor(zoom)) })
      }
    }
    return out.sort((a, b) => a.z - b.z)     // z 序渲染
  }
  function toleranceFor(zoom) { return zoom > 2 ? 0.5 : 2 }   // zoom 大容差小（保真）

  // op 压缩：逐点距离过滤（简版 DP）
  function compressPoints(points, tolerance) {
    if (points.length <= 2) return points
    const out = [points[0]]
    for (let i = 1; i < points.length - 1; i++) {
      const prev = out.at(-1), cur = points[i]
      const dist = Math.hypot(cur.x - prev.x, cur.y - prev.y)
      if (dist >= tolerance) out.push(cur)   // 距离小于容差：丢弃
    }
    out.push(points.at(-1))                  // 保留终点
    return out
  }

  return { upsert, bringToFront, visible, compressPoints, size: () => elements.size, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)) }
}

// 用例
const wb = createWhiteboard()
const pts = Array.from({ length: 100 }, (_, i) => ({ x: i, y: i % 3 }))   // 密集点
wb.upsert({ id: 's1', type: 'stroke', points: pts, z: 0 }, { opTs: 100 })
console.log(wb.upsert({ id: 's1', type: 'stroke', points: pts.slice(0, 5), z: 0 }, { opTs: 50 }))
// { applied: false, reason: 'STALE' }：旧时间戳 LWW 丢弃
wb.upsert({ id: 's1', type: 'stroke', points: pts.slice(0, 50), z: 0 }, { opTs: 200 })   // 新 op 胜

wb.bringToFront('s1')
const vis = wb.visible({ minX: -10, minY: -10, maxX: 200, maxY: 200 }, 1)
console.log(vis[0].points.length)     // 远小于 100：压缩
console.log(vis[0].z)                 // 1：置顶后 z 序

const compressed = wb.compressPoints(pts, 2)
console.log(compressed.length)        // ~17：容差 2 的简化
```
**评分点**：
- z 序 LWW（opTs 决胜 + STALE 丢弃）
- 可视裁剪（bbox 相交 + zoom 动态容差）
- op 压缩（逐点距离过滤，保留首尾终点）
- z 序渲染排序输出
