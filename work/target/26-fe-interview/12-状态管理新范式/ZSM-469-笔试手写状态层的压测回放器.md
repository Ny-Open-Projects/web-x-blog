## ZSM-469：手写状态层压测回放器
**需求**：实现 `createStressRunner(store)`：1) `run({ events, rate })`：以 rate（事件/秒）注入合成事件流（如 1000 事件/秒 × 5s）；2) 事件模式：burst（同 tick 50 条）/ steady（均匀）/ mixed；3) 度量：期间 setState P95、渲染通知总次数、丢帧检测（rAF 间隔 > 32ms 计一次）；4) 结束输出报告 { events, p95SetMs, notifyCount, droppedFrames }；5) 断言 API：expectP95Below(ms) 不达标抛错（CI 压测门禁）。

```javascript
export function createStressRunner(store, makeEvent) {
  let durations = [], notifyCount = 0, droppedFrames = 0

  const origSet = store.setState
  store.setState = (p) => { const t0 = performance.now(); const r = origSet(p)
    durations.push(performance.now() - t0); notifyCount++; return r }

  async function run({ count = 5000, rate = 1000, mode = 'steady' }) {
    durations = []; notifyCount = 0; droppedFrames = 0
    let lastFrame = performance.now()
    const raf = () => { const now = performance.now()
      if (now - lastFrame > 32) droppedFrames++; lastFrame = now
      if (!running) return
      requestAnimationFrame(raf) }
    let running = true
    requestAnimationFrame(raf)

    const interval = 1000 / rate
    const burst = mode === 'burst' ? 50 : 1
    for (let i = 0; i < count; i += burst) {
      for (let b = 0; b < burst && i + b < count; b++) store.setState(makeEvent(i + b))
      if (mode === 'steady' || mode === 'mixed')
        await new Promise((r) => setTimeout(r, interval * burst))
    }
    running = false
    const sorted = [...durations].sort((a, b) => a - b)
    return { events: count, p95SetMs: sorted[Math.floor(sorted.length * 0.95)] ?? 0,
      notifyCount, droppedFrames }
  }

  function expectP95Below(ms, report) {
    if (report.p95SetMs > ms) throw new Error(`压测不达标: p95=${report.p95SetMs}ms > ${ms}ms`)
  }

  return { run, expectP95Below }
}
```
**评分点**：
- 三种压力模式（burst 检验合并能力 / steady 检验持续吞吐 / mixed 综合）
- 三类度量齐备（P95、通知次数、rAF 丢帧——用户可感知的卡顿度量）
- rAF 丢帧检测实现（>32ms 即丢帧）
- CI 门禁断言（expectP95Below）与压测报告解耦
