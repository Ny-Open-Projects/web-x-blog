## ZSM-879：手写一页纸架构图数据生成器（四区装配 + 故障点标注）
**答**：878 落地——实现 `generateArchDiagram({ features })`：按场景特征装配四区内容与故障点：1) 区 A/B/C/D 按需裁剪（无实时流去区 C、无复杂流程去区 B）；2) 故障点按特征注入（有 WS 加断线追平、有乐观写加回滚、有远端依赖加熔断）；3) 输出结构化图数据（nodes/edges/annotations）可直接渲染；4) 附用例：完整场景 vs 轻量场景的裁剪对比。

```javascript
function generateArchDiagram({ realtime = false, optimistic = false, complexFlow = false, remote = true, governance = true } = {}) {
  const nodes = [], edges = [], annotations = [], faultPoints = []

  // 区 A：数据层（必选）
  nodes.push({ id: 'L1', zone: 'A', label: 'RSC 直读' })
  nodes.push({ id: 'L2', zone: 'A', label: '缓存层 staleTime/gcTime' })
  nodes.push({ id: 'L3', zone: 'A', label: '会话态 registry 分层' })
  edges.push({ from: 'L1', to: 'L2', kind: 'inject' })
  edges.push({ from: 'L2', to: 'L3', kind: 'inject' })

  // 区 B：流程层（按需）
  if (complexFlow) {
    nodes.push({ id: 'L4', zone: 'B', label: 'XState actor' })
    edges.push({ from: 'L3', to: 'L4', kind: 'event' })
    edges.push({ from: 'L4', to: 'L2', kind: 'invalidate' })
  }

  // 区 C：同步层（按需）
  if (realtime) {
    nodes.push({ id: 'WS', zone: 'C', label: 'WS 信封 seq/ack' })
    nodes.push({ id: 'HB', zone: 'C', label: 'hold-back buffer' })
    nodes.push({ id: 'CATCHUP', zone: 'C', label: '追平三策略' })
    edges.push({ from: 'WS', to: 'HB', kind: 'stream' })
    faultPoints.push({ at: 'WS', label: '断线→退避重连+追平' })
  }

  // 横切概念
  if (optimistic) {
    annotations.push('幂等键 + 乐观回滚（对账器）')
    faultPoints.push({ at: 'L3', label: '乐观失败→字段级回滚' })
  }
  if (remote) {
    annotations.push('版本裁决（version/baseVersion）')
    faultPoints.push({ at: 'L2', label: '远端故障→熔断降级' })
  }

  // 区 D：治理层（按需）
  if (governance) {
    nodes.push({ id: 'GOV', zone: 'D', label: '探针→门禁→告警→健康分' })
    annotations.push('ADR / 技术债登记')
  }

  return {
    nodes, edges, annotations, faultPoints,
    summary: `${nodes.length} 节点 / ${edges.length} 连接 / ${faultPoints.length} 故障点`,
  }
}

// 用例：完整场景
const full = generateArchDiagram({ realtime: true, optimistic: true, complexFlow: true, remote: true })
console.log(full.summary)                    // 9 节点 / 6 连接 / 3 故障点
console.log(full.faultPoints.map((f) => f.label))
// ['断线→退避重连+追平', '乐观失败→字段级回滚', '远端故障→熔断降级']

// 轻量场景：无实时/无流程/无乐观
const light = generateArchDiagram({ realtime: false, optimistic: false, complexFlow: false })
console.log(light.summary)                   // 3 节点 / 2 连接 / 1 故障点
console.log(light.nodes.map((n) => n.id))    // ['L1','L2','L3']：区 B/C 整体裁剪
```
**评分点**：
- 按特征裁剪区（无实时流去区 C、无流程去区 B）
- 故障点按特征注入（WS→追平 / 乐观→回滚 / 远端→熔断）
- 结构化输出（nodes/edges/annotations 可渲染）
- 横切概念标注（幂等键/版本裁决/ADR）
