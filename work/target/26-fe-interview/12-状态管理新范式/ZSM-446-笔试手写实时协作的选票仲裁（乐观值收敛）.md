## ZSM-446：手写乐观值收敛仲裁
**需求**：实现 `createValueArbiter(store, key)`：1) `propose(value)`：本地立即显示为「候选值」（标记 pending），同时提交服务端；2) 服务端返回权威值后：本地值=权威值，pending 清除；3) 多次快速 propose：只保留最后一次（前序提交取消）；4) 取消实现：前序提交的响应到达时若版本已旧则丢弃；5) 服务端失败：回退到「最后确认值」并标记 failed（用户可重试）；6) `get()` 返回 { value, status: 'confirmed'|'pending'|'failed' }。

```javascript
export function createValueArbiter(store, key, submit) {
  let confirmed = store.getState()[key]          // 最后确认值
  let pendingVersion = 0                          // 提交代次

  async function propose(value) {
    const v = ++pendingVersion
    setStatus('pending', value)                   // 本地立即候选
    try {
      const server = await submit(value)
      if (v !== pendingVersion) return            // 已被更新提交取代：丢弃
      confirmed = server.value
      store.setState({ [key]: confirmed })
      setStatus('confirmed', confirmed)
    } catch (e) {
      if (v !== pendingVersion) return
      store.setState({ [key]: confirmed })        // 回退最后确认值
      setStatus('failed', confirmed)
    }
  }

  function setStatus(status, value) {
    store.setState({ [`${key}__status`]: status })
  }

  return {
    propose,
    get: () => ({ value: store.getState()[key], status: store.getState()[`${key}__status`] ?? 'confirmed' }),
    get confirmed() { return confirmed },
  }
}
```
**评分点**：
- 代次（pendingVersion）取消语义：旧提交响应丢弃，只有最新提交生效
- 失败回退到「最后确认值」而非清空（用户不丢基线）；failed 状态驱动重试 UI
- confirmed/pending/failed 三态与 get() 投影一致（UI 只读状态机）
- 快速连续 propose 的提交风暴由代次+取消自然收敛（无需额外 debounce 也应说明可加）
