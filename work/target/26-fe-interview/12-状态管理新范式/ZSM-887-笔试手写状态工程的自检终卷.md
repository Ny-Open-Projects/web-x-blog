function autoGrade(answers) {
  // answers: { q: 'answer text' }
  const results = []
  let total = 0, count = 0
  for (const [cat, qs] of Object.entries(FINAL_EXAM)) {
    for (const { q, keys } of qs) {
      const text = answers[q] ?? ''
      const hit = keys.filter((k) => text.includes(k)).length
      const score = +(hit / keys.length * 100).toFixed(0)
      results.push({ cat, q, score, missing: score < 60 ? keys.filter((k) => !text.includes(k)) : [] })
      total += score; count++
    }
  }
  const byCat = {}
  results.forEach((r) => { byCat[r.cat] ??= { sum: 0, n: 0 }; byCat[r.cat].sum += r.score; byCat[r.cat].n++ })
  const gaps = Object.entries(byCat).map(([cat, { sum, n }]) => ({ cat, avg: Math.round(sum / n) }))
    .filter((c) => c.avg < 60).sort((a, b) => a.avg - b.avg)   // 缺口考点
  return {
    total: Math.round(total / count),
    gaps,
    weakest: gaps[0]?.cat,
    action: gaps[0] ? `优先补：${gaps[0].cat}（对应收官地图重读 + 笔试卡默写）` : '全考点达标',
  }
}
