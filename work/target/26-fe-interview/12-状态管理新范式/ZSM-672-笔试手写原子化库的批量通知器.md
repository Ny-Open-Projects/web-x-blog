## ZSM-672：手写原子化库的批量通知器（事务合并 + 依赖失效）
**答**：基于 Jotai 风格原子库：1) `batch(fn)`：fn 内多次 setAtom 合并——所有订阅者只收到一轮通知（通知在 batch 结束后统一 flush）；2) 嵌套 batch 支持内层合并外层提交；3) 通知时带上「本轮变更的 atom 集合」与「每个 atom 的新旧值」；4) atom 订阅可声明依赖派生（`subscribe(atom, fn, deps)`）——依赖 atom 未变不通知；5) 附用例：batch 内改 3 个 atom 验证一轮通知与依赖过滤。

```javascript
function createAtoms() {
  const values = new Map()      // atom -> value
  const subs = new Map()        // atom -> Set<{ fn, deps }>
  let batchDepth = 0
  let pending = null            // { changed: Map<atom, {prev,next}>, orderedAtoms: Set }

  function flush() {
    if (!pending) return
    const snapshot = pending
    pending = null
    // 一轮通知：遍历变更 atom 的订阅者，按依赖过滤
    for (const [atom, { prev, next }] of snapshot.changed) {
      for (const sub of subs.get(atom) ?? []) {
        if (sub.deps) {
          // 依赖过滤：声明的依赖 atom 未变则不通知
          const depsChanged = sub.deps.some((d) => snapshot.changed.has(d))
          if (!depsChanged && !snapshot.changed.has(atom)) continue
        }
        sub.fn(next, prev)
      }
    }
  }

  function setAtom(atom, updater) {
    const prev = values.get(atom)
    const next = typeof updater === 'function' ? updater(prev) : updater
    if (Object.is(prev, next)) return
    values.set(atom, next)
    if (batchDepth > 0) {
      pending ??= { changed: new Map(), orderedAtoms: new Set() }
      if (!pending.changed.has(atom)) pending.changed.set(atom, { prev, next })
      else pending.changed.get(atom).next = next      // batch 内多次改：记录最终值
    } else {
      // 无 batch：单次变更即一轮
      pending = { changed: new Map([[atom, { prev, next }]]), orderedAtoms: new Set([atom]) }
      flush()
    }
  }

  function batch(fn) {
    batchDepth++
    try { fn() } finally {
      batchDepth--
      if (batchDepth === 0) flush()      // 最外层才 flush
    }
  }

  function subscribe(atom, fn, deps) {
    const sub = { fn, deps }
    ;(subs.get(atom) ?? subs.set(atom, new Set()).get(atom)).add(sub)
    return () => subs.get(atom)?.delete(sub)
  }

  function get(atom) {
    if (!values.has(atom)) values.set(atom, atom.init)
    return values.get(atom)
  }

  return { get, setAtom, batch, subscribe }
}

// 用例
const countA = { init: 0 }, countB = { init: 0 }, noise = { init: 0 }
const atoms = createAtoms()

let notifications = 0
atoms.subscribe(countA, () => notifications++)   // 依赖声明版
atoms.subscribe(countA, () => notifications++, [countB])

atoms.batch(() => {
  atoms.setAtom(countA, 1)
  atoms.setAtom(noise, 9)      // 不影响 countA 订阅者（无依赖且未订阅）
  atoms.setAtom(countB, 2)
  atoms.setAtom(countA, 3)     // 同 atom 二次改：只记最终值
})
console.log(notifications)     // 1：一轮通知
console.log(atoms.get(countA)) // 3：最终值

// 嵌套 batch
notifications = 0
atoms.batch(() => {
  atoms.setAtom(countA, 10)
  atoms.batch(() => atoms.setAtom(countA, 11))   // 内层不单独 flush
})
console.log(notifications)     // 1
```
**评分点**：
- batch 计数器嵌套（最外层 flush）
- 同 atom 多次改记录最终值（prev 保留最初值）
- 通知带变更快照（changed Map 供依赖过滤）
- 无 batch 路径退化为单次通知（语义一致）
