## ZSM-785：手写架构健康分计算器（加权评分 + 等级 + 整改工单生成）
**答**：实现 `computeHealthScore(metrics)`：1) 十项指标带权重评分（每项 scoreFn 映射 0-10）；2) 加权总分映射 A/B/C 等级；3) C 级生成整改工单：低于 5 分的指标生成 `{ metric, score, suggestion, deadline }`（suggestion 按指标查建议表）；4) 历史对比：`compare(prev, curr)` 输出每项趋势（↑↓→）；5) 附用例：评分、工单生成、趋势对比。

```javascript
const SUGGESTIONS = {
  crossLayerViolations: '运行 eslint-boundaries，修复违规 import',
  concurrencyTestCoverage: '按审计八问补竞态用例（ZSM-766）',
  circularDeps: 'madge --circular 定位，抽公共包解环',
  heapGrowthSlope: 'ZSM-756 堆快照 diff 定位泄漏面',
  migrateChainIntegrity: '补齐 persist/IDB 迁移链版本',
}

function computeHealthScore(metrics) {
  const RULES = [
    { key: 'crossLayerViolations', weight: 2, score: (n) => n === 0 ? 10 : Math.max(0, 10 - n * 2) },
    { key: 'concurrencyTestCoverage', weight: 2, score: (r) => Math.round((r ?? 0) * 10) },
    { key: 'registryCoverage', weight: 1, score: (r) => Math.round((r ?? 0) * 10) },
    { key: 'adrCoverage', weight: 1, score: (r) => Math.round((r ?? 0) * 10) },
    { key: 'ciGatePassRate', weight: 1, score: (r) => Math.round((r ?? 0) * 10) },
    { key: 'heapGrowthSlope', weight: 1, score: (v) => (v ?? 0) < 0.5 ? 10 : Math.max(0, 10 - v * 4) },
    { key: 'runbookCompleteness', weight: 1, score: (r) => Math.round((r ?? 0) * 10) },
    { key: 'circularDeps', weight: 1, score: (n) => (n ?? 0) === 0 ? 10 : Math.max(0, 10 - n * 3) },
    { key: 'migrateChainIntegrity', weight: 1, score: (ok) => ok ? 10 : 0 },
    { key: 'fallbackDrillPass', weight: 1, score: (ok) => ok ? 10 : 0 },
  ]
  const totalWeight = RULES.reduce((a, r) => a + r.weight, 0)
  const scored = RULES.map((r) => {
    const s = r.score(metrics[r.key])
    return { key: r.key, weight: r.weight, score: s }
  })
  const total = +(scored.reduce((a, s) => a + s.weight * s.score, 0) / totalWeight * 10).toFixed(1)
  const grade = total >= 85 ? 'A' : total >= 70 ? 'B' : 'C'

  // C 级整改工单
  const tickets = []
  if (grade === 'C') {
    for (const s of scored) {
      if (s.score < 5) {
        tickets.push({
          metric: s.key,
          score: s.score,
          suggestion: SUGGESTIONS[s.key] ?? '人工评估',
          deadline: '两周',
        })
      }
    }
  }
  return { total, grade, scored, tickets }
}

function compare(prev, curr) {
  const prevMap = new Map(prev.scored.map((s) => [s.key, s.score]))
  return curr.scored.map((s) => {
    const p = prevMap.get(s.key) ?? 0
    const trend = s.score > p ? '↑' : s.score < p ? '↓' : '→'
    return { key: s.key, prev: p, curr: s.score, trend }
  })
}

// 用例
const curr = computeHealthScore({
  crossLayerViolations: 2, concurrencyTestCoverage: 0.4, registryCoverage: 0.9,
  adrCoverage: 0.5, ciGatePassRate: 0.95, heapGrowthSlope: 0.3,
  runbookCompleteness: 0.8, circularDeps: 1, migrateChainIntegrity: true, fallbackDrillPass: false,
})
console.log(curr.total, curr.grade)     // <70 'C'
console.log(curr.tickets.map((t) => `${t.metric}: ${t.suggestion}`))
// crossLayerViolations / concurrencyTestCoverage / circularDeps / fallbackDrillPass 四张工单

const prevScored = { scored: curr.scored.map((s) => ({ ...s, score: Math.max(0, s.score - 3) })) }
console.log(compare(prevScored, curr).filter((t) => t.trend === '↑').length)   // 全部 ↑
```
**评分点**：
- 加权评分（P0 相关指标权重 2x）
- C 级工单生成（score < 5 触发 + 建议表映射 + 限期）
- 历史趋势对比（↑↓→ 每项可追踪）
- 建议表与治理动作绑定（工单可执行）
