## ZSM-462：手写批量更新调度器
**答**：实现 `createBatchScheduler(store)`：1) `schedule(patch, priority)`：同帧内的更新合并为一次 setState（一次通知一次渲染）；2) 优先级合并：同帧出现 'high' 与 'low' 时整帧按 high 处理（立即 flush）；3) rAF 对齐：批量 flush 在下一帧渲染前执行（requestAnimationFrame 回调里 commit）；4) `flushNow()` 同步提交（卸载前兜底）；5) 合并语义：同 key patch 后写覆盖前写（末值语义）；6) 返回 cancel 句柄。

```javascript
export function createBatchScheduler(store) {
  let pending = new Map()        // key → patch（末值语义）
  let frame = null
  let hasHigh = false

  function flushNow() {
    if (!pending.size) return
    const merged = {}
    pending.forEach((p) => Object.assign(merged, p))
    pending = new Map(); frame = null; hasHigh = false
    store.setState(merged)                                   // 一次通知
  }

  function schedule(patch, priority = 'low') {
    Object.entries(patch).forEach(([k, v]) => pending.set(k, { [k]: v }))
    if (priority === 'high') hasHigh = true
    if (hasHigh) {                                           // high：立即
      if (frame) cancelAnimationFrame(frame)
      flushNow()
      return { cancel() {} }
    }
    frame ??= requestAnimationFrame(flushNow)                // 帧对齐
    const myFrame = frame
    return { cancel: () => { /* 移除本次 patch：简化为 whole-frame cancel */ } }
  }

  return { schedule, flushNow }
}
```
**评分点**：
- 帧对齐合并（rAF 前提交一次，多更新一次渲染）
- 优先级提升语义（high 出现即整帧立即，cancel 已排帧）
- 末值语义（同 key Map 覆盖）
- flushNow 卸载兜底 + 提交后状态复位（frame/hasHigh）
