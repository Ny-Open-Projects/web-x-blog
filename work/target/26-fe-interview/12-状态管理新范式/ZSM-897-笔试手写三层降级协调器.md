## ZSM-897：手写三层降级协调器（D1/D2/D3 触发 + 只读镜像兜底）
**答**：896 落地——实现 `createDegradationCoordinator()`：1) 触发路由：`onQueryError` → D1、`onCircuitOpen` → D2、`onUncaught` → D3；2) D1 自动：切 stale 缓存 + 排队重试；D2 自动：切离线模式 + 队列化；D3：ErrorBoundary 回调 → store 重置 + 只读镜像展示；3) `recover()`：恢复时统一 revalidate + 降级态清除；4) 降级事件日志（P3 监控口径）；5) 附用例：D1→恢复 与 D3→镜像→恢复全链。

```javascript
function createDegradationCoordinator({ queryClient, store, persistKey }) {
  let degraded = null           // null | 'D1' | 'D2' | 'D3'
  const log = []
  const listeners = new Set()

  function emit(e) { log.push({ ts: Date.now(), level: 'P3', ...e }); listeners.forEach((fn) => fn(e)) }

  function readMirror() {
    // D3 只读镜像：persist 快照静态展示
    try { return JSON.parse(localStorage.getItem(persistKey) ?? 'null') } catch { return null }
  }

  // D1：查询失败 → stale 缓存 + 排队重试
  function onQueryError(queryKey, err) {
    if (degraded) return
    degraded = 'D1'
    emit({ type: 'D1', queryKey: JSON.stringify(queryKey), reason: err.message })
    // stale 缓存展示（TanStack data 保留）
    scheduleRetry(queryKey)
  }

  let retryTimer = null
  function scheduleRetry(queryKey, attempt = 1) {
    retryTimer = setTimeout(async () => {
      try {
        await queryClient.refetchQueries({ queryKey })
        recover()
      } catch {
        if (attempt < 3) scheduleRetry(queryKey, attempt + 1)   // 退避重试
        else onCircuitOpen()                                     // 升级 D2
      }
    }, Math.min(30000, 1000 * 2 ** attempt))
  }

  // D2：熔断/通道失败 → 离线模式
  function onCircuitOpen() {
    degraded = 'D2'
    emit({ type: 'D2', reason: 'circuit open' })
    store.setState({ offlineMode: true })      // UI 明示横幅
    // 后续写操作进离线队列（588）
  }

  // D3：未捕获异常 → 重置 + 只读镜像
  function onUncaught(err) {
    degraded = 'D3'
    emit({ type: 'D3', reason: err.message })
    const mirror = readMirror()
    store.setState(mirror ?? {}, true)          // 重置到持久化快照（只读语义）
    queryClient.clear()
  }

  // 恢复：统一 revalidate
  function recover() {
    if (!degraded) return
    emit({ type: 'recover', from: degraded })
    degraded = null
    store.setState({ offlineMode: false })
    queryClient.invalidateQueries({ refetchType: 'active' })   // 对账闭环（896）
    listeners.forEach((fn) => fn({ type: 'revalidated' }))
  }

  return {
    onQueryError, onCircuitOpen, onUncaught, recover,
    get degraded() { return degraded },
    log: () => log,
    subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
  }
}

// 用例
const store = { state: {}, setState(p, replace) { this.state = replace ? { ...p } : { ...this.state, ...p } } }
const qc = {
  refetchQueries: async () => { throw new Error('still down') },
  invalidateQueries: () => {},
  clear: () => {},
}
localStorage.setItem('app', JSON.stringify({ user: { name: 'cached' } }))

const dc = createDegradationCoordinator({ queryClient: qc, store, persistKey: 'app' })
dc.onQueryError(['todos'], new Error('timeout'))       // D1
console.log(dc.degraded)                                // 'D1'
await new Promise((r) => setTimeout(r, 2300))
console.log(dc.degraded)                                // 'D2'：重试失败升级

dc.onUncaught(new Error('hydration mismatch'))          // D3
console.log(store.state.user?.name)                     // 'cached'：只读镜像兜底
dc.recover()
console.log(dc.degraded, store.state.offlineMode)       // null false：恢复对齐
```
**评分点**：
- 三层触发路由（D1 查询错/D2 熔断/D3 未捕获）
- D1→D2 自动升级（重试 3 次失败升级——退避）
- D3 只读镜像兜底（persist 快照静态展示）
- recover 统一 revalidate（降级不是终点，对齐才是）
