## ZSM-681：手写评论树状态引擎（嵌套回复 + 折叠 + 计数派生）
**答**：实现 `createCommentTree(comments)`：1) 归一化存储 `Map<id, { id, parentId, content, likeCount, deleted }>` + 子索引 `children: Map<parentId, id[]>`；2) `add(parentId, content)` / `remove(id)`——删除为软删（deleted 标记，子回复保留展示但父内容显示「已删除」）；3) `toggleCollapse(id)` 折叠状态；4) 派生 `visibleList(rootId)`：DFS 输出可展示序列（跳过折叠子树的内部节点但保留折叠锚点），带 depth 与 replyCount（递归计数含软删）；5) 附用例：软删父 + 折叠分支的派生结果。

```javascript
function createCommentTree(comments = []) {
  const nodes = new Map()     // id -> node
  const children = new Map()  // parentId -> id[]
  const collapsed = new Set()

  comments.forEach((c) => add(c.parentId ?? null, c.content, c.id, c.likeCount ?? 0))

  function add(parentId, content, id = crypto.randomUUID(), likeCount = 0) {
    nodes.set(id, { id, parentId, content, likeCount, deleted: false })
    if (parentId) {
      (children.get(parentId) ?? children.set(parentId, []).get(parentId)).push(id)
    }
    return id
  }

  function remove(id) {
    const n = nodes.get(id)
    if (!n) return
    n.deleted = true        // 软删：子回复保留
  }

  function toggleCollapse(id) {
    collapsed.has(id) ? collapsed.delete(id) : collapsed.add(id)
  }

  // 递归计数（含软删）
  function replyCount(id) {
    let n = 0
    for (const c of children.get(id) ?? []) {
      n += 1 + replyCount(c)
    }
    return n
  }

  // 可展示序列：折叠子树只留锚点；软删父保留占位
  function visibleList(rootId = null, depth = -1) {
    const out = []
    const walk = (pid, d) => {
      for (const id of children.get(pid) ?? []) {
        const n = nodes.get(id)
        if (collapsed.has(id)) {
          out.push({ ...n, depth: d + 1, replyCount: replyCount(id), collapsed: true })
          continue                                        // 折叠：不走子树
        }
        out.push({ ...n, depth: d + 1, replyCount: replyCount(id), collapsed: false,
          placeholder: n.deleted })
        walk(id, d + 1)
      }
    }
    walk(rootId, depth)
    return out
  }

  return { add, remove, toggleCollapse, visibleList, replyCount, nodes }
}

// 用例
const tree = createCommentTree()
const root = tree.add(null, '主评论')
const c1 = tree.add(root, '回复1')
tree.add(c1, '回复1-1')
tree.add(c1, '回复1-2')
const c2 = tree.add(root, '回复2')
tree.add(c2, '回复2-1')

tree.remove(root)                       // 软删主评论
tree.toggleCollapse(c1)                 // 折叠回复1分支
const list = tree.visibleList(null)
console.log(list.map((x) => `${x.placeholder ? '[已删除]' : x.content}(${x.replyCount})${x.collapsed ? '▸' : ''}`))
// [已删除](5)
// 回复2(1)
// 回复2-1(0)
console.log(tree.replyCount(root))      // 5：含软删的递归计数
```
**评分点**：
- 归一化双结构（nodes Map + children 索引）
- 软删语义（父删除子保留、placeholder 占位）
- 折叠为独立状态集（与数据结构解耦，DFS 跳过子树留锚点）
- replyCount 递归派生（含软删计数）
