## ZSM-699：手写搜索编排引擎（防抖 + 版本竞态 + 多源分组合并）
**答**：实现 `createSearchEngine(sources)`：1) `input(kw)`：空关键词回 idle；防抖 300ms 后执行；2) `#execute`：多源并发（各自 AbortController），**版本号竞态防御**（每 input 递增 ver，响应回报版本不符丢弃）；3) 分组状态 `{ phase, kw, sources: { name: { loading, items, error } } }`，各源独立完成独立通知；4) 全部完成后 phase=done，产出合并计数（总命中数）；5) 附用例：快速连续输入验证旧响应丢弃。

```javascript
function createSearchEngine(sources, { debounceMs = 300 } = {}) {
  let ver = 0
  let debounce = null
  let controllers = []
  const state = { phase: 'idle', kw: '', sources: {}, total: 0 }
  const subs = new Set()

  function notify() { subs.forEach((fn) => fn({ ...state, sources: structuredClone(state.sources) })) }

  function input(kw) {
    clearTimeout(debounce)
    state.kw = kw
    if (!kw.trim()) {
      state.phase = 'idle'
      state.sources = {}
      state.total = 0
      ver++                       // 使在途响应全部过期
      abortAll()
      notify()
      return
    }
    state.phase = 'typing'
    notify()
    debounce = setTimeout(() => execute(kw), debounceMs)
  }

  function abortAll() {
    controllers.forEach((c) => c.abort())
    controllers = []
  }

  async function execute(kw) {
    const myVer = ++ver
    abortAll()
    state.phase = 'searching'
    state.sources = Object.fromEntries(Object.keys(sources).map((k) => [k, { loading: true, items: [], error: null }]))
    notify()

    await Promise.all(Object.entries(sources).map(async ([name, fn]) => {
      const ac = new AbortController()
      controllers.push(ac)
      try {
        const items = await fn(kw, ac.signal)
        if (myVer !== ver) return                       // 过期响应：丢弃
        state.sources[name] = { loading: false, items, error: null }
      } catch (e) {
        if (e.name === 'AbortError') return
        state.sources[name] = { loading: false, items: [], error: e.message }
      }
      notify()                                          // 各源独立通知
    }))

    if (myVer !== ver) return                           // 全程被新输入取代
    state.phase = 'done'
    state.total = Object.values(state.sources)
      .reduce((a, s) => a + s.items.length, 0)
    notify()
  }

  return { input, subscribe: (fn) => (subs.add(fn), () => subs.delete(fn)), state: () => state }
}

// 用例：快速连续输入
const fakeSource = (delay) => (kw, signal) => new Promise((resolve, reject) => {
  const t = setTimeout(() => resolve([`${kw}-hit1`, `${kw}-hit2`]), delay)
  signal.addEventListener('abort', () => { clearTimeout(t); reject(Object.assign(new Error(), { name: 'AbortError' })) })
})
const engine = createSearchEngine({ docs: fakeSource(50), users: fakeSource(20) })
engine.input('react')
engine.input('react hooks')            // 300ms 内：第一次被防抖取消
await new Promise((r) => setTimeout(r, 400))
console.log(engine.state().kw)         // 'react hooks'（最新）
console.log(engine.state().phase)      // 'done'
console.log(engine.state().total)      // 4（两源各 2）
```
**评分点**：
- 版本号竞态防御（响应回报 ver 过期丢弃，防旧关键词闪现）
- 多源独立 AbortController + 独立通知（快的先渲染）
- 防抖与 idle 回退（空关键词使在途全部过期）
- 全部完成才置 done 并产出合并计数（阶段语义清晰）
