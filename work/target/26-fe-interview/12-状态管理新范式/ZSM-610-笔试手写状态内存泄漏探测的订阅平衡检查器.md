## ZSM-610：手写订阅平衡检查器（泄漏探测）
**答**：实现 `createLeakDetector()`：1) `instrument(store, label)`：包装 store 的 subscribe，记账每次 add/remove（含调用栈前 2 帧）；2) `check(label?)`：输出未平衡订阅清单（活跃订阅数 > 0 的 label + 栈），可选只查某个 label；3) `watchdog(interval)`：周期巡检——「活跃订阅数持续增长且无 remove」时告警（增长泄漏 vs 正常常驻订阅的区分：基线值 + 增长趋势）；4) `attach(store, label)` 包装组件用 hook（useEffect 自动平衡校验）；5) 附用例：三次 add 两次 remove 检出泄漏。

```javascript
import { useEffect, useRef } from 'react'

function createLeakDetector() {
  const ledger = new Map()   // label -> [{ id, stack, addedAt, removed: bool }]

  function instrument(store, label) {
    const origSub = store.subscribe
    store.subscribe = (...args) => {
      const entry = {
        id: Math.random().toString(36).slice(2),
        stack: new Error().stack.split('\n').slice(2, 4),
        addedAt: Date.now(),
        removed: false,
      }
      ;(ledger.get(label) ?? ledger.set(label, []).get(label)).push(entry)
      const unsub = origSub.apply(store, args)
      return () => {
        entry.removed = true
        entry.removedAt = Date.now()
        unsub()
      }
    }
    return store
  }

  function check(label) {
    const out = []
    for (const [l, entries] of ledger) {
      if (label && l !== label) continue
      const active = entries.filter((e) => !e.removed)
      if (active.length) {
        out.push({ label: l, active: active.length, samples: active.slice(-3).map((e) => ({ stack: e.stack, addedAt: e.addedAt })) })
      }
    }
    return out.length ? out : 'balanced'
  }

  function watchdog(interval, { baseline = 2 } = {}) {
    const history = []
    const timer = setInterval(() => {
      let active = 0
      ledger.forEach((entries) => entries.forEach((e) => { if (!e.removed) active++ }))
      history.push(active)
      // 增长趋势：最近 5 次单调递增且超基线
      if (history.length >= 5) {
        const last5 = history.slice(-5)
        const growing = last5.every((v, i) => i === 0 || v > last5[i - 1])
        if (growing && active > baseline) {
          console.warn(`[leak-watchdog] 活跃订阅持续增长: ${active} (基线 ${baseline})`, check())
        }
      }
    }, interval)
    return () => clearInterval(timer)
  }

  function useTrackedSubscribe(store, label, subscribeFn, deps) {
    const ref = useRef(null)
    useEffect(() => {
      const origSub = store.subscribe
      const entry = { stack: null, removed: false }
      ;(ledger.get(label) ?? ledger.set(label, []).get(label)).push(entry)
      const unsub = subscribeFn()
      return () => {
        entry.removed = true
        unsub()
      }
    }, deps)
  }

  return { instrument, check, watchdog, useTrackedSubscribe }
}

// 用例
const det = createLeakDetector()
const listeners = new Set()
const store = { subscribe(fn) { listeners.add(fn); return () => listeners.delete(fn) } }
det.instrument(store, 'cart')
const u1 = store.subscribe(() => {})
const u2 = store.subscribe(() => {})
const u3 = store.subscribe(() => {})
u1(); u2()                       // 只退订两个
console.log(det.check('cart'))   // [{ label: 'cart', active: 1, samples: [...] }]：检出泄漏
u3()
console.log(det.check())         // 'balanced'
```
**评分点**：
- 记账含调用栈与时间（泄漏定位到注册点）
- check 支持 label 过滤与全量审计
- watchdog 区分「常驻订阅」与「增长泄漏」（基线 + 趋势）
- hook 包装自动平衡（组件卸载即退订记账）
