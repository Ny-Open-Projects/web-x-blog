## ZSM-447：手写状态注释系统
**需求**：实现 `annotate(store)`：1) `annotate(path, meta)` 给状态路径挂标注 `{ owner, desc, warnWhen(v) }`；2) 每次 setState 后：对涉及路径执行 warnWhen，命中则产生警告事件（带 path/value/owner）+ 不阻断写入；3) `exportDoc()` 生成「状态字典」：路径/类型/owner/描述/最近变更时间（从变更历史提取）；4) 标注随状态删除自动失效（path 不存在则跳过并标记 orphan）；5) 支持通配路径 `items.*.qty`。

```javascript
export function annotate(store) {
  const notes = new Map()        // pathPattern → meta
  const history = new Map()      // path → lastTs
  const listeners = new Set()

  function add(pathPattern, meta) { notes.set(pathPattern, meta) }
  function onWarn(fn) { listeners.add(fn); return () => listeners.delete(fn) }

  function match(pat, path) {
    const ps = pat.split('.'), xs = path.split('.')
    return ps.length === xs.length && ps.every((s, i) => s === '*' || s === xs[i])
  }

  const origSet = store.setState
  store.setState = (patch) => {
    const before = structuredClone(store.getState())
    origSet(patch)
    const after = store.getState()
    const changed = collectPaths(before, after)                  // 变更路径
    changed.forEach((p) => history.set(p, Date.now()))
    notes.forEach((meta, pat) => {
      const targets = changed.filter((p) => match(pat, p))
      targets.forEach((p) => {
        const v = getPath(after, p)
        if (getPath(before, p) === undefined && !exists(after, p))
          meta.orphan = true                                     // 标注失效标记
        if (meta.warnWhen?.(v)) listeners.forEach((fn) => fn({ path: p, value: v, owner: meta.owner }))
      })
    })
  }

  function exportDoc() {
    return [...notes.entries()].map(([pat, m]) => ({
      path: pat, owner: m.owner, desc: m.desc, orphan: !!m.orphan,
      lastChangedAt: [...history.entries()].find(([p]) => match(pat, p))?.[1] ?? null,
    }))
  }

  return { add, onWarn, exportDoc }
}

const getPath = (o, p) => p.split('.').reduce((c, s) => c?.[s], o)
function exists(o, p) { return getPath(o, p) !== undefined }
function collectPaths(a, b, path = '', out = []) {
  if (Object.is(a, b)) return out
  if (!a || !b || typeof a !== 'object' || typeof b !== 'object') return (out.push(path), out)
  for (const k of new Set([...Object.keys(a), ...Object.keys(b)]))
    collectPaths(a[k], b[k], path ? `${path}.${k}` : k, out)
  return out
}
```
**评分点**：
- 标注与写入解耦（warnWhen 告警不阻断写入——观察者语义）
- 通配路径匹配（items.*.qty）与 orphan 标记（状态删除后标注失效可见）
- 状态字典导出（owner/描述/最近变更）——文档化的自动化
- 警告事件订阅制（onWarn 返回退订），不侵入业务
