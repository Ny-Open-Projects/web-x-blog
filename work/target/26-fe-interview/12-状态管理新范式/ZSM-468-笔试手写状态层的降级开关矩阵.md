## ZSM-468：手写状态层降级开关矩阵
**需求**：实现 `createDegradationMatrix()`：1) 注册能力依赖：`declare('search', { requires: ['api', 'indexReady'], fallback: () => '静态热门列表' })`；2) 能力事件 `setAbility('api', false)`：所有依赖 api 的能力自动进入降级态（fallback 生效）；3) 依赖恢复自动升回正常态；4) 级联依赖（A requires B，B 降级则 A 也降级）闭包计算；5) `status()` 返回各能力状态与降级原因链；6) 降级/恢复事件可订阅。

```javascript
export function createDegradationMatrix() {
  const caps = new Map()       // name → { requires, fallback, deps: Set, active: true }
  const abilities = new Map()  // ability → bool
  const listeners = new Set()

  function declare(name, { requires = [], fallback = null }) {
    caps.set(name, { requires, fallback, deps: new Set(requires) })
    reevaluate(name, new Set())
  }

  function reevaluate(name, seen) {
    if (seen.has(name)) return                        // 环保护
    seen.add(name)
    const cap = caps.get(name)
    if (!cap) return
    const wasActive = cap.active
    const down = [...cap.deps].filter((d) => abilities.get(d) === false)
    // 级联：依赖的能力若对应能力声明降级也纳入
    cap.active = down.length === 0
    if (wasActive !== cap.active) {
      listeners.forEach((fn) => fn({ name, active: cap.active, reasons: down }))
      caps.forEach((c, n) => { if (c.deps.has(name) || c.requires.includes(name)) reevaluate(n, new Set([...seen])) })
    }
  }

  function setAbility(ability, ok) {
    abilities.set(ability, ok)
    caps.forEach((_, n) => reevaluate(n, new Set()))
  }

  return {
    declare, setAbility,
    use(name) {
      const cap = caps.get(name)
      if (!cap) throw new Error(`unknown capability: ${name}`)
      return cap.active ? null : { fallback: cap.fallback, reasons: [...cap.deps].filter((d) => abilities.get(d) === false) }
    },
    status: () => Object.fromEntries([...caps.entries()].map(([n, c]) => [n, { active: c.active,
      reasons: [...c.deps].filter((d) => abilities.get(d) === false) }])),
    onChange: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
  }
}
```
**评分点**：
- 依赖闭包级联（B 降 → A 降），reevaluate 递归 + 环保护
- 恢复自动升回（active 重算对称）
- use() 返回 fallback + 原因链（降级 UI 可解释）
- onChange 订阅 + status 可观测
