## ZSM-842：手写状态联邦路由器（字段归属判定 + 跨层连接校验）
**答**：把 ZSM-841 的联邦架构落成可执行校验——实现 `createFederationRouter()`：1) `classify(field)`：字段五属性判定归层（ZSM-796 判定树扩展展示态分支）；2) `validateConnection(fromLayer, toLayer, kind)`：跨层连接白名单（注入/失效/事件三 kinds），非法连接拒绝；3) `lintImports(imports)`：检查跨层直接 import 深层字段（形如 `store.state.a.b` 的 import）违规；4) 附用例：五层判定与非法连接拦截。

```javascript
const LAYERS = ['L1-server-render', 'L2-server-cache', 'L3-client-session', 'L4-flow', 'L5-display']

function createFederationRouter() {
  // 跨层连接白名单：from → to 允许的 kinds
  const CONNECTIONS = {
    'L1->L2': ['inject'],                          // RSC 渲染产物 → 缓存 hydration
    'L1->L3': ['inject'],                          // SSR 初始值 → 会话 store
    'L2->L2': ['invalidate', 'inject'],            // 缓存内部失效传播
    'L3->L2': ['invalidate', 'event'],             // store 动作 → 缓存失效
    'L3->L4': ['event'],                           // 业务事件 → 状态机 send
    'L4->L2': ['invalidate'],                      // 流程完成 → 缓存失效
    'L4->L3': ['inject'],                          // 流程产出 → 会话态
  }

  function classify({ source, updatesInClient, cacheAcrossRequests, serverComputed, isUserIntent, isFlow, isDisplayOnly }) {
    if (isDisplayOnly) return { layer: 'L5-display', reason: '展示联动：CSS/平台原生' }
    if (isFlow) return { layer: 'L4-flow', reason: '多步流程：状态机' }
    if (!isUserIntent && serverComputed && !updatesInClient) return { layer: 'L1-server-render', reason: '渲染产物：RSC 直读' }
    if (cacheAcrossRequests && source === 'server') return { layer: 'L2-server-cache', reason: '跨请求缓存' }
    if (isUserIntent && updatesInClient) return { layer: 'L3-client-session', reason: '交互会话态' }
    return { layer: 'L1-server-render', reason: '默认：尽量不上客户端' }
  }

  function validateConnection(fromLayer, toLayer, kind) {
    const key = `${fromLayer}->${toLayer}`
    const allowed = CONNECTIONS[key]
    if (!allowed) return { ok: false, reason: `${key} 无合法连接（跨层直连禁止）` }
    if (!allowed.includes(kind)) return { ok: false, reason: `${key} 仅允许 ${allowed.join('/')}` }
    return { ok: true }
  }

  function lintImports(imports) {
    const violations = []
    for (const imp of imports) {
      // 形如 from '@/stores/domain' import { cartStoreApi } + 访问 .state.a.b 的深路径
      const layer = imp.from.match(/stores\/(\w+)/)?.[1]
      if (layer && imp.access?.match(/\.(state|getState\(\))\.[\w]+\./)) {
        violations.push({ from: imp.file, target: imp.from, reason: '跨层直接 import 深层字段——改走 selector/invalidate/event' })
      }
    }
    return { violations, ok: violations.length === 0 }
  }

  return { classify, validateConnection, lintImports }
}

// 用例
const fr = createFederationRouter()
console.log(fr.classify({ source: 'server', updatesInClient: false, cacheAcrossRequests: false, serverComputed: true, isUserIntent: false }))
// { layer: 'L1-server-render', ... }
console.log(fr.classify({ source: 'local', updatesInClient: true, cacheAcrossRequests: false, serverComputed: false, isUserIntent: true }))
// { layer: 'L3-client-session', ... }
console.log(fr.classify({ isDisplayOnly: true }))
// { layer: 'L5-display', ... }

console.log(fr.validateConnection('L3-client-session', 'L2-server-cache', 'invalidate'))   // ok
console.log(fr.validateConnection('L3-client-session', 'L1-server-render', 'event'))
// { ok: false, reason: 'L3->L1 无合法连接（跨层直连禁止）' }
console.log(fr.validateConnection('L2-server-cache', 'L3-client-session', 'inject'))
// { ok: false, reason: 'L2->L3 仅允许...' }

console.log(fr.lintImports([
  { file: 'a.tsx', from: '@/stores/domain', access: 'cartStoreApi.state.items[0].price' },
]).violations.length)   // 1：深层字段直连违规
```
**评分点**：
- 五层判定树（display/flow 优先，渲染产物归服务端）
- 连接白名单（三 kinds 注入/失效/事件，跨层直连拒绝）
- lintImports 深层字段检测（正则识别 .state.x.y 访问）
- 错误信息含合法 kinds（可行动）
