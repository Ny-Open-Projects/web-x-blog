## ZSM-891：手写面试剧本卡检索器（阶段→策略→应急路由）
**答**：890 落地——实现 `createScriptCards()`：1) `register(phase, strategies, contingencies)`：登记阶段剧本卡；2) `route(phase, signal)`：按临场信号路由策略（signal: 'normal'|'impatient'|'deep'|'challenge'）——impatient 切短答、deep 展开深度、challenge 先认口径；3) `timeline(minutes)`：输出 60 分钟阶段时间线（含每阶段时长与卡号）；4) 附用例：信号路由与超时应急。

```javascript
function createScriptCards() {
  const cards = new Map()   // phase -> { strategies: Map<signal, fn>, duration }

  function register(phase, { duration, strategies }) {
    cards.set(phase, { duration, strategies: new Map(Object.entries(strategies)) })
    return this
  }

  function route(phase, signal = 'normal', context = {}) {
    const card = cards.get(phase)
    const strategy = card?.strategies.get(signal) ?? card?.strategies.get('normal')
    if (!strategy) return { action: 'free talk', detail: null }
    return strategy(context)
  }

  function timeline(totalMinutes = 60) {
    const phases = [...cards.entries()]
    const totalWeight = phases.reduce((a, [, c]) => a + c.duration, 0)
    let cursor = 0
    return phases.map(([phase, card], i) => {
      const mins = Math.round(totalMinutes * card.duration / totalWeight)
      const seg = { index: i + 1, phase, start: cursor, end: cursor + mins, card: i + 1 }
      cursor += mins
      return seg
    })
  }

  return { register, route, timeline }
}

// 用例
const sc = createScriptCards()
sc.register('自我介绍', { duration: 5, strategies: {
  normal: () => ({ action: '埋两条叙事钩子', detail: '联邦化 + 门禁从0到1（845）' }),
} })
sc.register('项目深挖', { duration: 15, strategies: {
  normal: () => ({ action: '主线一四段式', detail: '801 骨架 + 留钩子' }),
  challenge: (ctx) => ({ action: '先认口径再辩内容', detail: `出示 ${ctx.metric || 'INP'} 的口径三要素（815）` }),
} })
sc.register('原理题', { duration: 15, strategies: {
  normal: () => ({ action: '三层法', detail: 'L1/L2/L3（803）' }),
  deep: () => ({ action: '展开段内深度', detail: 'L2 变量名证据 + L3 多反例' }),
} })
sc.register('设计题', { duration: 15, strategies: {
  normal: () => ({ action: '五段骨架+四图层', detail: '802/823' }),
  impatient: () => ({ action: '切短答模式', detail: '每答 30s 内 + 结论先行（808 止损）' }),
} })
sc.register('反问', { duration: 10, strategies: {
  normal: () => ({ action: '问治理现状/健康分', detail: '暗示治理经验' }),
} })

// 路由
console.log(sc.route('项目深挖', 'challenge', { metric: 'INP' }))
// { action: '先认口径再辩内容', detail: '出示 INP 的口径三要素（815）' }
console.log(sc.route('设计题', 'impatient'))
// { action: '切短答模式', ... }
console.log(sc.timeline().map((t) => `${t.phase}:${t.start}-${t.end}min(卡${t.card})`))
// 自我介绍:0-5min(卡1) 项目深挖:5-20min(卡2) ...
```
**评分点**：
- 信号路由（normal/impatient/deep/challenge 四信号策略分发）
- 缺策略 fallback（回 normal）
- timeline 权重分配时长（duration 加权）
- 阶段卡编号（与剧本卡索引对应）
