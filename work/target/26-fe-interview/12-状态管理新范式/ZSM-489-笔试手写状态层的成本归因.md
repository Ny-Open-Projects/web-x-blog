## ZSM-489：手写状态层成本归因
**需求**：实现 `createCostAttributor()`：1) `track(domain, { requests, bytes, ms })`：按业务域累计资源消耗；2) `attribute()`：把总成本按「资源权重」分摊到各域（请求 0.6 / 流量 0.3 / 耗时 0.1 权重可配）；3) 输出排序的成本榜（top 域与占比）+ 与上周期环比（涨跌标记）；4) `snapshot()` 保存周期基线（对比用）；5) 异常域检测：单域占比 > 40% 标记「成本热点」建议治理。

```javascript
export function createCostAttributor({ weights = { requests: 0.6, bytes: 0.3, ms: 0.1 } } = {}) {
  let current = new Map()      // domain → { requests, bytes, ms }
  let baseline = null

  function track(domain, m) {
    const cur = current.get(domain) ?? { requests: 0, bytes: 0, ms: 0 }
    cur.requests += m.requests ?? 0
    cur.bytes += m.bytes ?? 0
    cur.ms += m.ms ?? 0
    current.set(domain, cur)
  }

  function normalize(m, totals) {
    return weights.requests * (m.requests / (totals.requests || 1))
      + weights.bytes * (m.bytes / (totals.bytes || 1))
      + weights.ms * (m.ms / (totals.ms || 1))
  }

  function attribute() {
    const totals = { requests: 0, bytes: 0, ms: 0 }
    current.forEach((m) => { totals.requests += m.requests; totals.bytes += m.bytes; totals.ms += m.ms })
    const rows = [...current.entries()].map(([domain, m]) => ({
      domain, cost: normalize(m, totals), raw: m,
      hot: normalize(m, totals) > 0.4,                       // 成本热点
    })).sort((a, b) => b.cost - a.cost)
    // 环比
    if (baseline) rows.forEach((r) => {
      const prev = baseline.find((b) => b.domain === r.domain)
      r.delta = prev ? r.cost - prev.cost : r.cost
    })
    return rows
  }

  return {
    track,
    attribute,
    snapshot() { baseline = attribute() },
    report() {
      const rows = attribute()
      const top = rows[0]
      return { rows, topDomain: top?.domain, hotspots: rows.filter((r) => r.hot).map((r) => r.domain) }
    },
  }
}
```
**评分点**：
- 权重分摊模型（多资源归一到成本占比，权重可配）
- 环比基于 snapshot 基线（周期对比而非无限累计）
- 成本热点检测（单域 > 40% 标记）——归因的目的是可行动
- 排序输出（成本榜驱动治理优先级）
