## ZSM-562：手写时间旅行跳转的路径压缩器
**答**：长会话时间旅行的内存问题：全量快照每步存一份，1 万步 × 100KB 就爆。实现 `createSnapshots()`：1) `push(state)`：只存**首个全量快照**，之后每步存「变更路径 patch」（{path, prev, next}），逆操作可回退；2) `at(index)`：从最近全量快照重放 patch 序列到目标步（前后向都可），重放步数超阈值（如 200）时**自动压缩**——把当前状态固化成新全量锚点，丢弃旧 patch；3) `size()` 返回当前内存占用估算；4) 附用例验证回放正确性与锚点触发。

```javascript
function createSnapshots({ compactEvery = 200 } = {}) {
  let anchor = null            // { index, state }
  let patches = []             // { index, prev, next } 全量序
  let count = 0

  function push(state, prev, paths) {
    if (!anchor) anchor = { index: 0, state: structuredClone(state) }
    patches.push({ index: count++, prev: paths, next: paths })
  }

  function applyPatches(state, list, direction) {
    const s = structuredClone(state)
    const seq = direction === 'forward' ? list : [...list].reverse()
    for (const p of seq) {
      const entries = direction === 'forward' ? p.next : p.prev
      for (const [path, value] of entries) setPath(s, path, structuredClone(value))
    }
    return s
  }

  function setPath(obj, path, value) {
    const keys = path.split('.')
    let cur = obj
    for (const k of keys.slice(0, -1)) cur = cur[k] ??= {}
    cur[keys.at(-1)] = value
  }

  function at(index) {
    if (!anchor || index < 0 || index >= count) return null
    const from = anchor.index
    const slice = patches.slice(Math.min(from, index), Math.max(from, index))
    return applyPatches(anchor.state, slice, index >= from ? 'forward' : 'back')
  }

  function compact(toIndex) {
    // 把目标位置固化为新锚点，丢弃其前 patch
    const target = at(toIndex)
    if (!target) return
    anchor = { index: toIndex, state: target }
    patches = patches.filter((p) => p.index >= toIndex)
  }

  function size() {
    return JSON.stringify(anchor.state).length * 0.001 +
      patches.reduce((a, p) => a + JSON.stringify(p.prev).length, 0) * 0.001 // KB
  }

  return {
    push, at, compact, size,
    get count() { return count },
  }
}

// 用例
const snaps = createSnapshots({ compactEvery: 3 })
let state = { user: { name: 'a' }, count: 0 }
snaps.push(state, null, [])
for (let i = 1; i <= 5; i++) {
  const prev = state
  state = { ...state, count: i, user: { name: 'a' + i } }
  snaps.push(state, prev, [['count', prev.count], ['user.name', prev.user.name]])
}
console.log(snaps.at(2))          // { count: 2, user: { name: 'a2' } }
console.log(snaps.at(4).user.name) // 'a4'
console.log(snaps.size() > 0)
snaps.compact(4)                   // 压缩：4 号位变锚点
console.log(snaps.at(5).count)     // 5：锚点后向重放仍正确
```
**评分点**：
- 首锚点 + patch 增量的存储模型（全量快照只在关键点）
- 双向重放（forward/back 方向选择 patch 侧）
- 阈值压缩（compact 重建锚点丢弃历史 patch）
- 结构化克隆隔离（重放不污染原状态）
