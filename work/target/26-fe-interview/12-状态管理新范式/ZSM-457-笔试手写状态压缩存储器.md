## ZSM-457：手写状态压缩存储器
**答**：实现 `createCompactStore(storage)`：1) `save(key, state)`：与上次保存值做差量（diff 补丁），存储 `{ base, patches: [...] }`；2) 补丁链超过 10 条或累计体积超过全量 60% 时「重置基线」（全量重存，防补丁链膨胀）；3) 字典编码：重复字符串值入字典表（id 替换），读取时还原；4) `load(key)` 逆序还原（base + 依次 apply）；5) 存储体积统计 `size(key)`（压缩前后对比）。

```javascript
export function createCompactStore(storage, { maxPatches = 10 } = {}) {
  const dict = new Map()       // string → id

  function encode(v, out) {
    if (typeof v === 'string' && v.length > 8) {
      if (!dict.has(v)) { const id = `#${dict.size}`; dict.set(id, v); out.dict[id] = v }
      return { __d: [...dict.entries()].find(([id, s]) => s === v)?.[0] }
    }
    if (Array.isArray(v)) return v.map((x) => encode(x, out))
    if (v && typeof v === 'object')
      return Object.fromEntries(Object.entries(v).map(([k, x]) => [k, encode(x, out)]))
    return v
  }

  function save(key, state) {
    const prev = storage.get(key)
    let record
    const encoded = encode(state, { dict: {} })
    if (!prev || prev.patches.length >= maxPatches) {
      record = { base: encoded, patches: [] }                       // 重置基线
    } else {
      record = { base: prev.base, patches: [...prev.patches, diff(prev.current, encoded)] }
      if (record.patches.length * 10 > JSON.stringify(encoded).length * 6)   // 超 60% 重置
        record = { base: encoded, patches: [] }
    }
    record.current = encoded
    storage.set(key, record)
    return JSON.stringify(record).length
  }

  function load(key) {
    const r = storage.get(key)
    if (!r) return null
    let s = decode(r.base)
    r.patches.forEach((p) => (s = applyPatch(s, p)))
    return s
  }

  const decode = (v) => v && v.__d ? dict.get(v.__d)
    : Array.isArray(v) ? v.map(decode)
    : v && typeof v === 'object' ? Object.fromEntries(Object.entries(v).map(([k, x]) => [k, decode(x)]))
    : v

  return { save, load, size: (key) => ({ raw: JSON.stringify(storage.get(key)?.current ?? {}).length,
    stored: JSON.stringify(storage.get(key) ?? {}).length }) }
}
```
**评分点**：
- 差量链 + 重置基线策略（链长上限 + 体积比双重触发，防补丁链劣化）
- 字典编码针对重复字符串（长字符串场景收益最大），解码对称
- current 字段缓存最近值（下次 diff 的基准，避免重复重放）
- size 压缩前后对比可观测（收益可量化）
