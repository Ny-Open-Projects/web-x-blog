## ZSM-498：手写状态层发布预检器
**需求**：实现 `createReleasePreflight(checks)`：1) checks：发布前必过项数组 `{ name, run: async () => { ok, detail } , blocking }`（blocking=false 仅警告）；2) `preflight()`：并行跑全部检查（互相独立），汇总 { passed, warnings, failed }；3) 有 blocking 失败 → 拒绝发布（exit 非 0，CI 集成）；4) 检查项超时（30s）计失败（防挂死流水线）；5) 结果缓存：同 commit 短路复用（重复推送不重跑）；6) 输出人类可读报告。

```javascript
export function createReleasePreflight(checks, { timeoutMs = 30_000 } = {}) {
  const cache = new Map()      // commit → result

  async function runOne(check) {
    try {
      const r = await Promise.race([
        check.run(),
        new Promise((_, rej) => setTimeout(() => rej(new Error('timeout')), timeoutMs)),
      ])
      return { name: check.name, ok: r.ok, detail: r.detail, blocking: check.blocking ?? true, timedOut: false }
    } catch (e) {
      return { name: check.name, ok: false, detail: String(e), blocking: check.blocking ?? true, timedOut: true }
    }
  }

  async function preflight(commitSha) {
    if (cache.has(commitSha)) return cache.get(commitSha)     // 同 commit 短路
    const results = await Promise.all(checks.map(runOne))     // 并行独立
    const failed = results.filter((r) => !r.ok && r.blocking)
    const warnings = results.filter((r) => !r.ok && !r.blocking)
    const report = {
      passed: failed.length === 0,
      failed, warnings,
      md: [
        `# 发布预检`,
        ...results.map((r) => `- [${r.ok ? 'PASS' : r.blocking ? 'FAIL' : 'WARN'}] ${r.name}: ${r.detail}`),
        failed.length ? `\n发布被阻止：${failed.map((f) => f.name).join(', ')}` : '\n可以发布',
      ].join('\n'),
    }
    cache.set(commitSha, report)
    return report
  }

  return { preflight }
}
```
**评分点**：
- blocking 分级（阻断 vs 警告——不是所有失败都该拦发布）
- 并行独立执行 + 单项超时计失败（防挂死流水线）
- 同 commit 结果缓存（重复推送短路）
- 双输出（结构化 failed/warnings + 人类可读 md）
