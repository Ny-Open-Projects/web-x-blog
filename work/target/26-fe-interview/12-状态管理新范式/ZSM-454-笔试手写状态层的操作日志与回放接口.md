## ZSM-454：手写操作日志与回放接口
**需求**：实现 `createOpLog(store, applyPatch)`：1) `log(patch, meta)` 记录操作（环形 2000 条，带序号与时间）；2) `export()` 导出 `{ v, headSeq, baseline, ops }`（baseline 为最早条目前的快照）；3) `import(data)` 导入校验：版本匹配、序号连续（gap 检测）、patch 可解析；校验失败拒绝导入；4) `replayTo(seq)`：从 baseline 重建到指定序号（快照 + 前缀重放）；5) 回放确定性要求：patch 必须确定性（注册期校验 reject 非确定性 patch 类型）。

```javascript
export function createOpLog(store, applyPatch, { ring = 2000 } = {}) {
  let seq = 0
  let baseline = structuredClone(store.getState())
  let ops = []                 // { seq, patch, ts }
  let beforeBaseline = baseline

  function log(patch, meta = {}) {
    if (!isDeterministic(patch)) throw new Error('non-deterministic patch rejected')
    seq += 1
    ops.push({ seq, patch, ts: Date.now(), meta })
    if (ops.length > ring) {                        // 环形：最旧出队 → 新 baseline
      const dropped = ops.shift()
      beforeBaseline = replayAll(beforeBaseline, [dropped])
      baseline = structuredClone(beforeBaseline)
    }
    applyPatch(store, patch)
    return seq
  }

  function export_() {
    return { v: 1, headSeq: seq, baseline, ops: [...ops] }
  }

  function import_(data) {
    if (data.v !== 1) return { ok: false, reason: 'version' }
    const first = data.ops[0]?.seq ?? data.headSeq + 1
    if (data.ops.some((o, i) => o.seq !== first + i)) return { ok: false, reason: 'gap' }   // 序号连续
    if (data.ops.some((o) => !isDeterministic(o.patch))) return { ok: false, reason: 'nondeterministic' }
    baseline = data.baseline; ops = data.ops; seq = data.headSeq
    return { ok: true }
  }

  function replayTo(target) {
    let s = structuredClone(baseline)
    for (const o of ops) { if (o.seq > target) break; s = applyPatch({ state: s }, o.patch).state }
    return s
  }

  return { log, export: export_, import: import_, replayTo, get headSeq() { return seq } }
}

function isDeterministic(patch) {
  const s = JSON.stringify(patch, (k, v) =>
    typeof v === 'function' ? undefined : v)
  return !s.includes('"__now"') && !/function|Date\.now/.test(s)   // 简化校验
}
function replayAll(state, ops) { let s = state; for (const o of ops) s = applyPatch({ state: s }, o.patch).state; return s }
```
**评分点**：
- 环形出队时滚动 baseline（导出完整性：baseline + ops = 完整历史）
- 导入三重校验（版本/序号连续 gap/确定性）——回放可信的前提
- replayTo 目标序号前缀重放（O(分段)）
- 非确定性 patch 注册期拒绝（回放确定性的源头治理）
