## ZSM-422：手写带权限的状态访问层
**需求**：实现 `guardStore(store, rules)`：1) rules 定义每字段的读权限 `read: string[]`（角色码）与写权限 `write: string[]`；2) `can(role, field, 'read'|'write')` 判定；3) `view(role)` 返回按读权限裁剪的视图（无权字段不出现，不是 undefined——防止字段探测）；4) `write(role, patch)` 写权限校验：无权字段整个写入拒绝（带审计日志），有权部分成功；5) 未声明规则的字段默认私有（不可读不可写）。

```javascript
export function guardStore(store, rules) {
  const audit = []

  function can(role, field, action) {
    const r = rules[field]
    if (!r) return false                                    // 未声明默认私有
    return r[action]?.includes(role) ?? false
  }

  function view(role) {
    const s = store.getState()
    const out = {}
    for (const field of Object.keys(rules)) {
      if (can(role, field, 'read')) out[field] = s[field]
    }
    return out                                              // 无权字段不出现（防探测）
  }

  function write(role, patch) {
    const denied = Object.keys(patch).filter((f) => !can(role, f, 'write'))
    if (denied.length) {
      audit.push({ role, denied, ts: Date.now() })          // 审计
      throw new Error(`denied: ${denied.join(',')}`)
    }
    store.setState(patch)
    return true
  }

  return { view, write, can, getAudit: () => audit }
}

// rules
const rules = {
  nickname: { read: ['self', 'admin'], write: ['self'] },
  salary: { read: ['admin'], write: ['admin'] },
}
```
**评分点**：
- 无权字段「不出现」而非 undefined（防字段名探测）
- 默认私有（白名单模型，漏配置即拒绝）
- 写权限整体拒绝 + 审计日志（who/what/when）
- 读裁剪是投影层不删源数据（store 原始数据完整，权限在出口处执行）
