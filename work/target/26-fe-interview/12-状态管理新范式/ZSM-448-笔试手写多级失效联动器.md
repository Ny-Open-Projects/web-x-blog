## ZSM-448：手写多级失效联动器
**需求**：实现 `createInvalidationGraph(qc)`：1) `register(action, { affects, level })`：写操作登记影响域（key 前缀数组）与失效级别（'refetch'|'mark'|'clear'）；2) `run(action)`：计算影响闭包（affects 可引用其他 action 形成 DAG，拓扑展开）→ 按级别执行：refetch=失效+重拉、mark=只标 stale、clear=removeQueries；3) DAG 环检测（注册期抛错）；4) 同 action 重复 run 合并（100ms 窗口内只执行一次，取最高级别）；5) 执行报告（每个域的级别与耗时）。

```javascript
export function createInvalidationGraph(qc) {
  const nodes = new Map()      // action → { affects, level }
  let pending = null, timer = null

  function register(action, { affects, level = 'refetch' }) {
    detectCycle(action, affects)
    nodes.set(action, { affects, level })
  }

  function detectCycle(root, affects, stack = new Set()) {
    for (const a of affects) {
      if (a === root || stack.has(a)) throw new Error(`invalidation cycle: ${root}`)
      const n = nodes.get(a)
      if (n) detectCycle(root, n.affects, new Set([...stack, a]))
    }
  }

  const LEVEL = { mark: 1, refetch: 2, clear: 3 }

  function run(action) {
    pending ??= { actions: new Set(), level: 0 }
    const closure = expand(action, new Set())                 // DAG 闭包
    closure.forEach((a) => {
      pending.actions.add(a)
      pending.level = Math.max(pending.level, LEVEL[nodes.get(a)?.level ?? 'refetch'])
    })
    if (!timer) timer = setTimeout(flush, 100)
  }

  function expand(action, seen) {
    if (seen.has(action)) return seen
    seen.add(action)
    nodes.get(action)?.affects.forEach((a) => expand(a, seen))
    return seen
  }

  async function flush() {
    timer = null
    const { actions, level } = pending
    pending = null
    const report = []
    for (const a of actions) {
      const { affects } = nodes.get(a)
      const t0 = performance.now()
      affects.forEach((key) => {
        if (level >= 3) qc.removeQueries({ queryKey: keyOf(key) })
        else if (level >= 2) qc.invalidateQueries({ queryKey: keyOf(key) })
        else qc.getQueryCache().findAll({ queryKey: keyOf(key) }).forEach((q) => (q.isStale = true))
      })
      report.push({ action: a, level, ms: performance.now() - t0 })
    }
    return report
  }

  return { register, run }
}
const keyOf = (k) => Array.isArray(k) ? k : [k]
```
**评分点**：
- DAG 闭包展开 + 注册期环检测（联动失效的结构正确性）
- 级别合并取最高（同窗 mark+clear → clear），窗口合并防风暴
- 三级执行语义映射正确（mark 只标 stale 不发请求）
- report 可观测（每域级别/耗时）；pending 先清后执行（可重入）
