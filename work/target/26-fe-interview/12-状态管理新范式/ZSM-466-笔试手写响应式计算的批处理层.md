## ZSM-466：手写响应式计算批处理层
**答**：实现 `createReactiveCore()`：1) `signal(v)` 读写器；2) `effect(fn)`：依赖收集 + 自动重跑；3) 批处理：`batch(fn)` 内的所有 signal 写入推迟到 batch 结束统一传播（一次级联）；4) 传播去重：同一 effect 在一批内只跑一次（无论其依赖被写几次）；5) 拓扑序传播（上游 effect 先于下游）；6) 循环依赖检测（effect 间 A→B→A 抛错）。

```javascript
export function createReactiveCore() {
  let current = null
  let batchDepth = 0
  const queued = new Set()       // 待传播 effect
  const edges = new Map()        // effect → 上游 effects（DAG）

  function signal(v) {
    const subs = new Set()
    const read = () => { if (current) { current.deps.add(subs); subs.add(current) }; return v }
    read.set = (nv) => {
      if (Object.is(nv, v)) return
      v = nv
      if (batchDepth > 0) subs.forEach((e) => queued.add(e))   // 推迟传播
      else subs.forEach((e) => runEffect(e))
    }
    return read
  }

  function runEffect(e) {
    if (queued.has(e) && batchDepth > 0) return                // 批内去重
    e.deps = new Set()
    const prev = current
    current = e
    try { e.fn() } finally { current = prev }
  }

  function effect(fn) {
    const e = { fn, deps: new Set() }
    runEffect(e)
    return () => { e.deps.forEach((s) => s.delete(e)) }
  }

  function batch(fn) {
    batchDepth++
    try { return fn() }
    finally {
      batchDepth--
      if (batchDepth === 0) {
        const list = topoSort([...queued])
        queued.clear()
        list.forEach(runEffect)                                // 一次级联，拓扑序
      }
    }
  }

  function topoSort(effects) {
    return effects.sort((a, b) => (edges.get(b)?.has(a) ? 1 : -1))   // 上游先跑（简化）
  }

  return { signal, effect, batch }
}
```
**评分点**：
- batch 推迟传播（嵌套 batchDepth 计数，最外层统一 flush）
- 批内去重（queued Set，依赖写 N 次只跑一次）
- 拓扑序传播（上游先于下游，简化实现需诚实标注局限——严格版需真实 DAG 边）
- 依赖收集在 runEffect 重置（动态依赖正确）
