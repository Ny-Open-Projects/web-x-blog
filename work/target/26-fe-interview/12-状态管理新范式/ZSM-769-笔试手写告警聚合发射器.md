## ZSM-769：手写告警聚合发射器（分级阈值 + 上下文采样 + 风暴防御）
**答**：实现 `createAlertEmitter({ thresholds, sampleSize, windowMs })`：1) `emit(level, key, ctx)`：按 level 阈值聚合——P0 单条即发、P1 聚合 N 条、P2 计数但不打扰；2) 上下文采样：窗口内最多保留 sampleSize 条 ctx；3) 风暴防御：发出后窗口重置；窗口过期（windowMs）自动清零；4) `send` 可注入（测试友好）；5) `pending()`：未达阈值的聚合窗口清单；6) 附用例：P0 单发、P1 聚合发、窗口过期重置。

```javascript
function createAlertEmitter({ thresholds = { P0: 1, P1: 3, P2: 100 }, sampleSize = 3, windowMs = 60000, send = console.log } = {}) {
  const windows = new Map()      // key -> { level, count, firstTs, samples }

  function emit(level, key, ctx, now = Date.now()) {
    // 窗口过期清理
    for (const [k, w] of windows) {
      if (now - w.firstTs > windowMs) windows.delete(k)
    }
    let w = windows.get(key)
    if (!w || w.level !== level) {
      w = { level, count: 0, firstTs: now, samples: [] }
      windows.set(key, w)
    }
    w.count++
    if (w.samples.length < sampleSize) w.samples.push(ctx)    // 采样
    const threshold = thresholds[level] ?? Infinity
    if (w.count >= threshold) {
      fire(key, w, now)
      windows.delete(key)                                     // 风暴防御：重置
      return { fired: true }
    }
    return { fired: false }
  }

  function fire(key, w, now) {
    send({
      level: w.level, key,
      affectedSessions: w.count,
      windowMs: now - w.firstTs,
      samples: w.samples,                                     // 上下文包
    })
  }

  function pending() {
    return [...windows.entries()].map(([key, w]) => ({ key, level: w.level, count: w.count }))
  }

  return { emit, pending }
}

// 用例
const sent = []
const emitter = createAlertEmitter({
  thresholds: { P0: 1, P1: 3, P2: 100 },
  sampleSize: 2,
  windowMs: 5000,
  send: (a) => sent.push(a),
})

// P0：单条即发
emitter.emit('P0', 'order-stuck', { traceId: 't1', state: 'submitting' })
console.log(sent.length)                    // 1
console.log(sent[0].affectedSessions)       // 1

// P1：聚合 3 条发
emitter.emit('P1', 'state-storm', { sessionId: 'a' })
emitter.emit('P1', 'state-storm', { sessionId: 'b' })
console.log(emitter.pending())              // [{ key: 'state-storm', count: 2 }]：未达阈值
emitter.emit('P1', 'state-storm', { sessionId: 'c' })
console.log(sent.length)                    // 2：聚合发出
console.log(sent[1].samples.length)         // 2：采样上限
console.log(sent[1].affectedSessions)       // 3

// 窗口过期重置（虚拟时间推进）
emitter.emit('P2', 'selector-degraded', { label: 'x' })
await new Promise((r) => setTimeout(r, 5100))
console.log(emitter.pending().length)       // 0：窗口过期清零
```
**评分点**：
- 分级阈值（P0 单发 / P1 聚合 / P2 静默计数）
- 上下文采样上限（samples 不随 count 爆炸）
- 风暴防御（发出后窗口重置）+ 过期自动清零
- send 可注入（测试与接入解耦）
