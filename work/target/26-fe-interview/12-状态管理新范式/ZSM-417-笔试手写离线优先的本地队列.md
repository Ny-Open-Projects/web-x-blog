## ZSM-417：手写离线优先本地队列（持久化 + 顺序重放）
**需求**：实现 `createOfflineQueue(persist)`：1) `enqueue(op)`：离线时入队（persist 保存，恢复后可重放），在线直接执行；2) `flush()` 顺序重放队列：前一个成功才发下一个（顺序敏感操作）；3) 单个 op 失败重试 2 次后跳过并标记 dead（不阻塞后续）；4) 网络恢复事件触发 flush；5) `size()`/`dead` 可观测；6) 页面刷新后队列仍在（persist 恢复）。

```javascript
export function createOfflineQueue(persist, { retries = 2 } = {}) {
  let queue = persist.load() ?? []            // [{ id, op, tries }]
  let dead = []
  let flushing = false

  async function run(op) {
    for (let i = 0; i <= retries; i++) {
      try { await op.exec(); return true } catch (e) { if (i === retries) return false; await sleep(2 ** i * 500) }
    }
  }

  async function flush() {
    if (flushing || !queue.length) return
    flushing = true
    try {
      while (queue.length) {
        const item = queue[0]
        const ok = await run(item.op)
        if (ok) { queue.shift(); persist.save(queue) }
        else { dead.push(queue.shift()); persist.save(queue); persist.saveDead(dead) }  // 跳过不阻塞
      }
    } finally { flushing = false }
  }

  return {
    async enqueue(op) {
      if (navigator.onLine) { const ok = await run({ ...op, exec: () => op.exec() })
        if (!ok) { dead.push({ op }); persist.saveDead(dead) }
        return ok }
      queue.push({ id: crypto.randomUUID(), op, tries: 0 })
      persist.save(queue)
    },
    flush,
    size: () => queue.length,
    getDead: () => dead,
  }
}

const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
window.addEventListener('online', () => engine.flush())   // 网络恢复触发
```
**评分点**：
- 持久化恢复（刷新后队列仍在）；顺序敏感的串行重放
- 失败重试上限后转 dead 队列（不无限阻塞后续 op）
- flushing 锁防并发 flush（网络恢复 + 手动同时触发）
- online 事件监听触发；enqueue 在线直接执行失败也进 dead（不静默丢）
