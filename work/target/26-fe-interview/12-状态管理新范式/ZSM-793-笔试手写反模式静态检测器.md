## ZSM-793：手写反模式静态检测器（布尔爆炸/深路径/全局 mutation）
**答**：实现 `detectAntiPatterns(storeDef)`（输入结构化代码描述，聚焦检测算法）：1) `detectBooleanExplosion(exprs)`：统计同一组件描述内「≥3 个状态布尔组合判断」的出现次数；2) `detectDeepPaths(accesses)`：访问路径深度 ≥ 4 的硬编码（排除已封装 selector 白名单）；3) `detectGlobalMutation(ops)`：未走 setState 的外部引用修改（`arr.push/splice` 作用于 store 字段引用）；4) `report()`：坏味道清单 + 严重度（组合检出=死亡三件套升级）；5) 附用例：三坏味道检出与三件套判定。

```javascript
function createDetector() {
  const findings = []

  function detectBooleanExplosion(exprs, { threshold = 3 } = {}) {
    for (const [component, expr] of Object.entries(exprs)) {
      // 统计组合布尔判断中的不同状态布尔数量
      const bools = new Set(expr.match(/is[A-Z]\w+|has[A-Z]\w+|should[A-Z]\w+/g) ?? [])
      if (bools.size >= threshold) {
        findings.push({ pattern: 'boolean-explosion', target: component, detail: [...bools], severity: 'high' })
      }
    }
  }

  function detectDeepPaths(accesses, { whitelist = [], maxDepth = 4 } = {}) {
    for (const [file, paths] of Object.entries(accesses)) {
      for (const p of paths) {
        const depth = p.split('.').length
        if (depth >= maxDepth && !whitelist.some((w) => p.startsWith(w))) {
          findings.push({ pattern: 'deep-path', target: `${file}:${p}`, severity: 'medium' })
        }
      }
    }
  }

  function detectGlobalMutation(ops) {
    const MUTATORS = ['push', 'splice', 'pop', 'shift', 'sort', 'reverse']
    for (const [file, list] of Object.entries(ops)) {
      for (const op of list) {
        // 形如 storeState.items.push(...) 且未包 setState
        if (MUTATORS.some((m) => op.expr.includes(`.${m}(`)) && !op.inSetter) {
          findings.push({ pattern: 'global-mutation', target: `${file}:${op.expr}`, severity: 'high' })
        }
      }
    }
  }

  function report() {
    const byPattern = new Map()
    findings.forEach((f) => byPattern.set(f.pattern, (byPattern.get(f.pattern) ?? 0) + 1))
    // 死亡三件套判定：布尔爆炸+上帝store(简化为多high)+回调级联 同时存在
    const hasExplosion = findings.some((f) => f.pattern === 'boolean-explosion')
    const hasMutation = findings.some((f) => f.pattern === 'global-mutation')
    const cascade = findings.some((f) => f.pattern === 'cascade')
    return {
      findings, byPattern: Object.fromEntries(byPattern),
      deathTrio: hasExplosion && hasMutation && cascade,
      priority: hasExplosion && hasMutation ? '重构优先于新功能' : '常规',
    }
  }

  return { detectBooleanExplosion, detectDeepPaths, detectGlobalMutation, report }
}

// 用例
const d = createDetector()
d.detectBooleanExplosion({
  OrderCard: 'if (isLoading && isSubmitting && hasError && shouldRetry) {...}',
})
d.detectDeepPaths({
  'UserList.tsx': ['state.user.profile.settings.theme.value', 'state.cart.items'],
}, { whitelist: ['state.cart'] })
d.detectGlobalMutation({
  'Cart.tsx': [{ expr: 'storeState.items.push(newItem)', inSetter: false },
               { expr: 'draft.items.push(x)', inSetter: true }],   // setter 内合法
})
d.detectBooleanExplosion({ Editor: 'if (isLoading && isSaving && hasDraft) {}' })   // 级联模拟

const r = d.report()
console.log(r.byPattern)         // { 'boolean-explosion': 2, 'deep-path': 1, 'global-mutation': 1 }
console.log(r.deathTrio)         // false（缺 cascade 实证，此处演示判定逻辑）
console.log(r.priority)          // '重构优先于新功能'（explosion+mutation 同在）
```
**评分点**：
- 布尔爆炸检测（is/has/should 前缀布尔集合计数 + 阈值）
- 深路径检测（深度阈值 + 白名单豁免）
- 全局 mutation 检测（mutator 方法名 + inSetter 豁免——setter 内合法）
- 死亡三件套判定与重构优先级输出
