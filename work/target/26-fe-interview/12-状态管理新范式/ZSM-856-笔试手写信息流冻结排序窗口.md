## ZSM-856：手写信息流状态引擎（冻结排序窗口 + 互动分离 + 内存红线）
**答**：855 落地——实现 `createFeedEngine()`：1) `loadMore(cursor)`：append 幂等（695）+ 超过内存上限（如 500 条）卸载头部页（保留 anchor 游标）；2) `like(id)`：乐观 +1 互动 Map（不触发流重排）+ 超时回滚对账；3) `visibleWindow(start, end)`：可视区排序冻结——「热度分」变更只影响未冻结区；4) `restore(anchor)`：从 anchor 游标恢复（滚动位置恢复 772）；5) 附用例：可视区冻结与互动分离验证。

```javascript
function createFeedEngine({ maxItems = 500, likeTimeoutMs = 3000 } = {}) {
  const items = new Map()       // id -> { id, score, likes }
  const order = []              // 冻结的展示序
  const interactions = new Map() // id -> { likes: 本地增量, pending }
  const listeners = new Set()
  let anchor = null             // 头部 anchor（卸载后恢复用）
  let frozenRange = null        // { start, end, snapshot: id[] }

  function loadMore(cursor, page) {
    const seen = new Set(order)
    const fresh = page.filter((i) => !seen.has(i.id))
    fresh.forEach((i) => { items.set(i.id, { ...i, likes: i.likes ?? 0 }); order.push(i.id) })
    // 内存红线：卸载头部页（保留 anchor 游标）
    if (order.length > maxItems) {
      const remove = order.splice(0, order.length - maxItems)
      remove.forEach((id) => items.delete(id))
      anchor = cursor ?? anchor      // anchor 供服务端恢复（ZSM-772 会话恢复）
    }
    listeners.forEach((fn) => fn(view()))
    return { added: fresh.length }
  }

  // 互动：乐观 + 独立 Map（不触发流重排）
  const timers = new Map()
  function like(id) {
    const cur = interactions.get(id) ?? { delta: 0 }
    cur.delta++
    interactions.set(id, { ...cur, pending: true })
    const prev = cur.delta
    const timer = setTimeout(async () => {
      try {
        const res = await api.like(id, { delta: prev })
        interactions.set(id, { delta: res.likes - (items.get(id)?.likes ?? 0), pending: false })  // 对账
        items.set(id, { ...items.get(id), likes: res.likes })
      } catch {
        interactions.set(id, { delta: prev - prev, pending: false })   // 回滚
      }
      listeners.forEach((fn) => fn(view()))
    }, likeTimeoutMs)
    timers.set(id, timer)
    listeners.forEach((fn) => fn(view()))
  }

  // 可视区排序冻结：热度变更只影响未冻结区
  function freezeWindow(start, end) {
    const snapshot = order.slice(start, end)          // 快照 id 序
    frozenRange = { start, end, snapshot }
    listeners.forEach((fn) => fn(view()))
  }
  function unfreeze() { frozenRange = null; listeners.forEach((fn) => fn(view())) }

  function view() {
    // 展示序：冻结区用快照，其余按 score
    const sorted = [...order].sort((a, b) => (items.get(b)?.score ?? 0) - (items.get(a)?.score ?? 0))
    const display = frozenRange
      ? [...sorted.slice(0, frozenRange.start), ...frozenRange.snapshot, ...sorted.slice(frozenRange.end)]
      : sorted
    return display.map((id) => {
      const it = items.get(id)
      const inter = interactions.get(id)
      return { id, likes: (it?.likes ?? 0) + (inter?.delta ?? 0), pending: inter?.pending }
    })
  }

  function restore() { return { anchor } }   // 滚动恢复：anchor 交服务端重放（772）

  return { loadMore, like, freezeWindow, unfreeze, view, restore, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)) }
}

// 用例
const feed = createFeedEngine({ maxItems: 10, likeTimeoutMs: 50 })
feed.loadMore(null, [
  { id: 'a', score: 1, likes: 10 },
  { id: 'b', score: 3, likes: 5 },
  { id: 'c', score: 2, likes: 8 },
])
console.log(feed.view().map((x) => x.id))       // ['b','c','a']：按热度

feed.freezeWindow(0, 3)                          // 冻结可视区
// b 热度暴跌（服务端推送 score=0）
feed.loadMore(null, [{ id: 'b', score: 0, likes: 5 }])   // 幂等更新
console.log(feed.view().map((x) => x.id))        // 仍 ['b','c','a']：可视区冻结
feed.unfreeze()
console.log(feed.view().map((x) => x.id))        // ['c','a','b']：解冻后重排

feed.like('a')                                   // 互动：不重排只改计数
console.log(feed.view().find((x) => x.id === 'a').likes)   // 11（乐观 +1）
console.log(feed.restore())                      // { anchor: null }：恢复游标
```
**评分点**：
- 冻结窗口快照（可视区排序冻结，解冻后重排——交互稳定取舍）
- 互动分离（interactions Map 独立，不触发流重排）
- 内存红线（maxItems 卸载头部 + anchor 恢复）
- 乐观对账（超时回滚 delta 归零 + 服务端对账转正）
