## ZSM-310：手写跨标签页状态同步器（含全量快照与增量）
**需求**：实现 `createTabSync(store, channel)`：1) 任一标签页 store 变化以增量 `{ version, patch }` 广播；2) 新开标签页（version 落后）请求全量快照：广播 `sync-request`，持有最高版本的标签页响应 `sync-snapshot`；3) 增量按 version 严格递增消费，乱序/落后时回退到全量同步；4) 本地写入不回环。

```javascript
export function createTabSync(store, bus) {
  let version = 0
  let localWrite = false

  function applyFull(snapshot) {
    localWrite = true
    store.setState(snapshot.state)
    version = snapshot.version
    localWrite = false
  }

  bus.on('patch', (msg) => {
    if (msg.version !== version + 1) { bus.emit('sync-request'); return }  // 断档 → 全量
    localWrite = true
    store.setState(msg.patch)
    version = msg.version
    localWrite = false
  })
  bus.on('sync-request', () => {
    if (version > 0) bus.emit('sync-snapshot', { version, state: store.getState() })
  })
  bus.on('sync-snapshot', (snap) => { if (snap.version > version) applyFull(snap) })
  bus.emit('sync-request')                        // 新标签页启动即请求

  return store.subscribe(() => {
    if (localWrite) return                        // 远端来源不回写
    const state = store.getState()
    bus.emit('patch', { version: ++version, patch: state })
  })
}
```
**评分点**：
- 回环防护：远端应用时置 localWrite 标记，订阅回调里短路
- version 严格 +1 校验，断档时立即降级全量快照而不是盲应用
- 启动时主动 sync-request，快照响应带版本比较取最新
- 快照与增量共用同一版本号空间，避免恢复后版本错位
