## ZSM-616：手写性能预算门禁报告生成器
**答**：实现 `createBudgetGate(budget)`：1) budget 声明各场景预算 `{ scenario: { renderCount, maxCommitMs, blockingMs } }`；2) `record(scenario, metrics)`：登记一次测量；3) `evaluate()`：生成门禁报告——每场景 pass/fail、超预算项、与上次基线的趋势（上升红/下降绿）；4) 基线存取：`setBaseline`/导出 JSON 供 CI 对比；5) 输出人类可读报告字符串（表格化）；附用例验证 fail 项与趋势标记。

```javascript
function createBudgetGate(budget) {
  const metrics = new Map()       // scenario -> [measurements]
  let baseline = null

  function record(scenario, m) {
    const list = metrics.get(scenario) ?? []
    list.push(m)
    metrics.set(scenario, list)
  }

  function evaluate() {
    const rows = []
    for (const [scenario, items] of metrics) {
      const b = budget[scenario]
      if (!b) continue
      const last = items.at(-1)
      const checks = []
      for (const [k, limit] of Object.entries(b)) {
        const v = last[k]
        checks.push({ metric: k, value: v, limit, pass: v <= limit })
      }
      // 趋势：与基线同场景对比
      let trend = '='
      if (baseline?.[scenario]) {
        const prev = baseline[scenario].blockingMs ?? baseline[scenario]
        const cur = last.blockingMs
        trend = cur > prev ? '↑恶化' : cur < prev ? '↓改善' : '='
      }
      rows.push({
        scenario,
        pass: checks.every((c) => c.pass),
        checks,
        trend,
      })
    }
    return {
      pass: rows.every((r) => r.pass),
      rows,
      report() {
        const lines = ['| 场景 | 指标 | 值 | 预算 | 结果 |', '|---|---|---|---|---|']
        for (const r of rows) {
          for (const c of r.checks) {
            lines.push(`| ${r.scenario} | ${c.metric} | ${c.value} | ${c.limit} | ${c.pass ? '✓' : `✗(${r.trend})`} |`)
          }
        }
        lines.push(`\n总体: ${this.pass ? 'PASS' : 'FAIL — CI 阻断'}`)
        return lines.join('\n')
      },
    }
  }

  function setBaseline(data) { baseline = data }
  function exportBaseline() {
    const out = {}
    metrics.forEach((items, s) => { out[s] = items.at(-1) })
    return JSON.stringify(out)
  }

  return { record, evaluate, setBaseline, exportBaseline }
}

// 用例
const gate = createBudgetGate({
  '筛选交互': { renderCount: 12, maxCommitMs: 16, blockingMs: 50 },
  '加购提交': { renderCount: 30, maxCommitMs: 16, blockingMs: 80 },
})
gate.setBaseline({ '筛选交互': { blockingMs: 40 }, '加购提交': { blockingMs: 90 } })
gate.record('筛选交互', { renderCount: 15, maxCommitMs: 18, blockingMs: 62 })
gate.record('加购提交', { renderCount: 28, maxCommitMs: 12, blockingMs: 75 })
const r = gate.evaluate()
console.log(r.report())
// | 场景 | 指标 | 值 | 预算 | 结果 |
// | 筛选交互 | renderCount | 15 | 12 | ✗(↑恶化) |
// | 筛选交互 | maxCommitMs | 18 | 16 | ✗(↑恶化) |
// | 筛选交互 | blockingMs | 62 | 50 | ✗(↑恶化) |
// | 加购提交 | renderCount | 28 | 30 | ✓ |
// | 加购提交 | maxCommitMs | 12 | 16 | ✓ |
// | 加购提交 | blockingMs | 75 | 80 | ✓ |
// 总体: FAIL — CI 阻断
```
**评分点**：
- 多指标预算校验（pass = 全部指标达标）
- 基线对比趋势标记（恶化/改善方向）
- 人类可读表格报告（CI 注释直接可贴）
- 基线导出/导入（优化前后同口径对比）
