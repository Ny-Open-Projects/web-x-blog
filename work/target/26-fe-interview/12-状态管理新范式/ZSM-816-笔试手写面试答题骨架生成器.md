## ZSM-816：手写答题骨架生成器（按题型 × 考点自动装配答题结构）
**答**：实现 `createAnswerSkeleton()`：1) `register(type,考点, segments)`：登记答题段模板（题型：原理/手写/设计/排查；考点：源码/建模/跨端/性能/治理）；2) `generate(type, 考点, { timeLimit })`：装配骨架——按题型选段集合，按 timeLimit 比例分配每段秒数，输出 `{ segments: [{ name, seconds, anchor }] }`；3) `overlap()`：同一题型下考点段重叠分析（公共段复用，私有段标注）；4) 附用例：45min 设计题的段装配与 90s 组合题的裁剪。

```javascript
function createAnswerSkeleton() {
  const templates = new Map()   // `${type}:${考点}` -> segments

  function register(type, 考点, segments) {
    templates.set(`${type}:${考点}`, segments)
    return this
  }

  function generate(type, 考点, { timeLimit = 90 } = {}) {
    const key = `${type}:${考点}`
    const exact = templates.get(key)
    const segs = exact ?? fallback(type, 考点)
    const totalWeight = segs.reduce((a, s) => a + (s.weight ?? 1), 0)
    return {
      type, 考点,
      segments: segs.map((s) => ({
        name: s.name,
        seconds: Math.round(timeLimit * (s.weight ?? 1) / totalWeight),
        anchor: s.anchor,
      })),
    }
  }

  // 兜底：题型公共段 + 考点私有段
  function fallback(type, 考点) {
    const common = templates.get(`${type}:*`) ?? []
    const specific = templates.get(`*:${考点}`) ?? []
    return [...common, ...specific]
  }

  // 公共段重叠分析
  function overlap(type) {
    const by考点 = [...templates.entries()].filter(([k]) => k.startsWith(`${type}:`) && !k.endsWith(':*'))
    const count = new Map()
    by考点.forEach(([, segs]) => segs.forEach((s) => count.set(s.name, (count.get(s.name) ?? 0) + 1)))
    const total = by考点.length
    return [...count.entries()].map(([name, n]) => ({
      name, share: +(n / total).toFixed(2),
      isCommon: n === total,     // 所有考点都有 = 公共段，可抽到 type:*
    }))
  }

  return { register, generate, overlap }
}

// 用例：设计题的考点段注册
const sk = createAnswerSkeleton()
sk.register('design', '*', [
  { name: '澄清三问', weight: 1, anchor: '半径/规模/冲突语义' },
  { name: '故障闭环', weight: 1, anchor: '故障路径+可观测' },
])
sk.register('design', '跨端', [
  { name: '通道选型', weight: 1.2, anchor: 'WS+降级链+leader聚合' },
  { name: '可靠性三防', weight: 1.5, anchor: '去重/追平/乱序缓冲' },
  { name: '一致性分层', weight: 1.2, anchor: '服务端权威/LWW/op' },
])
sk.register('design', '建模', [
  { name: '分层建模', weight: 1.5, anchor: '归一化+状态机+五句式' },
])

// 45 分钟设计题（跨端）
const plan = sk.generate('design', '跨端', { timeLimit: 45 * 60 })
console.log(plan.segments.map((s) => `${s.name}:${Math.round(s.seconds / 60)}min`))
// 澄清三问:8min | 通道选型:10min | 可靠性三防:12min | 一致性分层:10min | 故障闭环:8min

// 公共段分析
console.log(sk.overlap('design'))
// [{ name: '澄清三问', share: 1, isCommon: true }, { name: '故障闭环', share: 1, isCommon: true }]
```
**评分点**：
- 权重比例分配秒数（timeLimit × weight/totalWeight）
- fallback 装配（题型公共段 + 考点私有段组合）
- overlap 公共段抽取分析（isCommon → 复用）
- 段锚点输出（每段一句话锚点可背）
