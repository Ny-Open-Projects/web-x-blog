## ZSM-473：手写状态层内存泄漏检测器
**需求**：实现 `createLeakDetector(store)`：1) 包装 subscribe：记录每个订阅的注册位置与存活时长；2) 检测器周期性（10s）对比「注册总量 - 退订总量」净增长趋势：连续 5 次采样净增长 > 0 判定疑似泄漏（输出 top 注册位置）；3) 检测「僵尸订阅」：注册超过 10 分钟且所属页面已卸载标记（需接入页面生命周期）→ 直接告警；4) `snapshot()` 返回当前订阅清单 { loc, ageMs }；5) dispose 完整清理检测器自身。

```javascript
export function createLeakDetector(store, { intervalMs = 10_000, zombieMs = 600_000 } = {}) {
  const subs = new Map()       // id → { loc, at, alive }
  let netGrowth = [], timer = null, disposed = false
  let nextId = 0

  const origSubscribe = store.subscribe
  store.subscribe = (fn, loc = callerLoc()) => {
    const id = ++nextId
    subs.set(id, { loc, at: Date.now(), alive: true })
    const unsub = origSubscribe(fn)
    return () => { subs.get(id).alive = false; subs.delete(id); return unsub() }
  }

  function sample() {
    const alive = [...subs.values()].filter((s) => s.alive)
    netGrowth.push(alive.length)
    if (netGrowth.length > 5) netGrowth.shift()
    if (netGrowth.length === 5 && netGrowth.every((v, i, a) => i === 0 || v > a[i - 1])) {
      const byLoc = {}
      alive.forEach((s) => (byLoc[s.loc] = (byLoc[s.loc] ?? 0) + 1))
      const top = Object.entries(byLoc).sort((a, b) => b[1] - a[1])[0]
      report('suspected_leak', { trend: netGrowth, topLoc: top })
    }
    // 僵尸订阅
    alive.forEach((s) => {
      if (Date.now() - s.at > zombieMs) report('zombie_sub', { loc: s.loc, ageMs: Date.now() - s.at })
    })
  }

  timer = setInterval(sample, intervalMs)

  return {
    snapshot: () => [...subs.values()].map((s) => ({ loc: s.loc, ageMs: Date.now() - s.at })),
    dispose: () => { disposed = true; clearInterval(timer); store.subscribe = origSubscribe },
  }
}

function callerLoc() { return new Error().stack?.split('\n')[3]?.trim().slice(0, 80) ?? '?' }
```
**评分点**：
- 趋势判定（连续 5 次净增长，单点波动不误报）
- top 注册位置定位（可行动的修复线索）
- 僵尸订阅检测（年龄超阈值的存活订阅）
- dispose 恢复原始 subscribe（检测器自身不留下痕迹）
