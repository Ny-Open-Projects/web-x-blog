## ZSM-179：手写拖拽排序的乐观更新提交（本地重排 + 服务端批量校准 + 失败回滚）
**需求**：实现 `reorderAndCommit(list, from, to)`：1) 本地立即按索引重排并渲染（乐观）；2) 服务端只接受增量 `{ id, beforeId, afterId }`（防并发冲突），实现从全量顺序到增量的转换；3) 提交中再拖拽要排队，不允许覆盖未确认结果；4) 失败回滚到提交前快照并提示；5) 快速连续拖拽合并为最后一次结果提交（防抖提交）。给出 Hook 形态实现。

```javascript
function useSortableList(initial) {
  const [list, setList] = useState(initial)
  const pending = useRef(null)          // 未确认的提交
  const snapshot = useRef(null)

  function reorder(from, to) {
    const next = [...list]
    const [moved] = next.splice(from, 1)
    next.splice(to, 0, moved)
    setList(next)                       // 乐观渲染
    snapshot.current = list             // 快照用于回滚
    scheduleCommit(next)
  }

  function scheduleCommit(order) {
    clearTimeout(pending.current?.timer)
    pending.current = {
      timer: setTimeout(() => commit(order), 300),   // 防抖：连续拖拽合并
    }
  }

  async function commit(order) {
    const target = order[0]
    const prevId = order[order.indexOf(target) - 1]?.id ?? null
    const nextId = order[order.indexOf(target) + 1]?.id ?? null
    try {
      await api.reorder({ id: target.id, beforeId: prevId, afterId: nextId })
    } catch (e) {
      setList(snapshot.current)        // 回滚到提交前快照
      toast.error('排序保存失败，已恢复')
    } finally {
      pending.current = null
    }
  }
  return { list, reorder }
}

// 评分点：
// 1. 本地重排先渲染（splice 正确性：先删后插）
// 2. 增量提交语义（beforeId/afterId）防并发覆盖
// 3. 防抖合并连续拖拽，避免一次拖拽一次请求
// 4. 快照回滚 + 用户提示
// 5. 提交中再拖拽的处理（排队/覆盖策略要能说清）
```
