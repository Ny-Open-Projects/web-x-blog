## ZSM-893：手写学习进度追踪器（三层内化的盲写/落地/表达打卡）
**答**：892 落地——实现 `createLearningTracker()`：1) `log(type, item, { score })`：打卡（type: blind-write/landing/expression）——盲写带自评分（对照评分点命中数）；2) `progress()`：三层达标判定——L2 达标 = 盲写 ≥10 且均分 ≥70 且落地 ≥3；L3 达标 = 表达 ≥1 次且自评 ≥70；3) `nextAction()`：按最弱层推荐下一步；4) 附用例：达标判定与推荐链。

```javascript
function createLearningTracker() {
  const logs = { 'blind-write': [], landing: [], expression: [] }

  function log(type, item, { score } = {}) {
    logs[type]?.push({ item, score: score ?? 0, at: Date.now() })
    return this
  }

  function progress() {
    const bw = logs['blind-write']
    const ld = logs.landing
    const ex = logs.expression
    const bwAvg = bw.length ? Math.round(bw.reduce((a, b) => a + b.score, 0) / bw.length) : 0
    const l1 = true                                                    // L1 读即达标（默认）
    const l2 = bw.length >= 10 && bwAvg >= 70 && ld.length >= 3
    const l3 = ex.length >= 1 && ex.every((e) => e.score >= 70)
    return { l1, l2, l3, blindWrites: bw.length, blindAvg: bwAvg, landings: ld.length, expressions: ex.length }
  }

  function nextAction() {
    const p = progress()
    if (!p.l2) {
      const needBlind = Math.max(0, 10 - p.blindWrites)
      const needLand = Math.max(0, 3 - p.landings)
      const needScore = p.blindAvg < 70 ? '重写低分卡（对照评分点找差）' : null
      return `L2 未达标：盲写还差 ${needBlind} 张${needScore ? '，' + needScore : ''}；落地还差 ${needLand} 个`
    }
    if (!p.l3) return 'L3 未达标：做一次团队分享或模拟面试录音（822 评分卡自评）'
    return '三层达标：维护个人概念矩阵（876）+ 季度更新技术雷达（880）'
  }

  return { log, progress, nextAction }
}

// 用例
const tr = createLearningTracker()
// L2 打卡：盲写 3 张（均分 65）、落地 1 个
tr.log('blind-write', 'ZSM-528 对账器', { score: 70 })
tr.log('blind-write', 'ZSM-531 命令栈', { score: 60 })
tr.log('blind-write', 'ZSM-537 三态选择', { score: 65 })
tr.log('landing', '购物车预算门禁', {})
console.log(tr.nextAction())
// L2 未达标：盲写还差 7 张，重写低分卡（对照评分点找差）；落地还差 2 个

// 补齐后
for (let i = 4; i <= 10; i++) tr.log('blind-write', `ZSM-卡${i}`, { score: 75 })
tr.log('landing', '契约总线', {})
tr.log('landing', '对账器', {})
console.log(tr.progress())       // { l1: true, l2: true, l3: false, blindWrites: 10, blindAvg: 74, landings: 3, expressions: 0 }
console.log(tr.nextAction())     // L3 未达标：做一次团队分享...
tr.log('expression', '状态联邦分享', { score: 80 })
console.log(tr.progress().l3)    // true
```
**评分点**：
- 三层达标量化（盲写数/均分/落地数/表达次的阈值组合）
- nextAction 按最弱层推荐（含具体差值）
- 盲写自评分驱动（低分卡重写建议）
- 达标后的进阶动作（矩阵维护 + 雷达更新）
