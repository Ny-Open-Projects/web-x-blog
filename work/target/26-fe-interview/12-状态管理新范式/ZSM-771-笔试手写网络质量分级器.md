## ZSM-771：手写网络质量分级器（RTT 滑动平均 + 滞回切换 + 策略表）
**答**：实现 `createNetworkTier()`：1) `probe(rtt | null)`：喂 RTT 样本（滑动窗口 5 个），null 表示失败；2) 分级规则：offline（连续 2 次失败）、slow（avg > 1000ms）、fast（avg < 150ms 且样本 ≥ 3）、normal 其余；3) **滞回**：新分级需连续 2 次评估一致才切换（防边界抖动）；4) `strategy()`：返回当前分级策略表；5) 附用例：慢网判定、滞回防抖、连续失败 offline。

```javascript
function createNetworkTier({ windowSize = 5, hysteresis = 2 } = {}) {
  let samples = []
  let failures = 0
  let current = 'normal'
  let pending = null
  let pendingStreak = 0
  const listeners = new Set()

  function classify(avg, failStreak) {
    if (failStreak >= 2) return 'offline'
    if (avg == null) return 'normal'
    if (avg > 1000) return 'slow'
    if (samples.length >= 3 && avg < 150) return 'fast'
    return 'normal'
  }

  function probe(rtt) {
    if (rtt == null) {
      failures++
    } else {
      failures = 0
      samples.push(rtt)
      if (samples.length > windowSize) samples.shift()
    }
    const avg = samples.length ? samples.reduce((a, b) => a + b) / samples.length : null
    const next = classify(avg, failures)

    // 滞回：连续 hysteresis 次一致才切换
    if (next === current) { pendingStreak = 0; pending = null }
    else if (next === pending) {
      pendingStreak++
      if (pendingStreak >= hysteresis) {
        const prev = current
        current = next
        pendingStreak = 0
        pending = null
        listeners.forEach((fn) => fn(current, prev))
      }
    } else {
      pending = next
      pendingStreak = 1
    }
    return current
  }

  function strategy() {
    const TIER_STRATEGY = {
      offline: { timeoutMs: 0, pollFactor: 0, queueOnly: true },
      slow: { timeoutMs: 8000, pollFactor: 3, aggressiveOptimistic: true },
      normal: { timeoutMs: 15000, pollFactor: 1 },
      fast: { timeoutMs: 15000, pollFactor: 1, prefetch: 'aggressive' },
    }
    return TIER_STRATEGY[current]
  }

  return {
    probe, strategy,
    subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
    get tier() { return current },
  }
}

// 用例
const net = createNetworkTier()
console.log(net.tier)                       // 'normal'
// 慢网样本：avg > 1000
net.probe(1200); net.probe(1300); net.probe(1100)
console.log(net.probe(1200))                // 第一次评估 slow：滞回不切
console.log(net.probe(1200))                // 连续 2 次：切换 'slow'
console.log(net.tier, net.strategy().timeoutMs)   // 'slow' 8000

// 恢复 normal：滞回同样生效
net.probe(100); net.probe(100); net.probe(100)
net.probe(100)
console.log(net.tier)                       // 'fast'（avg<150 且样本足够）

// 连续失败：offline
net.probe(null); net.probe(null)
console.log(net.probe(null))                // 'offline'（失败 ≥2）
console.log(net.strategy().queueOnly)       // true
```
**评分点**：
- RTT 滑动窗口平均（5 样本去抖）
- 分级规则四档（offline/slow/normal/fast，fast 需样本量门槛）
- 滞回切换（连续 2 次一致才切，防边界震荡）
- 策略表按级消费（timeout/轮询系数/乐观策略）
