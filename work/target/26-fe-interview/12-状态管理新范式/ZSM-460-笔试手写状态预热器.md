## ZSM-460：手写状态预热器
**需求**：实现 `createWarmer(queryClient, plans)`：1) plans：`[{ key, fn, priority, intervalMs? }]`；2) `warm()`：按优先级串行预热（低优先级不抢首屏带宽），每项之间让出主线程；3) 已有缓存且新鲜的 key 跳过；4) 支持 intervalMs 的周期性再热（保活型数据）；5) `stop()` 清全部周期定时器与事件；6) 页面隐藏暂停周期预热（省电），回前台补一轮。

```javascript
export function createWarmer(queryClient, plans) {
  const timers = new Map()
  let stopped = false

  async function warm() {
    const sorted = [...plans].sort((a, b) => (a.priority ?? 0) - (b.priority ?? 0))
    for (const plan of sorted) {
      if (stopped) return
      const st = queryClient.getQueryState(plan.key)
      if (st?.data && Date.now() - st.dataUpdatedAt < 60_000) continue    // 新鲜跳过
      await queryClient.prefetchQuery({ queryKey: plan.key, queryFn: plan.fn }).catch(() => {})
      await new Promise((r) => (requestIdleCallback ?? ((f) => setTimeout(f, 0)))(r))   // 让出主线程
    }
  }

  function onVis() { if (!stopped && document.visibilityState === 'visible') warm() }

  function start() {
    warm()
    plans.filter((p) => p.intervalMs).forEach((p) => {
      const t = setInterval(() => {
        if (document.visibilityState === 'hidden' || stopped) return      // 隐藏暂停
        queryClient.prefetchQuery({ queryKey: p.key, queryFn: p.fn }).catch(() => {})
      }, p.intervalMs)
      timers.set(p.key, t)
    })
    document.addEventListener('visibilitychange', onVis)
  }

  function stop() {
    stopped = true
    timers.forEach(clearInterval); timers.clear()
    document.removeEventListener('visibilitychange', onVis)
  }

  return { start, stop, warm }
}
```
**评分点**：
- 优先级串行 + 让出主线程（预热不卡首屏交互）
- 新鲜跳过（不重复拉取）；保活型周期预热与一次性预热分层
- 页面隐藏暂停 + 回前台补一轮；stop 清定时器 + 移除事件（完整清理）
- 预取失败静默（尽力而为语义，不阻塞正常功能）
