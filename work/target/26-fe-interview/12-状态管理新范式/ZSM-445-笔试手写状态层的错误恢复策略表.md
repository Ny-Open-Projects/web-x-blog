## ZSM-445：手写状态层错误恢复策略表
**答**：实现 `createRecoveryStore(store, policies)`：1) policies 按错误类型映射恢复动作：`{ ValidationError: 'reject', NetworkError: 'queue', QuotaError: 'evict', FatalError: 'reset' }`；2) `commit(op)` 执行 op，异常时按策略恢复：reject=原样抛出、queue=入离线队列、evict=清缓存重试一次、reset=恢复初始快照；3) 恢复动作链上限（每个 op 最多触发一次策略，防递归）；4) 未匹配策略的未知错误默认 reject；5) `simulate(errorType)` 测试助手：注入假错误验证策略路径。

```javascript
const STRATEGIES = {
  reject: async (op, err, ctx) => { throw err },
  queue: async (op, err, ctx) => { ctx.queue.push(op); return { recovered: 'queued' } },
  evict: async (op, err, ctx) => { ctx.store.clearCache?.(); return ctx.retry(op) },
  reset: async (op, err, ctx) => { ctx.store.setState(ctx.snapshot); return { recovered: 'reset' } },
}

export function createRecoveryStore(store, policies, extras = {}) {
  let snapshot = structuredClone(store.getState())
  let recovering = false
  const queue = []

  async function commit(op, depth = 0) {
    try {
      const r = await op.exec(store)
      snapshot = structuredClone(store.getState())           // 成功后刷新基线
      return r
    } catch (err) {
      const type = err?.name ?? 'Unknown'
      const action = policies[type] ?? 'reject'
      if (depth > 0) throw err                                // 策略链上限：一次
      if (recovering) { queue.push(op); return { queued: true } }
      recovering = true
      try {
        return await STRATEGIES[action](op, err, { store, queue, snapshot, retry: (o) => commit(o, depth + 1) })
      } finally { recovering = false }
    }
  }

  function simulate(errorType) {                              // 测试助手
    const bad = { exec: () => { const e = new Error(errorType); e.name = errorType; throw e } }
    return commit(bad)
  }

  return { commit, simulate, queue }
}
```
**评分点**：
- 策略表驱动（错误类型→动作映射），四种语义清晰且可扩展
- 恢复链上限（depth 防递归）+ recovering 锁（恢复期间新 op 入队不嵌套）
- reset 用快照恢复（错误前基线，不是清空——保留用户数据）
- simulate 测试助手（策略路径可单测）
