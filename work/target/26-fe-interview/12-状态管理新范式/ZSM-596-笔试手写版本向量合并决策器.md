## ZSM-596：手写版本向量合并决策器（含并发合并策略）
**答**：实现 `createSyncVV(deviceId)`：1) `localEdit(value)`：写入 + 向量 tick；2) `receive(remote)`：远端带 `{ vv, value }`，按向量关系决策——远端新→应用、本地新→忽略、相等→幂等、**并发→按合并策略**（策略：'max' 取大值 / 'concat' 数组并集 / 'custom' 用户函数）；3) `compare(v1, v2)` 返回关系与首个差异 device；4) `export()` 输出 `{ vv, value }`；5) 附用例：构造两副本并发编辑场景，验证 'max' 与 'concat' 两种策略的收敛结果一致。

```javascript
function createSyncVV(deviceId, { strategy = 'max' } = {}) {
  let value = null
  const vv = { [deviceId]: 0 }

  function localEdit(v) {
    value = v
    vv[deviceId]++
    return { vv: { ...vv }, value }
  }

  function compare(v1, v2) {
    let aG = false, bG = false
    const firstDiffDevice = {}
    for (const d of new Set([...Object.keys(v1), ...Object.keys(v2)])) {
      const a = v1[d] ?? 0, b = v2[d] ?? 0
      if (a !== b) firstDiffDevice.d ??= d
      if (a > b) aG = true
      if (b > a) bG = true
    }
    const rel = aG && bG ? 2 : aG ? 1 : bG ? -1 : 0
    return { rel, diffDevice: firstDiffDevice.d ?? null }   // rel: 1 本地新 / -1 远端新 / 2 并发 / 0 相同
  }

  function mergeValues(a, b) {
    if (strategy === 'max') return Math.max(a ?? -Infinity, b ?? -Infinity)
    if (strategy === 'concat') return [...new Set([...(a ?? []), ...(b ?? [])])]
    if (strategy === 'custom') return mergeValues.custom?.(a, b) ?? b
    return b
  }

  function receive(remote) {
    const { rel } = compare(vv, remote.vv)
    switch (rel) {
      case -1: {                          // 远端更新：应用并吸收向量
        value = remote.value
        for (const [d, c] of Object.entries(remote.vv)) vv[d] = Math.max(vv[d] ?? 0, c)
        return { action: 'applied', value }
      }
      case 0: return { action: 'dedupe', value }
      case 1: return { action: 'ignored', value }   // 本地更新
      case 2: {                           // 并发：合并 + 向量吸收
        value = mergeValues(value, remote.value)
        for (const [d, c] of Object.entries(remote.vv)) vv[d] = Math.max(vv[d] ?? 0, c)
        return { action: 'merged', value }
      }
    }
  }

  return { localEdit, receive, compare, export: () => ({ vv: { ...vv }, value }) }
}

// 用例：并发编辑 + 双策略收敛
// 设备 A 和 B 从同一基础状态分叉
const a = createSyncVV('A', { strategy: 'max' })
const b = createSyncVV('B', { strategy: 'max' })
const base = { vv: { A: 1, B: 1 }, value: 10 }
Object.assign(a, { vv: { ...base.vv }, value: base.value })   // 模拟同源
Object.assign(b, { vv: { ...base.vv }, value: base.value })
a.localEdit(20)     // A 并发改
b.localEdit(30)     // B 并发改
const ra = a.receive(b.export())
console.log(ra)     // { action: 'merged', value: 30 }：max 合并
const rb = b.receive(a.export())
console.log(rb)     // { action: 'merged', value: 30 }：两端收敛一致
console.log(JSON.stringify(a.export().vv) === JSON.stringify(b.export().vv)) // true

// concat 策略：标签并集
const c = createSyncVV('C', { strategy: 'concat' })
c.localEdit(['x'])
const r2 = c.receive({ vv: { D: 3 }, value: ['y'] })
console.log(r2)     // { action: 'merged', value: ['x','y'] }
```
**评分点**：
- 四种向量关系的完整决策（应用/忽略/幂等/合并）
- 并发合并的可交换性（两端策略一致则结果一致）
- 向量吸收在合并后也要做（防下次误判）
- compare 附带首个差异 device（可定位冲突来源）
