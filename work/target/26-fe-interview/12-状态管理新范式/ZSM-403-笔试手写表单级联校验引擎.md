## ZSM-403：手写表单级联校验引擎
**答**：实现 `createValidator(schema)`：1) schema 为 `{ field: { deps: string[], validate(values) => string|null } }`；2) `validate(values)` 按 deps 拓扑序校验：先校验无依赖字段，错误字段跳过其下游校验（下游视为 invalid，避免级联误报）；3) 返回 `{ valid, errors: {field: msg} }`；4) 循环依赖抛错；5) 支持异步 validate（Promise.all 收敛）。

```javascript
export function createValidator(schema) {
  // 环检测：DFS
  const checkCycle = (node, stack = new Set(), done = new Set()) => {
    if (stack.has(node)) throw new Error(`cycle at ${node}`)
    if (done.has(node)) return
    stack.add(node)
    schema[node].deps.forEach((d) => checkCycle(d, stack, done))
    stack.delete(node); done.add(node)
  }
  Object.keys(schema).forEach((f) => checkCycle(f))

  async function validate(values) {
    const errors = {}
    const invalid = new Set()
    const order = topoSort(Object.keys(schema), schema)          // deps 拓扑序
    for (const field of order) {
      if (schema[field].deps.some((d) => errors[d])) {           // 上游有错：跳过并标记
        errors[field] = 'depends on invalid field'
        invalid.add(field)
        continue
      }
      const msg = await schema[field].validate(values)
      if (msg) { errors[field] = msg; invalid.add(field) }
    }
    return { valid: Object.keys(errors).length === 0, errors }
  }
  return { validate }
}

function topoSort(fields, schema) {
  const indeg = {}; const adj = {}
  fields.forEach((f) => { indeg[f] = 0; adj[f] = [] })
  fields.forEach((f) => schema[f].deps.forEach((d) => { adj[d].push(f); indeg[f]++ }))
  const q = fields.filter((f) => indeg[f] === 0); const out = []
  while (q.length) { const f = q.shift(); out.push(f); adj[f].forEach((t) => --indeg[t] === 0 && q.push(t)) }
  return out
}
```
**评分点**：
- 上游错误阻断下游校验并显式标记（防级联误报），不是静默跳过
- 循环依赖在构造期检测（不是运行时死循环）
- 拓扑排序保证依赖先校验；无依赖字段可并行（优化点：同层 Promise.all）
- 异步 validate 全部收敛后才返回最终结果
