## ZSM-690：手写快捷键作用域栈（优先级消费 + 输入态豁免 + 泄漏检测）
**答**：实现 `createShortcutStack()`：1) `push(id, bindings)`：入栈作用域（bindings 为 `'mod+k' → fn` 简写解析）；2) `handle(event)`：从栈顶向下找第一个命中的作用域，命中即消费（返回 true），未命中向下冒泡；`mod` 自动区分平台（macOS metaKey / 其他 ctrlKey）；3) 输入态豁免：focus 在 input/textarea 且 binding 不含修饰键时跳过该作用域；4) `pop(id)`：清理（重复 pop 幂等）；5) `audit()`：栈中剩余作用域数（0 = 无泄漏）；6) 附用例：栈顶优先、豁免、幂等 pop。

```javascript
function createShortcutStack() {
  const stack = []

  function parseBinding(binding) {
    const [mods, ...keys] = binding.split('+')
    const key = keys.join('+').toLowerCase()
    const mod = /Mac|iPhone|iPad/.test(navigator.platform) ? 'metaKey' : 'ctrlKey'
    return {
      key,
      ctrlKey: mods.includes('ctrl'),
      metaKey: mods.includes('mod') ? (mod === 'metaKey') : mods.includes('meta'),
      shiftKey: mods.includes('shift'),
      altKey: mods.includes('alt'),
      hasMod: mods.includes('mod') || mods.includes('ctrl') || mods.includes('meta'),
    }
  }

  function push(id, bindings) {
    const handlers = new Map()
    for (const [binding, fn] of Object.entries(bindings)) {
      const spec = parseBinding(binding)
      handlers.set(`${spec.metaKey ? 'M' : ''}${spec.ctrlKey ? 'C' : ''}${spec.shiftKey ? 'S' : ''}${spec.altKey ? 'A' : ''}:${spec.key}`,
        { fn, hasMod: spec.hasMod })
    }
    stack.push({ id, handlers })
    return () => pop(id)
  }

  function keySpec(e) {
    return `${e.metaKey ? 'M' : ''}${e.ctrlKey ? 'C' : ''}${e.shiftKey ? 'S' : ''}${e.altKey ? 'A' : ''}:${e.key.toLowerCase()}`
  }

  function handle(e) {
    const isTyping = /INPUT|TEXTAREA/.test(e.target?.tagName ?? '') || e.target?.isContentEditable
    for (let i = stack.length - 1; i >= 0; i--) {
      const scope = stack[i]
      const hit = scope.handlers.get(keySpec(e))
      if (!hit) continue
      // 豁免：输入态且无修饰键
      if (isTyping && !hit.hasMod) continue
      hit.fn(e)
      return { consumed: true, scope: scope.id }
    }
    return { consumed: false }
  }

  function pop(id) {
    const idx = stack.findIndex((s) => s.id === id)
    if (idx >= 0) stack.splice(idx, 1)      // 幂等：不存在则忽略
  }

  return { push, handle, pop, audit: () => ({ remaining: stack.length, ids: stack.map((s) => s.id) }) }
}

// 用例
const stack = createShortcutStack()
const log = []
stack.push('global', { 'mod+k': () => log.push('global:search') })
stack.push('modal', { 'enter': () => log.push('modal:submit'), 'escape': () => log.push('modal:close') })

// 栈顶优先：modal 的 enter 先消费
let r = stack.handle({ key: 'Enter', ctrlKey: false, target: { tagName: 'DIV' } })
console.log(r)                        // { consumed: true, scope: 'modal' }
console.log(log)                      // ['modal:submit']

// mod 平台适配：macOS metaKey
r = stack.handle({ key: 'k', metaKey: true, target: { tagName: 'DIV' } })
console.log(r, log.at(-1))            // modal 无 mod+k → 冒泡到 global → 'global:search'

// 输入态豁免：Enter 在 input 中不触发 modal（无修饰键）
r = stack.handle({ key: 'Enter', target: { tagName: 'INPUT' } })
console.log(r)                        // { consumed: false }

// 幂等 pop + 泄漏审计
stack.pop('modal'); stack.pop('modal')
console.log(stack.audit())            // { remaining: 1, ids: ['global'] }
```
**评分点**：
- 栈顶优先语义冒泡（modal → global 层级消费）
- mod 绑定平台适配（Mac meta / 其他 ctrl）
- 输入态豁免仅针对无修饰键绑定
- pop 幂等 + audit 剩余作用域（泄漏检测闭环）
