## ZSM-451：手写乐观锁版本管理器
**答**：实现 `createVersionedStore(initial)`：1) 内部维护 `{ data, version }`；2) `casWrite(expectedVersion, patch)`：版本匹配才写入（CAS），版本不符抛 `ConflictError`（带服务端当前版本）；3) `write(patch)` 无条件写（版本 +1）；4) `casWrite` 成功返回新版本号；5) 冲突时提供 `resolve(strategy)`：'overwrite'（拉新版本重放 patch）/'abort'；6) 版本号单调递增不回退（undo 不减版本）。

```javascript
export class ConflictError extends Error {
  constructor(current) { super(`version conflict: expected != ${current}`); this.current = current }
}

export function createVersionedStore(initial) {
  let data = structuredClone(initial)
  let version = 1
  const listeners = new Set()

  function commit(patch) {
    data = typeof patch === 'function' ? patch(data) : { ...data, ...patch }
    version += 1                                    // 单调递增（undo 也不回退）
    listeners.forEach((l) => l({ data, version }))
    return version
  }

  return {
    get version() { return version },
    get data() { return data },
    subscribe(l) { listeners.add(l); return () => listeners.delete(l) },
    write: commit,
    casWrite(expected, patch) {
      if (expected !== version) throw new ConflictError(version)   // CAS 失败
      return commit(patch)
    },
    refresh() { return { data, version } },          // 冲突后拉最新
    resolveConflict(strategy, patch) {
      const cur = this.refresh()
      if (strategy === 'overwrite') return commit(patch)          // 新版本上重放
      return { aborted: true, current: cur }
    },
  }
}
```
**评分点**：
- CAS 语义正确（expected !== version 抛 ConflictError 带当前版本）
- 版本单调递增（undo/回滚不减版本——版本是事实序列号不是撤销计数）
- 冲突解决两策略（overwrite 重放 / abort），refresh 提供冲突现场
- 通知携带版本号（订阅方可做版本守卫）
