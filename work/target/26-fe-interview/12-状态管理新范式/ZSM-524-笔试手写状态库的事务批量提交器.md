## ZSM-524：手写状态库的事务批量提交器
**答**：实现 `createTransactionalStore(initial)`：1) `transact(fn)`：fn 里可多次 setState，全部成功才在结尾通知一次订阅者；2) fn 抛异常时**回滚到 transact 开始前的状态**（不通知）；3) 支持嵌套 transact（内层合并到外层，只有最外层提交才通知）；4) 提供 `log()` 返回最近一次事务的变更 keys 序列。附用例验证提交/回滚/嵌套。

```javascript
function createTransactionalStore(initial) {
  let state = initial
  const subs = new Set()
  const stack = []      // 嵌套事务快照栈
  let lastLog = []

  function setState(updater) {
    const next = typeof updater === 'function' ? updater(state) : updater
    if (stack.length) {
      // 事务内：只记变更不通知
      const keys = Object.keys(next).filter((k) => !Object.is(next[k], state[k]))
      lastLog.push(...keys)
    }
    state = next
    if (!stack.length) flush()
  }

  function flush() {
    subs.forEach((fn) => fn(state))
  }

  function transact(fn) {
    stack.push(state)                 // 快照用于回滚
    const parentLog = lastLog
    lastLog = []
    try {
      fn(setState, () => state)
      if (stack.length === 1) {       // 最外层才提交
        lastLog = [...new Set(lastLog)]
        flush()
        lastLog = parentLog.length ? [...parentLog, ...lastLog] : lastLog
      } else {
        parentLog.push(...lastLog)    // 内层变更上交外层
        lastLog = parentLog
      }
    } catch (e) {
      state = stack.pop()             // 回滚
      lastLog = parentLog
      throw e
    }
    stack.pop()
    return lastLog
  }

  return {
    transact,
    setState,
    getState: () => state,
    subscribe(fn) { subs.add(fn); return () => subs.delete(fn) },
    log: () => lastLog,
  }
}

// 用例
const store = createTransactionalStore({ a: 1, b: 2, c: 3 })
let notified = 0
store.subscribe(() => notified++)

store.transact((set) => { set({ a: 10 }); set({ b: 20 }) })
console.log(notified, store.log())        // 1, ['a','b']：两次变更一次通知

try {
  store.transact((set, get) => { set({ c: 30 }); throw new Error('boom') })
} catch {}
console.log(store.getState().c)           // 3：回滚成功
console.log(notified)                     // 仍为 1：回滚不通知

store.transact((set) => {
  set({ a: 100 })
  store.transact((set2) => set2({ b: 200 }))   // 嵌套：内层不单独通知
})
console.log(notified)                     // 2：仅最外层提交通知一次
```
**评分点**：
- 事务内抑制通知、最外层统一 flush（嵌套合并语义）
- 快照栈回滚（异常安全：finally/ catch 恢复）
- 变更 keys 日志去重聚合（lastLog 设计）
- 嵌套时内层变更上交外层日志（归因连续性）
