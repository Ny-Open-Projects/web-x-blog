## ZSM-181：手写 usePolling（可见性感知 + 指数间隔 + 竞态守卫的轮询 Hook）
**需求**：实现 `usePolling(fn, { interval, maxInterval, enabled })`：1) 定时轮询 fn 并保存结果；2) 页面隐藏（document.hidden）时暂停，恢复可见立即触发一次；3) 连续成功 N 次无变化则间隔指数增长到 maxInterval（退避省流量）；4) 卸载/依赖变化时取消进行中请求；5) 上一次响应未返回时不叠加发起。返回 { data, error, stop }。

```javascript
export function usePolling(fn, { interval = 3000, maxInterval = 30000, enabled = true }) {
  const [state, setState] = useState({ data: null, error: null })
  const timer = useRef(), ctrl = useRef(), inFlight = useRef(false)
  const delay = useRef(interval), miss = useRef(0)
  const fnRef = useRef(fn); fnRef.current = fn

  const tick = useCallback(async () => {
    if (document.hidden || inFlight.current) return   // 隐藏/进行中：跳过
    inFlight.current = true
    ctrl.current?.abort()
    ctrl.current = new AbortController()
    try {
      const data = await fnRef.current(ctrl.current.signal)
      if (deepEqual(data, state.data)) miss.current++   // 无变化累计
      else miss.current = 0
      delay.current = Math.min(interval * 2 ** miss.current, maxInterval)
      setState({ data, error: null })
    } catch (e) {
      if (e.name !== 'AbortError') setState(s => ({ ...s, error: e }))
    } finally {
      inFlight.current = false
      timer.current = setTimeout(tick, delay.current)   // 动态间隔重排
    }
  }, [state.data])

  useEffect(() => {
    if (!enabled) return
    tick()
    const onVisible = () => { if (!document.hidden) tick() }
    document.addEventListener('visibilitychange', onVisible)
    return () => {                                       // 清理：定时器+请求
      clearTimeout(timer.current)
      ctrl.current?.abort()
      document.removeEventListener('visibilitychange', onVisible)
    }
  }, [enabled, tick])

  return { ...state, stop: () => clearTimeout(timer.current) }
}

// 评分点：
// 1. visibilitychange 暂停/恢复，隐藏期零请求
// 2. inFlight 守卫：上一轮未结束不叠加发起
// 3. AbortController 随卸载/依赖变化取消
// 4. 无变化 miss 计数 → 指数退避到 maxInterval
// 5. AbortError 不作为业务 error 暴露
```
