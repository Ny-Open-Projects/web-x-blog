## ZSM-486：手写状态层告警规则引擎
**答**：实现 `createAlertEngine()`：1) 规则：`{ metric, op: '>'|'<'|'==', threshold, forMs, level, message }`——指标越过阈值持续 forMs 才触发（防瞬时抖动）；2) `eval(metric, value)`：喂入指标值，命中条件启动计时器，期间回落取消（pending 态）；3) 触发去重：告警 active 期间不重复触发，恢复后 re-arm；4) 事件：firing/resolved，带持续时间；5) `activeAlerts()` 当前告警列表。

```javascript
export function createAlertEngine() {
  const rules = new Map()      // id → rule
  const pending = new Map()    // id → { since, timer }
  const active = new Map()     // id → { since }
  const listeners = new Set()

  function addRule(id, rule) { rules.set(id, rule) }

  function test(rule, value) {
    switch (rule.op) {
      case '>': return value > rule.threshold
      case '<': return value < rule.threshold
      case '==': return value === rule.threshold
    }
  }

  function evalMetric(metric, value) {
    for (const [id, rule] of rules) {
      if (rule.metric !== metric) continue
      if (active.has(id)) {
        if (!test(rule, value)) resolve(id)                    // 恢复
        continue
      }
      if (pending.has(id)) {
        if (!test(rule, value)) { clearTimeout(pending.get(id).timer); pending.delete(id) }   // 回落取消
        continue
      }
      if (test(rule, value)) {
        const since = Date.now()
        const timer = setTimeout(() => {
          pending.delete(id); active.set(id, { since })
          listeners.forEach((fn) => fn({ type: 'firing', id, level: rule.level, message: rule.message }))
        }, rule.forMs)
        pending.set(id, { since, timer })
      }
    }
  }

  function resolve(id) {
    const a = active.get(id)
    active.delete(id)
    listeners.forEach((fn) => fn({ type: 'resolved', id, durationMs: Date.now() - a.since }))
  }

  return { addRule, evalMetric, activeAlerts: () => [...active.keys()],
    onChange: (fn) => (listeners.add(fn), () => listeners.delete(fn)) }
}
```
**评分点**：
- forMs 持续判定（pending 态 + 回落取消——防瞬时抖动的标准结构）
- 触发去重 + re-arm（active 期间不重复、恢复后重新武装）
- firing/resolved 事件带持续时间（告警质量指标）
- 规则与指标解耦（同指标多规则独立计时）
