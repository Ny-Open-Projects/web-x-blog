## ZSM-889：手写红线合规检查器（八红线扫描 + 豁免单核验）
**答**：888 落地——实现 `complianceScan(target)`：1) 输入目标的红线达标数据（探针计数/lint结果/门禁数据等），逐条 R1-R8 判定；2) 豁免单核验：违规项有豁免单（{reason, deadline, compensate} 三要素齐全且未过期）则标记 `waived`，否则 `violated`；3) 按 level 汇总：阻断级违规 → CI 阻断；4) 附用例：豁免过期判定与阻断汇总。

```javascript
const LEVELS = { R1: '阻断', R2: 'lint', R3: '阻断', R4: 'lint', R5: '告警', R6: 'lint', R7: '告警', R8: '阻断' }

function complianceScan(target, { now = Date.now } = {}) {
  // target: { r1: { setStatePerInteraction }, r2: { lintPass }, ..., waivers: { R5: { reason, deadline, compensate } } }
  const results = []
  const checks = {
    R1: (t) => t.r1?.setStatePerInteraction != null && t.r1.setStatePerInteraction <= 3,
    R2: (t) => t.r2?.lintPass === true,
    R3: (t) => t.r3?.balanced === true,
    R4: (t) => t.r4?.partialized === true && t.r4?.throttleMs >= 1000,
    R5: (t) => t.r5?.maxCommitMs <= 16 && t.r5?.maxComponents <= 30,
    R6: (t) => t.r6?.noSetStateInFrame === true,
    R7: (t) => t.r7?.cachesEvictable === true && t.r7?.snapshotCapped === true,
    R8: (t) => t.r8?.idempotent === true && t.r8?.optimisticRollback === true,
  }

  for (const [rid, check] of Object.entries(checks)) {
    const pass = check(target)
    if (pass) { results.push({ rid, status: 'pass' }); continue }
    // 违规：核验豁免单
    const waiver = target.waivers?.[rid]
    const valid = waiver
      && waiver.reason && waiver.compensate
      && waiver.deadline > now()                          // 豁免单三要素 + 未过期
    results.push({ rid, status: valid ? 'waived' : 'violated',
      waiver: valid ? `豁免至 ${new Date(waiver.deadline).toISOString().slice(0, 10)}` : null })
  }

  const blocking = results.filter((r) => r.status === 'violated' && LEVELS[r.rid] === '阻断')
  const lintViolations = results.filter((r) => r.status === 'violated' && LEVELS[r.rid] === 'lint')
  const alerts = results.filter((r) => r.status === 'violated' && LEVELS[r.rid] === '告警')

  return {
    results,
    ciBlocked: blocking.length > 0,                       // 阻断级 → CI 拒绝
    summary: { pass: results.filter((r) => r.status === 'pass').length, waived: results.filter((r) => r.status === 'waived').length, violated: results.filter((r) => r.status === 'violated').length, blocking: blocking.map((b) => b.rid), lint: lintViolations.map((b) => b.rid), alert: alerts.map((b) => b.rid) },
  }
}

// 用例
const t1 = {
  r1: { setStatePerInteraction: 2 }, r2: { lintPass: true }, r3: { balanced: true },
  r4: { partialized: true, throttleMs: 1500 }, r5: { maxCommitMs: 18, maxComponents: 25 },
  r6: { noSetStateInFrame: true }, r7: { cachesEvictable: true, snapshotCapped: true },
  r8: { idempotent: true, optimisticRollback: true },
  waivers: { R5: { reason: '大促临时图表', deadline: Date.now() + 864e5, compensate: 'worker 化计算' } },
}
const s1 = complianceScan(t1)
console.log(s1.ciBlocked)      // false：R5 豁免有效
console.log(s1.summary)        // { pass: 7, waived: 1, violated: 0 }

// 豁免过期：violated 且阻断
const t2 = { ...t1, r8: { idempotent: false, optimisticRollback: false },
  waivers: { R5: { reason: 'x', deadline: Date.now() - 1000, compensate: 'y' } } }
const s2 = complianceScan(t2)
console.log(s2.ciBlocked)      // true：R8 无豁免（阻断级）+ R5 豁免过期
console.log(s2.summary.blocking)   // ['R8']
```
**评分点**：
- 八红线逐条判定函数表（可扩展）
- 豁免单三要素核验（reason/compensate/deadline 未过期）
- level 汇总三态（阻断/lint/告警）→ ciBlocked 判定
- 豁免过期的 violated 转化
