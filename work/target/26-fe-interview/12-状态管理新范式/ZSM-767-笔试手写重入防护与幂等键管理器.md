## ZSM-767：手写重入防护与幂等键管理器（审计 #4 的完整实现）
**答**：实现 `createGuardedActions()`：1) `guard(name, fn, { mode: 'reject'|'queue'|'throttle', cooldown })`：重入三模式——reject（直接拒绝）、queue（排队，前一个完成后执行）、throttle（冷却窗口内合并参数取最后一次）；2) 幂等键：同 key 的重试沿用（防服务端重复），成功后销毁；3) 执行追踪：`inFlight()` 活跃 action 清单（可观测）；4) 附用例：三模式行为 + 幂等键沿用。

```javascript
function createGuardedActions({ onEvent } = {}) {
  const inFlight = new Map()      // name -> { startedAt, args }
  const idemKeys = new Map()      // name -> key（沿用直到成功）
  const queues = new Map()        // name -> 参数队列
  const lastRun = new Map()       // name -> ts（throttle）

  function makeIdemKey(name) {
    if (!idemKeys.has(name)) idemKeys.set(name, crypto.randomUUID())
    return idemKeys.get(name)     // 重试沿用：失败不清除
  }

  async function guard(name, fn, { mode = 'reject', cooldown = 300 } = {}) {
    if (inFlight.has(name)) {
      if (mode === 'reject') {
        onEvent?.({ type: 'guard.reject', name })
        return { ok: false, reason: 'IN_FLIGHT' }          // 直接拒绝
      }
      if (mode === 'queue') {
        // 排队：等前一个完成
        await new Promise((r) => {
          const q = queues.get(name) ?? queues.set(name, []).get(name)
          q.push(r)
        })
      }
      if (mode === 'throttle') {
        const last = lastRun.get(name) ?? 0
        if (Date.now() - last < cooldown) {
          return { ok: false, reason: 'THROTTLED' }        // 冷却窗口内拒绝（合并参数由调用方做）
        }
      }
    }
    inFlight.set(name, { startedAt: Date.now() })
    lastRun.set(name, Date.now())
    const idemKey = makeIdemKey(name)
    try {
      const result = await fn({ idemKey })                 // 幂等键传给 fn
      idemKeys.delete(name)                                // 成功销毁
      return { ok: true, result }
    } catch (e) {
      // 失败：幂等键保留（重试沿用）
      onEvent?.({ type: 'guard.fail', name, idemKey })
      return { ok: false, reason: 'FAILED', error: e }
    } finally {
      inFlight.delete(name)
      const q = queues.get(name)
      if (q?.length) q.shift()?.()                         // 唤醒排队者
    }
  }

  return {
    guard,
    inFlight: () => [...inFlight.entries()].map(([n, v]) => ({ name: n, startedAt: v.startedAt })),
    idemKeys: () => [...idemKeys.entries()],
  }
}

// 用例
const events = []
const ga = createGuardedActions({ onEvent: (e) => events.push(e) })

// 1) reject 模式：双击防护
const submit = (ga.guard('submit', async ({ idemKey }) => { await new Promise((r) => setTimeout(r, 50)); return idemKey }))
const r1 = await submit()
const r2 = await submit()        // 在途：被拒
console.log(r1.ok, r2)           // true { ok: false, reason: 'IN_FLIGHT' }

// 2) 失败后幂等键沿用
let attempts = 0
const flaky = ga.guard('pay', async ({ idemKey }) => {
  attempts++
  if (attempts === 1) throw new Error('timeout')
  return idemKey
})
await flaky()                    // 失败
const keys1 = ga.idemKeys()      // 键保留
await flaky()                    // 重试
const keys2 = ga.idemKeys()
console.log(keys1.length === 1, keys2.length === 0)   // true true：沿用后销毁

// 3) throttle：冷却窗口
const search = ga.guard('search', async () => 'ok', { mode: 'throttle', cooldown: 100 })
await search()
const r3 = await search()        // 冷却内：拒绝
console.log(r3.reason)           // 'THROTTLED'
```
**评分点**：
- 三种重入模式（reject/queue/throttle）语义分离
- 幂等键生命周期（失败沿用、成功销毁——防服务端重复下单）
- queue 模式的 FIFO 唤醒（前完成后执行）
- inFlight 可观测（活跃清单审计）
