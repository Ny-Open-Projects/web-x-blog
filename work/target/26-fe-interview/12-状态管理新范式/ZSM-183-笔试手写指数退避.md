## ZSM-183：手写指数退避重试（带抖动与可重试错误分类）
**答**：实现 `retry(fn, opts)`：1) 指数退避 `delay = base * 2^n`，上限 maxDelay；2) 每次延迟加 ±50% 抖动（防雪崩同步重试）；3) `isRetryable(err)` 决定是否重试（网络错误/5xx/429 重试，4xx 业务错直接抛）；4) 尊重 429 响应头的 Retry-After（若有则覆盖计算延迟）；5) 每次重试前调用 onRetry 回调；6) 总重试次数上限 attempts（不含首试）。

```javascript
const sleep = (ms) => new Promise(r => setTimeout(r, ms))

export async function retry(fn, {
  attempts = 3, base = 1000, maxDelay = 30000,
  isRetryable = defaultIsRetryable, onRetry = () => {},
} = {}) {
  let lastErr
  for (let n = 0; n <= attempts; n++) {
    try {
      return await fn()
    } catch (err) {
      lastErr = err
      if (n === attempts || !isRetryable(err)) throw err   // 到限或不可重试
      let delay = Math.min(base * 2 ** n, maxDelay)
      if (err.retryAfter) delay = err.retryAfter * 1000    // 429: 服务端说了算
      const jitter = delay * (0.5 + Math.random() * 0.5)   // +0~50% 抖动
      onRetry(err, n + 1, jitter)
      await sleep(jitter)
    }
  }
  throw lastErr
}

function defaultIsRetryable(err) {
  if (err.status === 429 || err.status >= 500) return true
  if (err.status >= 400 && err.status < 500) return false  // 业务错不重试
  return err.name === 'TypeError' || err.name === 'FetchError' // 网络层错误
}
```
