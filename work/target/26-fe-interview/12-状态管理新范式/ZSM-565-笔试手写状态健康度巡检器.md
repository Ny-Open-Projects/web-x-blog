## ZSM-565：手写状态健康度巡检器
**答**：实现 `createAuditor()` 对状态库做健康巡检：1) 注册巡检规则 `rule(name, check(state) => issues[])`（如未定义字段检测、NaN 检测、引用泄漏检测——数组长度超阈值、重复 key）；2) `audit(store)` 执行全部规则返回报告 `{ passed, issues: [{ rule, path, message }] }`；3) `monitor(store, interval)`：周期巡检 + 变更计数（1s 内 set 次数超阈值告警「状态风暴」）；4) 报告聚合：按规则统计 issue 数并排序；5) 附用例：注入 NaN 和超长数组验证检出。

```javascript
function createAuditor() {
  const rules = new Map()

  function rule(name, check) {
    rules.set(name, check)
    return () => rules.delete(name)
  }

  function audit(state) {
    const issues = []
    for (const [name, check] of rules) {
      const found = check(state) ?? []
      found.forEach((f) => issues.push({ rule: name, ...f }))
    }
    return {
      passed: issues.length === 0,
      issues,
      summary() {
        const byRule = new Map()
        for (const i of issues) byRule.set(i.rule, (byRule.get(i.rule) ?? 0) + 1)
        return [...byRule.entries()].sort((a, b) => b[1] - a[1])
      },
    }
  }

  function monitor(store, { interval = 5000, maxSetsPerSec = 30 } = {}) {
    let setCount = 0
    let windowStart = Date.now()
    const origSet = store.setState
    store.setState = (...args) => {
      const now = Date.now()
      if (now - windowStart > 1000) { windowStart = now; setCount = 0 }
      setCount++
      if (setCount > maxSetsPerSec) {
        console.warn(`[auditor] 状态风暴: ${setCount} sets/s`)
        setCount = 0 // 防连续告警刷屏
      }
      return origSet(...args)
    }
    const timer = setInterval(() => audit(store.getState()), interval)
    return () => clearInterval(timer)
  }

  return { rule, audit, monitor }
}

// 内置规则示例
const auditor = createAuditor()
auditor.rule('nan-check', (state) => {
  const out = []
  for (const [k, v] of Object.entries(state)) {
    if (typeof v === 'number' && Number.isNaN(v)) out.push({ path: k, message: `NaN at ${k}` })
  }
  return out
})
auditor.rule('array-bloat', (state) => {
  const out = []
  for (const [k, v] of Object.entries(state)) {
    if (Array.isArray(v) && v.length > 1000) out.push({ path: k, message: `${k} length=${v.length}` })
  }
  return out
})

// 用例
const store = { state: { price: NaN, cache: new Array(2000).fill(0), ok: 1 },
  setState(p) { Object.assign(this.state, p) } }
const report = auditor.audit(store.state)
console.log(report.passed)                  // false
console.log(report.summary())               // [['nan-check', 1], ['array-bloat', 1]]
const stop = auditor.monitor(store, { maxSetsPerSec: 3 })
for (let i = 0; i < 10; i++) store.setState({ ok: i })  // 10 次/秒 → 告警
stop()
```
**评分点**：
- 规则注册与报告结构（rule/path/message 可定位）
- 状态风暴检测：1s 窗口计数 + 阈值告警
- 巡检器对 setState 的侵入包装可拆卸（返回 stop）
- summary 按规则聚合排序（报告可读性）
