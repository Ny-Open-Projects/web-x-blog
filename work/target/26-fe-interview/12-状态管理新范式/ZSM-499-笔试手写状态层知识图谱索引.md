## ZSM-499：手写状态层知识图谱索引
**答**：实现团队状态实践的知识图谱 `createKnowledgeGraph()`：1) 节点：`addNode(id, { kind: 'pattern'|'pitfall'|'decision', title, links })`——边表示「适用/规避/替代」关系；2) `query(patternId)`：返回该模式的邻居（规避的坑/替代方案）按关系类型分组；3) 环检测（A 替代 B、B 替代 A 报错）；4) `path(from, to)`：BFS 找两个实践之间的关联路径（如「我用的模式 X 与新方案 Y 什么关系」）；5) `stats()` 节点/边/孤立节点统计。

```javascript
export function createKnowledgeGraph() {
  const nodes = new Map()      // id → { kind, title, links: [{to, rel}] }

  function addNode(id, { kind, title, links = [] }) {
    if (nodes.has(id)) throw new Error(`duplicate node ${id}`)
    nodes.set(id, { kind, title, links })
    // 环检测：替代关系成环报错
    if (links.some((l) => l.rel === '替代' && nodes.get(l.to)?.links.some((l2) => l2.to === id && l2.rel === '替代')))
      throw new Error(`替代关系成环: ${id}`)
  }

  function query(id) {
    const n = nodes.get(id)
    if (!n) return null
    const grouped = {}
    n.links.forEach((l) => { (grouped[l.rel] ??= []).push(nodes.get(l.to)) })
    return grouped
  }

  function path(from, to) {
    const visited = new Set([from])
    const queue = [[from, [from]]]
    while (queue.length) {
      const [cur, p] = queue.shift()
      if (cur === to) return p
      for (const l of nodes.get(cur)?.links ?? []) {
        if (!visited.has(l.to)) { visited.add(l.to); queue.push([l.to, [...p, l.to]]) }
      }
    }
    return null
  }

  return {
    addNode, query, path,
    stats: () => {
      const linked = new Set()
      nodes.forEach((n) => n.links.forEach((l) => linked.add(l.to)))
      let orphans = 0
      nodes.forEach((n, id) => { if (!n.links.length && !linked.has(id)) orphans++ })
      const edges = [...nodes.values()].reduce((a, n) => a + n.links.length, 0)
      return { nodes: nodes.size, edges, orphans }
    },
  }
}
```
**评分点**：
- 关系类型分组查询（适用/规避/替代——图谱的查询价值）
- 替代关系环检测（知识一致性）
- BFS 最短路径（两实践关联性查询）
- stats 含孤立节点（知识库健康度：孤立即未关联的碎片知识）
