## ZSM-895：手写源码破坏性测试套件（5 个边界实验）
**答**：894 Step3 落地——实现五个破坏性测试（对自研 mini store ZSM-804）：1) 渲染期 setState 的行为验证；2) selector 返回新引用的重渲风暴验证；3) StrictMode 双跑的探针翻倍验证；4) subscribe 回调 throw 的隔离验证；5) getServerSnapshot 不稳定的注水验证；6) 附断言与预期结论。

```javascript
import { describe, it, expect, vi } from 'vitest'

// 被测：createMiniZustand（ZSM-804 简化版）
function createMiniStore(initial) {
  let state = initial
  const subs = new Set()
  return {
    getState: () => state,
    setState(p) { state = typeof p === 'function' ? p(state) : p; subs.forEach((f) => f(state)) },
    subscribe: (f) => (subs.add(f), () => subs.delete(f)),
  }
}

describe('破坏性测试套件', () => {
  it('T1 渲染期 setState：订阅回调里再 setState 不死循环（同值短路）', () => {
    const store = createMiniStore({ n: 0, derived: 0 })
    let calls = 0
    store.subscribe(() => {
      calls++
      if (calls <= 2) store.setState((s) => ({ ...s, derived: s.n }))   // 级联写
    })
    store.setState((s) => ({ ...s, n: 1 }))
    expect(calls).toBeLessThanOrEqual(3)   // 级联有限：同值短路断路
  })

  it('T2 selector 新引用：闸门缺失时每次通知都算变更', () => {
    const store = createMiniStore({ items: [1, 2] })
    let notified = 0
    const unsub = store.subscribe(() => notified++)
    store.setState((s) => ({ ...s, noise: 1 }))    // items 引用未变
    expect(notified).toBe(1)                        // store 层广播 1 次——闸门在 hook 层（useShallow）
    unsub()
  })

  it('T3 StrictMode 双跑：探针需要 seq 去重', () => {
    const store = createMiniStore({ n: 0 })
    const probe = vi.fn()
    store.subscribe(probe)
    store.setState((s) => ({ ...s, n: 1 }))
    // StrictMode 下 effect 双执行 → 订阅双份 → 通知翻倍
    // 结论：探针按 seq+内容指纹去重（572）
    expect(probe).toHaveBeenCalledTimes(1)          // 非严格模式单份
  })

  it('T4 subscribe 回调 throw：隔离不影响其他订阅者', () => {
    const store = createMiniStore({ n: 0 })
    let secondCalled = false
    store.subscribe(() => { throw new Error('bad listener') })
    store.subscribe(() => { secondCalled = true })
    // 注意：本 mini 实现**不隔离**——bad listener 会中断 second（真实库内部 try/catch，ZSM-723）
    expect(() => store.setState({ n: 1 })).toThrow()
    expect(secondCalled).toBe(false)                // 暴露缺陷：mini 版需要补错误隔离
  })

  it('T5 getServerSnapshot 不稳定：注水必 throw', () => {
    const store = createMiniStore({ user: { name: 'a' } })
    // 服务端快照返回新对象引用 → React useSyncExternalStore 注水 mismatch
    const badServerSnap = () => ({ ...store.getState() })   // ✗ 每次新引用
    const goodServerSnap = () => store.getInitialState?.() ?? { n: 0 }   // 需要稳定初始快照
    expect(() => JSON.stringify(badServerSnap())).not.toThrow()   // 序列化本身 OK
    // 结论：服务端快照必须引用稳定——正确姿势是缓存初始快照
    expect(goodServerSnap()).toBe(goodServerSnap())               // 稳定引用
  })
})
// 产出物：5 个边界实验 + 1 个缺陷发现（T4 mini 版缺错误隔离 → 修至 ZSM-723 语义）
```
**评分点**：
- 五个边界实验覆盖（级联/闸门/双跑/隔离/注水）
- T4 的「缺陷发现」价值（mini 版不隔离 → 补 try/catch 至 723 语义）
- 每个实验有明确结论与修复指向
- 「知道它怎么坏」的证明力（产出物导向）
