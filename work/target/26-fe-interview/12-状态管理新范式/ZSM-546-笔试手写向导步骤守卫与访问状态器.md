## ZSM-546：手写向导步骤守卫与访问状态器
**答**：实现 `createWizard(steps)`：1) steps 为 `[{ key, canEnter, canLeave }]`，`canLeave(state)` 返回 false 时拦截离开（如「必填未完成」）；2) `go(index)` 任意跳转：路径上**逐个通过中间步骤的 canEnter/canLeave**（跳两步要过两道门）；3) 记录 `visited`（到过即算，含被拦截后的重试）与 `completed`（canLeave 通过过才算）；4) `back()` 不受 canEnter 限制（回退自由，前进受限）；5) `progress()` 返回 0~1 进度（按 completed 占比）。附用例验证「跳步逐门校验」与拦截。

```javascript
function createWizard(steps, getState) {
  let current = 0
  const visited = new Set([steps[0].key])
  const completed = new Set()

  function go(index, { forceBack = false } = {}) {
    if (index === current) return true
    if (index < current && !forceBack) return back(index)      // 回退不受 canEnter 限
    // 前进：逐门校验
    const dir = index > current ? 1 : -1
    let cur = current
    while (cur !== index) {
      const nextStep = steps[cur + dir]
      // 离开当前步：canLeave（前进方向才查，回退跳过）
      if (dir === 1 && steps[cur].canLeave && !steps[cur].canLeave(getState())) {
        return { blocked: steps[cur].key }
      }
      // 进入下一步：canEnter
      if (nextStep.canEnter && !nextStep.canEnter(getState())) {
        return { blocked: nextStep.key }
      }
      cur += dir
      visited.add(steps[cur].key)
    }
    current = index
    return true
  }

  function back(index) {
    current = Math.max(0, index)   // back 自由
    return true
  }

  function next() {
    return go(current + 1)
  }

  // 前进成功时标记当前步完成
  function tryComplete() {
    const res = next()
    if (res === true) completed.add(steps[current - 1]?.key)
    return res
  }

  return {
    go, next, tryComplete, back,
    current: () => current,
    visited: () => [...visited],
    completed: () => [...completed],
    progress: () => completed.size / (steps.length - 1 || 1),
  }
}

// 用例
const state = { name: '', plan: null, paid: false }
const wiz = createWizard([
  { key: 'info', canLeave: () => !!state.name },
  { key: 'plan', canEnter: () => !!state.name, canLeave: () => !!state.plan },
  { key: 'pay', canEnter: () => !!state.plan },
  { key: 'done', canEnter: () => state.paid },
], () => state)

console.log(wiz.go(3))          // { blocked: 'info' }：name 未填，第一道门就拦
state.name = '张三'
console.log(wiz.tryComplete())  // true → 进入 plan
console.log(wiz.go(3))          // { blocked: 'plan' }：plan 未选
state.plan = 'pro'
state.paid = true
console.log(wiz.go(3))          // true：逐门通过 info→plan→pay→done
console.log(wiz.progress())     // 0.75：info/plan/pay 完成，done 未走 canLeave
console.log(wiz.visited())      // ['info','plan','pay','done']
```
**评分点**：
- 逐门校验（跳步经过的每个中间步骤都过 canEnter/canLeave）
- 回退自由、前进受限的方向性差异
- visited 与 completed 语义分离（到过≠完成）
- blocked 返回具体步骤 key（可定位到哪道门拦截）
