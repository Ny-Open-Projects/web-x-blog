## ZSM-613：手写状态变更的火焰数据采集与聚合
**答**：实现 `createFlameProfiler()`：1) `record(action, fn)`：执行 fn 并用 performance.mark/measure 记录（命名 `state:{action}:{seq}`），返回耗时；2) `flame(action)`：输出该 action 的调用树（嵌套子段——`record` 内部可递归 `record(child)`，栈式记录父子关系）；3) `aggregate()`：把多次调用树聚合为「按名字合并的火焰图」——同层同名节点合并累计耗时与调用次数（自用时间 = 总时间 - 子节点时间和）；4) 输出 Top 自用耗时函数。附用例：两层嵌套的聚合验证。

```javascript
function createFlameProfiler() {
  const stack = []            // 调用栈：{ name, children, self }
  const roots = []

  function record(action, fn) {
    const name = `state:${action}:${seq}`
    const node = { name, children: [], self: 0, t0: performance.now(), count: 1 }
    stack.length ? stack.at(-1).children.push(node) : roots.push(node)
    stack.push(node)
    try {
      return fn()
    } finally {
      const total = performance.now() - node.t0
      // 自用时间 = 总时长 - 子节点总时长（在子节点结束后扣）
      stack.pop()
      const childSum = node.children.reduce((a, c) => a + (c.total ?? 0), 0)
      node.total = total
      node.self = Math.max(0, total - childSum)
      performance.measure(name, { start: node.t0 })   // Performance 面板可见
    }
  }
  let seq = 0

  // 聚合：同层同名合并
  function aggregate() {
    function mergeNodes(nodes) {
      const byName = new Map()
      for (const n of nodes) {
        const cur = byName.get(n.name) ?? { name: n.name, count: 0, total: 0, self: 0, children: [] }
        cur.count++; cur.total += n.total; cur.self += n.self
        cur.children.push(...n.children)
        byName.set(n.name, cur)
      }
      const merged = [...byName.values()]
      for (const m of merged) m.children = mergeNodes(m.children)
      return merged.sort((a, b) => b.self - a.self)   // 自用耗时降序
    }
    return mergeNodes(roots)
  }

  return { record, aggregate }
}

// 用例：嵌套段聚合
const fp = createFlameProfiler()
fp.record('submit', () => {
  fp.record('validate', () => { busy(2) })
  fp.record('api', () => { busy(10) })
  fp.record('validate', () => { busy(1) })   // 同名段出现两次
})
fp.record('submit', () => {
  fp.record('validate', () => { busy(3) })
})
function busy(ms) { const t = performance.now() + ms; while (performance.now() < t); }

const tree = fp.aggregate()
console.log(JSON.stringify(tree, null, 0).slice(0, 200))
// submit: count=2, total≈18ms, self≈0
//   ├─ validate: count=3, self≈6ms
//   └─ api: count=1, self≈10ms  ← 自用耗时 Top：优化目标
const top = tree[0].children.map((c) => `${c.name} self=${c.self.toFixed(1)}ms count=${c.count}`)
```
**评分点**：
- 栈式父子记录（递归 record 形成调用树）
- 自用时间 = 总时长 - 子节点合计（火焰图核心语义）
- 跨调用树的同名合并（count/total/self 累计）
- 输出按自用耗时排序（直接给出优化优先级）
