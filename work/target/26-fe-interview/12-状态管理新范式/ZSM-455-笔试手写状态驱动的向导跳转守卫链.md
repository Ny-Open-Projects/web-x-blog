## ZSM-455：手写向导跳转守卫链
**需求**：实现 `createGuardChain()`：1) `use(guard)` 注册守卫：`(to, from) => true | false | { redirect: string } | Promise`；2) `go(to)` 依次执行守卫：第一个返回 false 阻止（返回拒绝原因）、redirect 则改道（改道目标也过守卫，防环：改道深度上限 3）；3) 守卫支持异步（Promise.all 顺序不保证——需串行 await）；4) 守卫异常按「阻止」处理（安全默认）；5) `go` 返回最终结果 { ok, to, blockedBy? }。

```javascript
export function createGuardChain() {
  const guards = []

  function use(guard) { guards.push(guard); return () => guards.splice(guards.indexOf(guard), 1) }

  async function go(to, from = null, depth = 0) {
    if (depth > 3) return { ok: false, to, blockedBy: 'redirect-loop' }   // 改道环上限
    for (const g of guards) {
      try {
        const r = await g(to, from)                     // 串行：顺序语义明确
        if (r === false) return { ok: false, to, blockedBy: g.name || 'guard' }
        if (r && typeof r === 'object' && r.redirect) {
          if (r.redirect === to) return { ok: false, to, blockedBy: 'self-redirect' }
          return go(r.redirect, to, depth + 1)          // 改道也过守卫
        }
      } catch (e) {
        return { ok: false, to, blockedBy: `guard-error: ${g.name}` }   // 异常 = 阻止（安全默认）
      }
    }
    return { ok: true, to }
  }

  return { use, go }
}

// 用法
chain.use(async function auth(to) { return to !== 'pay' || !!session.user })
chain.use(async function stepGate(to) { return to !== 'step3' || wizard.step2Done })
```
**评分点**：
- 守卫串行 await（短路语义明确：第一个 false 即停）
- redirect 改道也过守卫 + 环防护（深度上限 + 自指检测）
- 异常按阻止处理（fail-safe 默认，而非 fail-open）
- use 返回退订（守卫可动态装卸）
