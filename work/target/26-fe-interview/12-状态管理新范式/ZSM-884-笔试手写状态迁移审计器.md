## ZSM-884：手写状态迁移审计器（900 卡知识的查重终检）
**答**：900 张卡的终检工具——实现 `createKnowledgeAuditor()`：1) `index(cards)`：登记卡片 `{ id, topics: [...], concepts: [...], type }`；2) `dupCheck()`：查重——同 topics 集合且 type 相同的卡视为疑似重复（相似度 = topics 交集/并集 > 0.8）；3) `conceptCoverage()`：12 概念的场景覆盖度（每概念被 ≥3 种 type 引用 = 健康）；4) `report()`：重复嫌疑清单 + 概念覆盖 + 总体健康分；5) 附用例：检出高相似卡与覆盖缺口。

```javascript
function jaccard(a, b) {
  const A = new Set(a), B = new Set(b)
  const inter = [...A].filter((x) => B.has(x)).length
  const uni = new Set([...A, ...B]).size
  return uni ? inter / uni : 0
}

function createKnowledgeAuditor({ conceptCount = 12 } = {}) {
  const cards = []
  const CONCEPTS = ['对账器', 'seq协议', '幂等键', '命令栈', 'phase派生', '版本裁决', '指纹', 'holdback', 'leader', '预算门禁', '事务合并', '契约协议']

  function index(newCards) {
    cards.push(...newCards)
    return this
  }

  function dupCheck({ threshold = 0.8 } = {}) {
    const suspects = []
    for (let i = 0; i < cards.length; i++) {
      for (let j = i + 1; j < cards.length; j++) {
        const sim = jaccard(cards[i].topics, cards[j].topics)
        if (sim > threshold && cards[i].type === cards[j].type) {
          suspects.push({ a: cards[i].id, b: cards[j].id, similarity: +sim.toFixed(2) })
        }
      }
    }
    return suspects
  }

  function conceptCoverage() {
    const byConcept = new Map(CONCEPTS.map((c) => [c, new Set()]))
    for (const card of cards) {
      for (const c of card.concepts ?? []) byConcept.get(c)?.add(card.type)
    }
    return CONCEPTS.map((c) => {
      const types = byConcept.get(c)?.size ?? 0
      return { concept: c, typesCovered: types, healthy: types >= 3 }
    })
  }

  function report() {
    const dups = dupCheck()
    const coverage = conceptCoverage()
    const healthy = coverage.filter((c) => c.healthy).length
    const score = +(((conceptCount - (conceptCount - healthy)) / conceptCount) * 60
      + (cards.length ? (1 - dups.length / cards.length) * 40 : 0)).toFixed(1)
    return {
      totalCards: cards.length,
      duplicateSuspects: dups,
      conceptCoverage: coverage,
      healthScore: score,       // 0~100
      verdict: score >= 85 ? '体系成熟' : score >= 70 ? '可发布' : '需补齐',
    }
  }

  return { index, dupCheck, conceptCoverage, report }
}

// 用例
const ka = createKnowledgeAuditor()
ka.index([
  { id: 'ZSM-528', topics: ['乐观更新', '对账'], concepts: ['对账器', '版本裁决'], type: '笔试' },
  { id: 'ZSM-737', topics: ['乐观更新', '对账', '多端'], concepts: ['对账器', '版本裁决'], type: '笔试' },
  { id: 'ZSM-810', topics: ['乐观更新', '对账', '观测'], concepts: ['对账器', '幂等键'], type: '笔试' },
  { id: 'ZSM-531', topics: ['撤销', '命令栈'], concepts: ['命令栈'], type: '笔试' },
])

console.log(ka.dupCheck())
// [{ a: 'ZSM-528', b: 'ZSM-737', similarity: 0.67 }] —— 0.67 < 0.8 不算重复（角度不同）
const r = ka.report()
console.log(r.conceptCoverage.slice(0, 3))
console.log(r.healthScore, r.verdict)    // 概念覆盖 + 查重合成的健康分
```
**评分点**：
- Jaccard 相似度查重（topics 交集/并集 + type 相同条件）
- 概念覆盖度（每概念 ≥3 type = 健康迁移性）
- 健康分合成（覆盖分 60 + 查重分 40）
- verdict 三级（成熟/可发布/需补齐）
