## ZSM-491：手写状态层巡检机器人
**答**：实现 `createPatrolBot(checks)`：1) checks：巡检项数组 `{ name, every: ms, run: async () => { ok, detail } }`（缓存命中率/订阅泄漏/错误率/内存水位等）；2) 调度：各项按自己的 every 独立调度（串行执行防互相打爆）；3) 失败处理：连续 2 次 fail 告警（连败计数），恢复自动清零；4) 巡检结果环形保留 50 条（历史可查）；5) `stop()` 清全部定时器；6) 机器人自身异常隔离（一个巡检项崩不影响其他）。

```javascript
export function createPatrolBot(checks) {
  const results = []       // { name, ok, detail, at } 环形 50
  const failCounts = new Map()
  const timers = new Map()
  let running = false
  const listeners = new Set()

  async function runCheck(check) {
    if (running) return                                    // 串行防打爆
    running = true
    try {
      const r = await check.run()
      results.push({ name: check.name, ...r, at: Date.now() })
      if (results.length > 50) results.shift()
      const fails = (failCounts.get(check.name) ?? 0)
      if (!r.ok) {
        if (fails + 1 >= 2) listeners.forEach((fn) => fn({ type: 'alert', name: check.name, detail: r.detail }))
        failCounts.set(check.name, fails + 1)
      } else failCounts.set(check.name, 0)                 // 恢复清零
    } catch (e) {
      listeners.forEach((fn) => fn({ type: 'bot-error', name: check.name, error: String(e) }))   // 隔离
    } finally { running = false }
  }

  function start() {
    checks.forEach((c) => timers.set(c.name, setInterval(() => runCheck(c), c.every)))
  }

  return {
    start,
    stop: () => { timers.forEach(clearInterval); timers.clear() },
    history: (name) => results.filter((r) => !name || r.name === name),
    onAlert: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
  }
}
```
**评分点**：
- 每项独立周期调度 + 全局串行锁（防巡检风暴，也防项间资源竞争）
- 连败告警 + 恢复清零（告警去噪的标准结构）
- 结果环形 50 条（历史可查不膨胀）；巡检项异常隔离
- stop 完整清理
