## ZSM-874：手写问卷回收引擎（版本化回收 + 实时看板 + 提交终审）
**答**：873 落地——实现 `createSurveyEngine()`：1) `submit(answers, schemaVersion)`：版本校验（过期版本拒绝并返回最新版本号）+ 服务端终审校验（必填/逻辑跳转）+ 幂等令牌；2) 回收计数实时看板：WS 增量 + 节流通知（1s 合并）；3) `materializeStats()`：统计物化（按快照计数——统计期间新回收不影响结果，一致性快照）；4) 附用例：版本拒绝、节流看板、快照一致性。

```javascript
function createSurveyEngine({ schemaVersion: currentV = 3 } = {}) {
  let version = currentV
  const submissions = []        // { version, answers, status, at }
  const tokenUsed = new Set()
  let liveCount = 0
  let notifyTimer = null
  const listeners = new Set()
  let counting = 0

  function submit({ answers, schemaVersion, idemToken }) {
    // 幂令牌
    if (tokenUsed.has(idemToken)) return { ok: false, code: 'DUPLICATE' }
    // 版本校验：过期 schema 拒绝
    if (schemaVersion !== version) {
      return { ok: false, code: 'SCHEMA_DRIFT', latestVersion: version }
    }
    // 服务端终审：必填校验（简版）
    const required = SCHEMA[version].filter((f) => f.required).map((f) => f.key)
    const missing = required.filter((k) => answers[k] === undefined || answers[k] === '')
    if (missing.length) return { ok: false, code: 'INVALID', missing }

    tokenUsed.add(idemToken)
    submissions.push({ version, answers, status: 'submitted', at: Date.now() })
    liveCount++
    scheduleNotify()          // 实时看板节流
    return { ok: true, id: submissions.length }
  }

  // 实时看板节流：1s 合并
  function scheduleNotify() {
    counting++
    clearTimeout(notifyTimer)
    notifyTimer = setTimeout(() => {
      listeners.forEach((fn) => fn({ type: 'live-count', count: liveCount, batch: counting }))
      counting = 0
    }, 1000)
  }

  // 统计物化：一致性快照（统计期间新回收不影响）
  function materializeStats() {
    const snapshotCount = liveCount               // 快照计数
    const snapshot = submissions.slice(0, snapshotCount)
    const stats = {}
    for (const f of SCHEMA[version]) {
      if (f.type !== 'enum') continue
      stats[f.key] = {}
      for (const s of snapshot) {
        const v = s.answers[f.key]
        if (v != null) stats[f.key][v] = (stats[f.key][v] ?? 0) + 1
      }
    }
    return { at: Date.now(), sampleCount: snapshotCount, stats }   // 物化结果自带快照标记
  }

  function upgradeSchema(next) { version = next }    // schema 升级：旧版本回收继续有效（版本化隔离）

  return { submit, materializeStats, upgradeSchema, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)), count: () => liveCount }
}

const SCHEMA = {
  3: [{ key: 'plan', required: true, type: 'enum' }, { key: 'note', required: false, type: 'text' }],
  4: [{ key: 'plan', required: true, type: 'enum' }, { key: 'budget', required: true, type: 'number' }],
}

// 用例
const eng = createSurveyEngine({ schemaVersion: 3 })
eng.subscribe((e) => { if (e.type === 'live-count') console.log('live', e.count) })

console.log(eng.submit({ answers: { plan: 'pro' }, schemaVersion: 3, idemToken: 't1' }))   // ok
console.log(eng.submit({ answers: { plan: 'pro' }, schemaVersion: 3, idemToken: 't1' }))   // DUPLICATE：令牌幂等
console.log(eng.submit({ answers: {}, schemaVersion: 3, idemToken: 't2' }))                // INVALID 缺 plan
console.log(eng.submit({ answers: { plan: 'free' }, schemaVersion: 2, idemToken: 't3' }))  // SCHEMA_DRIFT latestVersion 3

await new Promise((r) => setTimeout(r, 1100))
console.log(eng.count())                          // 1：实时计数

const stats = eng.materializeStats()
eng.submit({ answers: { plan: 'free' }, schemaVersion: 3, idemToken: 't4' })   // 物化后新回收
console.log(stats.sampleCount, eng.count())       // 1 2：快照一致性（统计不含新回收）
```
**评分点**：
- 三重提交防御（幂令牌/版本校验/必填终审）
- 版本化回收隔离（upgradeSchema 后旧回收有效）
- 看板节流 vs 物化快照的分级（实时计数/一致报表）
- materializeStats 自带快照标记（at/sampleCount 可审计）
