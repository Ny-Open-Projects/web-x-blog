## ZSM-304：手写带版本迁移与节流的状态持久化层
**答**：实现 `createPersistence(store, opts)`：1) 订阅 store，debounce 500ms 后把 `opts.pick(state)` 写入 localStorage；2) 写入时附带 `version`；3) 初始化时读取并按版本处理：相同版本合并恢复、旧版本走 opts.migrate、损坏 JSON 丢弃并回调 opts.onError；4) 提供 `flush()` 同步立即写（pagehide 兜底）；5) 配额溢出（QuotaExceededError）不抛出，清理后重试一次。

```javascript
export function createPersistence(store, { key, version, pick, migrate, onError }) {
  let timer = null

  function write() {
    try {
      localStorage.setItem(key, JSON.stringify({ v: version, state: pick(store.getState()) }))
    } catch (e) {
      if (e?.name === 'QuotaExceededError') {
        try { localStorage.clear(); localStorage.setItem(key, JSON.stringify({ v: version, state: pick(store.getState()) })) }
        catch (e2) { onError?.(e2) }                // 二次失败放弃
      } else onError?.(e)
    }
  }

  const unsub = store.subscribe(() => {
    clearTimeout(timer)
    timer = setTimeout(write, 500)                  // 节流
  })

  function restore() {
    try {
      const raw = localStorage.getItem(key)
      if (!raw) return
      const { v, state } = JSON.parse(raw)
      store.setState(v === version ? state : migrate?.(state, v) ?? undefined)  // 版本分支
    } catch (e) { onError?.(e); localStorage.removeItem(key) }                  // 损坏丢弃
  }
  restore()

  return {
    flush: () => { clearTimeout(timer); write() },
    destroy: () => { unsub(); clearTimeout(timer) },
  }
}
```
**评分点**：
- debounce 节流写盘 + flush 同步兜底（pagehide 不能异步）
- 版本三分支：相等合并 / 旧版本 migrate / 无法处理丢弃
- JSON.parse 必须 try-catch，损坏时 removeItem 防反复失败
- QuotaExceededError 单独处理不外抛
