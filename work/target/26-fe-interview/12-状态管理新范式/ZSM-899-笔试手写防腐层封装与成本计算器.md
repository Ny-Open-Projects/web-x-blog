## ZSM-899：手写状态库防腐层 + 引入成本计算器
**答**：898 落地——实现：1) `createOurStore()`：防腐层——内部用 zustand，暴露受限 API（get/set/subscribe/use），**禁止泄漏** zustand 特有 API（如 middleware 链）；类型上 T 泛型穿透保推导；2) `annualCost(lib)`：五项成本计算器（体积/学习/维护/迁移/风险）；3) `swap(newImpl)`：换库只改封装层（实现注入）——验证防腐有效；4) 附用例：成本对比与换库演示。

```javascript
// 1) 防腐层：实现注入 + 类型穿透
function createOurStore(initial) {
  let impl = null   // 延迟注入：zustand/自研都可
  const api = {
    _bind(implementation) { impl = implementation },
    get: () => impl.getState(),
    set: (updater, actionName) => impl.setState(updater, false, actionName),
    subscribe: (fn) => impl.subscribe(fn),
    getInitialState: () => impl.getInitialState?.() ?? initial,
  }
  return api
}

// zustand 实现（生产）
import { createStore } from 'zustand/vanilla'
const ourCart = createOurStore({ items: [] })
ourCart._bind(createStore((set) => ({ ...ourCart.getInitialState(), addItem: (i) => set((s) => ({ items: [...s.items, i] })) })))

// 2) 五项成本计算器
function annualCost(lib, { teamSize = 5, appLifeYears = 3 } = {}) {
  const sizeKB = lib.sizeGzipKB ?? 0
  const sizeCost = sizeKB * 0.05                       // 体积折算人日等价（粗略）
  const learnCost = (lib.rampUpDaysPerDev ?? 0) * teamSize
  const maintainCost = (lib.active ? 2 : 8)            // 活跃库年化 2 人日 / 停维 8 人日
  const migrateCost = (lib.migrateDays ?? 0)
  const riskCost = lib.exitDays ? lib.exitDays / appLifeYears : 0   // 退出成本年化摊销
  const total = +(sizeCost + learnCost + maintainCost + migrateCost / appLifeYears + riskCost).toFixed(1)
  return {
    breakdown: { sizeCost, learnCost, maintainCost, migrateCost: +(migrateCost / appLifeYears).toFixed(1), riskCost: +riskCost.toFixed(1) },
    totalAnnual: total,
    verdict: total < 5 ? '低成本：直接采纳' : total < 15 ? '中成本：Qualified 试点' : '高成本：需要 ROI 论证（774）',
  }
}

// 用例：zustand vs 自研 vs XState
console.log(annualCost({ sizeGzipKB: 1, rampUpDaysPerDev: 0.5, active: true, migrateDays: 1, exitDays: 2 }))
// totalAnnual ≈ 4.4 → '低成本：直接采纳'
console.log(annualCost({ sizeGzipKB: 8, rampUpDaysPerDev: 3, active: true, migrateDays: 10, exitDays: 15 }))
// totalAnnual ≈ 23.9 → '高成本：需要 ROI 论证'

// 3) 换库演示：只改封装层绑定
// ourCart._bind(myOwnTinyStore(...))   ← 业务代码零改动
console.log(typeof ourCart.set === 'function' && typeof ourCart.subscribe === 'function')   // true：API 面不变
```
**评分点**：
- 防腐层 API 受限（get/set/subscribe——zustand 特有 API 不泄漏）
- 实现注入（_bind 延迟绑定——换库零业务改动）
- 五项成本公式（体积/学习/维护/迁移摊销/风险年化）
- verdict 三档（直接采纳/试点/ROI 论证——接 774/880）
