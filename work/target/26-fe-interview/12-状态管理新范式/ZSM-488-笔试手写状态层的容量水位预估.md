## ZSM-488：手写状态层容量水位预估
**需求**：实现 `createCapacityForecaster(samples)`：1) samples：内存/存储用时序样本 `{ t, bytes }`；2) `trend()`：线性回归斜率（bytes/天）——用最小二乘法；3) `daysUntil(limit)`：按当前趋势到达 limit 的预计天数（斜率 <= 0 返回 Infinity）；4) R² 决定可信度（R² < 0.7 标记 unreliable——数据波动大不外推）；5) 异常样本剔除（MAD 离群值）后回归；6) `report(limit)` 输出 { slope, r2, days, reliable, advice }。

```javascript
export function createCapacityForecaster(samples, { r2Min = 0.7, madK = 3 } = {}) {
  function clean(arr) {                                      // MAD 剔除离群值
    const ys = arr.map((s) => s.bytes)
    const med = [...ys].sort((a, b) => a - b)[Math.floor(ys.length / 2)]
    const mad = [...ys.map((y) => Math.abs(y - med))].sort((a, b) => a - b)[Math.floor(ys.length / 2)]
    return arr.filter((s) => Math.abs(s.bytes - med) <= madK * (mad || 1))
  }

  function regress(arr) {
    const n = arr.length
    const sx = arr.reduce((a, s) => a + s.t, 0), sy = arr.reduce((a, s) => a + s.bytes, 0)
    const sxx = arr.reduce((a, s) => a + s.t * s.t, 0), sxy = arr.reduce((a, s) => a + s.t * s.bytes, 0)
    const slope = (n * sxy - sx * sy) / (n * sxx - sx * sx)
    const intercept = (sy - slope * sx) / n
    const meanY = sy / n
    const ssTot = arr.reduce((a, s) => a + (s.bytes - meanY) ** 2, 0)
    const ssRes = arr.reduce((a, s) => a + (s.bytes - (slope * s.t + intercept)) ** 2, 0)
    const r2 = ssTot ? 1 - ssRes / ssTot : 0
    return { slope, intercept, r2 }
  }

  function report(limit) {
    const arr = clean(samples)
    if (arr.length < 3) return { reliable: false, advice: '样本不足' }
    const { slope, intercept, r2 } = regress(arr)
    const reliable = r2 >= r2Min
    const days = slope > 0 ? (limit - (slope * arr[arr.length - 1].t + intercept)) / slope / 86400_000 : Infinity
    return {
      slopeBytesPerDay: slope * 86400_000, r2, days, reliable,
      advice: !reliable ? '数据波动大，不外推' : days < 30 ? '30 天内到限，立即治理' : '观察',
    }
  }

  return { report, trend: () => regress(clean(samples)).slope * 86400_000 }
}
```
**评分点**：
- 最小二乘回归实现正确（slope/intercept/R²）
- MAD 离群剔除（先清洗再回归——脏数据直接回归会误导）
- R² 可信度门槛（不可靠不外推——负责任的预估）
- daysUntil 公式与分级建议（<30 天立即治理）
