## ZSM-740：手写剪贴板三级降级链（权限探测 + 策略链 + 埋点上报）
**答**：实现 `createClipboardService()`：1) `probe()`：探测可用策略集（clipboard API / execCommand / manual）与权限状态；2) `copy(payload)`：按策略链依次尝试，每级失败降下级；返回 `{ ok, via, latency }`；3) 埋点：每次复制上报 `{ via, fallbacked, denied }`；4) `read()`：仅在权限 granted 时主动读，否则返回 null 引导手动粘贴；5) 策略可注册扩展（`use(strategyFn)`）；6) 附用例：mock 环境 API 缺失走 execCommand 降级。

```javascript
function createClipboardService({ onEvent } = {}) {
  const strategies = []
  let permState = 'unknown'

  function use(name, fn, { requiresPermission = false } = {}) {
    strategies.push({ name, fn, requiresPermission })
  }

  async function probe() {
    // 权限探测
    try {
      const st = await navigator.permissions.query({ name: 'clipboard-write' })
      permState = st.state
    } catch { permState = 'unsupported' }
    const available = strategies.filter((s) => {
      if (s.requiresPermission && (permState === 'denied' || permState === 'unsupported')) return false
      return s.available?.() ?? true
    })
    return { permState, strategies: available.map((s) => s.name) }
  }

  async function copy(payload) {
    const t0 = performance.now()
    for (const s of strategies) {
      try {
        await s.fn(payload)
        const result = { ok: true, via: s.name, latency: performance.now() - t0, fallbacked: s !== strategies[0] }
        onEvent?.({ type: 'clipboard.copy', ...result })
        return result
      } catch (e) {
        onEvent?.({ type: 'clipboard.fallback', via: s.name, error: e.message })
      }
    }
    const result = { ok: false, via: 'manual', denied: permState === 'denied' }
    onEvent?.({ type: 'clipboard.manual', ...result })
    return result
  }

  async function read() {
    if (permState !== 'granted' || !navigator.clipboard?.readText) return null   // 引导手动粘贴
    try { return await navigator.clipboard.readText() } catch { return null }
  }

  // 默认策略链
  use('clipboard-api', async (p) => {
    if (!navigator.clipboard?.writeText) throw new Error('unavailable')
    await navigator.clipboard.writeText(p.text)
  }, { requiresPermission: false })

  use('exec-command', async (p) => {
    const ta = document.createElement('textarea')
    ta.value = p.text
    ta.style.cssText = 'position:fixed;opacity:0'
    document.body.appendChild(ta)
    ta.select()
    const ok = document.execCommand('copy')
    ta.remove()
    if (!ok) throw new Error('execCommand failed')
  })

  return { use, probe, copy, read, get permState() { return permState } }
}

// 用例：mock 无 clipboard API 环境
const events = []
globalThis.navigator = { permissions: { query: async () => ({ state: 'granted', onchange: null }) } }  // 无 clipboard 对象
globalThis.document = {
  createElement: () => ({ style: {}, select: () => {}, remove: () => {} }),
  body: { appendChild: () => {}, },
  execCommand: () => true,
}

const svc = createClipboardService({ onEvent: (e) => events.push(e) })
const r = await svc.copy({ text: 'hello' })
console.log(r)                 // { ok: true, via: 'exec-command', fallbacked: true }
console.log(events.some((e) => e.type === 'clipboard.fallback'))   // true：降级埋点
```
**评分点**：
- 策略链注册制（use 扩展 + requiresPermission 过滤）
- 降级埋点（fallbacked/denied 降级率可监控）
- 权限探测与 read 保守语义（非 granted 返回 null 引导手动）
- latency 采集（策略性能可比较）
