## ZSM-449：手写状态回归快照测试器
**答**：实现 `createStateSnapshotRunner(store)`：1) `run(script)`：执行动作脚本 `[{ delay?, do: (store) => void }...]`，每步后截取快照（按 pick 白名单裁剪）；2) 快照序列化确定性：不稳定字段（时间戳/随机 id）按 maskFn 掩码；3) `compare(baseline)`：逐步对比快照，首个不一致处输出 step 索引与 diff；4) `saveBaseline()` 导出 JSON 基线文件（进 git）；5) 脚本执行支持异步步（await do）；6) 快照上限 50 步（防爆炸）。

```javascript
export function createStateSnapshotRunner(store, { pick, maskFn = (s) => s, maxSteps = 50 } = {}) {
  function snap() {
    const full = pick(store.getState())
    return JSON.parse(JSON.stringify(maskFn(full)))          // 掩码后确定性序列化
  }

  async function run(script) {
    const shots = []
    for (let i = 0; i < Math.min(script.length, maxSteps); i++) {
      await script[i].do?.(store)                             // 支持异步步
      if (script[i].delay) await sleep(script[i].delay)
      shots.push({ step: i, snap: snap() })
    }
    return shots
  }

  function compare(baseline, shots) {
    for (let i = 0; i < Math.max(baseline.length, shots.length); i++) {
      const a = JSON.stringify(baseline[i]?.snap), b = JSON.stringify(shots[i]?.snap)
      if (a !== b)
        return { pass: false, step: i, diff: { expected: baseline[i]?.snap, actual: shots[i]?.snap } }
    }
    return { pass: true }
  }

  return { run, compare, saveBaseline: (shots) => JSON.stringify(shots, null, 2) }
}

const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
```

**评分点**：
- 确定性三件套：pick 白名单裁剪 + maskFn 掩码不稳定字段 + 确定性序列化
- 逐步对比 + 首个失败定位（step 索引 + diff 输出）
- 异步步支持（await do）与 delay 步进（时序敏感逻辑可测）
- 基线进 git（可 review 的行为变更）、步数上限防爆炸
