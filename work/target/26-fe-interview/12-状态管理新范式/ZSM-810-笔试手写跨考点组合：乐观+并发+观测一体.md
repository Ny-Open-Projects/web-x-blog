## ZSM-810：手写跨考点组合手写（乐观更新 + 并发防护 + 可观测一体）
**答**：终极组合题——实现 `createLikeService()`：1) 乐观更新 + 字段级回滚对账（ZSM-528）；2) 重入防护（幂等键沿用，ZSM-767）；3) 超时自动 reject；4) 变更日志（可观测，ZSM-560 最小版）；5) 版本裁决（confirm 以服务端为准）；6) 附用例验证全链路：双击 → 乐观 → 超时回滚 → 幂等重试成功。

```javascript
function createLikeService({ timeoutMs = 3000 } = {}) {
  const state = { likes: 0 }
  const log = []
  const inFlight = new Set()
  const idemKeys = new Map()      // opId -> key（失败沿用）
  const listeners = new Set()
  let version = 0

  function emit(e) { log.push({ ts: Date.now(), ...e }); listeners.forEach((fn) => fn(e)) }

  async function like(postId) {
    const opId = `${postId}:${Date.now()}`
    // 重入防护：同帖在途拒绝
    if (inFlight.has(postId)) return { ok: false, reason: 'IN_FLIGHT' }
    inFlight.add(postId)
    // 幂等键：失败沿用
    if (!idemKeys.has(opId)) idemKeys.set(opId, crypto.randomUUID())

    // 乐观 +1
    const prev = state.likes
    state.likes++
    version++
    emit({ type: 'optimistic', opId, field: 'likes', prev, next: state.likes, version })

    try {
      const res = await Promise.race([
        api.like(postId, { IdempotencyKey: idemKeys.get(opId) }),   // 幂等键上送
        new Promise((_, rej) => setTimeout(() => rej(Object.assign(new Error('TIMEOUT'), { code: 'TIMEOUT' })), timeoutMs)),
      ])

      if (res.code === 'DUPLICATE') {          // 服务端幂等命中：转正即可
        emit({ type: 'confirm-dup', opId, version })
        return { ok: true, duplicated: true }
      }
      // confirm：以服务端为准（版本裁决）
      state.likes = res.likes
      version = res.version
      idemKeys.delete(opId)                     // 成功销毁幂等键
      emit({ type: 'confirm', opId, serverLikes: res.likes, version })
      return { ok: true }
    } catch (e) {
      // 字段级回滚（只回滚 likes）
      state.likes = prev
      version++
      emit({ type: 'reject', opId, reason: e.code ?? e.message, rollbackTo: prev, version })
      // 超时：幂等键保留供重试
      if (e.code !== 'TIMEOUT') idemKeys.delete(opId)
      return { ok: false, reason: e.code }
    } finally {
      inFlight.delete(postId)
    }
  }

  return {
    like,
    getState: () => ({ ...state, version }),
    subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
    log: () => log,
  }
}

// 用例：全链路
const svc = createLikeService({ timeoutMs: 50 })
// mock：首次超时、重试成功
let attempts = 0
globalThis.api = { like: async () => { attempts++; if (attempts === 1) await new Promise((r) => setTimeout(r, 100)); return { code: 'OK', likes: 42, version: 9 } } }

const r1 = await svc.like('p1')     // 超时 → 回滚
console.log(r1.ok, svc.getState().likes)   // false 0（回滚）
const r2 = await svc.like('p1')     // 重试 → 成功（服务端 42）
console.log(r2.ok, svc.getState().likes)   // true 42
console.log(svc.log().map((e) => e.type))  // ['optimistic','reject','optimistic','confirm']
```
**评分点**：
- 三考点融合（乐观对账 + 重入/幂等 + 日志）且互不干扰
- 幂等键生命周期（超时保留沿用、服务端 DUPLICATE 转正）
- 字段级回滚 + 版本递增（reject 也 bump version）
- 日志四事件类型（optimistic/confirm/reject/confirm-dup）可回放
