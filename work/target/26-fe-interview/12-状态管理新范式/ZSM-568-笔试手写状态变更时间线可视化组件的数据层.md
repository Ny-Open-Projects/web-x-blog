## ZSM-568：手写状态变更时间线可视化组件的数据层
**答**：实现时间线面板的数据层 `createTimeline()`：1) `push(event)`：事件入环形缓冲（上限 N），事件 `{ ts, action, keys, duration }`；2) `buckets(windowMs)`：按时间窗聚合（如每秒一个桶），返回 `{ ts, count, topActions, p95 }`——用于渲染柱状图；3) `actions()`：按 action 聚合统计（次数、平均耗时、max 耗时）；4) `select(seq)`：选中某事件返回详情与前后 3 条上下文；5) 附用例验证桶聚合与 p95 计算。

```javascript
function createTimeline({ max = 2000, windowMs = 1000 } = {}) {
  const events = []

  function push(e) {
    events.push({ seq: events.length, ...e })
    if (events.length > max) events.shift()
  }

  function buckets() {
    if (!events.length) return []
    const t0 = events[0].ts
    const byBucket = new Map()
    for (const e of events) {
      const b = Math.floor((e.ts - t0) / windowMs)
      const arr = byBucket.get(b) ?? []
      arr.push(e)
      byBucket.set(b, arr)
    }
    return [...byBucket.entries()]
      .sort((a, b) => a[0] - b[0])
      .map(([b, list]) => {
        // topActions：桶内频次最高的 3 个 action
        const freq = new Map()
        list.forEach((e) => freq.set(e.action, (freq.get(e.action) ?? 0) + 1))
        const topActions = [...freq.entries()].sort((a, b) => b[1] - a[1]).slice(0, 3)
        // p95：耗时升序取第 95 分位
        const durs = list.map((e) => e.duration).sort((a, b) => a - b)
        const p95 = durs[Math.min(durs.length - 1, Math.floor(durs.length * 0.95))]
        return { ts: t0 + b * windowMs, count: list.length, topActions, p95 }
      })
  }

  function actions() {
    const agg = new Map()
    for (const e of events) {
      const a = agg.get(e.action) ?? { count: 0, sum: 0, max: 0 }
      a.count++; a.sum += e.duration; a.max = Math.max(a.max, e.duration)
      agg.set(e.action, a)
    }
    return [...agg.entries()].map(([action, a]) => ({
      action, count: a.count, avg: a.sum / a.count, max: a.max,
    })).sort((x, y) => y.count - x.count)
  }

  function select(seq) {
    const i = events.findIndex((e) => e.seq === seq)
    if (i < 0) return null
    return { event: events[i], context: events.slice(Math.max(0, i - 3), i + 4) }
  }

  return { push, buckets, actions, select }
}

// 用例
const tl = createTimeline()
const t0 = Date.now()
for (let i = 0; i < 5; i++) tl.push({ ts: t0 + i * 100, action: 'scroll', keys: ['pos'], duration: 1 + (i % 3) })
for (let i = 0; i < 3; i++) tl.push({ ts: t0 + 500 + i * 50, action: 'addItem', keys: ['items'], duration: 20 })
const b = tl.buckets()
console.log(b.length >= 1 && b[0].count >= 5)     // 桶聚合
console.log(b[0].topActions[0])                    // ['scroll', 5]
console.log(tl.actions().at(-1).avg > 15)          // addItem 平均耗时高
console.log(tl.select(5).event.action)             // 'addItem'
```
**评分点**：
- 桶聚合的时间窗划分与排序（柱状图直接可渲染）
- p95 分位数计算（排序取分位，边界 min 处理）
- action 维度聚合三指标（count/avg/max）
- select 的前后上下文窗口（时间线排查的核心交互）
