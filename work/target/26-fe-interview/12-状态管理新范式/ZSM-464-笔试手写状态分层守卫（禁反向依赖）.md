## ZSM-464：手写状态分层守卫
**需求**：实现 `createLayerGuard(layers)`：1) layers 声明层序：`['server-cache', 'ui-state', 'ephemeral']`（下层不可依赖上层）；2) `registerAccess(fromLayer, toLayer)` 记录访问，反向依赖（高→低）在 dev 抛错并列出违规路径；3) `report()` 返回违规边列表（from→to/次数/首次位置）；4) 生产模式只记录不抛错（性能与体验优先）；5) 提供 `wrapGet(fromLayer, getter)` 包装读取访问做检测。

```javascript
export function createLayerGuard(layers, { strict = false } = {}) {
  const idx = Object.fromEntries(layers.map((l, i) => [l, i]))   // 下层 index 小
  const violations = new Map()   // edge → { count, firstLoc }

  function check(from, to, loc) {
    if (idx[from] === undefined || idx[to] === undefined) return
    if (idx[from] > idx[to]) {                                 // 上层读下层依赖？错向
      const edge = `${from}→${to}`
      const rec = violations.get(edge) ?? { count: 0, firstLoc: loc }
      rec.count++; violations.set(edge, rec)
      if (strict) throw new Error(`layer violation: ${edge} @ ${loc}`)
    }
  }

  function wrapGet(fromLayer, toLayer, getter) {
    return (...args) => {
      check(fromLayer, toLayer, callerLoc())
      return getter(...args)
    }
  }

  return {
    registerAccess: (f, t) => check(f, t, callerLoc()),
    wrapGet,
    report: () => [...violations.entries()].map(([edge, v]) => ({ edge, ...v })),
  }
}

function callerLoc() { return new Error().stack?.split('\n')[3]?.trim().slice(0, 80) ?? '?' }
// 层序语义：['server-cache','ui-state','ephemeral'] —— ephemeral(高) 可依赖 server-cache(低)？
// 依约定定义：index 小 = 底层。上层（大 index）依赖下层（小 index）为合法，反向为违规。
```
**评分点**：
- 层序与依赖方向约定清晰（题目需定义清楚 index 语义，答案里显式声明）
- dev strict 抛错 / 生产只记录（体验与治理的分级）
- 违规边聚合（次数 + 首次位置），report 可进 CI 断言
- wrapGet 透明包装（不改行为只观测）
