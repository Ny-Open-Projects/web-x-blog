## ZSM-556：手写条件断点式状态订阅
**答**：实现 `createDebugStore(initial)`：1) `watch(selector, { when, once, onHit })`：状态变更命中条件时触发 onHit（可断点 debugger、可截图上报）；2) `watch` 支持「进入条件」与「退出条件」（enter/leave），模拟断点的命中/释放；3) 命中后可 `pause()` 挂起 store（变更进入队列不执行），`resume()` 按序重放——用于「冻结现场」排查；4) `log()` 输出 watch 命中历史；5) 附用例验证 enter/leave 与 pause/resume 重放。

```javascript
function createDebugStore(initial) {
  let state = { ...initial }
  const subs = new Set()
  const watchers = new Set()
  let paused = false
  const queue = []
  const hits = []

  function setState(updater) {
    const next = typeof updater === 'function' ? updater(state) : updater
    if (paused) { queue.push(next); return }   // 冻结：入队不执行
    apply(next)
  }

  function apply(next) {
    const prev = state
    state = next
    // watch 条件判定
    watchers.forEach((w) => {
      const hitNow = w.when(state)
      if (w.active && hitNow && !w.wasHit) {
        hits.push({ ts: Date.now(), state: { ...state } })
        w.wasHit = true
        if (w.once) watchers.delete(w)
        w.onHit?.(state, prev)
        if (w.debugger) debugger
      } else if (w.wasHit && w.leave && w.leave(state)) {
        w.wasHit = false   // 退出条件满足：释放，可再次命中
      }
    })
    subs.forEach((fn) => fn(state, prev))
  }

  function watch(selector, { when, leave, once = false, debugger: useDebugger = false, onHit } = {}) {
    const w = { selector, when: when ?? selector, leave, once, debugger: useDebugger, onHit, active: true, wasHit: false }
    watchers.add(w)
    return () => watchers.delete(w)
  }

  function pause() { paused = true }
  function resume() {
    paused = false
    const q = queue.splice(0)
    q.forEach(apply)          // 按序重放冻结期间变更
  }

  return { setState, getState: () => state, subscribe: (fn) => (subs.add(fn), () => subs.delete(fn)), watch, pause, resume, log: () => hits }
}

// 用例
const store = createDebugStore({ balance: 100, locked: false })
store.watch(
  (s) => s.balance < 0,                       // 命中条件：余额为负
  { leave: (s) => s.balance >= 0,             // 退出条件：回正
    onHit: (s) => console.log('HIT negative balance', s.balance) }
)
store.setState((s) => ({ balance: s.balance - 150 }))   // -50 → HIT
store.setState({ balance: 50 })                          // 回正释放
store.setState((s) => ({ balance: s.balance - 100 }))    // 再负 → 再次 HIT（可重复命中）
store.pause()
store.setState((s) => ({ balance: s.balance + 999 }))    // 冻结：不生效
console.log(store.getState().balance)                    // -50
store.resume()
console.log(store.getState().balance)                    // 949：队列重放
console.log(store.log().length)                          // 2 次命中
```
**评分点**：
- enter/leave 双条件（可重复命中，非一次性 flag）
- once 语义与退订函数
- pause/resume 冻结现场 + 队列按序重放（不打乱变更顺序）
- onHit 快照留存（命中历史可回溯）
