## ZSM-496：手写状态层灾备切换器
**需求**：实现 `createFailoverManager(primary, backup)`：1) 健康探测：primary 失败 N 次自动切 backup（读请求走 backup，写请求排队）；2) 恢复探测：primary 恢复后延迟切换回（防止抖动：要求连续 5 次探测成功）；3) 切换期间写操作进队列（恢复后按序重放，幂等键去重）；4) 切换/回切事件可订阅（驱动 UI 提示「降级服务中」）；5) 数据不一致窗口处理：backup 服务期间产生的写，回切时提示「需同步确认」。

```javascript
export function createFailoverManager(primary, backup, { threshold = 3, stableChecks = 5 } = {}) {
  let active = primary
  let failCount = 0, okStreak = 0
  const writeQueue = []        // 幂等键去重
  const seenOps = new Set()
  const listeners = new Set()

  function switchTo(target, reason) {
    if (active === target) return
    active = target
    listeners.forEach((fn) => fn({ active: target === primary ? 'primary' : 'backup', reason }))
  }

  async function exec(op) {
    if (op.type === 'read') {
      try {
        const r = await active.read(op)
        if (active !== primary) { okStreak++; if (okStreak >= stableChecks) switchTo(primary, 'recovered') }   // 回切需稳定
        else { okStreak = 0; failCount = 0 }
        return r
      } catch (e) {
        if (active === primary && ++failCount >= threshold) switchTo(backup, 'primary-down')       // 灾备切换
        return backup.read(op)
      }
    }
    // 写：幂等去重 + 切换期间排队
    if (seenOps.has(op.id)) return { deduped: true }
    seenOps.add(op.id)
    if (active !== primary) { writeQueue.push(op); return { queued: true } }   // 降级期间写排队
    return primary.write(op)
  }

  async function drainQueue() {                                    // 回切后重放
    while (writeQueue.length) {
      const op = writeQueue.shift()
      await primary.write(op)
    }
  }

  return { exec, drainQueue, onSwitch: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
    get active() { return active === primary ? 'primary' : 'backup' } }
}
```
**评分点**：
- 切换/回切不对称语义（切换快、回切需连续稳定——防抖动震荡）
- 降级期间写排队（幂等键去重）+ 回切后 drain 重放
- 一致性窗口提示（backup 期间的写与 primary 可能分叉——诚实标注，不假装无缝）
- 切换事件可订阅（UI 感知）
