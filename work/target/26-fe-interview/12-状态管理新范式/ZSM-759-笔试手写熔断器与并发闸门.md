## ZSM-759：手写熔断器 + 并发闸门（三态机 + fail-fast + 恢复爬坡）
**答**：实现 `createCircuitBreaker({ threshold, cooldown })`：1) 三态 closed/open/half-open：closed 连续失败 ≥ threshold → open（fail-fast 抛 CircuitOpenError 不执行 fn）；open 冷却结束 → half-open 放行**单个**探测请求（期间其余请求仍 fail-fast）；探测成功 → closed、失败 → open 重新计时；2) 状态变更通知订阅者；3) `createGate(max)`：并发闸门——超限排队 FIFO，finally 释放；4) `stats()`：失败数/状态/在途数；5) 附用例：连续失败开闸、冷却半开、探测失败重回 open、闸门排队顺序。

```javascript
function createCircuitBreaker({ threshold = 5, cooldown = 1000 } = {}) {
  let state = 'closed'
  let failures = 0
  let openedAt = 0
  let probing = false
  const listeners = new Set()

  function setState(s) { state = s; listeners.forEach((fn) => fn(s, stats())) }

  async function exec(fn) {
    if (state === 'open') {
      if (Date.now() - openedAt >= cooldown) {
        setState('half-open')
        probing = false
      } else {
        const e = new Error('CIRCUIT_OPEN')
        e.code = 'CIRCUIT_OPEN'
        throw e                                  // fail-fast：不占服务
      }
    }
    if (state === 'half-open' && probing) {
      throw Object.assign(new Error('CIRCUIT_OPEN'), { code: 'CIRCUIT_OPEN' })   // 半开期间仅一个探测
    }
    if (state === 'half-open') probing = true
    try {
      const result = await fn()
      if (state === 'half-open') { failures = 0; setState('closed') }   // 探测成功：闭合
      else failures = 0
      return result
    } catch (e) {
      if (state === 'half-open') {
        openedAt = Date.now()                    // 探测失败：重回 open 重新计时
        setState('open')
      } else if (++failures >= threshold) {
        openedAt = Date.now()
        setState('open')
      }
      throw e
    } finally {
      if (state === 'half-open') probing = false
    }
  }

  function stats() { return { state, failures } }
  return { exec, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)), stats }
}

function createGate(max) {
  let inFlight = 0
  const queue = []
  async function run(fn) {
    if (inFlight >= max) {
      await new Promise((r) => queue.push(r))    // FIFO 排队
    }
    inFlight++
    try { return await fn() }
    finally {
      inFlight--
      queue.shift()?.()                          // 释放并唤醒队首
    }
  }
  return { run, stats: () => ({ inFlight, queued: queue.length }) }
}

// 用例
let calls = 0
const breaker = createCircuitBreaker({ threshold: 3, cooldown: 100 })
const gate = createGate(2)
breaker.subscribe((s) => console.log('state →', s))

const fail = () => { calls++; return Promise.reject(new Error('boom')) }
for (let i = 0; i < 3; i++) await breaker.exec(fail).catch(() => {})
console.log(breaker.stats().state)               // 'open'：连续失败开闸

const t0 = Date.now()
await breaker.exec(fail).catch((e) => console.log(e.code))   // 'CIRCUIT_OPEN'：fail-fast 未执行
console.log(calls)                               // 3（fail-fast 未调 fn）

await new Promise((r) => setTimeout(r, 150))     // 冷却结束
const ok = await breaker.exec(async () => 'ok')  // half-open 探测成功
console.log(ok, breaker.stats().state)           // 'ok' 'closed'

// 闸门排队
const order = []
const jobs = [1, 2, 3, 4].map((n) => gate.run(async () => {
  order.push(n)
  await new Promise((r) => setTimeout(r, 20))
}))
await Promise.all(jobs)
console.log(order.length)                        // 4：全部执行（FIFO 排队不丢）
```
**评分点**：
- 三态机完整（open fail-fast 不执行 fn、half-open 单探测）
- 探测失败重回 open 重新计时（openedAt 更新）
- 状态通知订阅者（全局横幅的数据源）
- 闸门 FIFO 排队与 finally 释放（恢复脉冲防打爆）
