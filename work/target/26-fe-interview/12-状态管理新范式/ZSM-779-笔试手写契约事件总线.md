## ZSM-779：手写契约事件总线（schema 校验 + qos 重试 + 写路径唯一）
**答**：实现 `createContractBus(contracts)`：1) `emit(key, payload, source)`：schema 校验（自写简单校验器：type/required/nullable）失败拦截并告警；2) qos：'at-least-once' 时投递失败自动重试（3 次退避），'best-effort' 丢弃；3) 写路径唯一：`emit` 校验 source 是否为 contract 的 from（非 owner 拒绝）；4) `subscribe(app, key, fn)`：只允许 to 列表内的应用订阅；5) 附用例：schema 拦截、非 owner 写拒绝、qos 重试。

```javascript
// 简单 schema 校验器
function validate(payload, schema) {
  for (const [field, rule] of Object.entries(schema)) {
    const v = payload?.[field]
    if (v === undefined || v === null) {
      if (rule.nullable || rule.optional) continue
      return `missing field ${field}`
    }
    if (!rule.nullable && v === null) return `field ${field} null not allowed`
    const t = Array.isArray(v) ? 'array' : typeof v
    if (t !== rule.type) return `field ${field} type ${t} != ${rule.type}`
  }
  return null
}

function createContractBus(contracts, { retryMs = 500 } = {}) {
  const subs = new Map()        // key -> Map<app, Set<fn>>
  const seen = new Set()

  function emit(key, payload, source) {
    const c = contracts[key]
    if (!c) return { ok: false, reason: 'UNKNOWN_CONTRACT' }
    if (source !== c.from) return { ok: false, reason: 'NOT_OWNER' }   // 写路径唯一
    const err = validate(payload, c.schema)
    if (err) {
      console.warn(`[bus] ${key} schema 违反: ${err}`)
      return { ok: false, reason: 'SCHEMA_VIOLATION', detail: err }
    }
    const dedupe = `${key}:${payload.version ?? Date.now()}`
    if (seen.has(dedupe)) return { ok: false, reason: 'DUP' }
    seen.add(dedupe)

    return deliver(c.to, key, payload, c.qos)
  }

  async function deliver(targets, key, payload, qos) {
    for (const app of targets) {
      const fns = subs.get(key)?.get(app)
      if (!fns?.size) continue
      for (const fn of fns) {
        if (qos === 'at-least-once') {
          for (let i = 0; i < 3; i++) {                 // 重试退避
            try { await fn(payload); break }
            catch { if (i < 2) await new Promise((r) => setTimeout(r, retryMs * 2 ** i)) }
          }
        } else {
          try { await fn(payload) } catch { /* best-effort 丢弃 */ }
        }
      }
    }
    return { ok: true }
  }

  function subscribe(app, key, fn) {
    const c = contracts[key]
    if (!c?.to.includes(app)) throw new Error(`app ${app} 无权订阅 ${key}`)   // 订阅白名单
    ;(subs.get(key) ?? subs.set(key, new Map()).get(key))
      .set(app, (subs.get(key).get(app) ?? new Set()).add(fn))
    return () => subs.get(key)?.get(app)?.delete(fn)
  }

  return { emit, subscribe }
}

// 用例
const bus = createContractBus({
  'cart:updated': {
    schema: { version: { type: 'number' }, count: { type: 'number' } },
    from: 'app-order', to: ['shell'],
    qos: 'at-least-once',
  },
})

// 非 owner 写拒绝
console.log(bus.emit('cart:updated', { version: 1, count: 2 }, 'shell'))
// { ok: false, reason: 'NOT_OWNER' }

// schema 拦截
console.log(bus.emit('cart:updated', { version: '1', count: 2 }, 'app-order'))
// { ok: false, reason: 'SCHEMA_VIOLATION' }（version 应为 number）

// 合法投递 + qos 重试
let calls = 0
bus.subscribe('shell', 'cart:updated', async (p) => {
  calls++
  if (calls === 1) throw new Error('transient')       // 首次失败 → 重试
})
const r = bus.emit('cart:updated', { version: 2, count: 3 }, 'app-order')
await new Promise((r) => setTimeout(r, 600))
console.log(r.ok, calls)                              // true 2：重试成功
```
**评分点**：
- 写路径唯一（from owner 校验，非 owner 拒绝）
- schema 校验拦截（type/nullable 简版校验器）
- qos 语义（at-least-once 重试退避 / best-effort 丢弃）
- 订阅白名单（to 列表外的应用无权订阅）
