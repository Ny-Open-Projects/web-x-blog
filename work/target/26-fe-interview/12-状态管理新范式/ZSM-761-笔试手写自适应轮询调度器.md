## ZSM-761：手写自适应轮询调度器（降档阶梯 + 终止语义 + 可见性暂停）
**答**：实现 `createAdaptivePoller({ task, ladder, idleThreshold, hasChanged, isDone })`：1) 频率阶梯：连续 idleThreshold 次无变化升档（5s→15s→60s），有变化回基础档；2) 终止语义：isDone() 为真停止（不再调度）并触发 onDone；3) 可见性：隐藏时暂停（clearTimeout），恢复时立即执行一次（追平）并重置档位；4) `stats()`：当前档位/idle 计数/执行次数；5) 附用例：无变化降档、有变化回档、done 终止。

```javascript
function createAdaptivePoller({ task, ladder = [100, 300, 1000], idleThreshold = 2, hasChanged = () => false, isDone = () => false, onDone, onTick } = {}) {
  let timer = null
  let idleCount = 0
  let paused = false
  let stopped = false
  let ticks = 0
  const listeners = new Set()

  function level() { return ladder[Math.min(idleCount, ladder.length - 1)] }

  function schedule() {
    if (stopped) return
    timer = setTimeout(run, level())
  }

  async function run() {
    if (stopped || paused) return
    if (isDone()) {
      stopped = true
      clearTimeout(timer)
      onDone?.()
      notify()
      return
    }
    ticks++
    try {
      const data = await task()
      if (hasChanged(data)) idleCount = 0          // 有变化：回基础档
      else idleCount++                             // 无变化：升档
    } catch {
      idleCount = Math.max(0, idleCount - 1)       // 失败：轻微回档
    }
    onTick?.(ticks, level())
    notify()
    schedule()
  }

  function onVisibility() {
    if (typeof document === 'undefined') return
    if (document.visibilityState === 'visible') {
      paused = false
      idleCount = 0
      if (!stopped) run()                          // 恢复：立即追平一次
    } else {
      paused = true
      clearTimeout(timer)                          // 隐藏：暂停
    }
  }
  if (typeof document !== 'undefined') {
    document.addEventListener('visibilitychange', onVisibility)
  }

  function notify() { listeners.forEach((fn) => fn(stats())) }
  function stats() { return { level: level(), idleCount, ticks, paused, stopped } }

  function stop() {
    stopped = true
    clearTimeout(timer)
    document.removeEventListener('visibilitychange', onVisibility)
  }

  return { start: run, stop, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)), stats }
}

// 用例（虚拟时间）
let changes = 0
let done = false
let callCount = 0
const poller = createAdaptivePoller({
  task: async () => { callCount++; return { v: callCount } },
  ladder: [50, 150, 500],
  idleThreshold: 2,
  hasChanged: (d) => d.v <= changes,
  isDone: () => done,
})
poller.subscribe((s) => console.log('tick', s.ticks, 'level', s.level))

changes = 1                     // 第一次有变化：回档
await poller.start()
await new Promise((r) => setTimeout(r, 120))
changes = 99                    // 之后无变化：降档
await new Promise((r) => setTimeout(r, 600))
console.log(poller.stats())     // idleCount 增长 → level 升至 500
done = true
await new Promise((r) => setTimeout(r, 600))
console.log(poller.stats().stopped)   // true：done 终止
poller.stop()
```
**评分点**：
- 阶梯降档与回档（idle 计数驱动）
- 终止语义（isDone → 停止调度 + onDone 回调）
- 可见性暂停/恢复（隐藏 clearTimeout，恢复立即追平 + 重置档位）
- stats 可观测（level/idle/ticks）
