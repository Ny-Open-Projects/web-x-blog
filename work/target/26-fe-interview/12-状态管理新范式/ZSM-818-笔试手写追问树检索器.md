## ZSM-818：手写追问树检索器（主题→分支匹配 + 备答压缩）
**答**：实现 `createFollowupIndex()`：1) `index(topic, branches)`：登记主题追问树；2) `query(text)`：文本匹配主题（关键词命中计分）+ 命中分支（分支关键词加权）——返回 `{ topic, main, matchedBranches }`；3) `compress(branches, maxLen)`：备答压缩——每分支截取锚点到 maxLen（保留句首锚定词）；4) `hitStats()`：分支命中率统计（哪些分支从未被命中=冗余）；5) 附用例：模糊匹配「列表很卡怎么办」命中大列表主题与扇出分支。

```javascript
function createFollowupIndex() {
  const topics = new Map()      // topic -> { keywords, main, branches: Map<name, {keywords, anchor, full}> }
  const hits = new Map()

  function index(topic, { keywords, main, branches }) {
    topics.set(topic, {
      keywords,
      main,
      branches: new Map(Object.entries(branches).map(([name, b]) => [
        name, { keywords: b.keywords ?? [name], anchor: b.anchor ?? b, full: typeof b === 'object' ? b.full : b },
      ])),
    })
    return this
  }

  function score(text, keywords) {
    return keywords.reduce((a, k) => a + (text.includes(k) ? k.length : 0), 0)
  }

  function query(text) {
    // 主题匹配：关键词命中计分
    const ranked = [...topics.entries()]
      .map(([topic, t]) => ({ topic, s: score(text, t.keywords) }))
      .sort((a, b) => b.s - a.s)
    const best = ranked[0]
    if (!best || best.s === 0) return null
    const t = topics.get(best.topic)
    // 分支匹配：分支关键词加权
    const matched = [...t.branches.entries()]
      .map(([name, b]) => ({ name, s: score(text, b.keywords) }))
      .filter((b) => b.s > 0)
      .sort((a, b) => b.s - a.s)
      .slice(0, 3)
    matched.forEach((m) => hits.set(`${best.topic}:${m.name}`, (hits.get(`${best.topic}:${m.name}`) ?? 0) + 1))
    return {
      topic: best.topic,
      main: t.main,
      matchedBranches: matched.map((m) => ({ name: m.name, anchor: t.branches.get(m.name).anchor })),
    }
  }

  function compress(branches, maxLen = 40) {
    return branches.map((b) => {
      const a = typeof b.anchor === 'string' ? b.anchor : JSON.stringify(b.anchor)
      return a.length <= maxLen ? a : a.slice(0, maxLen) + '…'
    })
  }

  function hitStats() {
    const all = []
    topics.forEach((t, topic) => t.branches.forEach((_, name) => all.push(`${topic}:${name}`)))
    const neverHit = all.filter((k) => !hits.has(k))
    return { total: all.length, hit: hits.size, neverHit }
  }

  return { index, query, compress, hitStats }
}

// 用例
const idx = createFollowupIndex()
idx.index('大列表性能', {
  keywords: ['列表', '卡', '性能', '渲染'],
  main: '四段式骨架：定性→建模→方案→闭环',
  branches: {
    虚拟滚动: { keywords: ['滚动', '万条'], anchor: '窗口三数 scrollTop/viewportH/rowH + overscan' },
    扇出: { keywords: ['重渲', '通知', '订阅'], anchor: 'observer 通知闸门 + select 粒度' },
    排序: { keywords: ['排序', '切换'], anchor: 'order 与实体分离，行引用稳定' },
  },
})
idx.index('多端同步', {
  keywords: ['同步', '多端', 'tab'],
  main: '协议栈五层',
  branches: {
    leader: { keywords: ['多tab', '连接'], anchor: 'WebLocks 选举 + 转发聚合' },
  },
})

const r = idx.query('商品列表很卡，滚动和重渲都慢')
console.log(r.topic)                    // '大列表性能'
console.log(r.matchedBranches.map((b) => b.name))   // ['虚拟滚动', '扇出']
console.log(idx.compress(r.matchedBranches, 30))
console.log(idx.hitStats())             // 排序分支未命中 → neverHit 含 '大列表性能:排序'
```
**评分点**：
- 主题关键词计分匹配（模糊命中）
- 分支二级匹配（加权排序 Top3）
- 备答压缩（锚点截断，保留句首锚定词）
- hitStats 冗余分支审计（neverHit = 无用备答）
