## ZSM-307：手写缓存失效调度器（合并 + 分域去重）
**答**：实现 `createInvalidationScheduler(execute)`：1) `invalidate(domain, id?)` 入队；2) 100ms 窗口内的失效合并——同域多个 id 去重为一次批量失效；3) 全域失效（无 id）覆盖该域所有精确 id 失效；4) 到期后按域分组调用 `execute(batch)`，batch 形如 `{ 'order': { all: true } }` 或 `{ 'order': { ids: [1,2] } }`；5) 支持同步 flush。

```javascript
export function createInvalidationScheduler(execute, delay = 100) {
  const pending = new Map()   // domain → { all: boolean, ids: Set }
  let timer = null

  function add(domain, id) {
    if (!pending.has(domain)) pending.set(domain, { all: false, ids: new Set() })
    const cur = pending.get(domain)
    if (id == null) cur.all = true            // 全域失效覆盖精确失效
    else if (!cur.all) cur.ids.add(id)
    if (!timer) timer = setTimeout(flush, delay)
  }

  function flush() {
    clearTimeout(timer); timer = null
    if (!pending.size) return
    const batch = {}
    pending.forEach((v, domain) => {
      batch[domain] = v.all || !v.ids.size ? { all: true } : { ids: [...v.ids] }
    })
    pending.clear()                            // 先清后执行，执行中的新失效进下一轮
    execute(batch)
  }

  return { invalidate: add, flush }
}
```
**评分点**：
- all 标记吸收后续精确 id（幂等升级）
- 先 clear 再 execute：执行期间产生的新失效不被吞也不重复
- 去重用 Set，同 id 多次入队只失效一次
- timer 单例管理，flush 可同步调用且可重入
