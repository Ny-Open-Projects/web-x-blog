## ZSM-404：手写跨标签页互斥锁（心跳 + 自动释放）
**需求**：实现 `createTabLock(name)`：1) `acquire()` 尝试获取锁，成功返回 true；2) 锁状态写 localStorage（`{holder, ts}`）；获取条件：无锁或锁「过期」（心跳超 3s 未续）；3) 持锁期间每 1s 续约心跳；4) `release()` 释放；5) 页面卸载（pagehide）自动释放；6) 过期锁可被其他标签页抢占。

```javascript
export function createTabLock(name) {
  const key = `lock:${name}`
  const id = Math.random().toString(36).slice(2)
  let held = false, heartbeat = null

  const read = () => { try { return JSON.parse(localStorage.getItem(key)) } catch { return null } }

  function acquire() {
    const cur = read()
    const expired = !cur || Date.now() - cur.ts > 3000        // 心跳超时 = 过期
    if (cur && !expired) return false
    localStorage.setItem(key, JSON.stringify({ holder: id, ts: Date.now() }))
    // 写后回读（多标签同时抢：后写覆盖，需二次确认——简化约定）
    held = true
    heartbeat = setInterval(() => {
      localStorage.setItem(key, JSON.stringify({ holder: id, ts: Date.now() }))
    }, 1000)
    return true
  }

  function release() {
    if (!held) return
    clearInterval(heartbeat); held = false
    const cur = read()
    if (cur?.holder === id) localStorage.removeItem(key)      // 只删自己的锁
  }

  window.addEventListener('pagehide', release)
  return { acquire, release }
}
```
**评分点**：
- 过期判定依赖心跳时间戳（持锁崩溃的标签页锁可被抢占，不会死锁）
- release 只删除自己持有的锁（不误删抢占者的锁）
- pagehide 自动释放（sync 可行）；心跳 interval 随 release 清理
- 并发抢占的竞态边界要点出（回读确认/Lock API 升级是加分项）
