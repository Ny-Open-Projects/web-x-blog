## ZSM-860：手写课堂答题状态机 + 进度心跳上报
**答**：859 落地——实现 `createClassroom()`：1) 答题窗口状态机：`open → collecting(收答案) → closed(截止，迟到拒绝) → revealed(公布)`，教师端控制迁移，学生端消费广播；2) 迟到防御：closed 后收到的答案入「迟到桶」并标记 late（展示但不计分）；3) 进度心跳：每 60s 上报（观看秒数累计），服务端权威校验返回合法时长——本地展示派生；4) 断线重连后课件页追平（lastPage seq）；5) 附用例：迟到答案与心跳校验。

```javascript
function createClassroom({ now = Date.now } = {}) {
  const quiz = { phase: 'idle', answers: new Map(), late: [], openAt: 0 }
  const progress = { watchedSec: 0, serverConfirmed: 0, lastBeat: 0 }
  let currentPage = { seq: 0, page: 0 }
  const listeners = new Set()

  // 教师端：控制答题窗口
  function openQuiz(quizId, durationMs = 30000) {
    quiz.phase = 'collecting'
    quiz.openAt = now()
    quiz.answers.clear()
    quiz.late = []
    quiz._closeAt = quiz.openAt + durationMs
    emit({ type: 'quiz-open', quizId })
  }
  function closeQuiz() {
    if (quiz.phase !== 'collecting') return
    quiz.phase = 'closed'
    emit({ type: 'quiz-closed', count: quiz.answers.size, late: quiz.late.length })
  }
  function revealQuiz() {
    quiz.phase = 'revealed'
    emit({ type: 'quiz-revealed', answers: [...quiz.answers.entries()] })
  }

  // 学生端：提交答案（迟到防御）
  function submitAnswer(studentId, answer) {
    if (quiz.phase !== 'collecting') {
      quiz.late.push({ studentId, answer, at: now() })      // 迟到桶：展示不计分
      emit({ type: 'answer-late', studentId })
      return { ok: false, reason: 'LATE' }
    }
    quiz.answers.set(studentId, answer)                     // 幂等：覆盖即改答案
    emit({ type: 'answer', studentId })
    return { ok: true }
  }

  // 进度心跳：服务端权威校验
  function tickWatched(sec) {
    progress.watchedSec += sec                              // 本地展示累计
    if (now() - progress.lastBeat >= 60000) {
      progress.lastBeat = now()
      api.reportProgress({ watched: progress.watchedSec, at: progress.lastBeat })
        .then(({ confirmed }) => {
          progress.serverConfirmed = confirmed              // 服务端权威（防挂机）
          emit({ type: 'progress-confirmed', confirmed })
        })
        .catch(() => { /* 离线队列兜底（588） */ })
    }
  }

  // 课件追平：断线重连后
  function syncPage(remote) {
    if (remote.seq > currentPage.seq) {
      currentPage = remote
      emit({ type: 'page-sync', page: remote.page })
    }
  }

  function emit(e) { listeners.forEach((fn) => fn(e)) }

  return {
    openQuiz, closeQuiz, revealQuiz, submitAnswer, tickWatched, syncPage,
    quizState: () => ({ ...quiz, answers: [...quiz.answers.entries()], _closeAt: undefined }),
    progress: () => ({ display: progress.watchedSec, confirmed: progress.serverConfirmed }),
    subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)),
  }
}

// 用例
let t = 0
const cls = createClassroom({ now: () => t })
cls.subscribe((e) => console.log(e.type))

cls.openQuiz('q1', 30000)
cls.submitAnswer('s1', 'B')                     // ok
t += 31000                                      // 超过窗口
cls.closeQuiz()
const r = cls.submitAnswer('s2', 'A')           // 迟到
console.log(r, cls.quizState().late.length)     // { ok: false, reason: 'LATE' } 1

cls.tickWatched(61)                             // 触发心跳（61s ≥ 60s）
console.log(cls.progress().display > 0)         // true：本地展示派生
cls.syncPage({ seq: 5, page: 3 })
cls.syncPage({ seq: 3, page: 2 })               // 旧 seq 丢弃
console.log(cls.quizState()._closeAt === undefined)   // 内部字段不外泄
```
**评分点**：
- 答题窗口四态机（collecting/closed/revealed + 迟到桶展示不计分）
- 答案幂等覆盖（同学生改答案）
- 心跳权威校验（本地展示与服务端确认分离——防挂机）
- seq 追平的旧包丢弃（过期同步包防御）
