## ZSM-559：手写状态指纹与一致性校验器
**答**：实现状态指纹工具，用于多端/回放后的一致性校验：1) `fingerprint(state)`：对状态树做稳定序列化（**键排序**、数组保序、忽略黑名单字段如时间戳）后 hash，输出 16 进制短指纹；2) `assertEqual(f1, f2, { ignore })`：不等时输出首个差异路径（用路径级指纹二分定位）；3) `stableStringify` 处理循环引用（记 `[Circular]`）、undefined（跳过键）；4) `streamFingerprint()`：增量指纹——边序列化边 hash，超大状态不用全量字符串。附用例验证键序无关性与差异定位。

```javascript
function stableStringify(v, { ignore = [], seen = new WeakSet() } = {}) {
  if (v === null || typeof v !== 'object') return JSON.stringify(v) ?? 'undefined'
  if (seen.has(v)) return '"[Circular]"'
  seen.add(v)
  if (Array.isArray(v)) {
    return '[' + v.map((x) => stableStringify(x, { ignore, seen })).join(',') + ']'
  }
  const keys = Object.keys(v).filter((k) => !ignore.includes(k)).sort()  // 键排序
  return '{' + keys.map((k) => `${JSON.stringify(k)}:${stableStringify(v[k], { ignore, seen })}`).join(',') + '}'
}

function fingerprint(state, opts = {}) {
  const s = stableStringify(state, opts)
  // FNV-1a hash → 16 进制（真实生产可用 crypto.subtle，这里同步可测）
  let h = 0x811c9dc5
  for (let i = 0; i < s.length; i++) {
    h ^= s.charCodeAt(i)
    h = (h * 0x01000193) >>> 0
  }
  return h.toString(16).padStart(8, '0')
}

function assertEqual(fp1, fp2, stateA, stateB, { ignore = [] } = {}) {
  if (fp1 === fp2) return { equal: true }
  // 路径级二分定位首个差异
  function firstDiff(a, b, path) {
    const ka = new Set(Object.keys(a ?? {})), kb = new Set(Object.keys(b ?? {}))
    for (const k of [...ka].sort()) {
      if (!(k in (b ?? {}))) return `${path}.${k} (removed)`
      if (typeof a[k] === 'object' && a[k]) {
        const r = firstDiff(a[k], b[k], `${path}.${k}`)
        if (r) return r
      } else if (!Object.is(a[k], b[k]) && !ignore.includes(k)) return `${path}.${k}`
    }
    for (const k of [...kb].sort()) if (!(k in (a ?? {}))) return `${path}.${k} (added)`
    return null
  }
  return { equal: false, diff: firstDiff(stateA, stateB, '') }
}

// 用例
const s1 = { b: 2, a: 1, meta: { ts: 999, arr: [1, { y: 2, x: 1 }] } }
const s2 = { a: 1, b: 2, meta: { arr: [{ x: 1, y: 2 }], ts: 12345 } }
const f1 = fingerprint(s1, { ignore: ['ts'] })
const f2 = fingerprint(s2, { ignore: ['ts'] })
console.log(f1 === f2)              // true：键序无关 + 黑名单字段忽略
const f3 = fingerprint({ ...s2, b: 3 }, { ignore: ['ts'] })
console.log(assertEqual(f1, f3, s1, { ...s2, b: 3 }, { ignore: ['ts'] }))
// { equal: false, diff: '.b' }：定位到差异路径
```
**评分点**：
- 稳定序列化三要素（键排序/数组保序/黑名单忽略）
- 循环引用与 undefined 的处理（WeakSet、跳过键）
- 差异定位到具体路径（二分递归首个不等字段）
- 指纹算法同步且确定性（同状态同指纹）
