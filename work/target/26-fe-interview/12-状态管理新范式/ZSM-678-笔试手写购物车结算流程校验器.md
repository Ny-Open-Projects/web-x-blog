## ZSM-678：手写购物车结算校验器（多条件门禁 + 差额提示）
**答**：实现 `createCheckoutValidator()`：1) `validate(cart, rules)`：rules 含——库存校验（item.stock >= qty）、限购（item.limit >= qty）、金额门槛（满 X 才可用某支付方式）、地址校验（缺省字段报错）；2) 返回结构化结果 `{ passed, errors: [{code, field, item?}], warnings: [{code, message}] }`——errors 阻断，warnings 放行（如库存紧张提示）；3) `suggestFix(cart, result)`：对可修复项给建议（qty 调整为 min(stock, limit)、金额差 X 元凑单）；4) 分单位整数运算；5) 附用例：库存不足/限购超量/门槛差额的建议生成。

```javascript
function createCheckoutValidator() {
  function validate(cart, rules) {
    const errors = []
    const warnings = []
    const items = cart.items

    for (const it of items) {
      // 库存校验
      if (it.qty > (it.stock ?? Infinity)) {
        errors.push({ code: 'OUT_OF_STOCK', field: `items.${it.goodsId}.qty`, item: it.goodsId,
          message: `库存仅剩 ${it.stock}` })
      }
      // 限购
      if (rules.limits?.[it.goodsId] != null && it.qty > rules.limits[it.goodsId]) {
        errors.push({ code: 'LIMIT_EXCEED', field: `items.${it.goodsId}.qty`, item: it.goodsId,
          message: `限购 ${rules.limits[it.goodsId]} 件` })
      }
      // 库存紧张 warning（通过但提示）
      if (it.stock != null && it.qty <= it.stock && it.stock - it.qty < 3) {
        warnings.push({ code: 'STOCK_LOW', item: it.goodsId, message: '库存紧张，建议尽快下单' })
      }
    }

    // 支付方式门槛（单位：分）
    const total = items.reduce((a, it) => a + it.price * it.qty, 0)
    for (const pay of rules.paymentRules ?? []) {
      if (pay.method === cart.payment && total < pay.threshold) {
        errors.push({ code: 'THRESHOLD', field: 'payment', message: `${pay.method} 需满 ¥${(pay.threshold / 100).toFixed(2)}` })
      }
    }

    // 地址必填
    for (const f of rules.addressFields ?? []) {
      if (!cart.address?.[f]) errors.push({ code: 'MISSING_FIELD', field: `address.${f}`, message: `地址缺少 ${f}` })
    }

    return { passed: errors.length === 0, errors, warnings, total }
  }

  // 自动修复建议
  function suggestFix(cart, result) {
    const suggestions = []
    for (const e of result.errors) {
      if (e.code === 'OUT_OF_STOCK') {
        const it = cart.items.find((i) => i.goodsId === e.item)
        suggestions.push({ action: 'reduceQty', goodsId: e.item, to: it.stock, message: `调整为库存上限 ${it.stock} 件` })
      }
      if (e.code === 'LIMIT_EXCEED') {
        const to = (rules => 0)(e) ?? 0
        const limit = cart.limits?.[e.item] ?? e.message.match(/\d+/)?.[0]
        suggestions.push({ action: 'reduceQty', goodsId: e.item, to: Number(limit), message: `限购调整为 ${limit} 件` })
      }
      if (e.code === 'THRESHOLD') {
        const need = Number(e.message.match(/¥([\d.]+)/)[1] * 100) - result.total
        suggestions.push({ action: 'addItems', need: Math.ceil(need), message: `再凑 ¥${(Math.ceil(need) / 100).toFixed(2)} 即可用该支付方式` })
      }
      if (e.code === 'MISSING_FIELD') {
        suggestions.push({ action: 'fillAddress', field: e.field, message: `请补全 ${e.field}` })
      }
    }
    return suggestions
  }

  return { validate, suggestFix }
}

// 用例（金额单位：分）
const rules = { limits: { g1: 3 }, paymentRules: [{ method: 'cod', threshold: 5000 }], addressFields: ['city', 'street'] }
const v = createCheckoutValidator()
const cart = { payment: 'cod', items: [{ goodsId: 'g1', qty: 5, price: 1000, stock: 2 }], address: { city: '北京' } }
const result = v.validate(cart, rules)
console.log(result.passed)   // false
console.log(result.errors.map((e) => e.code))   // ['OUT_OF_STOCK','LIMIT_EXCEED','THRESHOLD','MISSING_FIELD']
console.log(v.suggestFix(cart, result).map((s) => s.action))   // ['reduceQty','reduceQty','addItems','fillAddress']
console.log(result.warnings.length)   // 0（库存 2 < 3 → 触发 STOCK_LOW？qty 5 > stock 2 已 error）
```
**评分点**：
- errors/warnings 分级（阻断 vs 放行提示）
- 结构化错误（code/field/item 可定位可 i18n）
- suggestFix 可执行建议（差额凑单、qty 收敛）
- 分单位整数运算
