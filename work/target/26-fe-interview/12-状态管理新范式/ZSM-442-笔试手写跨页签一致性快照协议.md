## ZSM-442：手写跨页签一致性快照协议
**答**：实现 `createConsensusTabs(bus, store)`：1) 多标签页写同一状态，每次写广播 `{ version, patch }`；2) 冲突仲裁：收到 version 不连续时，请求各页版本号，最高版本的页成为「源」，广播全量快照；低版本页接受快照（丢本地未广播修改并告警）；3) 版本号由「逻辑时钟」（各页 max seen + 1）维护，避免时钟偏移；4) 页关闭释放；5) 写锁：仲裁期间暂停本地写。

```javascript
export function createConsensusTabs(bus, store) {
  let version = 0, electing = false, localWritesPaused = false
  let candidates = new Map()

  bus.on('patch', (m) => {
    if (m.version === version + 1) { version = m.version; store.setState(m.patch) }
    else if (m.version > version + 1) elect(m.version)        // 断档 → 选举
  })
  bus.on('election-request', (m) => bus.emit('election-vote', { tabId, version }))
  bus.on('election-vote', (m) => {
    if (!electing) return
    candidates.set(m.tabId, m.version)
    if (candidates.size >= expectedTabs()) {
      const winner = [...candidates.entries()].sort((a, b) => b[1] - a[1])[0]
      electing = false; localWritesPaused = false
      if (winner[0] === tabId) bus.emit('snapshot', { version, state: store.getState() })   // 最高版本广播
    }
  })
  bus.on('snapshot', (s) => {
    if (s.version >= version) { version = s.version; store.setState(s.state) }   // 仲裁采纳
  })

  function elect(remoteVersion) {
    electing = true; localWritesPaused = true                  // 仲裁期写锁
    candidates.clear()
    bus.emit('election-request', {})
    setTimeout(() => {                                         // 选举超时兜底
      if (electing) { electing = false; localWritesPaused = false }
    }, 1000)
  }

  function write(patch) {
    if (localWritesPaused) throw new Error('consensus in progress')   // 写锁
    bus.emit('patch', { version: ++version, patch })
    store.setState(patch)
  }

  const expectedTabs = () => 2   // 简化：由心跳统计；实际实现需动态
  const tabId = crypto.randomUUID()
  return { write, getVersion: () => version }
}
```
**评分点**：
- 逻辑时钟版本号（不用 Date.now 防时钟偏移）——分布式共识的入门正确姿势
- 断档触发选举：最高版本页成为快照源，低版本页接受（last-writer-wins 的页级版）
- 仲裁期写锁（防止选举期间继续产生分裂版本）+ 超时兜底防死锁
- 边界诚实：expectedTabs 动态化、本地未广播修改丢失需告警——不假装完美
