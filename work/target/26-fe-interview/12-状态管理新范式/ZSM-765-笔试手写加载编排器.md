## ZSM-765：手写加载编排器（关键/非关键分级 + 超时降级 + idle 补拉）
**答**：实现 `createOrchestrator()`：1) `critical(name, promise)` / `deferred(name, promise)`：注册两类任务；2) `run({ timeoutMs })`：关键任务并行 + 逐个超时降级（超时的标记 degraded 但不阻塞其他），全部 settle 后回调 onCriticalReady（带 degraded 清单）；3) deferred 在关键完成后 idle 补拉（rIC 或 setTimeout），结果互不阻塞；4) `retry(name)`：对降级任务重试（带退避）；5) 附用例：关键任务超时降级 + 非关键不阻塞 + 重试恢复。

```javascript
function createOrchestrator({ onCriticalReady } = {}) {
  const criticals = new Map()      // name -> { promise, result, degraded }
  const deferreds = new Map()

  function critical(name, promise) {
    criticals.set(name, { promise, result: null, degraded: false })
    return this
  }
  function deferred(name, promise) {
    deferreds.set(name, { promise, result: null })
    return this
  }

  function withTimeout(p, ms) {
    return Promise.race([
      p,
      new Promise((_, reject) => setTimeout(() => reject(new Error('TIMEOUT')), ms)),
    ])
  }

  async function run({ timeoutMs = 3000 } = {}) {
    // 关键任务并行：逐个超时降级
    await Promise.all([...criticals.entries()].map(async ([name, entry]) => {
      try {
        entry.result = await withTimeout(entry.promise, timeoutMs)
        entry.degraded = false
      } catch (e) {
        entry.result = null
        entry.degraded = true                  // 降级：不阻塞其他关键任务
      }
    }))
    const degraded = [...criticals.entries()].filter(([, e]) => e.degraded).map(([n]) => n)
    onCriticalReady?.({ degraded, snapshot: snapshot() })

    // 非关键：idle 补拉（关键完成后）
    scheduleDeferred()
    return { degraded }
  }

  function scheduleDeferred() {
    const run = () => {
      Promise.allSettled([...deferreds.entries()].map(async ([name, entry]) => {
        entry.result = await entry.promise
      }))
    }
    if (typeof requestIdleCallback !== 'undefined') requestIdleCallback(run, { timeout: 2000 })
    else setTimeout(run, 100)
  }

  async function retry(name, { backoffMs = 500 } = {}) {
    const entry = criticals.get(name) ?? deferreds.get(name)
    if (!entry?.degraded && criticals.has(name) && entry.result) return entry.result
    await new Promise((r) => setTimeout(r, backoffMs))   // 退避重试
    entry.result = await entry.promise
    entry.degraded = false
    return entry.result
  }

  function snapshot() {
    return Object.fromEntries([...criticals.entries()].map(([n, e]) => [n, { value: e.result, degraded: e.degraded }]))
  }

  return { critical, deferred, run, retry, snapshot }
}

// 用例
let ready = null
const orch = createOrchestrator({ onCriticalReady: (r) => (ready = r) })
const slow = new Promise((r) => setTimeout(() => r('slow-data'), 500))     // 超时降级
orch
  .critical('feed', Promise.resolve('feed-data'))
  .critical('stats', slow)
  .deferred('recommend', Promise.resolve('rec-data'))

const { degraded } = await orch.run({ timeoutMs: 100 })
console.log(degraded)                        // ['stats']：关键任务超时降级
console.log(ready.snapshot.feed.value)       // 'feed-data'：其他关键任务不受阻
await new Promise((r) => setTimeout(r, 600)) // idle 补拉完成
const recovered = await orch.retry('stats')  // 重试恢复
console.log(recovered)                       // 'slow-data'
```
**评分点**：
- 关键任务并行 + 逐个超时降级（一个慢接口不拖全局）
- degraded 清单回调（UI 可渲染降级提示）
- deferred idle 补拉（不与关键路径争带宽）
- retry 带退避恢复（降级任务的闭环）
