## ZSM-619：手写状态热点交互的自动基准测试器
**答**：实现 `createBench()`：1) `case(name, setupFn, runFn)`：注册基准用例（setup 建状态、run 执行热点交互）；2) `run(times)`：预热 N 次后计时 M 次，输出 p50/p95/mean/opsPerSec（操作吞吐）；3) 对比模式：`compare(a, b)` 输出两个用例的相对性能（百分比）；4) GC 抖动防御：运行间强制 `global.gc?.()`（--expose-gc 下）且丢弃首个异常值（离群点剔除：超过 p95×3 丢弃）；5) 附用例：对比两种数组更新写法的基准。

```javascript
function createBench() {
  const cases = new Map()

  function case(name, setupFn, runFn) {
    cases.set(name, { setupFn, runFn })
    return () => cases.delete(name)
  }

  async function run(name, { warmup = 20, times = 200 } = {}) {
    const c = cases.get(name)
    if (!c) throw new Error(`unknown case ${name}`)
    const samples = []
    // 预热：JIT 稳定
    for (let i = 0; i < warmup; i++) {
      const ctx = c.setupFn()
      const t0 = performance.now()
      c.runFn(ctx)
      performance.now() - t0
    }
    for (let i = 0; i < times; i++) {
      const ctx = c.setupFn()
      const t0 = performance.now()
      c.runFn(ctx)
      const dt = performance.now() - t0
      samples.push(dt)
      globalThis.gc?.()          // GC 抖动防御
      await new Promise((r) => setTimeout(r, 0))   // 让出事件循环
    }
    // 离群点剔除：> p95×3 丢弃
    const sorted = [...samples].sort((a, b) => a - b)
    const p95v = sorted[Math.floor(sorted.length * 0.95)]
    const kept = samples.filter((s) => s <= p95v * 3)
    kept.sort((a, b) => a - b)
    const sum = kept.reduce((a, b) => a + b, 0)
    return {
      name,
      mean: +(sum / kept.length).toFixed(3),
      p50: +kept[Math.floor(kept.length * 0.5)].toFixed(3),
      p95: +kept[Math.floor(kept.length * 0.95)].toFixed(3),
      opsPerSec: +(1000 / (sum / kept.length)).toFixed(0),
      samples: kept.length,
    }
  }

  async function compare(a, b, opts) {
    const [ra, rb] = await Promise.all([run(a, opts), run(b, opts)])
    const diff = +(rb.mean / ra.mean).toFixed(2)
    return {
      [a]: ra, [b]: rb,
      relative: `b 比 a 慢 ${diff}x` + (diff > 1 ? '' : `（快 ${(1 / diff).toFixed(2)}x）`),
    }
  }

  return { case: case, run, compare }
}

// 用例：两种数组更新写法
const bench = createBench()
bench.case('immutable-append',
  () => ({ arr: new Array(5000).fill(0) }),
  (ctx) => { for (let i = 0; i < 50; i++) ctx.arr = [...ctx.arr, i] })
bench.case('mutate-append',
  () => ({ arr: new Array(5000).fill(0), v: 0 }),
  (ctx) => { for (let i = 0; i < 50; i++) { ctx.arr.push(i); ctx.v++ } })

bench.compare('immutable-append', 'mutate-append', { times: 100 })
  .then((r) => console.log(r))
// { 'immutable-append': { mean: 1.8, p50: 1.7, p95: 2.4, opsPerSec: 555, ... },
//   'mutate-append': { mean: 0.02, ... }, relative: 'b 比 a 慢 0.01x（快 90x）' }
```
**评分点**：
- 预热 + 让出事件循环（JIT 与微任务队列稳定）
- GC 防御（--expose-gc 可用则触发）与 p95×3 离群剔除
- 分位数输出（p50/p95 比均值更能反映用户体感）
- compare 相对倍数（优化决策的直接依据）
