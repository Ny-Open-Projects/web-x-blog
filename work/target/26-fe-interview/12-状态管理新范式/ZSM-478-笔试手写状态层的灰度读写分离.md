## ZSM-478：手写状态层灰度读写分离
**需求**：实现 `createGrayReaderWriter(oldStore, newStore)`：1) 读灰度：`read()` 按 bucket 返回对应 store 视图（bucket 函数注入）；2) 写双投：`write(patch)` 总是写 old（回滚安全），灰度桶内同时写 new；3) 一致性校验：双写后按 viewFn 对比，不一致计数（只统计灰度桶内）；4) 读路径一致性风险点处理：灰度读 new miss 时回退 old（soft read）；5) `stats()` 按桶统计 { reads, writes, softFallback, mismatch }；6) 全量切换接口 `promote()`（写 new only，保留 old 只读一周期）。

```javascript
export function createGrayReaderWriter(oldStore, newStore, viewFn, bucketFn) {
  const stats = { reads: 0, writes: 0, softFallback: 0, mismatch: 0 }

  function read() {
    stats.reads++
    const bucket = bucketFn()
    if (bucket === 'new') {
      const nv = viewFn(newStore.getState())
      if (Object.keys(nv).length === 0) {                    // new miss → soft read
        stats.softFallback++
        return viewFn(oldStore.getState())
      }
      return nv
    }
    return viewFn(oldStore.getState())
  }

  function write(patch) {
    stats.writes++
    oldStore.setState(patch)                                 // 回滚安全：old 必写
    if (bucketFn() === 'new') {
      newStore.setState(patch)
      if (JSON.stringify(viewFn(oldStore.getState())) !== JSON.stringify(viewFn(newStore.getState())))
        stats.mismatch++
    }
  }

  function promote() {
    // 全量切 new：old 转只读观察一周期
    const oldWrite = oldStore.setState
    oldStore.setState = () => { throw new Error('old store readonly after promote') }
    return () => (oldStore.setState = oldWrite)              // 观察期结束恢复（回滚开关）
  }

  return { read, write, promote, stats: () => ({ ...stats }) }
}
```
**评分点**：
- 写双投、old 必写（回滚安全第一原则）；读按桶 + miss 软回退
- 一致性校验只在灰度桶内统计（非灰度流量不产生噪音）
- promote 后 old 只读 + 可恢复（观察期设计——不是删掉就完事）
- stats 按维度可观测（softFallback 是灰度读的风险指标）
