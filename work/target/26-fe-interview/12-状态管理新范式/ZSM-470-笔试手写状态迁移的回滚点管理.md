## ZSM-470：手写状态迁移回滚点管理
**答**：实现 `createMigrationManager(store, storage)`：1) `beforeMigrate()`：迁移前保存回滚点（当前持久化数据快照 + 迁移版本标记）；2) `migrate(fn)`：执行迁移函数；3) 失败自动回滚：恢复回滚点数据 + 标记迁移失败原因；4) `rollback()`：手动回滚到最近回滚点（校验「迁移后无新写入」——有新写入拒绝回滚并提示）；5) 回滚点过期清理（7 天）；6) `verify(fn)` 迁移后校验函数（schema 检查），失败也触发自动回滚。

```javascript
export function createMigrationManager(store, storage) {
  const KEY = 'migration:rollback:v1'

  async function beforeMigrate() {
    const snap = { data: storage.get('app:state'), at: Date.now(), toVersion: store.version + 1 }
    storage.set(KEY, snap)
    // 过期清理
    const old = storage.get(KEY)
    if (old && Date.now() - old.at > 7 * 86400_000) storage.remove(KEY)
    return snap
  }

  async function migrate(fn, verify) {
    const point = storage.get(KEY)
    if (!point) throw new Error('call beforeMigrate first')
    try {
      await fn(store)
      if (verify) {
        const ok = await verify(store.getState())
        if (!ok) throw new Error('verify failed')           // 校验失败 → 自动回滚
      }
      storage.remove(KEY)                                   // 成功清除回滚点
      return { ok: true }
    } catch (e) {
      await rollback({ force: true })                       // 失败自动回滚
      report('migrate_fail', e)
      return { ok: false, reason: String(e) }
    }
  }

  async function rollback({ force = false } = {}) {
    const point = storage.get(KEY)
    if (!point) return { ok: false, reason: 'no rollback point' }
    const now = storage.get('app:state')
    if (!force && now !== point.data && hasNewWrites())     // 有新写入拒绝回滚
      return { ok: false, reason: 'new writes after migrate, use force' }
    storage.set('app:state', point.data)
    store.hydrate?.(point.data)
    storage.remove(KEY)
    return { ok: true }
  }

  const hasNewWrites = () => false   // 由写入计数器接入：迁移后写操作计数 > 0
  return { beforeMigrate, migrate, rollback }
}
```
**评分点**：
- 回滚点三要素（数据快照 + 时间戳 + 目标版本）
- 失败自动回滚 + verify 校验失败也回滚（迁移的安全性闭环）
- 「迁移后新写入拒绝回滚」守卫（防丢用户数据）——核心考点
- 回滚点过期清理（7 天）防存储堆积
