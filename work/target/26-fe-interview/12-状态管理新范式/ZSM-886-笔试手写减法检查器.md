## ZSM-886：手写「不需要状态」减法检查器（五项判定 + 交互密度门槛）
**答**：885 落地——实现 `checkSubtraction(field)`：对候选状态字段跑五项减法检查（服务端展示/展示联动/一次性/导航态/可派生），命中即「不需要建 store 状态」；`interactionDensity(page)`：页面交互密度评分（工具型 > 阈值时减法建议降权）；附用例：五项命中与密度门槛。

```javascript
function checkSubtraction(field) {
  // field: { source, serverOnly, pureDisplay, oneShot, isNavState, derivableFrom, deps }
  const hits = []
  // 1) 纯服务端展示
  if (field.serverOnly && !field.updatesInClient) {
    hits.push({ check: 'server-only', solution: 'RSC 直读/片段替换', store: false })
  }
  // 2) 纯展示联动
  if (field.pureDisplay && !(field.deps?.some((d) => d.kind === 'business'))) {
    hits.push({ check: 'display-linkage', solution: 'CSS :has / scroll-driven', store: false })
  }
  // 3) 一次性请求
  if (field.oneShot) {
    hits.push({ check: 'one-shot', solution: 'use(promise)', store: false })
  }
  // 4) 导航态
  if (field.isNavState) {
    hits.push({ check: 'nav-state', solution: 'Navigation API entry.state', store: false })
  }
  // 5) 可派生
  if (field.derivableFrom?.length) {
    hits.push({ check: 'derivable', solution: `派生自 ${field.derivableFrom.join('+')}，不存储`, store: false })
  }
  return {
    needStore: hits.length === 0,       // 全部不命中才需要状态
    subtraction: hits,
    suggestion: hits.length ? `零状态方案：${hits.map((h) => h.solution).join(' / ')}` : '保留 store 状态',
  }
}

// 交互密度评分：工具型页面减法建议降权
function interactionDensity(page) {
  const { interactiveComponents, totalComponents, statefulOps } = page
  const density = interactiveComponents / Math.max(1, totalComponents)
  const opsPerView = statefulOps / Math.max(1, totalComponents)
  return {
    density: +density.toFixed(2),
    opsPerView: +opsPerView.toFixed(1),
    type: density > 0.6 || opsPerView > 3 ? '工具型（减法降权）' : '内容型（减法优先）',
  }
}

// 用例
console.log(checkSubtraction({ serverOnly: true, updatesInClient: false, derivableFrom: [] }))
// { needStore: false, subtraction: [{ check: 'server-only', ... }] }
console.log(checkSubtraction({ serverOnly: false, derivableFrom: ['items', 'coupon'] }))
// { needStore: false, subtraction: [{ check: 'derivable', solution: '派生自 items+coupon，不存储' }] }
console.log(checkSubtraction({ serverOnly: false, updatesInClient: true, isUserIntent: true, derivableFrom: [] }))
// { needStore: true, suggestion: '保留 store 状态' }

console.log(interactionDensity({ interactiveComponents: 8, totalComponents: 10, statefulOps: 25 }))
// 工具型（减法降权）：编辑器类页面状态库刚需
console.log(interactionDensity({ interactiveComponents: 2, totalComponents: 15, statefulOps: 2 }))
// 内容型（减法优先）：资讯类页面零状态空间大
```
**评分点**：
- 五项减法判定（server/display/one-shot/nav/derivable）
- needStore 语义（全部不命中才建状态）
- 交互密度门槛（工具型降权 vs 内容型优先——885 边界落地）
- suggestion 输出零状态方案组合
