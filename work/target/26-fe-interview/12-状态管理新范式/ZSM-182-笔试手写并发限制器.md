## ZSM-182：手写并发限制器 pLimit（取数层的请求排队）
**答**：实现 `pLimit(concurrency)` 返回 `limit(fn, ...args)`：1) 同时最多 concurrency 个任务执行，超出的排队等待；2) 返回值是任务完成后的结果（Promise 语义不变）；3) `limit.activeCount/ pendingCount` 可查询；4) 支持清空队列 `limit.clearQueue()`（等待中的任务以「已取消」错误 reject，进行中的不受影响）。边界：fn 同步抛错也要正确释放占位。

```javascript
export function pLimit(concurrency) {
  if (!Number.isInteger(concurrency) || concurrency < 1)
    throw new RangeError('concurrency must be a positive integer')
  const queue = []
  let active = 0

  function next() {
    active--
    if (queue.length) queue.shift()()        // 唤醒队首
  }

  function run(fn, args, resolve, reject) {
    active++
    Promise.resolve()
      .then(() => fn(...args))
      .then(resolve, reject)                 // 同步抛错也进入此链
      .finally(next)                         // 无论成败都释放占位
  }

  function limit(fn, ...args) {
    if (active < concurrency) {
      return new Promise((resolve, reject) => run(fn, args, resolve, reject))
    }
    return new Promise((resolve, reject) => {
      queue.push(() => {
        if (limit.cancelled) { reject(new Error('cancelled')); return }
        run(fn, args, resolve, reject)
      })
    })
  }

  limit.clearQueue = () => {
    limit.cancelled = true
    queue.splice(0).forEach(wake => wake())   // 等待中的全部取消
    limit.cancelled = false
  }
  limit.activeCount = () => active
  limit.pendingCount = () => queue.length
  return limit
}
```
