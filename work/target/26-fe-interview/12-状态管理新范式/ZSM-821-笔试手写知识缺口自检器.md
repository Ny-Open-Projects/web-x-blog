## ZSM-821：手写知识缺口自检器（考点覆盖矩阵 + 边界标注）
**答**：实现 `createKnowledgeAudit()`：1) `audit(items)`：items 为自评清单 `{ topic, depth: 'api'|'impl'|'edge'|'none', evidence }`——按三层法（803）评估每主题深度；2) `matrix()`：输出覆盖矩阵（主题 × 深度），标记缺口（面试高危：depth='none' 或 'api' 无 evidence）；3) `priority()`：缺口优先级排序——按「面试高频 × 深度不足」加权；4) 附用例：六考点自评的缺口报告。

```javascript
const TOPIC_WEIGHTS = {   // 面试高频权重（状态域）
  源码级: 3, 业务建模: 3, 跨端同步: 2, 性能方法论: 2, 工具链: 1, 治理: 1,
}
const DEPTH_SCORE = { none: 0, api: 1, impl: 2, edge: 3 }

function createKnowledgeAudit() {
  let items = []

  function audit(newItems) {
    items = newItems.map((i) => ({
      ...i,
      depthScore: DEPTH_SCORE[i.depth] ?? 0,
      weak: i.depth === 'none' || (i.depth === 'api' && !i.evidence),
    }))
    return this
  }

  function matrix() {
    // 主题 × 深度覆盖矩阵
    const byTopic = new Map()
    for (const it of items) {
      const topic = it.topic.split('/')[0]      // 一级主题
      const cur = byTopic.get(topic) ?? { depths: new Set(), evidence: 0, weak: 0, total: 0 }
      cur.depths.add(it.depth)
      if (it.evidence) cur.evidence++
      if (it.weak) cur.weak++
      cur.total++
      byTopic.set(topic, cur)
    }
    return [...byTopic.entries()].map(([topic, m]) => ({
      topic,
      coverage: [...m.depths].join('/'),
      evidenceRate: m.total ? +(m.evidence / m.total).toFixed(2) : 0,
      weakCount: m.weak,
      risk: m.weak > 0 ? '高危' : 'ok',          // 有 weak 项 = 面试高危
    }))
  }

  function priority() {
    return items.filter((i) => i.weak)
      .map((i) => ({
        topic: i.topic,
        gap: i.depth,
        weight: TOPIC_WEIGHTS[i.topic.split('/')[0]] ?? 1,
        priorityScore: (TOPIC_WEIGHTS[i.topic.split('/')[0]] ?? 1) * (3 - DEPTH_SCORE[i.depth]),
        suggestion: i.depth === 'none' ? '从对应笔试卡默写起步' : '补源码证据（变量名级）+ 边界反例',
      }))
      .sort((a, b) => b.priorityScore - a.priorityScore)
  }

  return { audit, matrix, priority }
}

// 用例
const ka = createKnowledgeAudit()
ka.audit([
  { topic: '源码级/Zustand-selector', depth: 'edge', evidence: 'inst.hasValue 源码 + tearing 案例' },
  { topic: '源码级/XState-actor', depth: 'api' },                                    // 无证据：weak
  { topic: '跨端同步/CRDT', depth: 'none' },                                          // weak
  { topic: '性能方法论/九步法', depth: 'impl', evidence: '购物车优化叙事带数字' },
  { topic: '治理/健康分', depth: 'impl' },
])

console.log(ka.matrix())
// [{ topic: '源码级', coverage: 'edge/api', evidenceRate: 0.5, weakCount: 1, risk: '高危' },
//  { topic: '跨端同步', coverage: 'none', evidenceRate: 0, weakCount: 1, risk: '高危' },
//  ...]

console.log(ka.priority().map((p) => `${p.topic}(${p.priorityScore}): ${p.suggestion}`))
// 跨端同步/CRDT(6): 从笔试卡默写起步
// 源码级/XState-actor(4): 补源码证据
```
**评分点**：
- 三层深度自评（api/impl/edge 对应三层法）
- 高危判定（none 或 api 无 evidence）
- 优先级 = 面试高频权重 × 深度差（加权排序）
- suggestion 分级（none→默写起步 / api→补证据）
