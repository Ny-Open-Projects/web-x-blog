## ZSM-537：手写跨页全选的三态选择器
**答**：实现 `createSelector()` 管理批量操作：1) `toggleRow(pageId, rowId, total)`：单行勾选/取消；2) `togglePage(pageId, rowIds)`：整页全选——全选语义支持「仅本页」与「跨页累计」两种模式（mode: 'page' | 'global'）；3) 三态派生 `state(pageId, pageRowIds, total)`：返回 `'none' | 'some' | 'all'`（all 含跨页模式「本页全选但还有其他页选中」的情况应为 some）；4) 黑名单排除：`disabledIds` 里的行不可选且不计入统计；5) `clear()` 与 `selectedCount()`。附用例验证跨页三态。

```javascript
function createSelector({ mode = 'global' } = {}) {
  const selected = new Map()   // pageId -> Set<rowId>
  let disabled = new Set()

  const setOf = (p) => selected.get(p) ?? new Set()

  function toggleRow(pageId, rowId, total) {
    if (disabled.has(rowId)) return
    const s = new Set(setOf(pageId))
    s.has(rowId) ? s.delete(rowId) : s.add(rowId)
    if (s.size) selected.set(pageId, s)
    else selected.delete(pageId)
  }

  function togglePage(pageId, rowIds) {
    const valid = rowIds.filter((r) => !disabled.has(r))
    const cur = setOf(pageId)
    const allSelected = valid.every((r) => cur.has(r))
    if (allSelected) {
      selected.delete(pageId)                     // 取消本页
    } else if (mode === 'page') {
      selected.set(pageId, new Set(valid))        // 仅本页模式：替换
    } else {
      const s = new Set(cur)                      // 跨页模式：合并
      valid.forEach((r) => s.add(r))
      selected.set(pageId, s)
    }
  }

  function state(pageId, pageRowIds, total) {
    const selectable = pageRowIds.filter((r) => !disabled.has(r))
    const pageSel = selectable.filter((r) => setOf(pageId).has(r))
    const totalSel = [...selected.values()].reduce((a, s) => a + s.size, 0)
    if (totalSel === 0) return 'none'
    // global 模式下：本页全选≠全局全选
    if (pageSel.length === selectable.length && totalSel === total - disabled.size) return 'all'
    return 'some'
  }

  return {
    toggleRow, togglePage, state,
    selectedCount: () => [...selected.values()].reduce((a, s) => a + s.size, 0),
    selectedAll: () => [...selected.entries()].flatMap(([p, s]) => [...s].map((r) => `${p}:${r}`)),
    disable: (ids) => { disabled = new Set([...disabled, ...ids]) },
    clear: () => selected.clear(),
  }
}

// 用例
const sel = createSelector()   // global 模式
sel.togglePage('p1', ['a', 'b', 'c'])
console.log(sel.state('p1', ['a', 'b', 'c'], 5))  // 'some'：本页全选但共 5 行
sel.togglePage('p2', ['d', 'e'])
console.log(sel.state('p2', ['d', 'e'], 5))       // 'all'：全局 5 行全选
console.log(sel.selectedCount())                  // 5
sel.disable(['e'])
console.log(sel.state('p2', ['d', 'e'], 5))       // 'some'：e 被排除
```
**评分点**：
- 三态派生的「全局 vs 本页」区分（global 模式本页全选是 some）
- 黑名单行不可选且不计入 total 统计
- page/global 两种全选模式的语义差异（替换 vs 合并）
- 全空页/禁选行的边界处理（无可选行不误报 all）
