## ZSM-438：手写状态库内存水位控制
**答**：实现 `createMemoryGuard(store)`：1) 估算 store 内存占用（JSON 长度近似）；2) 三级水位：黄线（5MB）开始淘汰「未订阅缓存」（LRU）；橙线（10MB）强制淘汰最旧缓存（无论订阅，计数上报）；红线（20MB）触发「紧急瘦身」：清全部缓存数据仅保留运行必需态 + 告警；3) `track(key, size)` 登记条目尺寸；4) 每次 setState 后检查水位；5) `level()` 返回当前水位级别。

```javascript
export function createMemoryGuard(store, { yellow = 5 * 1024 * 1024, orange = 10 * 1024 * 1024, red = 20 * 1024 * 1024 } = {}) {
  const entries = new Map()   // key → { size, subs, ts }
  let evicted = 0

  function estimate() {
    let total = 0
    entries.forEach((e) => (total += e.size))
    return total
  }

  function evictLRU(limit) {
    const sorted = [...entries.entries()].filter(([, e]) => e.subs === 0).sort((a, b) => a[1].ts - b[1].ts)
    for (const [k] of sorted) {
      if (estimate() <= limit) break
      entries.delete(k); store.drop?.(k); evicted++
    }
  }

  function check() {
    const size = estimate()
    if (size > red) { entries.clear(); store.clearCache?.(); report('mem_red', { size }) }        // 紧急瘦身
    else if (size > orange) { evictLRU(orange); report('mem_orange', { size }) }                  // 强制淘汰
    else if (size > yellow) evictLRU(yellow)                                                      // 温和淘汰
    return size
  }

  return {
    track(key, sizeFn, subs = 0) {
      entries.set(key, { get size() { return sizeFn() }, subs, ts: Date.now() })
      return () => entries.delete(key)
    },
    touch(key) { const e = entries.get(key); if (e) e.ts = Date.now() },
    check,
    level: () => { const s = estimate(); return s > red ? 'red' : s > orange ? 'orange' : s > yellow ? 'yellow' : 'green' },
    get evicted() { return evicted },
  }
}
```
**评分点**：
- 三级水位分级响应（温和 LRU → 强制 → 紧急清空），每级动作与代价明确
- 订阅保护（有活引用的条目黄线不淘汰；橙线强制时上报）
- 水位检查挂在 setState 后（写入驱动）+ size 惰性计算（sizeFn）
- level()/evicted 计数可观测；store.drop/clearCache 的集成钩子设计
