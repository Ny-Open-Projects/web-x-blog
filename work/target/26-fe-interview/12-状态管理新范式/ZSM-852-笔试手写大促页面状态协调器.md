## ZSM-852：手写大促页面状态协调器（五域联动 + 按钮四态派生）
**答**：实现 `createPromoCoordinator()`：1) 注册五域状态（activity/stock/coupon/order）与派生按钮四态：`'upcoming'|'can-seize'|'sold-out'|'joined'`（判定链：phase→已参与→库存→可抢）；2) `seize(userId)`：门禁链（851）+ 本地库存乐观扣减 + 幂等键 + 超时回滚；3) `applyStockPush(list)`：推送节流合并（1s 窗口取最后值）；4) `couponUsable(couponId)`：券与活动价互斥派生；5) 附用例：四态判定链与乐观回滚。

```javascript
function createPromoCoordinator({ now = () => Date.now(), nowServer }) {
  const activity = { startAt: 1000, endAt: 3000 }
  let stock = new Map([['g1', { total: 10, shown: 10 }]])
  const joined = new Set()
  const coupons = new Map([['c1', { amount: 500, usedWithPromo: true }]])
  let inFlight = false
  let pushTimer = null
  const pendingPush = new Map()
  const log = []

  function phase(t = now()) {
    return t < activity.startAt ? 'upcoming' : t < activity.endAt ? 'active' : 'ended'
  }

  // 按钮四态：判定链 phase → joined → stock → can-seize
  function buttonState(userId, goodsId = 'g1') {
    const p = phase()
    if (p === 'upcoming') return 'upcoming'
    if (p === 'ended') return 'sold-out'
    if (joined.has(userId)) return 'joined'
    if ((stock.get(goodsId)?.shown ?? 0) <= 0) return 'sold-out'
    return 'can-seize'
  }

  function seize(userId, goodsId = 'g1') {
    if (buttonState(userId, goodsId) !== 'can-seize') return { ok: false, reason: 'GATE' }
    if (inFlight) return { ok: false, reason: 'IN_FLIGHT' }       // 重入防护
    inFlight = true
    const idemKey = crypto.randomUUID()

    // 乐观扣减（紧张感）
    const s = stock.get(goodsId)
    s.shown--
    emit({ type: 'optimistic-dec', goodsId, shown: s.shown })

    return {
      ok: true,
      exec: async () => {
        try {
          const res = await api.seize({ goodsId, userId, idemKey })
          if (res.code === 'SOLD_OUT') { s.shown = 0; emit({ type: 'sold-out-sync' }); return { ok: false, code: 'SOLD_OUT' } }
          joined.add(userId)                                       // 服务端确认后标记参与
          emit({ type: 'joined', userId, order: res.order })
          return { ok: true, order: res.order }
        } catch (e) {
          s.shown++                                                // 失败回滚乐观
          emit({ type: 'rollback', goodsId, shown: s.shown, reason: e.code })
          return { ok: false, reason: e.code }
        } finally {
          inFlight = false
        }
      },
    }
  }

  // 推送节流合并：1s 窗口取最后值
  function applyStockPush(list) {
    for (const it of list) pendingPush.set(it.goodsId, it.shown)
    clearTimeout(pushTimer)
    pushTimer = setTimeout(() => {
      for (const [goodsId, shown] of pendingPush) {
        stock.set(goodsId, { ...stock.get(goodsId), shown })       // 推送覆盖本地乐观（权威）
      }
      pendingPush.clear()
      emit({ type: 'stock-sync' })
    }, 1000)
  }

  // 券与活动价互斥派生
  function couponUsable(couponId) {
    const c = coupons.get(couponId)
    return c && !c.usedWithPromo
  }

  function emit(e) { log.push({ ts: now(), ...e }) }

  return { buttonState, seize, applyStockPush, couponUsable, phase, log: () => log, stock: () => stock }
}

// 用例
let t = 1500     // active 窗口
const pc = createPromoCoordinator({ now: () => t })
console.log(pc.buttonState('u1'))              // 'can-seize'
console.log(pc.buttonState('u0'))              // 'can-seize'（另一用户）
const s1 = pc.seize('u1')
s1.exec().then(() => {
  console.log(pc.buttonState('u1'))            // 'joined'：已参与
  console.log(pc.buttonState('u2'))            // 'can-seize'（库存 9）
  t = 4000
  console.log(pc.buttonState('u3'))            // 'sold-out'：ended
  pc.applyStockPush([{ goodsId: 'g1', shown: 0 }])
  setTimeout(() => console.log(pc.buttonState('u4')), 1100)   // 'sold-out'：推送节流后
})
```
**评分点**：
- 四态判定链（phase→joined→stock→can-seize 短路）
- 乐观扣减与推送权威覆盖（紧张感 + 终局一致）
- 重入防护 + 幂等键 + 失败回滚（851 全链）
- 推送节流合并（1s 窗口取最后值防风暴）
