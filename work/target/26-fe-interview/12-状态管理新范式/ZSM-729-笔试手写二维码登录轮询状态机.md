## ZSM-729：手写二维码登录状态机（轮询驱动 + 过期刷新 + 竞态防御）
**答**：实现 `createQrLogin(api)`：1) `start()`：生成二维码（qrcodeId）并启动 2s 轮询；2) 状态迁移由轮询响应驱动：`valid → scanned(带设备名) → confirmed`，`expired` 时自动重新生成（regen 计数上限 3 次）；3) 竞态防御：轮询响应按 **qrcodeId 匹配**——旧二维码的响应（已刷新后到达）丢弃；轮询在 stop 后不发请求；4) `stop()`：清理定时器与在途请求；5) 附用例：过期自动重生 + 旧响应丢弃。

```javascript
function createQrLogin(api, { intervalMs = 2000, maxRegen = 3 } = {}) {
  let state = 'idle'          // idle|valid|scanned|confirmed|expired|regenerating
  let qrcodeId = null
  let device = null
  let regenCount = 0
  let pollTimer = null
  let stopped = true
  const listeners = new Set()

  function setState(s) { state = s; listeners.forEach((fn) => fn(snapshot())) }

  function snapshot() { return { state, qrcodeId, device, regenCount } }

  async function generate() {
    if (regenCount >= maxRegen) { setState('failed'); return }
    const res = await api.createQr()
    qrcodeId = res.id
    regenCount++
    setState('valid')
    startPolling()
  }

  function startPolling() {
    clearTimeout(pollTimer)
    pollTimer = setTimeout(poll, intervalMs)
  }

  async function poll() {
    if (stopped || !qrcodeId) return
    try {
      const res = await api.qrStatus(qrcodeId)
      if (stopped) return
      // 竞态防御：响应的 qrcodeId 必须与当前一致（旧码响应丢弃）
      if (res.qrcodeId !== qrcodeId) return startPolling()

      switch (res.status) {
        case 'scanned':
          if (state !== 'scanned') { device = res.device; setState('scanned') }
          break
        case 'confirmed':
          setState('confirmed')
          stopped = true
          return                       // 终态：停止轮询
        case 'expired':
          await regenerate()
          return
      }
      startPolling()
    } catch {
      if (!stopped) startPolling()     // 网络错误：继续轮询
    }
  }

  async function regenerate() {
    setState('regenerating')
    clearTimeout(pollTimer)
    await generate()
  }

  function start() {
    stopped = false
    regenCount = 0
    return generate()
  }

  function stop() {
    stopped = true                     // 轮询循环与在途响应双双拦截
    clearTimeout(pollTimer)
    setState('idle')
  }

  return { start, stop, subscribe: (fn) => (listeners.add(fn), () => listeners.delete(fn)), snapshot }
}

// 用例
let currentQr = 'qr1'
const api = {
  createQr: async () => ({ id: currentQr = `qr${Math.random().toString(36).slice(2, 6)}` }),
  qrStatus: async (id) => {
    // 模拟：qr1 过期 → 重生成 qr2；qr2 被扫 → 确认
    if (id === 'qr1') return { qrcodeId: 'qr1', status: 'expired' }
    if (id === 'qr2' && pollCount++ > 2) return { qrcodeId: 'qr2', status: 'confirmed' }
    return { qrcodeId: 'qr2', status: 'scanned', device: { name: 'iPhone 15' } }
  },
}
let pollCount = 0

const login = createQrLogin(api)
login.subscribe((s) => console.log(s.state, s.qrcodeId))
await login.start()
// valid(qr1) → regenerating → valid(qr2) → scanned(iPhone) → confirmed
login.stop()    // 清理：轮询停止 + 在途响应丢弃
```
**评分点**：
- 轮询响应按 qrcodeId 匹配（旧码响应丢弃——竞态防御核心）
- 过期自动重生成 + regen 上限（failed 终态）
- confirmed 终态停止轮询（不泄漏定时器）
- stop 双重拦截（stopped 标志拦循环 + 拦在途响应）
