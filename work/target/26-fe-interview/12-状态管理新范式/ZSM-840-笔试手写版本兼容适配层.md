## ZSM-840：手写版本兼容适配层（v4/v5 双栈共存与渐进迁移）
**答**：zustand v4→v5 灰度迁移——实现 `createCompat()`：1) 兼容 v4 签名 `useStore(selector, equalityFn)` → 自动转 v5 的 `useStore(useShallow(selector))` 语义（equalityFn 提供时走自定义比较）；2) 默认 export 兼容：`import zustand from 'zustand'` → 命名导出重导出（迁移期 polyfill，lint 标记 deprecated）；3) 迁移进度跟踪：`compat.stats()` 记录已迁移/未迁移文件（用调用点标记）；4) 附用例：equalityFn 兼容路径与 stats。

```javascript
import { useSyncExternalStore, useCallback, useRef } from 'react'
import { useStore as useV5Store } from 'zustand'

// v4 签名兼容：useStore(api, selector, equalityFn)
function useCompatStore(api, selector = (s) => s, equalityFn) {
  // equalityFn 提供时：包装 selector 做结果比较闸门（v5 语义）
  if (equalityFn) {
    const selectorRef = useRef(selector)
    selectorRef.current = selector
    const stableSelector = useCallback((state) => {
      // 内部缓存：浅相等返回旧引用
      const next = selectorRef.current(state)
      compatCache.current = compatCache.current ?? { value: next }
      if (equalityFn(compatCache.current.value, next)) return compatCache.current.value
      compatCache.current = { value: next }
      return next
    }, [equalityFn])
    return useV5Store(api, stableSelector)
  }
  return useV5Store(api, selector)
}

const compatCache = { current: null }

// 默认 export 兼容（迁移期 polyfill）
// polyfills/zustand-default.js
// export { create, createStore } from 'zustand'          ← v5 删除了默认 export
// export default { create, createStore }                   ← 兼容旧 import
// lint 规则：no-restricted-imports 标记 deprecated 路径

// 迁移进度跟踪
const compatStats = { migrated: new Set(), pending: new Set() }
function markMigrated(file) { compatStats.pending.delete(file); compatStats.migrated.add(file) }
function markPending(file) { compatStats.pending.add(file) }
function stats() {
  return {
    migrated: compatStats.migrated.size,
    pending: compatStats.pending.size,
    progress: compatStats.migrated.size / (compatStats.migrated.size + compatStats.pending.size),
  }
}

// 用例
const listeners = new Set()
const api = {
  state: { items: ['a', 'b'] },
  getState: () => api.state,
  setState(p) { Object.assign(api.state, typeof p === 'function' ? p(api.state) : p); listeners.forEach((f) => f()) },
  subscribe(f) { listeners.add(f); return () => listeners.delete(f) },
}

// v4 签名：带 equalityFn（shallow）
function shallow(a, b) {
  const ka = Object.keys(a ?? {}), kb = Object.keys(b ?? {})
  return ka.length === kb.length && ka.every((k) => Object.is(a[k], b[k]))
}

// v4 用法模拟：selector 返回数组（新引用）
let renderCount = 0
function renderOnce() {
  renderCount++
  // useCompatStore(api, s => s.items, shallow)
  const sel = (s) => s.items
  const eq = shallow
  const next = sel(api.getState())
  compatCache.current = compatCache.current ?? { value: next }
  const stable = eq(compatCache.current.value, next)
      ? compatCache.current.value
      : (compatCache.current = { value: next }).value
  return stable
}
const v1 = renderOnce()
api.setState({ noise: 1 })                  // items 引用未变
const v2 = renderOnce()
console.log(v1 === v2, renderCount)         // true：equalityFn 闸门稳定
api.setState((s) => ({ items: [...s.items, 'c'] }))
const v3 = renderOnce()
console.log(v3 !== v2)                      // true：items 真变更产生新引用

// stats
markPending('a.ts'); markPending('b.ts'); markMigrated('c.ts')
console.log(stats())                        // { migrated: 1, pending: 2, progress: 0.33 }
```
**评分点**：
- v4 签名兼容（equalityFn 走自定义闸门，无 equalityFn 直通 v5）
- 迁移进度跟踪（migrated/pending 集合 + progress 比率）
- 默认 export polyfill 的 lint deprecated 标记
- 渐进迁移语义（同文件可逐步迁移，双栈共存不冲突）
