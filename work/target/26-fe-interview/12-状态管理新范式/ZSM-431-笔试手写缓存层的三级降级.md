## ZSM-431：手写三级缓存读取链（内存→会话→网络）
**答**：实现 `createTieredCache({ memory, session, fetcher })`：1) `get(key)` 依次读内存（同步最快）→ sessionStorage（JSON，带 TTL）→ 网络；2) 低层命中后回填高层（会话命中回填内存）；3) 网络 fetcher 失败但会话有过期数据 → 返回过期数据并标 `stale: true`（可用性优先）；4) `set(key, value)` 三层同写；5) 每层命中计数 `stats()`；6) 内存层上限 LRU 50。

```javascript
export function createTieredCache({ fetcher, ttl = 60_000, memLimit = 50 }) {
  const mem = new Map()        // key → { value, ts }
  const stats = { mem: 0, session: 0, network: 0, stale: 0 }

  function memGet(key) {
    const e = mem.get(key)
    if (!e) return null
    if (Date.now() - e.ts > ttl) return null            // 内存层同样受 TTL
    mem.delete(key); mem.set(key, e)                    // LRU 刷新
    return e
  }
  function memSet(key, value) {
    if (mem.size >= memLimit) mem.delete(mem.keys().next().value)   // 最旧淘汰
    mem.set(key, { value, ts: Date.now() })
  }

  async function get(key) {
    const m = memGet(key)
    if (m) { stats.mem++; return { value: m.value, source: 'memory' } }

    try {
      const raw = sessionStorage.getItem(`c:${key}`)
      if (raw) {
        const e = JSON.parse(raw)
        if (Date.now() - e.ts <= ttl) { memSet(key, e.value); stats.session++
          return { value: e.value, source: 'session' } }
        // 过期：先试网络，失败回退过期值
        try {
          const v = await fetcher(key); memSet(key, v)
          sessionStorage.setItem(`c:${key}`, JSON.stringify({ value: v, ts: Date.now() }))
          stats.network++; return { value: v, source: 'network' }
        } catch { stats.stale++; return { value: e.value, source: 'session-stale', stale: true } }
      }
    } catch { /* session 异常继续走网络 */ }

    const v = await fetcher(key)                        // 网络层
    memSet(key, v)
    sessionStorage.setItem(`c:${key}`, JSON.stringify({ value: v, ts: Date.now() }))
    stats.network++
    return { value: v, source: 'network' }
  }

  return { get, set: (key, value) => { memSet(key, value)
      sessionStorage.setItem(`c:${key}`, JSON.stringify({ value, ts: Date.now() })) },
    stats: () => stats }
}
```
**评分点**：
- 三层顺序与回填（低层命中回填高层，下次更快）
- 可用性优先：网络失败回退过期值并标 stale（可用性 > 新鲜性的明确取舍）
- TTL 双层生效（内存也过期，防内存层永远新鲜）
- LRU 淘汰 + 命中统计可观测；sessionStorage 异常不炸主流程
