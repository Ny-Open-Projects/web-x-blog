## ZSM-553：手写状态录制回放器（bug 复现工具）
**答**：实现 `createRecorder(store)`：1) `start()`：录制所有 setState（action 名+参数）与关键事件（可插桩），环形缓冲上限可配；2) `stop()`/`export()`：导出 JSON（含初始快照+事件序列），体积敏感——大参数截断；3) `replay(recording, targetApi)`：在新 store 上重放事件序列，重放速率可配（含逐帧单步）；4) `replayUntil(predicate)`：重放到谓词为真停止（复现 bug 前一刻）；5) 附用例验证导出与重放一致性。

```javascript
function createRecorder(store, { max = 1000 } = {}) {
  let recording = false
  let events = []
  let initial = null
  let origSet = null

  function start() {
    initial = structuredCloneSafe(store.getState())
    events = []
    recording = true
    origSet = store.setState
    store.setState = (updater, replace, name) => {
      if (!recording) return origSet(updater, replace, name)
      const args = typeof updater === 'function'
        ? { __fn: true, captured: null }   // 函数型 updater 无法序列化：重放时用捕获模式
        : updater
      const r = origSet(updater, replace, name)
      events.push({ i: events.length, ts: Date.now(), name: name ?? 'anon', args: truncate(args) })
      if (events.length > max) events.shift()
      return r
    }
  }

  function stop() {
    recording = false
    if (origSet) { store.setState = origSet; origSet = null }
  }

  function truncate(v, depth = 0) {
    if (typeof v === 'string' && v.length > 200) return v.slice(0, 200) + '…'
    if (depth > 4) return '[deep]'
    if (Array.isArray(v)) return v.slice(0, 50).map((x) => truncate(x, depth + 1))
    if (v && typeof v === 'object') return Object.fromEntries(
      Object.entries(v).slice(0, 50).map(([k, val]) => [k, truncate(val, depth + 1)]))
    return v
  }

  function structuredCloneSafe(v) {
    return JSON.parse(JSON.stringify(v))   // demo 简化；生产用 structuredClone
  }

  function export() { return JSON.stringify({ initial, events, exportedAt: Date.now() }) }

  async function replay(json, targetApi, { speed = 0, onStep } = {}) {
    const { initial: ini, events: evs } = JSON.parse(json)
    targetApi.setState(ini, true)
    for (const e of evs) {
      targetApi.setState({ __replayed: e.name }, false, e.name)
      onStep?.(e)
      if (speed > 0) await new Promise((r) => setTimeout(r, speed))
    }
  }

  function replayUntil(json, targetApi, predicate) {
    const { initial: ini, events: evs } = JSON.parse(json)
    targetApi.setState(ini, true)
    for (const e of evs) {
      targetApi.setState({ __replayed: e.name }, false, e.name)
      if (predicate(targetApi.getState())) return e   // 停在触发点
    }
    return null
  }

  return { start, stop, export, replay, replayUntil }
}

// 用例
const store = { state: { n: 0 }, getState: () => store.state, setState(p, _r, name) { Object.assign(store.state, typeof p === 'function' ? p(store.state) : p) } }
const rec = createRecorder(store)
rec.start()
store.setState({ n: 1 }, false, 'inc')
store.setState({ n: 2 }, false, 'inc')
rec.stop()
const data = rec.export()
const clone = { state: { n: 0 }, getState: () => clone.state, setState(p) { Object.assign(clone.state, p) } }
rec.replayUntil(data, clone, (s) => s.n >= 2)   // 重放到 n=2
console.log(clone.getState())                   // { n: 2, __replayed: 'inc' }
```
**评分点**：
- 环形缓冲录制（上限截断防内存爆）
- 导出体积控制（参数截断 + 初始快照）
- replayUntil 谓词停点（bug 触发前一刻定位）
- 函数型 updater 的序列化边界处理（显式标注不可序列化）
