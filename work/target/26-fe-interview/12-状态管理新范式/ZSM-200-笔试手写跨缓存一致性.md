## ZSM-200：手写双缓存一致性同步器（Query 缓存 ↔ Zustand 派生冗余的受控同步）
**需求**：业务要求把服务端数据 `['goods']` 的一个子集镜像进 Zustand（组件外高频读取）。实现 `createMirror({ queryKey, pick, store, write })`：1) Query 缓存变化（成功/失效重取）时自动把 `pick(data)` 写入 store（write 提供自定义写入动作）；2) 去抖：200ms 内多次缓存更新合并写入一次；3) 单向同步铁律：store 侧写入镜像字段以外的字段不触发回写（防环）；4) 提供 `stop()` 停止镜像；5) 缓存清空（gc/remove）时写入 fallback。核心考点：单向数据流 + 变化检测 + 防环。

```javascript
export function createMirror({ queryCache, queryKey, pick, write, fallback = null }) {
  let stopped = false
  let lastPicked; let timer = null

  const unsubscribe = queryCache.subscribe((event) => {
    if (stopped) return
    const q = queryCache.find({ queryKey })
    const isSuccess = event.type === 'success' ||
      (q?.state.status === 'success' && event.type === 'updated')
    if (!isSuccess) return
    const picked = pick(q.state.data)
    if (deepEq(picked, lastPicked)) return               // 变化检测：pick 结果没变不写
    lastPicked = picked
    clearTimeout(timer)
    timer = setTimeout(() => write(picked), 200)         // 去抖合并写入
  })

  function onRemoved() {
    write(fallback)                                      // 缓存被清：镜像兜底
  }
  queryCache.subscribe(e => { if (e.type === 'removed') onRemoved() })

  return {
    stop: () => { stopped = true; clearTimeout(timer); unsubscribe() },
  }
}
```
