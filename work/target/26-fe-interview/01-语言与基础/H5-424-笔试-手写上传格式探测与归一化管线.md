## H5-424：笔试：手写上传格式探测与归一化管线

**答**：实现 `normalizeUpload(file)`：① 读文件头探测真实格式（JPEG/PNG/HEIC/WEBP/GIF），未知格式拒绝；② HEIC：尝试 createImageBitmap 解码（iOS 支持），失败提示「请改用 JPEG」；③ 全部图片统一归一化：EXIF 方向修正（from-image）→ 重编码 webp（q0.85）；④ GIF 保留动图不重编码（直接透传但校验大小 ≤5MB）；⑤ 输出归一化报告 { from, to, sizeBefore, sizeAfter, exifFixed }。写出核心实现。

```javascript
async function sniff(file) {
  const b = new Uint8Array(await file.slice(0, 16).arrayBuffer());
  const hex = (a) => [...a].map(x => x.toString(16).padStart(2, '0')).join(' ');
  if (b[0] === 0xFF && b[1] === 0xD8) return 'jpeg';
  if (b[0] === 0x89 && b[1] === 0x50) return 'png';
  if (hex(b.slice(4, 12)).includes('66 74 79 70') &&
      (hex(b.slice(8, 12)).includes('68 65 69 63') || hex(b.slice(8, 12)).includes('6d 69 66 31'))) return 'heic';
  if (b[8] === 0x57 && b[9] === 0x45 && b[10] === 0x42 && b[11] === 0x50) return 'webp';
  if (b[0] === 0x47 && b[1] === 0x49 && b[2] === 0x46) return 'gif';
  return null;
}

async function normalizeUpload(file) {
  const fmt = await sniff(file);
  if (!fmt) throw new Error('unknown format');
  if (fmt === 'gif') {                                       // 动图透传
    if (file.size > 5 * 1024 * 1024) throw new Error('gif too large');
    return { blob: file, report: { from: fmt, to: 'gif', sizeBefore: file.size, sizeAfter: file.size, exifFixed: false } };
  }
  let bmp;
  try { bmp = await createImageBitmap(file, { imageOrientation: 'from-image' }); }
  catch { if (fmt === 'heic') throw new Error('HEIC 解码失败, 请改用 JPEG'); throw new Error('decode fail'); }
  const off = new OffscreenCanvas(bmp.width, bmp.height);
  off.getContext('2d').drawImage(bmp, 0, 0);
  bmp.close();
  const blob = await off.convertToBlob({ type: 'image/webp', quality: 0.85 }); // EXIF 归一化
  return { blob, report: { from: fmt, to: 'webp', sizeBefore: file.size, sizeAfter: blob.size, exifFixed: true } };
}
```

**评分点**：
- 魔数探测覆盖 HEIC（ftyp box 特征）/WEBP（RIFF 偏移）/GIF
- HEIC 解码失败的用户提示路径
- GIF 动图透传不重编码（帧丢失防护）
- EXIF 修正 + 统一 webp 输出 + 报告完整
