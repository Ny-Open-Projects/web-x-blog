## JS-288：笔试：防御性正则封装

**题目**：实现 `safeMatch(pattern, input, opts)`：用户输入可作正则；限制输入长度、限制回溯风险（try/catch + 超时保护）、自动转义开关；正则编译失败返回 null 而非抛错。

**答**：
```javascript
function buildRe(pattern, { caseInsensitive = true } = {}) {
  try {
    return new RegExp(pattern, caseInsensitive ? 'iu' : 'u');
  } catch {
    return null; // 非法正则降级为字面量搜索
  }
}

function safeMatch(pattern, input, opts = {}) {
  const { maxLen = 5000, timeoutMs = 100, autoEscape = true } = opts;
  if (input.length > maxLen) return null;             // 长度闸门
  const src = autoEscape
    ? pattern.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')  // 元字符转义
    : pattern;
  const re = buildRe(src, opts);
  if (!re) return null;
  // 超时保护：正则同步执行无法中断，用「分块 + 长度探测」近似
  const chunks = Math.ceil(input.length / 1000);
  const results = [];
  for (let i = 0; i < chunks; i++) {
    const seg = input.slice(i * 1000, (i + 1) * 1000 + 50); // 重叠防漏
    const m = seg.match(re);
    if (m) results.push({ index: i * 1000 + m.index, match: m[0] });
    if (results.length >= 20) break; // 结果上限
  }
  return results;
}

safeMatch('订单号: 123', '订单号: 123 出货'); // [{ index: 0, match: '订单号: 123' }]
safeMatch('((((((((((((((((', 'x'.repeat(3000)); // null（元字符被转义，安全）
```

**评分点**：
- 长度闸门是 ReDoS 第一防线
- autoEscape 默认开启（用户输入永不作为正则语法）
- 编译失败降级（null 而非抛错）
- 诚实说明 JS 正则无法中断、分块是近似手段（完整方案要 Worker + 计时，加分）
