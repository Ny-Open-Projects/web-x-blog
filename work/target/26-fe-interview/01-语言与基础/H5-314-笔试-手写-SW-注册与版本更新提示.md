## H5-314：笔试：手写 SW 注册与版本更新提示

**题目**：实现 `setupSW()`：注册 sw.js；周期检查更新（60s）；发现 waiting 版本时显示「更新可用」toast（带刷新按钮）；用户点击后 skipWaiting 并在 controllerchange 后 reload；防重复 reload。

**答**：
```javascript
let reloading = false;
async function setupSW() {
  if (!('serviceWorker' in navigator)) return;
  const reg = await navigator.serviceWorker.register('/sw.js');

  navigator.serviceWorker.addEventListener('controllerchange', () => {
    if (reloading) return;
    reloading = true;       // 防重复 reload
    location.reload();
  });

  async function check() {
    await reg.update();
    if (reg.waiting) showUpdateToast(async () => {
      reg.waiting.postMessage('SKIP_WAITING'); // 激活后触发 controllerchange
    });
  }
  check();
  setInterval(check, 60000);
}
function showUpdateToast(onRefresh) {
  const el = document.createElement('div');
  el.style.cssText = 'position:fixed;bottom:20px;left:50%;transform:translateX(-50%);padding:12px 20px;background:#333;color:#fff;border-radius:8px;display:flex;gap:12px;z-index:9999';
  el.innerHTML = '<span>新版本已就绪</span><button style="color:#4fc3f7;background:none;border:none;cursor:pointer">立即刷新</button>';
  el.querySelector('button').onclick = () => { el.remove(); onRefresh(); };
  document.body.appendChild(el);
}
```

**评分点**：
- controllerchange 的防重入（reloading 标志）——skipWaiting 可能多次触发事件
- 周期 update 检查（默认只导航时检查）
- 用户主动刷新而非强制 reload
- SW 不支持的环境静默退出
