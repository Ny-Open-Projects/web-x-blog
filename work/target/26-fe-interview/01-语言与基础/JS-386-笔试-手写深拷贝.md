## JS-386：笔试：手写深拷贝

**答**：实现 `deepClone(obj)`：① 支持循环引用（WeakMap 记录已拷贝对象）；② 正确处理 Date/RegExp/Map/Set/ArrayBuffer/TypedArray；③ 函数与 Symbol 属性处理（可枚举 Symbol 拷贝，函数引用共享）；④ 不可拷贝类型（DOM 节点/window）抛错或返回引用；⑤ 数组保持数组。写出核心实现。

```javascript
function deepClone(obj, seen = new WeakMap()) {
  if (obj === null || typeof obj !== 'object') return obj;      // 原始值/函数引用
  if (obj instanceof Date) return new Date(obj);
  if (obj instanceof RegExp) return new RegExp(obj.source, obj.flags);
  if (obj instanceof Map) {
    const m = new Map();
    seen.set(obj, m);
    obj.forEach((v, k) => m.set(deepClone(k, seen), deepClone(v, seen)));
    return m;
  }
  if (obj instanceof Set) {
    const s = new Set();
    seen.set(obj, s);
    obj.forEach(v => s.add(deepClone(v, seen)));
    return s;
  }
  if (ArrayBuffer.isView(obj)) {                                 // TypedArray/DataView
    return new obj.constructor(obj);                             // 拷贝 backing buffer
  }
  if (obj instanceof ArrayBuffer) return obj.slice(0);
  if (typeof Node !== 'undefined' && obj instanceof Node)
    throw new Error('DOM 节点不可克隆');

  if (seen.has(obj)) return seen.get(obj);                       // 循环引用
  const out = Array.isArray(obj) ? [] : Object.create(Object.getPrototypeOf(obj));
  seen.set(obj, out);
  for (const key of [...Object.keys(obj), ...Object.getOwnPropertySymbols(obj)
      .filter(k => obj.propertyIsEnumerable(k))]) {
    out[key] = deepClone(obj[key], seen);
  }
  return out;
}
```

**评分点**：
- WeakMap 处理循环引用（记录时机：先 set 再递归）
- 特殊类型分发完整（Date/RegExp/Map/Set/TypedArray/ArrayBuffer）
- Symbol 可枚举属性拷贝 + 原型链保留（Object.create）
- DOM/window 类型的防御性处理
