## JS-315：笔试：手写深度比较

**题目**：实现 `deepEqual(a, b)`：原始值（含 NaN 视为相等）、Date/RegExp、数组与对象（键集合相等即相等，不比较顺序）、循环引用（双 WeakMap 记忆化）。

**答**：
```javascript
function deepEqual(a, b, seen = new Map()) {
  if (Object.is(a, b)) return true;              // 含 NaN 相等、+0/-0 区分
  if (a === null || b === null || typeof a !== 'object' || typeof b !== 'object') return false;

  const key = [a, b];
  if (seen.has(JSON.stringify(key))) return true; // 循环引用：已在比较中视为相等
  seen.set(JSON.stringify(key), true);

  if (a instanceof Date && b instanceof Date) return +a === +b;
  if (a instanceof RegExp && b instanceof RegExp) return a.source === b.source && a.flags === b.flags;
  if (a instanceof Map && b instanceof Map) {
    if (a.size !== b.size) return false;
    for (const [k, v] of a) {
      if (!b.has(k) || !deepEqual(v, b.get(k), seen)) return false;
    }
    return true;
  }
  if (Array.isArray(a) !== Array.isArray(b)) return false;
  if (Array.isArray(a)) {
    return a.length === b.length && a.every((v, i) => deepEqual(v, b[i], seen));
  }
  // 普通对象：键集合相等
  const ka = Object.keys(a), kb = Object.keys(b);
  if (ka.length !== kb.length) return false;
  return ka.every(k => k in b && deepEqual(a[k], b[k], seen));
}

deepEqual({ a: 1, b: { c: [1, NaN] } }, { a: 1, b: { c: [1, NaN] } }); // true（NaN 相等）
deepEqual([1, 2], [2, 1]);   // false（顺序敏感）
deepEqual({ a: 1 }, { a: 1, b: undefined }); // false（键数不同）
```

**评分点**：
- Object.is 处理 NaN——高频判分点
- 键集合比较（长度 + 每键递归），数组顺序敏感
- 循环引用的记忆化（Map 存配对）
- Date/RegExp/Map 特殊分支
