## JS-290：笔试：带清理的生命周期管理器

**题目**：实现 `Lifecycle` 类：`on(target, type, fn)`、`timer(fn, ms, interval)`、`raf(fn)` 三种注册都自动登记；`destroy()` 一次性清理全部（解绑监听、清定时器、停 rAF、断 Observer）；destroy 后再注册抛错。

**答**：
```javascript
class Lifecycle {
  #cleanups = [];
  #dead = false;

  on(target, type, fn, opts) {
    this.#check();
    target.addEventListener(type, fn, opts);
    this.#cleanups.push(() => target.removeEventListener(type, fn, opts));
    return () => this.#cleanups = this.#cleanups.filter(c => { c(); return false; });
  }
  timer(fn, ms, interval = false) {
    this.#check();
    const id = interval ? setInterval(fn, ms) : setTimeout(fn, ms);
    this.#cleanups.push(() => (interval ? clearInterval : clearTimeout)(id));
  }
  raf(fn) {
    this.#check();
    let id, alive = true;
    const loop = (t) => { fn(t); if (alive) id = requestAnimationFrame(loop); };
    id = requestAnimationFrame(loop);
    this.#cleanups.push(() => { alive = false; cancelAnimationFrame(id); });
  }
  observe(target, opts, cb) {
    this.#check();
    const io = new IntersectionObserver(cb, opts);
    io.observe(target);
    this.#cleanups.push(() => io.disconnect());
  }
  #check() { if (this.#dead) throw new Error('Lifecycle already destroyed'); }
  destroy() {
    this.#dead = true;
    this.#cleanups.splice(0).forEach(c => c()); // 逆序或顺序均可，登记时成对
    this.#cleanups = [];
  }
}
// 组件卸载：lifecycle.destroy() —— 一行清空所有异步资源
```

**评分点**：
- 「注册即登记清理函数」的统一模型（核心）
- destroy 后再注册抛错（防静默泄漏）
- rAF 循环的 alive 标志（cancel 只停当前帧）
- 与 useEffect/组件卸载的对接说明（加分）
