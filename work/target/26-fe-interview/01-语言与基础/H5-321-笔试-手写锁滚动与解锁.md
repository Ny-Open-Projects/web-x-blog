## H5-321：笔试：手写锁滚动与解锁

**题目**：实现 `lockScroll() / unlockScroll()`：锁定期间背景不滚动（含 iOS）、不跳动、恢复后滚动位置还原；弹层内部仍可正常滚动（含到达边界时不穿透）。

**答**：
```javascript
let savedY = 0, savedStyle = '';

function lockScroll() {
  savedY = window.scrollY;
  savedStyle = document.body.getAttribute('style') || '';
  // 方案：body 固定在当前位置，滚动彻底冻结（iOS 最稳）
  document.body.style.cssText += `;position:fixed;top:${-savedY}px;left:0;right:0;overflow:hidden;`;
}
function unlockScroll() {
  document.body.setAttribute('style', savedStyle);
  window.scrollTo(0, savedY); // 还原位置
}

// 弹层内部滚动防穿透：overscroll-behavior + 兜底 touchmove 控制
function containScroll(el) {
  el.style.overscrollBehavior = 'contain'; // 现代浏览器防滚动链
  // 兜底：到边界时阻止默认行为（老 iOS）
  el.addEventListener('touchmove', (e) => {
    const { scrollTop, scrollHeight, clientHeight } = el;
    const atTop = scrollTop <= 0 && e.touches[0].clientY > lastY;
    const atBottom = scrollTop + clientHeight >= scrollHeight && e.touches[0].clientY < lastY;
    if (atTop || atBottom) e.preventDefault();
    lastY = e.touches[0].clientY;
  }, { passive: false });
}
let lastY = 0;
```

**评分点**：
- body fixed 方案的位置保存与还原（scrollY 取负放到 top）——iOS 下的标准解
- overscroll-behavior: contain 阻断滚动链（现代浏览器）
- 边界处 preventDefault 的兜底（passive: false 是必须的）
- 还原时 scrollTo 恢复位置，避免「弹窗关闭回到顶部」
