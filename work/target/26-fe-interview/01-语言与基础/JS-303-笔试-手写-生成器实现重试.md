## JS-303：笔试：生成器实现重试

**题目**：实现 `retryGen(genFn, { retries, backoff })`：驱动生成器；yield 出的 Promise 失败时自动重试当前步（指数退避），成功继续下一步；重试耗尽抛最后一错。

**答**：
```javascript
const sleep = (ms) => new Promise(r => setTimeout(r, ms));

async function retryGen(genFn, { retries = 3, base = 300 } = {}) {
  const gen = genFn();
  let input;
  while (true) {
    const { value, done } = gen.next(input);
    if (done) return value;
    let attempt = 0;
    while (true) {
      try {
        input = await value;      // 等结果，成功传回生成器
        break;
      } catch (e) {
        if (attempt >= retries) throw e;      // 耗尽：抛给生成器（其可 catch）或直接抛出
        await sleep(base * 2 ** attempt);     // 指数退避
        attempt++;
        // 重试同一个 value（Promise 已 reject，需重新创建——此处约定 yield 返回任务函数）
        input = undefined;
        const nv = gen.next();                // 约定：失败后生成器重新 yield 新任务
        if (nv.done) throw e;
        // 简化版：直接要求 yield 出「任务工厂」以保证每次重试拿到新 Promise
        value = nv.value;
      }
    }
  }
}
// 用法：yield 出任务工厂（每次调用创建新 Promise）
const result = await retryGen(function* () {
  const a = yield () => flakyFetch('/a');     // 失败自动重试
  const b = yield () => fetch('/b').then(r => r.json());
  return b;
});
```

**评分点**：
- 关键认知：reject 过的 Promise 重试无效——必须 yield 任务工厂每次新建 Promise（本题核心判分）
- 驱动循环与结果回传 next(input)
- 指数退避与重试上限
- 失败后与生成器的交互约定说明
