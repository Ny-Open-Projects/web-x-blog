## H5-384：笔试：手写深链降级与白名单校验器

**答**：实现 `DeepLinker`：① `open(path, params, { fallback })`：先校验 path 在路由白名单（通配模式数组如 `/goods/:id`），非法直接走 fallback；② scheme 唤起 + 800ms visibility 探测降级：未离开页面则跳 fallback（fallback 必须同域白名单内）；③ iOS Universal Link 域名下直接用 https 链接；④ 参数校验：id 类参数必须为数字、redirect 类参数必须同域；⑤ 记录降级事件与最终去向（归因）。写出核心实现。

```javascript
const ROUTES = ['/goods/:id', '/activity/:slug', '/home'];
const SAFE_HOST = 'm.example.com';

function matchRoute(path) {
  return ROUTES.some(r => {
    const rp = r.split('/'), pp = path.split('/');
    return rp.length === pp.length && rp.every((seg, i) => seg.startsWith(':') ? !!pp[i] : seg === pp[i]);
  });
}
function safeParam(k, v) {
  if (/^id$|Id$/.test(k)) return /^\d+$/.test(v);            // id 必须数字
  if (/redirect|url/i.test(k)) { try { return new URL(v).host === SAFE_HOST; } catch { return false; } }
  return typeof v === 'string' && v.length < 200;
}

class DeepLinker {
  open(path, params = {}, { fallback } = {}) {
    if (!matchRoute(path) || Object.entries(params).some(([k, v]) => !safeParam(k, v))) {
      this._log(path, 'rejected'); return location.assign(fallback || '/home');
    }
    const qs = new URLSearchParams(params);
    const scheme = `myapp://${path}?${qs}`;
    const start = Date.now();
    location.href = scheme;
    setTimeout(() => {
      if (document.visibilityState !== 'hidden' && Date.now() - start < 2000) {
        this._log(path, 'downgrade');
        location.assign(fallback || `https://${SAFE_HOST}${path}?${qs}`);
      }
    }, 800);
  }
  _log(path, evt) { navigator.sendBeacon('/api/mon/deeplink', JSON.stringify({ path, evt, t: Date.now() })); }
}
```

**评分点**：
- 路由白名单模式匹配 + 参数类型校验（id 数字/redirect 同域）
- 降级判定用 visibility（未离开页面即失败），超时保护
- fallback 自身走同域约束，防开放重定向
- 降级事件埋点归因（rejected/downgrade）
