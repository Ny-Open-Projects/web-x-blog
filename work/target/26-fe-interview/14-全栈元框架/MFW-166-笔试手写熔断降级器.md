## MFW-166：笔试：手写一个熔断降级器（错误率超阈值切兜底）
**答**：实现 `CircuitBreaker`（SSR 取数保护）：① 滑动窗口统计成功率；② 连续失败率超 50%（窗口 ≥10 次）→ 熔断 OPEN，直接走 fallback；③ 熔断 30s 后进 HALF-OPEN：放行一次探测，成功 → CLOSE，失败 → 重新 OPEN（重置计时）；④ CLOSE 下全量放行；⑤ fallback 未提供时熔断期间直接抛错。

```ts
type State = 'CLOSED' | 'OPEN' | 'HALF_OPEN';

export class CircuitBreaker {
  private state: State = 'CLOSED';
  private results: Array<{ ok: boolean; at: number }> = [];
  private openedAt = 0;
  private probing = false;

  constructor(
    private opts: {
      windowMs?: number;        // 滑动窗口
      minSamples?: number;      // 最小样本数
      errorThreshold?: number;  // 失败率阈值
      openMs?: number;          // 熔断时长
      fallback?: () => unknown;
    } = {},
  ) {
    this.opts = {
      windowMs: 10_000, minSamples: 10, errorThreshold: 0.5, openMs: 30_000, ...opts,
    };
  }

  private record(ok: boolean): void {
    const now = Date.now();
    this.results.push({ ok, at: now });
    // 滑动窗口清理过期样本
    this.results = this.results.filter((r) => now - r.at < this.opts.windowMs!);
  }

  private errorRate(): number {
    if (this.results.length < this.opts.minSamples!) return 0; // 样本不足不熔断
    return this.results.filter((r) => !r.ok).length / this.results.length;
  }

  async exec<T>(fn: () => Promise<T>): Promise<T> {
    // HALF-OPEN 探测：只放一个请求
    if (this.state === 'OPEN') {
      if (Date.now() - this.openedAt < this.opts.openMs!) {
        if (this.opts.fallback) return this.opts.fallback() as T;
        throw new Error('CIRCUIT_OPEN');
      }
      if (this.probing) { // 已有探测在途，其余请求走兜底
        if (this.opts.fallback) return this.opts.fallback() as T;
        throw new Error('CIRCUIT_HALF_OPEN');
      }
      this.probing = true;
      this.state = 'HALF_OPEN';
    }

    try {
      const result = await fn();
      this.record(true);
      if (this.state === 'HALF_OPEN') {
        this.state = 'CLOSED';       // 探测成功 -> 恢复
        this.results = [];
      } else if (this.state === 'CLOSED' && this.errorRate() > this.opts.errorThreshold!) {
        this.trip();                 // 错误率超标 -> 熔断
      }
      return result;
    } catch (e) {
      this.record(false);
      if (this.state === 'HALF_OPEN') {
        this.trip();                 // 探测失败 -> 重新熔断
      } else if (this.errorRate() > this.opts.errorThreshold!) {
        this.trip();
      }
      throw e;
    } finally {
      if (this.state === 'HALF_OPEN') { this.probing = false; this.state = 'OPEN'; this.openedAt = Date.now(); }
    }
  }

  private trip(): void {
    this.state = 'OPEN';
    this.openedAt = Date.now();
  }
}
```

**评分点**：
1. 三态机完整：CLOSED/OPEN/HALF_OPEN 转换正确（30s 后单次探测、成功闭合、失败重开）。
2. 滑动窗口有最小样本数保护（minSamples 前 10 次全失败才熔断，避免低流量误触发）。
3. HALF-OPEN 期间只放一个探测请求（probing 门禁），其余走兜底。
4. fallback 缺失时抛错而非挂起——能说清「熔断器不吞错误，只是换一种失败方式」。
5. finally 中探测结束后恢复 OPEN 与计时器重置。
