## MFW-651：笔试：手写埋点收集端点

**答**：实现 `createCollector({ schemaRegistry, writeTo, { flushMs, maxBatch, sampleRate, now } })`。规则：1) `collect(events: unknown[])`：校验（schema：事件注册 + 字段白名单——RBL-645）→ 合规的最小化（IP 截断/PII 剥离）→ 缓冲；2) 采样：过载时（buffer > maxBatch × 3）→ 采样（按事件名 hash 采样——保留固定子集而非随机（统计一致性））；3) flush：定时（flushMs）或满批（maxBatch）→ writeTo(batch)；写失败：重新入队（头部——不丢已缓冲）+ 指数退避；4) 返回 `{ collect, stats }`（stats: received/rejected/sampled/flushed）。边界：events 空；全部不合规（buffer 不增）；writeTo 持续失败（退避加大 + 死信上限丢弃最旧）。

```javascript
function createCollector({ schemaRegistry = {}, writeTo, flushMs = 5000, maxBatch = 1000, sampleRate = 0.5, now = Date.now } = {}) {
  const buffer = []; const deadLetters = [];
  const stats = { received: 0, rejected: 0, sampled: 0, flushed: 0 };
  let timer = null; let backoff = 0; let lastFlushFail = 0;
  function validate(e) {
    const fields = schemaRegistry[e?.event];
    if (!fields) return false;
    return Object.keys(e).every(k => k === 'event' || fields.includes(k));
  }
  function minimize(e) {
    const { ip, email, phone, ...rest } = e;               // PII 剥离
    return rest;
  }
  function scheduleFlush() {
    if (timer || buffer.length === 0) return;
    timer = setTimeout(async () => {
      timer = null;
      const batch = buffer.splice(0, maxBatch);
      if (!batch.length) return;
      try { await writeTo(batch); stats.flushed += batch.length; backoff = 0; }
      catch {
        buffer.unshift(...batch);                          // 写失败回队头
        backoff = Math.min(backoff ? backoff * 2 : 1000, 30000);
        if (buffer.length > maxBatch * 5) deadLetters.push(...buffer.splice(0, maxBatch));   // 死信上限
        setTimeout(scheduleFlush, backoff);
        return;
      }
      if (buffer.length) scheduleFlush();                  // 还有剩余继续
    }, flushMs + backoff);
  }
  function collect(events = []) {
    const overloaded = buffer.length > maxBatch * 3;
    for (const e of events ?? []) {
      stats.received++;
      if (!validate(e)) { stats.rejected++; continue; }
      if (overloaded) {
        const h = parseInt(String(e.event ?? '').split('').reduce((a, c) => a + c.charCodeAt(0), 0));
        if (h % 100 > sampleRate * 100) { stats.sampled++; continue; }   // 事件名 hash 采样
      }
      buffer.push(minimize(e));
    }
    if (buffer.length >= maxBatch) scheduleFlush();        // 满批即刷
    else scheduleFlush();
  }
  return { collect, stats, _buffer: buffer, _deadLetters: deadLetters };
}

// 自测
const written = [];
let fail = 0;
const c = createCollector({
  schemaRegistry: { page_view: ['path', 'ts'] },
  writeTo: async (batch) => { if (fail > 0) { fail--; throw new Error('wh down'); }
    written.push(...batch); },
  flushMs: 10, maxBatch: 3, now: () => 1,
});
c.collect([
  { event: 'page_view', path: '/', ts: 1 },               // 合规
  { event: 'page_view', path: '/', email: 'x' },          // PII 剥离
  { event: 'unknown', x: 1 },                             // 拒绝
  { event: 'page_view', extra: 1 },                       // 字段不合规拒绝
]);
await sleep(30);
console.log(c.stats.received === 4 && c.stats.rejected === 2);
console.log(written.length === 2);                       // 2 条合规入仓（email 已剥离）
console.log(written.every(e => !e.email));

// 写失败重入队
fail = 1;
c.collect([{ event: 'page_view', path: '/a', ts: 2 }]);
await sleep(10);                                         // 第一次 flush 失败
console.log(c._buffer.length === 1);                     // 回队头
await sleep(20);                                         // 退避后重试成功
console.log(written.length === 3);
```

**评分点**：
- fire-and-forget（204 即回）+ 批量 flush（写放大控制 RBL-650）；失败回队头 + 指数退避
- 采样的一致性（事件名 hash 固定子集——非随机）；PII 剥离（最小化 RBL-601）
- 死信上限（防内存膨胀 RBL-619）；边界（空/全不合规）
- 能说清「收集与处理解耦是埋点架构基本盘——同步写 DB 会让收集端被 DB 拖垮」属于加分
