## MFW-782：笔试：手写乐观锁冲突

**答**：实现 `withOptimisticLock(db, table, { maxRetries })`。db: 注入（模拟 DB）。方法：`update(id, patch, expectedVersion)`。规则：1) WHERE version = expectedVersion AND id = id → 成功返回 `{ ok: true, newVersion }`；2) 0 行更新 = 冲突 → 查当前 version → 返回 `{ conflict: true, currentVersion, currentData }`；3) **自动重试**：maxRetries > 0 时，冲突后自动读最新 version 重试（期望他人修改是瞬态的）；重试耗尽 → 返回 conflict；4) delete 同理（WHERE version = ? AND id = ?）；5) 返回统一结构。边界：maxRetries=0（不重试）；record 不存在（404 语义——非 conflict）；version=0（初始——合法）。

```javascript
function withOptimisticLock(db, table, { maxRetries = 2 } = {}) {
  async function attempt(id, patch, expectedVersion) {
    const rows = db.updateWhere(table, { id, version: expectedVersion }, patch);
    if (rows.length > 0)
      return { ok: true, newVersion: expectedVersion + 1, data: rows[0] };
    const current = db.find(table, id);
    if (!current) return { notFound: true };
    return { conflict: true, currentVersion: current.version, currentData: current };
  }
  return {
    async update(id, patch, expectedVersion) {
      let ver = expectedVersion;
      for (let i = 0; i <= maxRetries; i++) {
        const r = await attempt(id, patch, ver);
        if (r.ok || r.notFound) return r;
        if (r.conflict && i < maxRetries) ver = r.currentVersion;   // 重试：用最新 version
        else return r;                                     // 重试耗尽或最后一次
      }
      return { conflict: true };
    },
    async delete(id, expectedVersion) {
      const r = db.deleteWhere(table, { id, version: expectedVersion });
      return r > 0 ? { ok: true } : { conflict: true };
    },
  };
}

// 自测
function mockDb() {
  const tables = new Map();
  return {
    tables,
    seed: (t, rows) => tables.set(t, rows.map(r => ({ ...r }))),
    updateWhere: (t, where, patch) => {
      const rows = tables.get(t) ?? [];
      const matches = rows.filter(r => r.id === where.id && r.version === where.version);
      matches.forEach(r => { Object.assign(r, patch); r.version++; });
      return matches;
    },
    deleteWhere: (t, where) => {
      const rows = tables.get(t) ?? [];
      const before = rows.length;
      tables.set(t, rows.filter(r => !(r.id === where.id && r.version === where.version)));
      return before - tables.get(t).length;
    },
    find: (t, id) => (tables.get(t) ?? []).find(r => r.id === id),
  };
}
const db = mockDb();
db.seed('users', [{ id: 'u1', name: 'old', version: 1 }]);
let lock = withOptimisticLock(db, 'users', { maxRetries: 0 });
// 成功
let r = lock.update('u1', { name: 'new' }, 1);
console.log(r.ok === true && r.newVersion === 2);
// 冲突：期望 version=1 但已是 2
r = lock.update('u1', { name: 'other' }, 1);
console.log(r.conflict === true && r.currentVersion === 2);
// 重试后成功
lock = withOptimisticLock(db, 'users', { maxRetries: 1 });
r = lock.update('u1', { name: 'final' }, 1);             // 冲突 → 重试用 v2 → 成功
console.log(r.ok === true && r.newVersion === 3);
// 不存在
r = lock.update('ghost', { name: 'x' }, 1);
console.log(r.notFound === true);
// 删除
r = lock.delete('u1', 3);
console.log(r.ok === true);
```

**评分点**：
- WHERE version 的原子更新（SQL 层 OCC——0 行 = 冲突）；自动重试（最新 version 重新尝试）
- notFound 区分（404 语义 vs 409 conflict——不同错误类型）；delete 的乐观锁
- 边界（maxRetries=0/version=0/不存在）；冲突时返回当前数据（client 合并决策）
- 能说清「乐观锁 = WHERE version 原子更新：0 行 = 冲突返回当前数据让 client 合并——无锁竞争的优势」属于加分
