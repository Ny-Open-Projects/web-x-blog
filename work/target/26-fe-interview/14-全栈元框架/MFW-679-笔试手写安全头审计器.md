## MFW-679：笔试：手写安全头审计器

**答**：实现 `auditSecurityHeaders(headers, { route })`。headers: `Record<string, string>`（实际响应头）。规则：1) 必需头缺失：HSTS/X-Frame-Options/nosniff/Referrer-Policy → 'missing'（error）；2) 值检查：HSTS max-age < 31536000 → 'weak-hsts'；X-Frame-Options 不是 DENY/SAMEORIGIN → 'weak-frame'；CSP 存在但含 'unsafe-inline'（且无 nonce）→ 'weak-csp'；3) CSP 检查：script-src 无 nonce 且无 hash 且无 strict-dynamic → 'weak-csp'；4) 豁免：/api/** 路由的 CSP/frame-ancestors 豁免（JSON 无渲染面）。返回 `{ findings: Array<{ header, type, severity }>, ok }`（ok = 无 error）。边界：headers 空（全 missing）；/api 路由的豁免生效；CSP 不存在也报 weak-csp。

```javascript
function auditSecurityHeaders(headers = {}, { route = '/', isApi = false } = {}) {
  const findings = [];
  const get = (k) => headers[k.toLowerCase()];
  // HSTS
  const hsts = get('strict-transport-security');
  if (!hsts) findings.push({ header: 'HSTS', type: 'missing', severity: 'error' });
  else {
    const maxAge = /max-age=(\d+)/.exec(hsts)?.[1];
    if (+maxAge < 31536000) findings.push({ header: 'HSTS', type: 'weak-hsts', severity: 'warn' });
  }
  // X-Frame-Options
  if (!isApi) {
    const fo = get('x-frame-options');
    if (!fo) findings.push({ header: 'X-Frame-Options', type: 'missing', severity: 'error' });
    else if (!['DENY', 'SAMEORIGIN'].includes(fo.toUpperCase()))
      findings.push({ header: 'X-Frame-Options', type: 'weak-frame', severity: 'warn' });
  }
  // nosniff
  if (get('x-content-type-options') !== 'nosniff')
    findings.push({ header: 'X-Content-Type-Options', type: 'missing', severity: 'error' });
  // Referrer-Policy
  if (!get('referrer-policy'))
    findings.push({ header: 'Referrer-Policy', type: 'missing', severity: 'error' });
  // CSP
  const csp = get('content-security-policy');
  if (!isApi) {
    if (!csp) findings.push({ header: 'CSP', type: 'weak-csp', severity: 'warn' });
    else if (csp.includes("'unsafe-inline'") && !csp.includes("'nonce-") &&
             !csp.includes('strict-dynamic'))
      findings.push({ header: 'CSP', type: 'weak-csp', severity: 'warn' });
  }
  return { findings, ok: !findings.some(f => f.severity === 'error') };
}

// 自测
// 全缺失
let r = auditSecurityHeaders({}, { route: '/home' });
console.log(r.findings.filter(f => f.type === 'missing').length >= 4);
console.log(r.ok === false);

// 完整
const good = {
  'strict-transport-security': 'max-age=31536000; includeSubDomains',
  'x-frame-options': 'DENY', 'x-content-type-options': 'nosniff',
  'referrer-policy': 'strict-origin-when-cross-origin',
  'content-security-policy': "script-src 'self' 'nonce-abc'",
};
r = auditSecurityHeaders(good, { route: '/home' });
console.log(r.ok === true && r.findings.length === 0);

// 弱 HSTS
r = auditSecurityHeaders({ ...good, 'strict-transport-security': 'max-age=86400' },
  { route: '/home' });
console.log(r.findings.some(f => f.type === 'weak-hsts' && f.severity === 'warn'));

// weak-csp：unsafe-inline 无 nonce
r = auditSecurityHeaders({ ...good,
  'content-security-policy': "script-src 'self' 'unsafe-inline'" }, { route: '/home' });
console.log(r.findings.some(f => f.type === 'weak-csp'));

// API 豁免 CSP/frame
r = auditSecurityHeaders({ ...good,
  'content-security-policy': "script-src 'self' 'unsafe-inline'" }, { route: '/api/x', isApi: true });
console.log(!r.findings.some(f => f.header === 'CSP' && f.type === 'weak-csp'));
console.log(!r.findings.some(f => f.header === 'X-Frame-Options'));
```

**评分点**：
- 必需头四项 + CSP 深检（unsafe-inline 无 nonce/hash 的弱 CSP）；HSTS 的 max-age 检查
- API 路由豁免（JSON 无渲染面——CSP/frame 的豁免语义）；error/warn 分级
- 边界（空 headers 全 missing/完整通过）；ok 仅由 error 决定
- 能说清「安全头的遗漏是静默漏洞——middleware 统一注入 + 审计器门禁是防遗漏的双防线」属于加分
