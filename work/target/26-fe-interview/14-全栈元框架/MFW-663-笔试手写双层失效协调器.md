## MFW-663：笔试：手写双层失效协调器

**答**：实现 `createDualInvalidator({ frameworkRevalidate, cdnPurge, { onPartialFail } })`。规则：1) invalidate(tags)：**并行**执行框架 revalidate + CDN purge（两层独立——不串行（一个慢不拖另一个））；2) **部分失败的补偿**：一层成功一层失败 → onPartialFail（记录哪层失败——重试只重试失败层）；3) 重试：失败层指数退避重试 2 次（RBL-590）；4) 全失败 → dead letter；5) 批量失效：多 tag 合并为一次 CDN purge（purge 的批量 API——配额节省 RBL-662）。返回 `{ invalidate(tags), stats: { frameworkOk, cdnOk, partialFails, dead } }`。边界：单层注入（只测一层）；tags 空（跳过）。

```javascript
function createDualInvalidator({ frameworkRevalidate, cdnPurge, onPartialFail = () => {} } = {}) {
  const stats = { frameworkOk: 0, cdnOk: 0, partialFails: 0, dead: 0 };
  const sleep = (ms) => new Promise(r => setTimeout(r, ms));
  async function retry(fn, attempts = 3) {
    for (let i = 0; i < attempts; i++) {
      try { await fn(); return true; } catch { if (i < attempts - 1) await sleep(1000 * 2 ** i); }
    }
    return false;
  }
  async function invalidate(tags = []) {
    if (!tags.length) return;
    // 并行双层
    const [fwOk, cdnOk] = await Promise.all([
      frameworkRevalidate ? retry(() => frameworkRevalidate(tags)) : Promise.resolve(true),
      cdnPurge ? retry(() => cdnPurge(tags)) : Promise.resolve(true),
    ]);
    if (fwOk) stats.frameworkOk++;
    if (cdnOk) stats.cdnOk++;
    if (fwOk !== cdnOk) {
      stats.partialFails++;
      onPartialFail({ tags, layer: fwOk ? 'cdn' : 'framework' });   // 单层失败：补偿信号
    }
    if (!fwOk && !cdnOk) stats.dead++;
    return { fwOk, cdnOk };
  }
  return { invalidate, stats };
}

// 自测
const log = [];
const inv = createDualInvalidator({
  frameworkRevalidate: async (tags) => { log.push(['fw', ...tags]); },
  cdnPurge: async (tags) => { log.push(['cdn', ...tags]); },
});
await inv.invalidate(['product:1', 'products']);
console.log(log.length === 2);                           // 两层都调用
console.log(log[0][0] === 'fw' && log[1][0] === 'cdn');  // 并行（顺序不保证但都在）
console.log(inv.stats.frameworkOk === 1 && inv.stats.cdnOk === 1);

// 部分失败
const partials = [];
let cdnFails = 0;
const inv2 = createDualInvalidator({
  frameworkRevalidate: async () => {},
  cdnPurge: async () => { if (cdnFails++ < 2) throw new Error('cdn down'); },   // 前 2 次失败
  onPartialFail: (info) => partials.push(info),
});
const r2 = await inv2.invalidate(['p:1']);
console.log(r2.fwOk === true && r2.cdnOk === false);     // 重试后仍失败
console.log(partials[0].layer === 'cdn');                // 部分失败记录
console.log(inv2.stats.partialFails === 1);

// 批量：多 tag 一次 purge
const purges = [];
const inv3 = createDualInvalidator({
  frameworkRevalidate: async (tags) => tags.forEach(t => log.push(['fw', t])),
  cdnPurge: async (tags) => purges.push(tags),           // 一次收到全部 tags（批量）
});
await inv3.invalidate(['product:1', 'product:2', 'products']);
console.log(purges.length === 1 && purges[0].length === 3);   // 合并一次调用

// 空 tags
await inv.invalidate([]);
console.log(inv.stats.frameworkOk === 1);                // 未变化
```

**评分点**：
- 双层并行（框架 + CDN 独立执行——互不阻塞）；部分失败补偿（单层重试）
- 批量合并（多 tag 一次 CDN purge——配额节省 RBL-662）；dead letter 全失败兜底
- 边界（单层注入/空 tags）；stats 可观测
- 能说清「双层失效必须并行+同步：漏 CDN 失效 = 边缘旧内容屏蔽框架新数据（RBL-581）」属于加分
