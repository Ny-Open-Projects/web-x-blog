## MFW-629：笔试：手写语义等待辅助器

**答**：实现 `waitForSemantics(page, events, { timeout })`。events: `Array<{ marker: string, phase: 'before'|'after' }>`（按序断言：骨架 marker 先 attach、内容 marker 后 visible）。规则：1) 顺序断言：依次 waitForSelector（before 用 attached——DOM 存在即可（不可见）；after 用 visible——渲染完成）；2) 超时即 fail（附失败点：哪个 marker 等不到——诊断信息）；3) 全部通过返回耗时 `Array<{ marker, ms }>`（各阶段耗时——流式时序的可观测）；4) 总超时：各 marker 独立 timeout（不是共享——每个阶段 5s）。边界：events 空（返回 []）；marker 从未出现（超时报错含 marker 名）。

```javascript
async function waitForSemantics(page, events = [], { timeout = 5000 } = {}) {
  const timings = [];
  const t0 = Date.now();
  for (const ev of events) {
    const start = Date.now();
    try {
      await page.waitForSelector(`[data-marker="${ev.marker}"]`, {
        state: ev.phase === 'before' ? 'attached' : 'visible',
        timeout,
      });
      timings.push({ marker: ev.marker, ms: Date.now() - start });
    } catch (e) {
      throw new Error(`语义等待失败: marker=${ev.marker} phase=${ev.phase}` +
        ` (等了 ${timeout}ms)。已完成: ${timings.map(t => t.marker).join(',') || '无'}`);
    }
  }
  return { timings, total: Date.now() - t0 };
}

// 自测（mock page）
function mockPage(markers, delays) {
  const appeared = new Set();
  return {
    async waitForSelector(sel, opts) {
      const marker = sel.match(/data-marker="([^"]+)"/)[1];
      const delay = delays[marker] ?? Infinity;
      if (delay > (opts.timeout ?? 5000)) throw new Error('timeout');
      await sleep(Math.min(delay, 10));
      appeared.add(marker);
    },
  };
}
// 骨架 10ms → 内容 50ms
let r = await waitForSemantics(mockPage(null, { skeleton: 10, content: 50 }), [
  { marker: 'skeleton', phase: 'before' },
  { marker: 'content', phase: 'after' },
]);
console.log(r.timings.length === 2 && r.total >= 50);
console.log(r.timings[0].ms <= 15 && r.timings[1].ms >= 40);

// 超时诊断
let e;
try { await waitForSemantics(mockPage(null, { skeleton: 99999 }), [
  { marker: 'skeleton', phase: 'before' },
  { marker: 'content', phase: 'after' },
], { timeout: 50 }); } catch (x) { e = x; }
console.log(/marker=skeleton/.test(e.message));

// 空 events
r = await waitForSemantics(mockPage(null, {}), []);
console.log(r.timings.length === 0 && r.total >= 0);
```

**评分点**：
- 顺序语义断言（before=attached/after=visible——流式时序的表达式化）；失败诊断（已完成 marker 列表——定位卡在哪阶段）
- 各阶段独立计时（流式时序可观测——RBL-540 双渲染断言的时序版）；独立超时
- 边界（空 events/超时诊断信息）；反模式对比（waitForTimeout 的 flaky）
- 能说清「SSR E2E 断言用语义标记而非网络状态——networkidle 在流式页是反模式」属于加分
