## MFW-800：笔试：手写三方合并

**答**：实现 `threeWayMerge(base, mine, theirs)`。三个对象同结构。规则：1) **三方比较**：每个字段检查 base/mine/theirs——我改了 & 他人未改 → 用我的；他人改了 & 我未改 → 用他人的；都改了 → 冲突（值不同）；都改了且值相同 → 用值（无冲突）；2) **冲突返回**：`{ merged, conflicts: string[] }`（merged 含非冲突字段合并结果 + conflicts 列出冲突字段名）；3) **嵌套对象**：递归合并（嵌套字段也做三方比较）；4) **新增/删除检测**：base 没有的字段（mine 或 theirs 新增——取新增方；双方都新增且值不同 → 冲突）；base 有的字段 mine/theirs 删除 → 删除语义。边界：三者全同（merged = base）；mine/theirs 相同（无冲突）；空对象。

```javascript
function threeWayMerge(base, mine, theirs) {
  const merged = {}; const conflicts = [];
  const keys = new Set([...Object.keys(base ?? {}),
    ...Object.keys(mine ?? {}), ...Object.keys(theirs ?? {})]);
  for (const key of keys) {
    const b = base?.[key], m = mine?.[key], t = theirs?.[key];
    const mineChanged = m !== b;
    const theirsChanged = t !== b;
    if (!mineChanged && !theirsChanged) { merged[key] = b; continue; }     // 均未改
    if (mineChanged && !theirsChanged) { merged[key] = m; continue; }      // 我改
    if (!mineChanged && theirsChanged) { merged[key] = t; continue; }      // 他人改
    // 都改了
    if (m === t) { merged[key] = m; continue; }            // 值相同：无冲突
    if (typeof m === 'object' && typeof t === 'object' &&
        m !== null && t !== null && !Array.isArray(m)) {
      // 嵌套递归合并
      const sub = threeWayMerge(b ?? {}, m, t);
      if (sub.conflicts.length) conflicts.push(`${key}.${sub.conflicts.join(`,${key}.`)}`);
      merged[key] = sub.merged;
      continue;
    }
    conflicts.push(key);                                   // 冲突
    merged[key] = m;                                       // 默认用我的（可覆盖）
  }
  return { merged, conflicts };
}

// 自测
const base = { name: 'original', age: 25, city: 'NYC', nested: { x: 1, y: 2 } };
// 我改 name，他人改 age，city 都改但值相同
let r = threeWayMerge(base,
  { name: 'my-name', age: 25, city: 'LA', nested: { x: 1, y: 2 } },
  { name: 'original', age: 30, city: 'LA', nested: { x: 1, y: 3 } });
console.log(r.merged.name === 'my-name');                // 我改 → 用我的
console.log(r.merged.age === 30);                        // 他人改 → 用他人的
console.log(r.merged.city === 'LA');                     // 都改同值 → 无冲突
console.log(r.merged.nested.y === 3);                    // 嵌套递归
console.log(r.conflicts.length === 0);                   // 无冲突

// 冲突：都改 name 且值不同
r = threeWayMerge(base,
  { name: 'alice', age: 25, city: 'NYC', nested: { x: 1, y: 2 } },
  { name: 'bob', age: 25, city: 'NYC', nested: { x: 1, y: 2 } });
console.log(r.conflicts.join(',') === 'name');
console.log(r.merged.name === 'alice');                  // 默认用我的

// 嵌套冲突
r = threeWayMerge(base,
  { name: 'original', age: 25, city: 'NYC', nested: { x: 99, y: 2 } },
  { name: 'original', age: 25, city: 'NYC', nested: { x: 88, y: 2 } });
console.log(r.conflicts.join(',') === 'nested.x');       // 嵌套冲突路径

// 新增字段
r = threeWayMerge({ name: 'x' }, { name: 'x', new1: 'mine' }, { name: 'x', new2: 'theirs' });
console.log(r.merged.new1 === 'mine' && r.merged.new2 === 'theirs');   // 独立新增合并
console.log(r.conflicts.length === 0);

// 全同
r = threeWayMerge(base, base, base);
console.log(r.conflicts.length === 0 && r.merged.name === 'original');
```

**评分点**：
- 三方比较四态：未改/我改/他人改/都改——each 的合并语义正确
- 嵌套递归合并（嵌套字段的冲突路径 `nested.x`——层级冲突定位）；新增/删除语义
- 边界（都改同值=无冲突/空对象/全同）；冲突字段默认用我的（可覆盖）
- 能说清「三方合并 = git merge 的对象版：base 作参照检测双方变更——不同字段自动合并、同字段不同值才是真冲突」属于加分
