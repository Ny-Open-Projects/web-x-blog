## MFW-611：笔试：手写段树审计器

**答**：实现 `buildSegmentTree(routes)` 与 `auditSegments(tree, { maxDepth, maxSegmentKB })`。routes: `Record<routePath, { hasLayout, hasPage, hasLoading, sizeKB }>`（如 '/shop/[id]': { hasLayout: true, hasPage: true, sizeKB: 80 }）。buildSegmentTree：按 '/' 切段构建树，每节点聚合 children 的 page/layout 计数。auditSegments：1) 孤段：node 有 layout 但（无 page 且无任何子树 page）→ 'orphan'；2) 深嵌套：节点距根的 layout 深度 > maxDepth → 'deep'；3) 重段：sizeKB > maxSegmentKB → 'heavy'。返回 findings：`{ path, issue, detail }`。边界：空 routes（空树无 findings）；根节点不算深嵌套。

```javascript
function buildSegmentTree(routes = {}) {
  const root = { segment: '', path: '', children: new Map(), hasLayout: false, hasPage: false, sizeKB: 0, depth: 0 };
  for (const [route, meta] of Object.entries(routes)) {
    let cur = root;
    const segs = route.split('/').filter(Boolean);
    segs.forEach((seg, i) => {
      if (!cur.children.has(seg))
        cur.children.set(seg, { segment: seg, path: '/' + segs.slice(0, i + 1).join('/'),
          children: new Map(), hasLayout: false, hasPage: false, sizeKB: 0, depth: i + 1 });
      cur = cur.children.get(seg);
      cur.hasLayout = cur.hasLayout || meta.hasLayout;
      cur.hasPage = cur.hasPage || meta.hasPage;
      cur.sizeKB = Math.max(cur.sizeKB, meta.sizeKB);
    });
  }
  function hasPageBelow(n) {
    if (n.hasPage) return true;
    return [...n.children.values()].some(hasPageBelow);
  }
  root._hasPageBelow = hasPageBelow(root);
  return root;
}

function auditSegments(tree, { maxDepth = 4, maxSegmentKB = 150 } = {}) {
  const findings = [];
  function walk(node) {
    for (const child of node.children.values()) {
      if (child.hasLayout && !child.hasPage &&
          ![...child.children.values()].some(c => walkHasPage(c)))
        findings.push({ path: child.path, issue: 'orphan',
          detail: '有 layout 无 page 子树——404 死区' });
      if (child.depth > maxDepth)
        findings.push({ path: child.path, issue: 'deep',
          detail: `layout 嵌套 ${child.depth} 层 > ${maxDepth}` });
      if (child.sizeKB > maxSegmentKB)
        findings.push({ path: child.path, issue: 'heavy',
          detail: `段 bundle ${child.sizeKB}KB > ${maxSegmentKB}KB` });
      walk(child);
    }
  }
  function walkHasPage(n) { return n.hasPage || [...n.children.values()].some(walkHasPage); }
  walk(tree);
  return findings;
}

// 自测
const routes = {
  '/shop': { hasLayout: true, hasPage: true, sizeKB: 80 },
  '/shop/[id]': { hasLayout: true, hasPage: true, sizeKB: 120 },
  '/shop/[id]/reviews': { hasLayout: true, hasPage: true, sizeKB: 30 },
  '/admin/ghost': { hasLayout: true, hasPage: false, sizeKB: 10 },   // 孤段
  '/dashboard/deep/l3/l4/l5': { hasLayout: true, hasPage: true, sizeKB: 200 },  // 深 + 重
};
const tree = buildSegmentTree(routes);
const findings = auditSegments(tree, { maxDepth: 4, maxSegmentKB: 150 });
console.log(findings.some(f => f.path === '/admin/ghost' && f.issue === 'orphan'));
console.log(findings.some(f => f.issue === 'deep' && f.path.includes('l5')));
console.log(findings.some(f => f.issue === 'heavy' && f.path.includes('l5')));
console.log(findings.length === 3);
// 正常路由无告警
console.log(!findings.some(f => f.path === '/shop/[id]'));
// 空树
console.log(auditSegments(buildSegmentTree({})).length === 0);
```

**评分点**：
- 段树构建正确（路径切段聚合 hasLayout/hasPage/size）；孤段判定（含子树递归 hasPage）
- 深嵌套的深度语义（layout 层数而非路径段数）；heavy 独立于深度；边界
- 能说清「段树审计把结构成本可见化：layout 深度 = payload 与水合成本——阈值是成本约束不是风格」属于加分
