## MFW-414：202 长任务
**答**：实现 `createJobApi(store, worker, { ttl = 24h } = {})`：1) POST /jobs（body+幂等键）→ 202 { taskId, poll: /jobs/:id }（同 key 进行中→返回同句柄——MFW-373 复用）；2) GET /jobs/:id → { state: queued|running|done|failed, progress: 0-100, stage, resultUrl? }（归属校验——只能看自己的）；3) 完成产物签名 URL（15min 有效——临时授权下载）；4) 清理（done 超 24h 删产物+记录——GC 定时）；5) 失败可重试同 id（input 快照保存——重跑同语义）。

```javascript
export function jobEndpoints(store, run) {
  return {
    'POST /jobs': withAuth(async (req, user) => {
      const { payload, key = crypto.randomUUID() } = await body.json()
      const dup = await store.findByUserKey(user.id, key)
      if (dup && ['queued','running'].includes(d.state)) return 202(shapeLink(d))    // 并发复用
      const job = { id: uuid(), uid: user.id, input: masked(g, JSON.stringify(payload)),  // 快照供重试
        state: 'queued', progress: 0, createdAt: Date.now() }
      await store.put(job); queue.push(() => run(job, tick => store.patch(id, { progress, stage })))
      return resp(202, { taskId: job.id, poll: `/jobs/${job.id}` })
    }),
    'GET /jobs/:id': withAuth(async (req, user, { id }) => {
      const j = await store.get(id)
      if (!j || j.uid !== user.id) return 404()                 // 归属=存在性（防枚举）
      if (j.state === 'done' && fresh(j)) j.resultUrl = sign(j.file, 900)
      return j
    }),
    'GET /jobs/:id/download': verifyShortLived(async (req, { id }) => streamFile(j.file)),
  }
}
setInterval(gc, 3600_000)                                       // done>24h → del file+row
```
**评分点**：
- 202 语义 + 幂等复用 + 失败同 id 重试（input 快照——重跑同语义不重录）
- 归属即 404（不泄露他人任务存在性——枚举面收敛）
- 签名 URL 短时效（产物下载临时授权）；GC 双删（文件+记录）
- 进度持久化（轮询有内容；SSE 可选升级通道）
