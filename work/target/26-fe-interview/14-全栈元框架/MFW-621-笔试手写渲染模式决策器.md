## MFW-621：笔试：手写渲染模式决策器

**答**：实现 `decideRenderMode(page)`。page: `{ personalization: number(0~1), cacheable: boolean, changeFreqSec: number, entityCount?: number, criticalLCP?: boolean }`. 规则（按序）：1) personalization ≥ 0.5 且 criticalLCP → 'ppr'；≥0.5 否则 'ssr'；2) !cacheable → 'ssr'；3) changeFreqSec ≤ 5 → 'ssr'；≤ 600 → 'isr'；4) entityCount > 100000 → 'isr-on-demand'；5) 其余 'ssg'。返回 `{ mode, reason }`。边界：personalization 恰 0.5（≥ 含等于——走 ppr/ssr 分支）；cacheable=false 且 personalization=1（先走个性化分支——顺序敏感）；changeFreqSec=0（实时→ssr）。

```javascript
function decideRenderMode(page) {
  const p = { entityCount: 0, criticalLCP: false, ...page };
  if (p.personalization >= 0.5)
    return { mode: p.criticalLCP ? 'ppr' : 'ssr',
      reason: `个性化 ${p.personalization}——${p.criticalLCP ? 'LCP 关键：壳静态+洞动态' : '纯动态'}` };
  if (!p.cacheable)
    return { mode: 'ssr', reason: '不可缓存' };
  if (p.changeFreqSec <= 5)
    return { mode: 'ssr', reason: `变更频率 ${p.changeFreqSec}s——实时性要求` };
  if (p.changeFreqSec <= 600)
    return { mode: 'isr', reason: `变更 ${p.changeFreqSec}s——TTL 窗口` };
  if (p.entityCount > 100000)
    return { mode: 'isr-on-demand', reason: `${p.entityCount} 页——热点预渲染+按需` };
  return { mode: 'ssg', reason: `变更 ${p.changeFreqSec}s——全静态最优` };
}

// 自测
// 首页
console.log(decideRenderMode({ personalization: 0, cacheable: true, changeFreqSec: 86400 }).mode === 'ssg');
// 商品页
console.log(decideRenderMode({ personalization: 0, cacheable: true, changeFreqSec: 60 }).mode === 'isr');
// 购物车
console.log(decideRenderMode({ personalization: 1, cacheable: false }).mode === 'ssr');
// 搜索页（LCP 关键 + 个性化）
console.log(decideRenderMode({ personalization: 0.6, cacheable: false, criticalLCP: true }).mode === 'ppr');
// 百万商品
console.log(decideRenderMode({ personalization: 0, cacheable: true, changeFreqSec: 3600,
  entityCount: 500000 }).mode === 'isr-on-demand');
// 实时
console.log(decideRenderMode({ personalization: 0, cacheable: true, changeFreqSec: 0 }).mode === 'ssr');
// 顺序敏感：个性化优先于 cacheable
console.log(decideRenderMode({ personalization: 0.8, cacheable: false }).mode === 'ssr');   // 无 LCP → ssr
// 边界：0.5 恰好
console.log(decideRenderMode({ personalization: 0.5, cacheable: true, changeFreqSec: 3600, criticalLCP: true }).mode === 'ppr');
```

**评分点**：
- 决策树顺序正确（个性化 > cacheable > 变更频率 > 规模 > 默认——成本与体验的优先级链）
- ppr 的触发条件（个性化 + LCP 关键——静态壳价值成立的条件）；isr-on-demand 的规模阈值
- 边界（0.5 恰好/实时 0s/顺序敏感）
- 能说清「渲染模式是成本与体验的架构决策——路由级 ADR 记录让模式可复审可迁移」属于加分
