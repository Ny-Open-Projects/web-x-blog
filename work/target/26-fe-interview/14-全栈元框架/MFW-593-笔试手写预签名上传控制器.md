## MFW-593：笔试：手写预签名上传控制器

**答**：实现 `createUploadController({ s3, auth, db, { maxMB, allowedTypes, now } })`。方法：1) `presign(fileName, size, mime, session)`：鉴权 → 校验类型白名单 → 大小上限 → 生成 key（UUID 防枚举 + 用户目录隔离）→ s3.presign（含 conditions：content-length-range + content-type）→ 登记 pending 上传（db.files.create status=pending）；2) `confirm(key, etag, session)`：鉴权 + pending 校验（防伪造确认）→ s3.head 校验 etag/size → 状态改 confirmed → 返回 meta；3) `pending 过期清理`（TTL：确认窗口 1h——过期的 pending 回收）。边界：未授权抛错；类型不在白名单拒绝；key 不可预测（防遍历他人文件）。

```javascript
function createUploadController({ s3, auth, db, maxMB = 100, allowedTypes = ['image/png', 'image/jpeg', 'application/pdf'], now = Date.now } = {}) {
  async function presign(fileName, size, mime, session) {
    const user = await auth(session);                    // 鉴权
    if (!user) throw new AuthError();
    if (!allowedTypes.includes(mime)) throw new Error(`类型不允许: ${mime}`);
    if (size > maxMB * 1024 * 1024) throw new Error(`超上限 ${maxMB}MB`);
    const key = `uploads/${user.id}/${crypto.randomUUID()}-${fileName}`;   // 用户隔离 + UUID 防枚举
    const url = await s3.presign(key, { expiresIn: 3600,
      conditions: [['content-length-range', 1, size], ['eq', '$Content-Type', mime]] });
    await db.files.create({ key, status: 'pending', userId: user.id,
      size, mime, createdAt: now() });
    return { url, key };
  }
  async function confirm(key, etag, session) {
    const user = await auth(session);
    if (!user) throw new AuthError();
    const rec = await db.files.find(key);
    if (!rec || rec.userId !== user.id) throw new Error('no such upload');   // 防伪造确认
    if (now() - rec.createdAt > 3600_000) throw new Error('确认窗口过期');
    const meta = await s3.head(key);
    if (meta.etag !== etag) throw new Error('integrity fail');
    await db.files.update(key, { status: 'confirmed', etag });
    return { key, size: meta.size, mime: rec.mime };
  }
  return { presign, confirm };
}

// 自测
const s3 = { store: new Map(),
  async presign(key, opts) { return `https://s3/${key}?sig=ok`; },
  async head(key) { return this.store.has(key) ? this.store.get(key) : Promise.reject(new Error('nf')); } };
const db = { files: new Map(),
  async create(rec) { db.files.set(rec.key, rec); },
  async find(key) { return db.files.get(key); },
  async update(key, patch) { Object.assign(db.files.get(key), patch); } };
const auth = async (session) => session?.userId ? { id: session.userId } : null;
const ctl = createUploadController({ s3, auth, db, maxMB: 10 });

const s = { userId: 'u1' };
const { url, key } = await ctl.presign('a.png', 2048, 'image/png', s);
console.log(url.includes(key) && key.startsWith('uploads/u1/'));
console.log(db.files.get(key).status === 'pending');

// 类型拒绝
let e; try { await ctl.presign('a.exe', 100, 'application/x-exe', s); } catch (x) { e = x; }
console.log(/类型不允许/.test(e.message));
// 大小拒绝
e = null; try { await ctl.presign('b.png', 20 * 1024 * 1024, 'image/png', s); } catch (x) { e = x; }
console.log(/超上限/.test(e.message));
// 未授权
e = null; try { await ctl.presign('c.png', 1, 'image/png', {}); } catch (x) { e = x; }
console.log(/Auth|unauth/i.test(e.message));
// 确认
s3.store.set(key, { etag: 'e123', size: 2048 });
const meta = await ctl.confirm(key, 'e123', s);
console.log(meta.size === 2048 && db.files.get(key).status === 'confirmed');
// 伪造确认（他人 key）
e = null; try { await ctl.confirm(key, 'e123', { userId: 'u2' }); } catch (x) { e = x; }
console.log(/no such upload/.test(e.message));
```

**评分点**：
- 鉴权在 presign 与 confirm 双侧（上传也是受控资源）；key 防枚举（UUID + 用户目录隔离）
- 预签名 conditions（大小/类型进签名——S3 层强制而非仅应用层）；pending 状态机（presign→confirm）
- 伪造确认防护（key 归属校验）+ etag 完整性；边界（类型/大小/未授权/过期）
- 能说清「server = 授权者而非中转：预签名直传绕开 serverless 限制并省双倍带宽成本」属于加分
