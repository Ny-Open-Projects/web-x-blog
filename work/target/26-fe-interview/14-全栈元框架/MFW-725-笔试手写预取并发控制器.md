## MFW-725：笔试：手写预取并发控制器

**答**：实现 `createPrefetchControl({ maxConcurrent, prefetchFn })`。prefetchFn: `(url, signal) => Promise<void>`. 方法：`queue(url)`（入队）、`cancel(url)`（取消）、`stats()`。规则：1) **并发上限**：同时 prefetch ≤ maxConcurrent（FIFO 队列——先入先出）；2) **取消**：cancel(url) → AbortController.abort → 从 active 和 queue 中移除；3) 重复 queue（已 active 或已 queued → 忽略——去重）；4) prefetch 完成/失败 → 释放槽位 → pump 队列下一个；5) 返回 stats（active/queued/cancelled）。边界：queue 空；cancel 不存在的 url（忽略）；maxConcurrent 0（全部排队）。

```javascript
function createPrefetchControl({ maxConcurrent = 3, prefetchFn } = {}) {
  const active = new Map();                              // url -> AbortController
  const queue = []; const cancelled = new Set();
  const stats = { active: 0, queued: 0, cancelled: 0, done: 0 };
  function pump() {
    while (active.size < maxConcurrent && queue.length) {
      const url = queue.shift();
      if (cancelled.has(url)) { cancelled.delete(url); continue; }
      const controller = new AbortController();
      active.set(url, controller);
      prefetchFn(url, controller.signal)
        .catch(() => {})                                 // prefetch 失败静默
        .finally(() => {
          active.delete(url); stats.done++;
          pump();                                        // 释放槽位 → 下一个
        });
    }
  }
  return {
    queue(url) {
      if (active.has(url) || queue.includes(url) || cancelled.has(url)) return;   // 去重
      queue.push(url);
      pump();
    },
    cancel(url) {
      const controller = active.get(url);
      if (controller) { controller.abort(); active.delete(url); }
      const qIdx = queue.indexOf(url);
      if (qIdx !== -1) queue.splice(qIdx, 1);
      cancelled.add(url); stats.cancelled++;
    },
    stats: () => ({ ...stats, active: active.size, queued: queue.length }),
  };
}

// 自测
const prefetched = []; const aborted = [];
const ctl = createPrefetchControl({
  maxConcurrent: 2,
  prefetchFn: (url, signal) => new Promise((res) => {
    prefetched.push(url);
    signal.addEventListener('abort', () => aborted.push(url));
    setTimeout(res, 30);
  }),
});
// 入队 5 个：只有 2 个 active
['/a', '/b', '/c', '/d', '/e'].forEach(u => ctl.queue(u));
console.log(ctl.stats().active === 2 && ctl.stats().queued === 3);

// 去重：重复 queue 忽略
ctl.queue('/a');
console.log(ctl.stats().queued === 3);                   // 不重复

// cancel 未开始的
ctl.cancel('/d');
console.log(ctl.stats().cancelled === 1);
await sleep(50);                                         // 等完成
console.log(prefetched.length === 4);                    // a,b,c,e（d 被取消）
console.log(aborted.length === 0);                       // d 未开始未 abort

// cancel 进行中的
const ctl2 = createPrefetchControl({ maxConcurrent: 1,
  prefetchFn: (url, signal) => new Promise((res) => {
    signal.addEventListener('abort', () => aborted.push(url));
    setTimeout(res, 50);
  }) });
ctl2.queue('/x');                                        // active
ctl2.queue('/y');                                        // queued
ctl2.cancel('/x');                                       // abort 进行中的
await sleep(10);
console.log(aborted.includes('/x'));                     // abort 传播
await sleep(60);
console.log(ctl2.stats().done === 1);                    // y 完成

// maxConcurrent=0
const zero = createPrefetchControl({ maxConcurrent: 0, prefetchFn: async () => {} });
zero.queue('/a');
console.log(zero.stats().active === 0 && zero.stats().queued === 1);
```

**评分点**：
- 并发上限正确（FIFO 队列 + pump 释放槽位）；cancel 双路径（active abort + queued 移除）
- 去重（active/queued/cancelled 三态检查）；prefetch 失败静默（不阻塞队列）
- 边界（maxConcurrent=0/不存在 cancel/abort 传播）；stats 可观测
- 能说清「长列表 prefetch 风暴的多层防线：并发限制 + 取消 + hover 意图——带宽预算管理」属于加分
