## MFW-731：笔试：手写流中错误注入器

**答**：实现 `createErrorInjector(handlers)`。handlers: `Record<path, handler>`（正常 handler）。方法：`inject(path, error: 'network'|'timeout'|'500')`, `restore(path)`。返回 MSW 兼容的 handler 函数。规则：1) inject：指定 path 的请求 → 返回错误（networkError/延迟/500——三种注入形态）；2) restore：恢复原 handler；3) 注入计数：`stats.injected[path]`（注入次数——验证注入生效）；4) **恢复验证**：restore 后请求正常返回（原 handler 执行）；5) 超时注入的实现：延迟 N 秒后返回（模拟慢——RBL-655 超时降级的测试）。边界：inject 未注册的 path（忽略）；restore 未注入的 path（忽略）；重复 inject（覆盖——后注入生效）。

```javascript
function createErrorInjector(handlers = {}) {
  const injected = new Map();                            // path -> errorType
  const stats = {};
  function handler(req) {
    const path = new URL(req.url).pathname;
    const errorType = injected.get(path);
    if (!errorType) {
      const normal = handlers[path];
      return normal ? normal(req) : new Response('Not Found', { status: 404 });
    }
    stats[path] = (stats[path] ?? 0) + 1;
    switch (errorType) {
      case 'network': return Response.error();           // 网络错误
      case 'timeout': return new Promise((res) =>
        setTimeout(() => res(new Response('timeout', { status: 504 })), 10000));   // 超时
      case '500': return new Response(JSON.stringify({ error: 'internal' }),
        { status: 500, headers: { 'content-type': 'application/json' } });
    }
  }
  return {
    handler,
    inject(path, errorType) { injected.set(path, errorType); },
    restore(path) { injected.delete(path); },
    stats: () => ({ ...stats }),
  };
}

// 自测
const injector = createErrorInjector({
  '/api/products': () => new Response(JSON.stringify([{ name: 'test' }]),
    { headers: { 'content-type': 'application/json' } }),
  '/api/reviews': () => new Response(JSON.stringify({ reviews: [] }),
    { headers: { 'content-type': 'application/json' } }),
});

// 正常请求
let res = await injector.handler(new Request('http://x/api/products'));
console.log(res.status === 200);
const data = await res.json();
console.log(data[0].name === 'test');

// 注入 network 错误
injector.inject('/api/reviews', 'network');
res = await injector.handler(new Request('http://x/api/reviews'));
console.log(res.status === 0);                           // Response.error
console.log(injector.stats()['/api/reviews'] === 1);

// 其他 path 正常（隔离验证）
res = await injector.handler(new Request('http://x/api/products'));
console.log(res.status === 200);                         // products 不受影响

// 注入 500
injector.inject('/api/products', '500');
res = await injector.handler(new Request('http://x/api/products'));
console.log(res.status === 500);

// 恢复
injector.restore('/api/products');
res = await injector.handler(new Request('http://x/api/products'));
console.log(res.status === 200);                         // 原 handler 恢复

// 未注册 path
res = await injector.handler(new Request('http://x/unknown'));
console.log(res.status === 404);
```

**评分点**：
- 三种注入形态（network/timeout/500——不同错误类型的测试）；注入计数验证
- 隔离验证（注入一个 path 其他 path 正常——错误只影响一个段）；restore 恢复
- 边界（未注册 path 404/重复 inject 覆盖）；stats 可观测
- 能说清「错误注入器 = 流中错误的 E2E 测试工具：隔离验证（一个段失败不影响其他）+ 恢复验证（reset 后正常）」属于加分
