## MFW-756：笔试：手写全链路预算门禁

**答**：实现 `fullChainGate(metrics, budget, { warnRatio })`。metrics: `{ ttfbMs, lcpMs, inpMs, cls, jsKB, cacheHitRate, regenQps }`。budget: PROMO_BUDGET 结构（RBL-755）。规则：1) 逐项比对：metrics 超预算 → finding `{ layer, metric, current, budget, ratio }`（ratio = current/budget）；2) 分级：ratio > warnRatio（默认 1.2）→ 'error'（超 20%）；ratio > 1 → 'warn'（超但 <20%）；ratio ≤ 1 → pass；3) cacheHitRate 是「越高越好」——低于 budget.hitRate → 超标（反向指标：hitRate 0.9 < 0.95 → 超标）；4) 返回 `{ findings, verdict: 'pass'|'warn'|'fail' }`（fail = 有 error；warn = 有 warn 无 error）。边界：metrics 空（pass——无数据不误报）；regenQps 无 budget（跳过）；反向指标的处理。

```javascript
function fullChainGate(metrics = {}, budget = {}, { warnRatio = 1.2 } = {}) {
  const findings = [];
  const check = (layer, metric, current, target, higher = false) => {
    if (current == null || target == null) return;
    let ratio, ok;
    if (higher) {                                          // 反向：越高越好
      ratio = +(target / current).toFixed(2);              // ratio < 1 = 超标
      ok = current >= target;
    } else {
      ratio = +(current / target).toFixed(2);
      ok = current <= target;
    }
    if (!ok) {
      const severity = ratio > warnRatio ? 'error' : 'warn';
      findings.push({ layer, metric, current, budget: target, ratio, severity });
    }
  };
  // CDN 层
  check('cdn', 'cacheHitRate', metrics.cacheHitRate, budget.cdn?.hitRate, true);
  // SSR 层
  check('ssr', 'ttfbMs', metrics.ttfbMs, budget.ssr?.ttfbMs);
  // client 层
  check('client', 'lcpMs', metrics.lcpMs, budget.client?.lcpMs);
  check('client', 'inpMs', metrics.inpMs, budget.client?.inpMs);
  check('client', 'cls', metrics.cls, budget.client?.cls);
  check('client', 'jsKB', metrics.jsKB, budget.client?.jsKB);
  // 大促特殊
  check('invalidation', 'regenQps', metrics.regenQps, budget.invalidation?.regenQpsLimit);
  const hasError = findings.some(f => f.severity === 'error');
  const hasWarn = findings.some(f => f.severity === 'warn');
  return { findings, verdict: hasError ? 'fail' : hasWarn ? 'warn' : 'pass' };
}

// 自测
const budget = {
  cdn: { hitRate: 0.95 }, ssr: { ttfbMs: 300 },
  client: { lcpMs: 2500, inpMs: 200, cls: 0.1, jsKB: 200 },
  invalidation: { regenQpsLimit: 50 },
};
// 全达标
let r = fullChainGate({ ttfbMs: 250, lcpMs: 2000, inpMs: 150, cls: 0.05,
  jsKB: 180, cacheHitRate: 0.97, regenQps: 30 }, budget);
console.log(r.verdict === 'pass' && r.findings.length === 0);

// LCP 超标 20%+
r = fullChainGate({ lcpMs: 3500, ttfbMs: 200 }, budget);
console.log(r.verdict === 'fail');
console.log(r.findings.some(f => f.metric === 'lcpMs' && f.severity === 'error' && f.ratio === 1.4));

// 轻微超标
r = fullChainGate({ lcpMs: 2600, ttfbMs: 200 }, budget);
console.log(r.verdict === 'warn');                       // 1.04 ≤ 1.2 → warn

// 反向指标：命中率低
r = fullChainGate({ cacheHitRate: 0.85, ttfbMs: 100 }, budget);
console.log(r.findings.some(f => f.metric === 'cacheHitRate' && f.severity === 'error'));
// 0.85/0.95 = 0.89 < 1 → 超标 → ratio = 0.95/0.85 = 1.12 → warn
console.log(r.findings.find(f => f.metric === 'cacheHitRate').severity === 'warn');

// 空 metrics
console.log(fullChainGate({}, budget).verdict === 'pass');
```

**评分点**：
- 逐项比对 + 分级（ratio > 1.2 error / > 1 warn / ≤ 1 pass——阈值分级）
- 反向指标处理（cacheHitRate 越高越好——ratio 计算取 target/current）；空 metrics 不误报
- 边界（全达标/超标 20%/轻微超标/反向/空）；verdict 三级
- 能说清「全链路门禁的分层预算（CDN/SSR/client）——反向指标（命中率）的正向逻辑差异」属于加分
