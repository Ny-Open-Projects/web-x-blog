## MFW-661：笔试：手写表单状态机

**答**：实现 `createFormStateMachine(actions, { onRedirect })`。actions: `{ submit: (fd) => Promise<{ ok, error?, redirect? }>, validate?: (fd) => string | null }`. 状态机：idle → validating（有 validate）→ submitting → done/failed/redirected。规则：1) submit(fd)：先 validate（失败 → failed，不调 action——client 快速反馈）；2) **pending 互斥**：submitting 中再次 submit → reject 'in-flight'（双击防 RBL-567）；3) redirect：调 onRedirect(url) 并转 redirected（后续 submit 全拒绝——终态）；4) action 抛错 → failed；5) 返回 `{ state(), submit(fd), reset() }`（reset 回 idle）。状态变化回调：`onChange(state)`。边界：validate 抛错（视为 validate 失败——不调 action）；redirect 后 reset（允许重新提交——reset 解锁终态）。

```javascript
function createFormStateMachine(actions, { onRedirect = () => {}, onChange = () => {} } = {}) {
  let state = 'idle'; let lastResult = null;
  const setState = (s) => { state = s; onChange(s); };
  return {
    state: () => state,
    reset() { lastResult = null; setState('idle'); },
    async submit(fd) {
      if (state === 'submitting') throw new Error('in-flight');   // 互斥
      if (state === 'redirected') throw new Error('redirected-terminal');
      try {
        if (actions.validate) {
          setState('validating');
          const err = actions.validate(fd);
          if (err) { lastResult = { error: err }; setState('failed'); return lastResult; }
        }
        setState('submitting');
        const result = await actions.submit(fd);
        lastResult = result;
        if (result.redirect) {
          setState('redirected');
          onRedirect(result.redirect);
          return result;
        }
        setState(result.ok ? 'done' : 'failed');
        return result;
      } catch (e) {
        lastResult = { error: e.message };
        setState('failed');
        return lastResult;
      }
    },
    get lastResult() { return lastResult; },
  };
}

// 自测
const log = [];
let redirects = 0;
const fsm = createFormStateMachine({
  validate: (fd) => fd.get('qty') > 0 ? null : 'qty 必须为正',
  submit: async (fd) => {
    log.push('submit');
    if (fd.get('redirect')) return { ok: true, redirect: '/orders' };
    return { ok: true };
  },
}, { onRedirect: () => redirects++, onChange: (s) => log.push(s) });

// validate 拦截
let r = await fsm.submit({ get: (k) => k === 'qty' ? 0 : null });
console.log(r.error === 'qty 必须为正' && fsm.state() === 'failed');
console.log(log.filter(x => x === 'submit').length === 0);   // 未调 action

// 重置后正常提交
fsm.reset();
r = await fsm.submit({ get: () => 1 });
console.log(r.ok === true && fsm.state() === 'done');
console.log(log.filter(x => x === 'submit').length === 1);

// pending 互斥
fsm.reset();
fsm.setState; // 无——直接并发
const p1 = fsm.submit({ get: () => 1 });
let e; try { await fsm.submit({ get: () => 1 }); } catch (x) { e = x; }
console.log(e.message === 'in-flight');
await p1;
console.log(fsm.state() === 'done');

// redirect 终态
fsm.reset();
const fdRedirect = { get: (k) => k === 'redirect' ? '1' : 1 };
r = await fsm.submit(fdRedirect);
console.log(fsm.state() === 'redirected' && redirects === 1);
e = null; try { await fsm.submit(fdRedirect); } catch (x) { e = x; }
console.log(e.message === 'redirected-terminal');
```

**评分点**：
- 状态机完整（idle→validating→submitting→done/failed/redirected）；validate 前置（client 快速反馈）
- pending 互斥（in-flight 防双击）+ redirected 终态（导航后拒绝）；reset 解锁终态
- 边界（validate 抛错→failed 不调 action）；onChange 回调（UI 绑定）
- 能说清「表单提交的状态机是乐观 UI 与 server 真值的交接协议——redirect 终态与 pending 互斥是防重复提交的架构约束」属于加分
