## VIZ-474：手写一个渲染健康分（多指标加权）

**答**：
- 需求：implement renderHealth(s)。s 为 { fps, drawCalls, overdraw, texMB, firstRenderMs, gcMin }；每项阈值打分 0~1（线性区间），权重 fps 0.3/draw 0.2/overdraw 0.15/tex 0.15/first 0.1/gc 0.1；总分 0~100；输出 { score, worst, items }。
- 边界：缺失项权重重分配；fps 直接 /60 归一；score 保留整数；items 附每项得分。
- 评分点：线性归一；权重重分配；worst 选取；整数分

```javascript
function renderHealth(s) {
  const W = { fps: 0.3, drawCalls: 0.2, overdraw: 0.15,
    texMB: 0.15, firstRenderMs: 0.1, gcMin: 0.1 };
  const norm = {
    fps: v => Math.min(1, v / 60),
    drawCalls: v => Math.max(0, 1 - v / 2000),
    overdraw: v => Math.max(0, 1 - v / 4),
    texMB: v => Math.max(0, 1 - v / 200),
    firstRenderMs: v => Math.max(0, 1 - v / 2000),
    gcMin: v => Math.max(0, 1 - v / 5),
  };
  const parts = []; let wSum = 0;
  for (const [k, w] of Object.entries(W)) {
    if (s[k] == null) continue;
    parts.push({ metric: k, score: +norm[k](s[k]).toFixed(2), w });
    wSum += w;
  }
  if (!parts.length) return { score: null, worst: null, items: [] };
  const score = Math.round(parts.reduce((x, p) => x + p.score * (p.w / wSum), 0) * 100);
  return { score, worst: parts.sort((a, b) => a.score - b.score)[0].metric,
    items: parts };
}
```
