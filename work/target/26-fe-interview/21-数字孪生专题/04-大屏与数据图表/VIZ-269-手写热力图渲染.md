## VIZ-269：手写热力图渲染：Canvas ImageData 逐像素与色标映射

**答**：
- 题目：热力图——矩阵值 → 颜色 → ImageData 逐像素写入（比 fillRect 快 10 倍）。色标插值（多段色标）。
- 思路：ImageData 直接操作 RGBA 数组；色标 = 多段线性插值（蓝→绿→红）；NaN/缺值 → 透明。
- 评分点：ImageData 直写；多段色标插值；缺值透明处理。

```javascript
function renderHeatmap(ctx, matrix, colorScale, { cellW = 1, cellH = 1 } = {}) {
  const w = matrix[0].length, h = matrix.length;
  const img = ctx.createImageData(w, h);                   // 逐像素
  for (let y = 0; y < h; y++)
    for (let x = 0; x < w; x++) {
      const v = matrix[y][x];
      const i = (y * w + x) * 4;
      if (v == null || Number.isNaN(v)) { img.data[i + 3] = 0; continue; }   // 缺值透明
      const [r, g, b] = colorScale(v);                     // 色标插值
      img.data[i] = r; img.data[i + 1] = g; img.data[i + 2] = b; img.data[i + 3] = 255;
    }
  ctx.putImageData(img, 0, 0);                             // 一次写入
  // 放大：ctx.drawImage(canvas, 0, 0, w, h, 0, 0, w * cellW, h * cellH)（imageSmoothing 关）
}
function multiStopScale(stops) {                           // 多段色标 [[t, [r,g,b]]]
  return (v) => {
    for (let i = 1; i < stops.length; i++)
      if (v <= stops[i][0]) {
        const t = (v - stops[i - 1][0]) / (stops[i][0] - stops[i - 1][0]);
        return stops[i - 1][1].map((c, j) => Math.round(c + (stops[i][1][j] - c) * t));
      }
    return stops.at(-1)[1];
  };
}
```
