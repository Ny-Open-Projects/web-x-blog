## VIZ-465：手写一个视锥剔除器：包围球与包围盒双模式

**答**：
- 需求：实现 cull(objects, planes)。objects 为 [{ id, sphere: { c, r }, aabb: { min, max } }]；mode 为 sphere 用球判定（快）、mode 为 aabb 用盒判定（8 角点或 slab）；输出保留的 objects。
- 边界：planes 为 6 个 { n, d }；部分相交（跨边界）保留；objects 为空返回空；mode 非法按 sphere。
- 评分点：球/盒双模式；部分相交保留；平面判定；mode 兜底

```javascript
function cull(objects, planes, mode = 'sphere') {
  const inPlane = (p, n, d, r = 0) =>
    n[0]*p[0] + n[1]*p[1] + n[2]*p[2] + d >= -r;
  return objects.filter(o => {
    if (mode === 'aabb' && o.aabb) {
      const { min, max } = o.aabb;
      return planes.every(pl =>
        inPlane([max[0], max[1], max[2]], pl.n, pl.d, radiusOf(min, max)) ||
        inPlane([min[0], min[1], min[2]], pl.n, pl.d) ||
        (inPlane(center(min, max), pl.n, pl.d,
          radiusOf(min, max))));
    }
    return planes.every(pl => inPlane(o.sphere.c, pl.n, pl.d, o.sphere.r));
  });
}
function center(min, max) { return min.map((v, i) => (v + max[i]) / 2); }
function radiusOf(min, max) { return Math.hypot(...max.map((v, i) => (v - min[i]) / 2)); }
```
