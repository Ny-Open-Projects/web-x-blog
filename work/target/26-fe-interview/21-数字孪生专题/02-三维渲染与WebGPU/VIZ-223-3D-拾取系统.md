## VIZ-223：3D 拾取系统：射线求交与最近命中排序

**答**：
- 题目：实现 3D 场景点选：屏幕坐标 → 射线 → 与场景对象包围盒求交 → 最近命中。要求先包围盒粗筛再三角形精筛。
- 思路：unproject 生成射线（near/far 两点）；粗筛：射线与 AABB（slab 法）；精筛：射线-三角形（Möller–Trumbore）。
- 评分点：unproject 推导；slab 法求交；粗筛+精筛两阶段。

```javascript
function rayFromScreen(camera, x, y, canvas) {           // 屏幕坐标 → 射线
  const ndcX = (x / canvas.width) * 2 - 1, ndcY = 1 - (y / canvas.height) * 2;
  const origin = unproject(camera, ndcX, ndcY, 0);        // near 面
  const target = unproject(camera, ndcX, ndcY, 1);        // far 面
  const dir = normalize(sub(target, origin));
  return { origin, dir };
}
function rayAABB(ray, box) {                              // slab 法
  let tmin = 0, tmax = Infinity;
  for (const axis of ['x', 'y', 'z']) {
    const invD = 1 / ray.dir[axis];
    let t1 = (box.min[axis] - ray.origin[axis]) * invD;
    let t2 = (box.max[axis] - ray.origin[axis]) * invD;
    if (t1 > t2) [t1, t2] = [t2, t1];
    tmin = Math.max(tmin, t1); tmax = Math.min(tmax, t2);
    if (tmin > tmax) return null;                          // 无交
  }
  return tmin;                                            // 最近进入距离
}
function pick(scene, ray) {
  const hits = scene.objects
    .map(o => ({ o, t: rayAABB(ray, o.aabb) }))
    .filter(h => h.t !== null)
    .sort((a, b) => a.t - b.t);                            // 最近优先
  for (const h of hits) {                                  // 粗筛序精筛
    const tri = rayTrianglePrecise(ray, h.o);
    if (tri) return tri;
  }
  return null;
}
```
