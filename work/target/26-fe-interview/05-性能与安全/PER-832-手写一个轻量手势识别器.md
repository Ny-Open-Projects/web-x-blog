## PER-832：手写一个轻量手势识别器：tap、longpress、swipe 的识别与手势竞争仲裁

**答**：
- 需求：在元素上实现三种手势：tap（位移 < 10px 且 < 250ms 松开）、longpress（按住 >= 500ms 且未超位移）、swipe（位移 > 30px，输出方向 up/down/left/right）。同一触摸序列只触发一种（竞争仲裁：longpress 触发后 tap 作废；swipe 触发后 longpress 取消）。
- 边界：多点触控只取第一指；touchcancel 等同 touchend 但不触发 tap；快速连续 tap 需正确复位状态。
- 评分点：竞争仲裁（一种手势独占）；longpress 计时器在位移超标时清除；touchcancel 处理。

```javascript
function gestureRecognizer(el, { tapTh = 10, lpMs = 500, swipeTh = 30 } = {}) {
  let sx, sy, st, timer, fired = false;
  const reset = () => { clearTimeout(timer); fired = false; };
  el.addEventListener('touchstart', e => {
    if (e.touches.length > 1) return reset();              // 多指只取第一指
    sx = e.touches[0].clientX; sy = e.touches[0].clientY; st = Date.now(); fired = false;
    timer = setTimeout(() => {                             // longpress 候选
      if (fired) return;
      fired = true; el.dispatchEvent(new CustomEvent('longpress'));
    }, lpMs);
  }, { passive: true });
  el.addEventListener('touchmove', e => {
    const dx = e.touches[0].clientX - sx, dy = e.touches[0].clientY - sy;
    if (Math.hypot(dx, dy) > tapTh) clearTimeout(timer);   // 位移超标：longpress 取消
    if (Math.hypot(dx, dy) > swipeTh && !fired) {
      fired = true;
      const dir = Math.abs(dx) > Math.abs(dy)
        ? (dx > 0 ? 'right' : 'left') : (dy > 0 ? 'down' : 'up');
      el.dispatchEvent(new CustomEvent('swipe', { detail: { dir, dx, dy } }));
    }
  }, { passive: true });
  const end = e => {
    if (fired) return reset();                             // 已触发他手势：tap 作废
    const dt = Date.now() - st;
    const dx = e.changedTouches[0].clientX - sx, dy = e.changedTouches[0].clientY - sy;
    if (Math.hypot(dx, dy) < tapTh && dt < 250)
      el.dispatchEvent(new CustomEvent('tap'));
    reset();
  };
  el.addEventListener('touchend', end, { passive: true });
  el.addEventListener('touchcancel', reset, { passive: true });
}
```
