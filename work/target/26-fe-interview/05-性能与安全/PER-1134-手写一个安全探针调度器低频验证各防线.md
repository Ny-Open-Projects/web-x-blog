## PER-1134：手写一个安全探针调度器：低频验证各防线

**答**：
- 需求：实现 probeScheduler(probes, opts)。probes 为 [{ id, fn, intervalMs }]；opts: { jitter=0.1, onResult }；按各自 interval 调度（加随机抖动防同步）；上一次未结束则跳过本次（防堆积）；连续 fail >= 3 触发 onAlert 并暂停该探针 10 分钟；stop() 停止全部。
- 边界：fn 返回 { pass, detail }；抛错视为 fail；调度精度用 setTimeout 链而非 setInterval；stop 后 pending 回调不再执行。
- 评分点：抖动调度；防堆积；连续失败暂停；stop 清理

```javascript
function probeScheduler(probes, { jitter = 0.1, onResult, onAlert } = {}) {
  const timers = new Map(); const stats = new Map();
  function run(p) {
    const st = stats.get(p.id) ?? { fails: 0, running: false };
    if (st.running) { schedule(p); return; }               // 防堆积
    st.running = true;
    Promise.resolve().then(p.fn)
      .then(r => {
        st.fails = r.pass ? 0 : st.fails + 1;
        onResult?.({ id: p.id, ...r, ts: Date.now() });
        if (st.fails >= 3) {
          st.fails = 0; st.pausedUntil = Date.now() + 10 * 60_000;
          onAlert?.({ id: p.id, action: 'paused-10min' });
        }
      })
      .catch(e => { st.fails++; onResult?.({ id: p.id, pass: false, error: e.message }); })
      .finally(() => { st.running = false; schedule(p); });
    stats.set(p.id, st);
  }
  function schedule(p) {
    const st = stats.get(p.id);
    const wait = p.intervalMs * (1 + Math.random() * jitter);
    const t = setTimeout(() => {
      if (st.pausedUntil && Date.now() < st.pausedUntil) { schedule(p); return; }
      run(p);
    }, wait);
    timers.set(p.id, t);
  }
  probes.forEach(schedule);
  return { stop() { timers.forEach(clearTimeout); timers.clear(); } };
}
```
