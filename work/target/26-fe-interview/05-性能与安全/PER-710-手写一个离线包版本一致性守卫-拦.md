## PER-710：手写一个离线包版本一致性守卫：拦截层校验 buildId + 错配样本统计 + 兜底真网

**答**：
- 实现：拦截函数收到请求 -> 取 HTML 的 buildId（启动参数注入）与当前包 buildId 对比 -> 一致才查本地映射 -> 未命中/错配返回 null 走真网；错配事件计数上报。
- 要点：错配统计分类型（html新包旧/包新html旧）；兜底要防循环（真网失败降级旧包可用版本而不是白屏）；统计进发版看板（错配率 > 0.5% 告警）。
- 边界：buildId 缺失（老版本 HTML）默认放行真网（保守）；静态公共资源（不带 buildId）按文件 hash 匹配。
- 评分点：错配分型与兜底防循环。

```javascript

function offlineGuard(req, ctx) {                               // ctx: {htmlBuildId, pkg}
  const mismatches = ctx.mismatches ??= { htmlNewPkgOld: 0, pkgNewHtmlOld: 0 };
  // 公共资源按 hash 匹配，不比 buildId
  if (!req.path.includes('/common/')) {
    if (ctx.htmlBuildId && ctx.pkg.buildId !== ctx.htmlBuildId) {
      const newer = cmpVersion(ctx.pkg.buildId, ctx.htmlBuildId) > 0;
      mismatches[newer ? 'pkgNewHtmlOld' : 'htmlNewPkgOld']++;
      if (mismatches.pkgNewHtmlOld + mismatches.htmlNewPkgOld > 50)  // 错配率告警阈值
        report('offline-version-mismatch', mismatches);
      return null;                                              // 放行真网
    }
  }
  const local = ctx.pkg.files[req.path];
  if (local) return serveLocal(local);
  // 未命中：真网兜底，失败降级旧包（防循环，真网失败不重试离线同版本）
  return fallbackNet(req, { onFail: () => serveOlderPkg(req, ctx) });
}
function cmpVersion(a, b) { const [x, y] = [a.split('.'), b.split('.')];
  for (let i = 0; i < 3; i++) if ((+x[i] || 0) !== (+y[i] || 0)) return (+x[i] || 0) - (+y[i] || 0);
  return 0; }
```
