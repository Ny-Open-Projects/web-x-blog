## PER-876：手写一个白屏自检恢复器：关键节点检测、超时重载与恢复原因上报

**答**：
- 需求：实现 WhiteScreenGuard：watch(selectors, {timeout})——页面加载后定时检查关键节点可见性；不可见持续 > timeout 触发恢复序列：① 重新拉取数据（回调 onRecover(1)）② 整页重载（location.reload，URL 带 recover 参数）③ 兜底展示错误卡片；每次动作上报原因（缺失节点清单+页面时长+内存信号）。
- 边界：recover 参数存在时跳过整页重载（防死循环，只做数据级恢复）；用户交互（有点击）后重置计时器（用户主动操作视为页面活着）；检查用 rAF/空闲时机。
- 评分点：recover 参数防循环；交互重置；原因上报完整。

```javascript
function whiteScreenGuard(selectors, { timeout = 3000, checkInterval = 500 } = {}) {
  let timer = null, lastActive = Date.now();
  const hasRecover = /[?&]recover/.test(location.search);      // 已恢复过：禁重载
  const interactReset = () => { lastActive = Date.now(); };
  ['pointerdown', 'keydown'].forEach(ev =>
    addEventListener(ev, interactReset, { passive: true }));
  function visible(sel) {
    const el = document.querySelector(sel);
    return el && el.getBoundingClientRect().width > 0;
  }
  function check() {
    const missing = selectors.filter(s => !visible(s));
    if (!missing.length || Date.now() - lastActive < timeout) return;
    const reason = { missing, pageMs: Date.now() - navStart(),
      heapMB: memoryMB(), hadRecover: hasRecover };
    report('white-screen', reason);                            // 原因上报
    if (!hasRecover && reason.pageMs > 2 * timeout) {
      location.href += (location.search ? '&' : '?') + 'recover=1';   // 二级恢复
    } else onRecoverData?.();                                  // 一级：重载数据
    stop();
  }
  timer = setInterval(check, checkInterval);
  function stop() { clearInterval(timer); }
  return { stop };
}
```
