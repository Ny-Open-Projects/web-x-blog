## PER-548：手写一个自动化交互回放器：按脚本回放点击/输入/滚动序列，输出每步交互的 INP 表现

**答**：
- 思路：用 puppeteer 按"步骤剧本"（click/input/scroll/wait）回放真实用户路径，页面内装 event timing 探针，回放完汇总每个交互 id 的 INP 值并找出最差交互。
- 要点：每步交互后要等页面稳定（固定 wait 或等 networkidle）再触发下一步，否则连续交互互相污染时长统计；给每个交互起语义名（如"加入购物车"）方便归因。
- 边界：找不到选择器要 fail 并留截图；滚动步要触发 rAF 采样掉帧。
- 评分点：剧本驱动结构、event timing 的 interactionId 关联与最差交互定位。

```javascript

const puppeteer = require('puppeteer');
const SCRIPT = [
  { act: 'click', sel: '[data-testid="search-input"]', name: '点搜索框' },
  { act: 'type', sel: '[data-testid="search-input"]', text: '耳机', name: '输入关键词' },
  { act: 'click', sel: '[data-testid="search-btn"]', name: '触发搜索' },
  { act: 'scroll', by: 800, name: '下滑列表' },
];
async function replay(url) {
  const browser = await puppeteer.launch();
  const page = await browser.newPage();
  await page.evaluateOnNewDocument(() => {
    window.__inps = [];
    new PerformanceObserver(l => l.getEntries().forEach(e =>
      window.__inps.push({ name: e.name, dur: e.duration })))
      .observe({ type: 'event', durationThreshold: 16, buffered: true });
  });
  await page.goto(url, { waitUntil: 'networkidle2' });
  for (const step of SCRIPT) {
    if (step.act === 'click') await page.click(step.sel).catch(e => { throw new Error(`${step.name} 失败: ${e.message}`); });
    if (step.act === 'type') await page.type(step.sel, step.text);
    if (step.act === 'scroll') { await page.evaluate(by => scrollBy(0, by), step.by); }
    await page.waitForTimeout(800);                     // 等稳定再下一步
  }
  const inps = await page.evaluate(() => window.__inps);
  await browser.close();
  const worst = inps.sort((a, b) => b.dur - a.dur)[0];
  return { count: inps.length, worst, inpP: Math.round(worst?.dur ?? 0), pass: (worst?.dur ?? 0) < 200 };
}
replay('https://example.com').then(console.log);
```
