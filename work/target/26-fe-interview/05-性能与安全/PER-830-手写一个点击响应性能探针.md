## PER-830：手写一个点击响应性能探针：touchstart 到 click 的延迟测量与慢响应元素检测

**答**：
- 需求：全局监听 touchstart/touchend/click，对带 data-tap 标记的元素记录：touchstart→click 延迟（ms）、期间主线程长任务（>50ms 的任务数）；输出报告：慢响应元素 TopN（延迟 > 150ms）及其长任务数。
- 边界：滚动导致的 touch 取消（touchmove 触发）不算 click 延迟样本；同一元素采样聚合（p75 + 次数）；报告按 p75 降序。
- 评分点：touchmove 取消判定；元素级聚合；p75 与长任务关联。

```javascript
function createTapProbe({ slowMs = 150 } = {}) {
  const samples = new Map();                 // sel -> [{ms, longTasks}]
  let start = null, moved = false, longTasks = 0;
  new PerformanceObserver(list => {          // 主线程长任务计数
    for (const e of list.getEntries()) if (e.duration > 50) longTasks++;
  }).observe({ entryTypes: ['longtask'] });
  document.addEventListener('touchstart', e => {
    start = performance.now(); moved = false;
    start && (start = { t: start, sel: e.target.closest('[data-tap]')?.dataset.tap });
  }, { passive: true });
  document.addEventListener('touchmove', () => { moved = true; }, { passive: true });   // 滚动取消
  document.addEventListener('click', () => {
    if (!start || moved || !start.sel) return;
    const arr = (samples.get(start.sel) ??= []);
    arr.push({ ms: performance.now() - start.t, longTasks });
    longTasks = 0;
  });
  return { report(topN = 5) {
    return [...samples].map(([sel, arr]) => ({ sel, count: arr.length,
      p75: p75(arr.map(a => a.ms)),
      avgLongTasks: +(arr.reduce((s, a) => s + a.longTasks, 0) / arr.length).toFixed(1) }))
      .filter(r => r.p75 > slowMs).sort((a, b) => b.p75 - a.p75).slice(0, topN);
  } };
}
function p75(a) { const s = [...a].sort((x, y) => x - y); return s[Math.floor(s.length * 0.75)]; }
```
