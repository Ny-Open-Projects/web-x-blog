## PER-1014：手写一个虚拟滚动渲染器（合成器友好版）

**答**：
- 需求：实现 VirtualList(container, items, rowHeight, renderRow)。利用 content-visibility 或绝对定位 + transform 平移：总高度固定撑开滚动，只渲染可视窗口 ± overscan（默认 5）行，滚动用 rAF 合帧，行 DOM 复用（池化）。
- 边界：items 动态变化时重算；rowHeight 不齐时退化为按最大高度估算并提示；容器 resize 用 ResizeObserver 重算。
- 评分点：窗口计算；DOM 池化复用；rAF 合帧；resize 重算

```javascript
function VirtualList(container, items, rowHeight, renderRow, overscan = 5) {
  const total = items.length;
  const spacer = document.createElement('div');
  spacer.style.height = total * rowHeight + 'px';
  container.style.position = 'relative';
  container.appendChild(spacer);
  const pool = new Map();                    // index -> node
  let raf = 0;
  function paint() {
    raf = 0;
    const top = container.scrollTop;
    const h = container.clientHeight;
    const start = Math.max(0, Math.floor(top / rowHeight) - overscan);
    const end = Math.min(total, Math.ceil((top + h) / rowHeight) + overscan);
    const alive = new Set();
    for (let i = start; i < end; i++) {
      alive.add(i);
      let node = pool.get(i);
      if (!node) { node = renderRow(items[i]); pool.set(i, node); }
      node.style.transform = `translateY(${i * rowHeight}px)`;
      node.style.position = 'absolute';
      if (!node.isConnected) container.appendChild(node);
    }
    for (const [i, node] of pool)
      if (!alive.has(i) && node.isConnected) node.remove();  // 池外回收
  }
  container.addEventListener('scroll', () => {
    if (!raf) raf = requestAnimationFrame(paint);          // rAF 合帧
  }, { passive: true });
  new ResizeObserver(paint).observe(container);
  paint();
}
```
