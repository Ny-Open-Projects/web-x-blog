## PER-1265：手写一个接口降级矩阵执行器

**答**：
- 需求：实现 degradeExecutor(matrix, ctx)。matrix 为 [{ api, fallback: cache/default/hide, default: any }]；请求失败时按 fallback 策略：cache 读上次成功值（store 抽象）、default 用默认值、hide 移除该数据（返回 undefined）；输出每项 { api, mode, value, degraded }。
- 边界：cache 策略无缓存时降级为 default（有 default）否则 hide；matrix 未知 api 报错；全部降级时整体标记 allDegraded。
- 评分点：三种降级；缓存兜底；成功入缓存；整体标记

```javascript
function degradeExecutor(matrix, store) {
  const results = [];
  let allDegraded = true;
  for (const m of matrix) {
    let value, degraded = false, mode = 'fresh';
    try { value = fetchSync(m.api); }                 // 同步简化，实际为 async
    catch (e) {
      degraded = true; mode = m.fallback;
      if (m.fallback === 'cache' && store.has(m.api)) value = store.get(m.api);
      else if (m.fallback === 'default') value = m.default;
      else value = undefined;                         // hide
    }
    if (value !== undefined && (mode === 'fresh' || store.has(m.api) || m.default !== undefined))
      allDegraded = false;
    if (!degraded) store.set(m.api, value);           // 成功值入缓存
    results.push({ api: m.api, mode, value, degraded });
  }
  return { results, allDegraded };
}
function fetchSync(api) { return api === '/ok' ? { ok: 1 } : (() => { throw new Error('fail'); })(); }
```
