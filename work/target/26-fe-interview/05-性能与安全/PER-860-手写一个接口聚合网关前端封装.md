## PER-860：手写一个接口聚合网关前端封装：并发小请求合批、超时独立与部分失败容忍

**答**：
- 需求：实现 gateway(requests)：并发小请求合批——同窗口（默认 16ms）内到达的请求合并为一次批量调用（后端 /batch 接口），每个请求独立超时（超时不影响同批其他请求），部分失败容忍（失败请求返回 {error}，成功的正常返回）；提供 abortAll 清空窗口并拒绝未发请求。
- 边界：批量请求自身超时后，批内所有未完成请求按各自超时判定；请求参数序列化后相同视为同一请求（去重共享结果）。
- 评分点：合批窗口；请求级独立超时；去重共享。

```javascript
function createGateway({ batchUrl, windowMs = 16 } = {}) {
  let pending = [], timer = null, inflight = [];
  function gateway(req, { timeout = 5000 } = {}) {
    return new Promise((resolve, reject) => {
      const dup = pending.find(p => JSON.stringify(p.req) === JSON.stringify(req));
      if (dup) { dup.then(resolve, reject); return; }            // 去重共享
      const p = { req, resolve, reject, timeout, timer: setTimeout(() => {
        settle(p, { error: 'timeout' }); }, timeout) };
      pending.push(p);
      if (!timer) timer = setTimeout(flush, windowMs);
    });
  }
  function settle(p, result) { clearTimeout(p.timer); result.error ? p.reject(result) : p.resolve(result); }
  async function flush() {
    timer = null;
    if (!pending.length) return;
    const batch = pending.splice(0);
    const ctrl = new AbortController();
    inflight.push(ctrl);
    try {
      const res = await fetch(batchUrl, { method: 'POST', signal: ctrl.signal,
        body: JSON.stringify(batch.map(p => p.req)) });
      const data = await res.json();                             // data: [{ok?, error?}]
      batch.forEach((p, i) => settle(p, data[i] ?? { error: 'missing-result' }));
    } catch (e) {
      batch.forEach(p => settle(p, { error: 'batch-failed:' + String(e) }));   // 部分失败容忍
    } finally { inflight = inflight.filter(c => c !== ctrl); }
  }
  gateway.abortAll = () => {                                     // 清窗口 + 中断
    clearTimeout(timer); timer = null;
    pending.splice(0).forEach(p => settle(p, { error: 'aborted' }));
    inflight.forEach(c => c.abort()); inflight = [];
  };
  return gateway;
}
```
