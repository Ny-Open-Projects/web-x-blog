## PER-793：手写一个离线包版本兼容校验器：manifest 版本矩阵校验与不匹配降级

**答**：
- 需求：输入页面版本、多个包 manifest（含 requiresPage / requiresBase 区间），输出每个包的兼容判定：ok / downgrade-online / skip-preload；基础包不兼容时所有依赖它的业务包连锁降级。
- 边界：区间语法支持 ^ 与 >=；版本号解析要处理 prerelease（如 1.2.3-beta 排在 1.2.3 之前）；无 requires 字段视为全兼容。
- 评分点：连锁降级（基础包不兼容传导到业务包）；版本区间判定正确性。

```javascript
const cmp = (a, b) => {                                   // 简化 semver 比较（含 prerelease）
  const p = v => v.split('-')[0].split('.').map(Number);
  const [x, y] = [p(a), p(b)];
  for (let i = 0; i < 3; i++) if (x[i] !== y[i]) return x[i] - y[i];
  return (a.includes('-') ? -1 : 1) - (b.includes('-') ? -1 : 1);   // beta < 正式
};
function satisfies(version, range) {
  if (!range) return true;
  if (range.startsWith('^')) { const b = range.slice(1); const n = b.split('.').map(Number);
    n[0]++; return cmp(version, b) >= 0 && cmp(version, n.map((v, i) => i ? 0 : v).join('.')) < 0; }
  if (range.startsWith('>=')) return cmp(version, range.slice(2)) >= 0;
  return false;
}
function validateBundles(pageV, manifests) {
  const baseOk = satisfies(pageV, manifests.find(m => m.id === 'base')?.requiresPage);
  return manifests.map(m => ({
    id: m.id,
    verdict: m.id === 'base' ? (baseOk ? 'ok' : 'downgrade-online')
      : !baseOk ? 'skip-preload'                             // 基础包不兼容：业务包连锁跳过
      : satisfies(pageV, m.requiresPage) && satisfies(m.requiresBase, versions.base) ? 'ok' : 'downgrade-online',
  }));
}
```
