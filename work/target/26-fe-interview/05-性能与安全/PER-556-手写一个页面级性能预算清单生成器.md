## PER-556：手写一个页面级性能预算清单生成器：根据路由表与团队归属自动生成 budget.json 并校验

**答**：
- 输入是路由表（路由 -> 团队 -> 流量占比），输出是每路由的 JS/图片/请求数预算文件，供 Lighthouse CI 与体积脚本消费；生成器还要做一致性校验（各团队预算之和不超过站点总量）。
- 要点：预算按流量占比加权分配（流量大的页面拿更多预算），再加最低保底（小流量页不低于 80KB 防止预算小到没法开发）；校验失败要给明确的冲突报告。
- 边界：路由增删要保持预算文件幂等（重跑生成器不产生无意义 diff）。
- 评分点：加权分配算法与超总量校验。

```javascript

const fs = require('fs');
const routes = [
  { path: '/home',  team: 'home',  traffic: 0.4 },
  { path: '/pay',   team: 'pay',   traffic: 0.3 },
  { path: '/order', team: 'order', traffic: 0.3 },
];
const TOTAL = { jsKB: 800, imgKB: 1500 }, MIN_JS = 80;
function genBudget(routes, total) {
  const trafficSum = routes.reduce((s, r) => s + r.traffic, 0);
  const byTeam = {};
  const budget = {};
  for (const r of routes) {
    const w = r.traffic / trafficSum;                    // 流量加权
    const jsKB = Math.max(MIN_JS, Math.round(total.jsKB * w)); // 最低保底
    budget[r.path] = { jsKB, imgKB: Math.round(total.imgKB * w), owner: r.team };
    byTeam[r.team] = (byTeam[r.team] ?? 0) + jsKB;
  }
  const allocated = Object.values(budget).reduce((s, b) => s + b.jsKB, 0);
  if (allocated > total.jsKB)                            // 保底可能撑爆总量
    throw new Error(`预算超总量: ${allocated} > ${total.jsKB}KB，需调整流量占比或总量`);
  fs.writeFileSync('budget.json', JSON.stringify(budget, null, 2));
  return budget;
}
console.log(genBudget(routes, TOTAL));
```
