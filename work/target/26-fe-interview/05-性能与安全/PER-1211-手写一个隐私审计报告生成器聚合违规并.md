## PER-1211：手写一个隐私审计报告生成器：聚合违规并输出整改单

**答**：
- 需求：实现 auditReport(scans, registry)。scans 为 [{ type: code/runtime/log, findings: [{ rule, detail, field }] }]；对每个 finding 分类：field 未注册 -> P1（若 field 级别 >= 3 则 P0）；日志类 -> 按规则严重度；输出 { violations: [{ level, rule, fix, owner }], summary }；owner 由 rule 映射（缺省 platform）。
- 边界：重复 (rule+field) 去重取严重级更高；summary 一句话含各级计数；无 findings 输出 pass。
- 评分点：分类分级；去重取高；owner 映射；人话 summary

```javascript
function auditReport(scans, registry, owners = {}) {
  const seen = new Map();
  const SEV = { 'sensitive-log': 2, 'undeclared-field': 1,
    'consent-violation': 4, 'unreviewed-vendor': 3 };
  for (const scan of scans)
    for (const f of scan.findings) {
      const key = f.rule + '|' + (f.field ?? '');
      const level = f.rule === 'undeclared-field'
        ? ((registry[f.field]?.level ?? 0) >= 3 ? 'P0' : 'P1')
        : (SEV[f.rule] >= 3 ? 'P0' : 'P1');
      const cur = seen.get(key);
      if (!cur || levelRank(level) > levelRank(cur.level))
        seen.set(key, { rule: f.rule, field: f.field ?? '', level,
          fix: fixOf(f.rule), owner: owners[f.rule] ?? 'platform' });
    }
  const violations = [...seen.values()]
    .sort((a, b) => levelRank(b.level) - levelRank(a.level));
  const p0 = violations.filter(v => v.level === 'P0').length;
  return { violations, summary: violations.length
    ? `发现 ${violations.length} 项（P0 ${p0}，P1 ${violations.length - p0}）` : 'pass' };
}
function levelRank(l) { return l === 'P0' ? 2 : 1; }
function fixOf(rule) {
  return { 'undeclared-field': '登记字段并声明级别',
    'sensitive-log': '接入脱敏管道并删除存量',
    'consent-violation': '立即停采并评估通报',
    'unreviewed-vendor': '补第三方评审' }[rule] ?? '待评估';
}
```
