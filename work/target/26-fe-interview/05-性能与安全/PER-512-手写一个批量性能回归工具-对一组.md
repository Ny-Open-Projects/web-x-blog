## PER-512：手写一个批量性能回归工具：对一组页面跑新旧两个版本，输出逐页 diff 并标记回归页

**答**：
- 核心是"A/B 双跑"：新旧版本在**同一时段、同一环境、同一采样数**下交叉采样（交替跑而非先旧后新），抵消时段性网络抖动。
- 聚合用中位数，diff 用相对百分比，回归阈值要考虑抖动底噪（如 > 8% 才判回归，避免误报）。
- 边界：页面列表要从路由表自动生成而不是手写，防漏页；单页失败不能中断整批，最后汇总失败清单。
- 评分点：交替采样与抖动容差这两处是否体现"工程防误报"意识。

```javascript

const puppeteer = require('puppeteer');

async function measureOnce(browser, url) {
  const page = await browser.newPage();
  await page.emulate({ viewport: { width: 412, height: 823 }, isMobile: true });
  await page.goto(url, { waitUntil: 'load', timeout: 30000 });
  const lcp = await page.evaluate(() => new Promise(res => {
    new PerformanceObserver(l => res(l.getEntries().at(-1).startTime))
      .observe({ type: 'largest-contentful-paint', buffered: true });
    setTimeout(() => res(-1), 15000);
  }));
  await page.close();
  return lcp;
}

async function abRegression(oldBase, newBase, pages) {
  const browser = await puppeteer.launch();
  const med = a => a.sort((x, y) => x - y)[a.length >> 1];
  const report = [];
  for (const p of pages) {                                   // pages 由路由表自动生成
    const o = [], n = [];
    for (let i = 0; i < 3; i++) {
      o.push(await measureOnce(browser, oldBase + p));       // 交替采样抵消时段抖动
      n.push(await measureOnce(browser, newBase + p));
    }
    const [om, nm] = [med(o.filter(v => v > 0)), med(n.filter(v => v > 0))];
    const delta = (nm - om) / om;
    report.push({ p, om, nm, delta: +(delta * 100).toFixed(1),
      regression: delta > 0.08 });                           // 8% 抖动容差防误报
  }
  await browser.close();
  return report;
}
abRegression('https://old.example.com', 'https://new.example.com',
  ['/home', '/list', '/detail']).then(r => console.table(r));
```
