## PER-813：手写一个跨端启动性能对比报告生成器：多端分段数据对齐成对比矩阵

**答**：
- 需求：输入各端的分段耗时记录（端、页面、阶段、耗时 ms），按统一阶段映射表（如 bundle-load ↔ logic-boot ↔ js-parse 都归"代码加载"）对齐，输出矩阵：行=页面，列=统一阶段，值为各端 p75；同阶段多端数据取并列展示并标注端名。
- 边界：某端缺某阶段数据输出占位符；样本 < 5 的格标 low；矩阵按"代码加载"列耗时降序。
- 评分点：映射表可配置；缺数据与低样本处理；p75 计算正确。

```javascript
const STAGE_MAP = {                     // 各端阶段 → 统一阶段
  'bundle-load': '代码加载', 'logic-boot': '代码加载', 'js-parse': '代码加载',
  'first-frame': '首帧', 'first-wxml': '首帧', 'first-paint': '首帧',
  'native-init': '引擎启动', 'shell-init': '引擎启动', 'container-warm': '引擎启动',
};
function crossPlatformMatrix(records, { minSample = 5 } = {}) {
  const cell = new Map();               // key: page|stage|platform
  for (const r of records) {
    const stage = STAGE_MAP[r.stage];
    if (!stage) continue;
    const k = `${r.page}|${stage}|${r.platform}`;
    (cell.get(k) ??= []).push(r.ms);
  }
  const stages = ['引擎启动', '代码加载', '首帧'];
  const pages = [...new Set(records.map(r => r.page))];
  return pages.map(page => ({
    page,
    row: stages.map(st => {
      const entries = Object.keys(cell).length && ['rn', 'mp', 'h5']
        .map(pf => { const arr = cell.get(`${page}|${st}|${pf}`) || [];
          return arr.length ? `${pf}:${p75(arr)}${arr.length < minSample ? '(low)' : ''}` : null; })
        .filter(Boolean);
      return entries.length ? entries.join(' / ') : '-';
    }),
  })).sort((a, b) => (parseInt(b.row[1]) || 0) - (parseInt(a.row[1]) || 0));
}
function p75(a) { const s = [...a].sort((x, y) => x - y); return s[Math.floor(s.length * 0.75)]; }
```
