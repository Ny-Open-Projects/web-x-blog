## PER-1068：手写一个并发限流调度器：按优先级与域名分组

**答**：
- 需求：实现 createScheduler(rules)。rules: { perHost: { cdn.com: 6, default: 4 } }；push(url, init, priority) 返回 Promise；同 host 并发不超过 perHost；priority=high 的先出队；active 集合请求完成触发 drain。
- 边界：未匹配 perHost 用 default；priority 缺省 normal；url 非法时立即 reject；同 host 任务在 high 队列内部 FIFO。
- 评分点：host 分组限流；优先级先出；满 host 跳过轮转；非法 url

```javascript
function createScheduler(rules = {}) {
  const q = { high: [], normal: [] };
  const active = new Map();                      // host -> Set
  function hostOf(url) {
    try { return new URL(url).host; } catch { return null; }
  }
  function limitOf(host) {
    return (rules.perHost && (rules.perHost[host] ?? rules.perHost.default)) ?? 4;
  }
  function drain() {
    const pool = q.high.length ? q.high : q.normal;
    while (pool.length) {
      const task = pool[0];
      const host = hostOf(task.url);
      if (!host) { pool.shift(); task.reject(new Error('bad url')); continue; }
      const cur = active.get(host) ?? new Set();
      if (cur.size >= limitOf(host)) {
        // 该 host 已满，尝试其他队列里的任务（跳过）
        if (pool === q.normal) break;
        const idx = pool.findIndex(t => (active.get(hostOf(t.url)) ?? new Set()).size < limitOf(hostOf(t.url)));
        if (idx < 0) break;
        pool.push(pool.splice(idx, 1)[0]);       // 轮转跳过满 host
        continue;
      }
      pool.shift(); cur.add(task.url); active.set(host, cur);
      fetch(task.url, task.init).then(task.resolve, task.reject)
        .finally(() => { cur.delete(task.url); drain(); });
      if (pool === q.high && !q.high.length) break;
    }
  }
  return { push(url, init, priority = 'normal') {
    return new Promise((resolve, reject) => {
      q[priority === 'high' ? 'high' : 'normal'].push({ url, init, resolve, reject });
      drain();
    }); } };
}
```
