**场景**：轮播/卡片手势：pointerdown/up 序列判定——swipe（位移 > 30px 且主轴占比 > 0.6）、longpress（按住 > 600ms 且位移 < 10px）、doubletap（两次 tap 间隔 < 300ms）——阈值参数化（不同组件不同灵敏度）。

**思路**：pointerdown 记起点+计时 → pointermove 累计位移（超 swipe 阈值即判定，不等 up——滑动跟手）→ pointerup 终判（tap/doubletap/longpress）。

```javascript
function gesture(el, { swipeDist = 30, axisRatio = 0.6, longpressMs = 600, doubletapMs = 300,
                        onSwipe, onLongpress, onTap, onDoubleTap } = {}) {
  let start = null, timer = null, lastTap = 0, fired = false;
  el.addEventListener('pointerdown', (e) => {
    start = { x: e.clientX, y: e.clientY, t: Date.now() };
    fired = false;
    timer = setTimeout(() => {                            // 长按：600ms 且未滑动
      if (!fired && start) { fired = true; onLongpress?.(start); }
    }, longpressMs);
  });
  el.addEventListener('pointermove', (e) => {
    if (!start || fired) return;
    const dx = e.clientX - start.x, dy = e.clientY - start.y;
    if (Math.hypot(dx, dy) > swipeDist) {                 // 超阈值：立即判定 swipe（跟手）
      clearTimeout(timer); fired = true;
      const horizontal = Math.abs(dx) > Math.abs(dy) * (axisRatio / (1 - axisRatio));
      onSwipe?.(horizontal ? (dx > 0 ? 'right' : 'left') : (dy > 0 ? 'down' : 'up'), { dx, dy });
    }
  });
  el.addEventListener('pointerup', (e) => {
    clearTimeout(timer);
    if (!start || fired) { start = null; return; }
    const now = Date.now();
    if (now - lastTap < doubletapMs) { onDoubleTap?.(e); lastTap = 0; }   // 双击：间隔窗口
    else { lastTap = now; setTimeout(() => { if (lastTap === now) onTap?.(e); }, doubletapMs); }   // 单击延迟判定
    start = null;
  });
  for (const ev of ['pointercancel']) el.addEventListener(ev, () => { clearTimeout(timer); start = null; });
}
// swipe 跟手判定：move 阶段超阈值立即触发（不等 up）—— 滑动动画与手指同步
// 主轴占比：|dx| > |dy|×ratio —— 斜滑不误判方向
// 单击延迟判定：doubletap 窗口内等第二击（300ms 单击延迟是双击识别的代价）
// pointercancel 清理（来电/手势抢占——HW-156 同款纪律）
```

**评分点**：
- 四种手势判定阈值参数化（swipe/longpress/doubletap/tap）——本题核心。
- swipe 跟手（move 阶段立即判定）。
- 主轴占比防斜滑误判。
- 单击延迟判定的代价说明 + pointercancel 清理。
