## HW-342：手写一个场景图：节点树、变换矩阵与命中测试

**答**：
- 需求：实现 SceneGraph：节点 {id, x, y, scale, rotation, children}；worldMatrix(node) 沿父链累积变换矩阵；hitTest(x, y) 从根 DFS，把点逆变换到节点局部坐标判断是否在包围盒内（取最上层命中）；z 序按 children 顺序。
- 边界：逆变换计算（旋转+缩放+平移复合）；命中测试顺序与渲染顺序一致（后渲染的在上层优先命中）；缓存世界矩阵（节点未变时复用）。
- 评分点：矩阵累积与逆变换；DFS 逆序命中；矩阵缓存。

```javascript
class SceneNode {
  constructor(props = {}) { Object.assign(this, { x: 0, y: 0, scale: 1, rotation: 0, children: [] }, props); }
  add(child) { child.parent = this; this.children.push(child); }
}
function worldMatrix(node) {                               // 沿父链累积
  if (node.__wm && !node.__dirty) return node.__wm;
  const cos = Math.cos(node.rotation), sin = Math.sin(node.rotation);
  const m = { a: node.scale * cos, b: node.scale * sin, c: -node.scale * sin,
    d: node.scale * cos, e: node.x, f: node.y };
  if (node.parent) {
    const p = worldMatrix(node.parent);
    const r = { a: p.a * m.a + p.c * m.b, b: p.b * m.a + p.d * m.b,
      c: p.a * m.c + p.c * m.d, d: p.b * m.c + p.d * m.d,
      e: p.a * m.e + p.c * m.f + p.e, f: p.b * m.e + p.d * m.f + p.f };
    node.__wm = r;
  } else node.__wm = m;
  node.__dirty = false;
  return node.__wm;
}
function hitTest(root, px, py) {                           // 逆序 DFS：上层优先
  const stack = [root];
  const hits = [];
  while (stack.length) {
    const n = stack.pop();
    const m = worldMatrix(n);
    const det = m.a * m.d - m.b * m.c;
    const lx = ((px - m.e) * m.d - (py - m.f) * m.c) / det;   // 逆变换到局部
    const ly = (-(px - m.e) * m.b + (py - m.f) * m.a) / det;
    if (lx >= 0 && ly >= 0 && lx <= n.w && ly <= n.h) hits.push(n);
    [...n.children].reverse().forEach(c => stack.push(c));
  }
  return hits.at(-1) ?? null;                              // 最后渲染的优先
}
function invalidate(node) { node.__dirty = true;
  node.children?.forEach(c => invalidate(c)); }
```
