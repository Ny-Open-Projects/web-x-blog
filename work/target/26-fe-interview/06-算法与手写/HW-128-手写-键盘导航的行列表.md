**答**：可键盘导航列表：roving tabindex（列表容器 tabbable，行内 tabindex=-1，焦点程序化移动）；↑↓ 移动、Home/End 到首尾、PageUp/Down 翻页（10 行）；滚动跟随（focus 行滚入可视区）；aria 语义（role=listbox/option + aria-selected）。

```jsx
function KeyboardList({ items, selected, onSelect }) {
  const ref = useRef(null);
  const focusRow = (idx) => {
    const clamped = Math.max(0, Math.min(items.length - 1, idx));
    const el = ref.current?.children[clamped];
    el?.focus();                                                     // roving tabindex：程序化聚焦
    el?.scrollIntoView({ block: 'nearest' });                        // 滚动跟随（nearest 不跳动）
    onSelect(clamped);
  };
  function onKeyDown(e) {
    const cur = items.findIndex((_, i) => ref.current.children[i] === document.activeElement);
    const step = { ArrowDown: 1, ArrowUp: -1, PageDown: 10, PageUp: -10 }[e.key];
    if (step != null) { e.preventDefault(); focusRow(cur + step); }
    if (e.key === 'Home') { e.preventDefault(); focusRow(0); }
    if (e.key === 'End') { e.preventDefault(); focusRow(items.length - 1); }
  }
  return (
    <ul ref={ref} role="listbox" onKeyDown={onKeyDown} tabIndex={0}
        aria-label="商品列表" aria-activedescendant={`row-${selected}`}>
      {items.map((item, i) => (
        <li key={item.id} id={`row-${i}`} role="option"
            aria-selected={i === selected} tabIndex={-1}
            onClick={() => focusRow(i)}>
          {item.name}
        </li>
      ))}
    </ul>
  );
}
// roving tabindex：容器 tabIndex=0 一次 Tab 进入，行内 -1 由程序导航（WAI-ARIA 规范做法）
// scrollIntoView({block:'nearest'})：跟随但不强制居中（PageUp/Down 不跳屏）
```

**评分点**：
- roving tabindex 模式（容器可 Tab、行 -1 程序化聚焦）——规范做法是本题核心。
- 全键盘动作齐备（↑↓/Home/End/PageUp/Down）+ preventDefault 防页面滚动。
- scrollIntoView block:nearest（跟随不跳屏）；aria listbox/option/selected 语义。
- 虚拟列表下焦点管理需结合渲染窗口（能提一句加分）。
