**场景**："iphon" → 纠错 "iphone"：全表编辑距离 O(n·len)（5 万词）太慢——首字符索引 + 长度窗口剪枝：候选 = 同首字母 ± 长度差 ≤ 2 的词，编辑距离计算量缩到几百。

**思路**：倒排（首字母 + 长度桶）→ 候选集 → 编辑距离（DP 带提前终止——距离超 k 剪枝）→ Top-3。

```javascript
function suggest(query, index, { maxDist = 2, topK = 3 } = {}) {
  const candidates = new Set();
  for (let len = query.length - maxDist; len <= query.length + maxDist; len++) {
    for (const w of index.byLength.get(len) ?? []) candidates.add(w);   // 长度窗口剪枝
  }
  const scored = [];
  for (const w of candidates) {
    const d = boundedEditDistance(query, w, maxDist);      // DP + 提前终止
    if (d <= maxDist) scored.push({ word: w, d });
  }
  return scored.sort((a, b) => a.d - b.d).slice(0, topK);
}
function boundedEditDistance(a, b, k) {                   // 带阈值的 Levenshtein（超 k 提前终止）
  const dp = Array.from({ length: a.length + 1 }, () => new Uint16Array(b.length + 1));
  for (let j = 0; j <= b.length; j++) dp[0][j] = j;
  for (let i = 1; i <= a.length; i++) {
    let rowMin = dp[i][0] = i;
    for (let j = 1; j <= b.length; j++) {
      dp[i][j] = Math.min(dp[i - 1][j] + 1, dp[i][j - 1] + 1,
        dp[i - 1][j - 1] + (a[i - 1] === b[j - 1] ? 0 : 1));
      rowMin = Math.min(rowMin, dp[i][j]);
    }
    if (rowMin > k) return k + 1;                         // 整行最小值 > k：提前终止（剪枝）
  }
  return dp[a.length][b.length];
}
// 剪枝两级：长度窗口（±2）+ 行最小值提前终止 —— 5 万词 → 数百候选 → O(候选×len²)
// 正确性：编辑距离 ≥ |len(a)-len(b)|（插入删除至少贡献长度差）—— 长度窗口不漏解
```

**评分点**：
- 两级剪枝（长度窗口 + 行最小值提前终止）——本题核心。
- 剪枝正确性证明（编辑距离 ≥ 长度差）。
- 与 BK-树的对比（实现简单 vs 查询更优）。
- Top-3 按距离排序。
