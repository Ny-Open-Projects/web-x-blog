## HW-316：手写一个任务编排引擎：DAG 依赖调度与失败传播

**答**：
- 需求：实现 DAGRunner：任务 [{id, run, deps: [id]}]；runAll：入度为 0 的先跑，完成解锁下游（可并行）；任务失败传播（下游跳过标 skipped，其余分支不受影响）；输出每任务状态与耗时。
- 边界：环检测（拓扑排序失败）；失败传播只影响直接依赖链；并发不限（每批入度 0 全跑）；重试策略可挂（每任务 retries）。
- 评分点：拓扑调度；失败传播语义；并行批次。

```javascript
async function dagRun(tasks, { retries = 0 } = {}) {
  const map = new Map(tasks.map(t => [t.id, t]));
  const indeg = new Map(tasks.map(t => [t.id, t.deps.length]));
  const dependents = new Map();
  tasks.forEach(t => t.deps.forEach(d =>
    dependents.set(d, [...(dependents.get(d) ?? []), t.id])));
  if (topoFail(tasks, indeg)) throw new Error('dag-cycle');
  const state = new Map();                                  // pending/running/done/failed/skipped
  tasks.forEach(t => state.set(t.id, 'pending'));
  const ready = tasks.filter(t => !indeg.get(t.id)).map(t => t.id);
  const waiters = [];
  function runTask(id) {
    const t = map.get(id);
    state.set(id, 'running');
    const t0 = performance.now();
    return attempt(t.run, retries).then(r => {
      state.set(id, 'done');
      results.set(id, { status: 'done', ms: performance.now() - t0, value: r });
      (dependents.get(id) ?? []).forEach(dec => {
        indeg.set(dec, indeg.get(dec) - 1);
        if (indeg.get(dec) === 0 && state.get(dec) === 'pending') ready.push(dec);
      });
    }).catch(e => {
      state.set(id, 'failed');
      results.set(id, { status: 'failed', error: String(e) });
      propagateSkip(id);                                    // 失败传播下游
    });
  }
  function propagateSkip(id) {
    (dependents.get(id) ?? []).forEach(d => {
      if (state.get(d) === 'pending') { state.set(d, 'skipped'); results.set(d, { status: 'skipped' }); propagateSkip(d); }
    });
  }
  const results = new Map();
  const loop = () => { const batch = ready.splice(0);
    return Promise.all(batch.map(runTask)).then(loop); };
  await loop();
  return Object.fromEntries(results);
}
function attempt(run, n) { let i = 0; const go = () => run().catch(e => i++ < n ? go() : Promise.reject(e)); return go(); }
function topoFail(tasks, indeg) { const c = new Map(indeg);
  const q = tasks.filter(t => !c.get(t.id)); let n = 0;
  while (q.length) { const t = q.pop(); n++;
    tasks.forEach(x => x.deps.includes(t.id) && c.set(x.id, c.get(x.id) - 1));
    tasks.forEach(x => c.get(x.id) === 0 && !q.includes(x.id) && q.push(x)); }
  return n !== tasks.length; }
```
