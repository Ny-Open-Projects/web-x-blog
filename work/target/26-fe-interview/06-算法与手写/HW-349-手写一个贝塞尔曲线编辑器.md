## HW-349：手写一个贝塞尔曲线编辑器：控制点拖拽与曲线采样

**答**：
- 需求：实现 BezierEditor：三次贝塞尔（P0 P1 P2 P3），控制点可拖拽；曲线渲染按采样点折线（自适应采样：曲率大处密）；路径测量（长度近似，采样累计）；控制柄渲染（P1P0、P2P3 连线）。
- 边界：采样密度按曲率自适应（平坦段粗采样）；控制点拖拽实时重采样；锚点与控制柄区分（锚点可拖位置，柄只调方向）。
- 评分点：曲率自适应采样；长度近似；柄连线渲染。

```javascript
function cubicPoint(p, t) {                                // 三次贝塞尔
  const u = 1 - t;
  return {
    x: u ** 3 * p.p0.x + 3 * u * u * t * p.p1.x + 3 * u * t * t * p.p2.x + t ** 3 * p.p3.x,
    y: u ** 3 * p.p0.y + 3 * u * u * t * p.p1.y + 3 * u * t * t * p.p2.y + t ** 3 * p.p3.y,
  };
}
function sampleAdaptive(p, { tol = 0.5 } = {}) {           // 曲率自适应采样
  const out = [];
  let t = 0, prev = cubicPoint(p, 0);
  out.push(prev);
  let step = 0.05;
  while (t < 1) {
    t = Math.min(1, t + step);
    const pt = cubicPoint(p, t);
    const d = Math.hypot(pt.x - prev.x, pt.y - prev.y);
    if (d > tol * 4 && step > 0.005) { t -= step; step /= 2; continue; }   // 曲率大：加密
    if (d < tol && step < 0.1) step *= 2;                  // 平坦：放粗
    out.push(pt); prev = pt;
  }
  return out;
}
function bezierLength(p) {                                  // 长度近似（累计）
  return sampleAdaptive(p).reduce((s, pt, i, arr) =>
    i ? s + Math.hypot(pt.x - arr[i - 1].x, pt.y - arr[i - 1].y) : 0, 0);
}
class BezierEditor {
  drag(handle, pos) {                                       // 控制点拖拽
    Object.assign(this.curve[handle], pos);
    this.samples = sampleAdaptive(this.curve);              // 实时重采样
  }
  render(ctx) {                                             // 柄连线
    const c = this.curve;
    ctx.beginPath();
    ctx.moveTo(c.p0.x, c.p0.y); ctx.lineTo(c.p1.x, c.p1.y); ctx.stroke();   // P0-P1
    ctx.beginPath(); ctx.moveTo(c.p2.x, c.p2.y); ctx.lineTo(c.p3.x, c.p3.y); ctx.stroke();
    ctx.beginPath();
    ctx.moveTo(this.samples[0].x, this.samples[0].y);
    this.samples.forEach(pt => ctx.lineTo(pt.x, pt.y));
    ctx.stroke();
  }
}
```
