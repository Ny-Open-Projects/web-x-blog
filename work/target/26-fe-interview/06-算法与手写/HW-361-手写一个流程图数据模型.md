## HW-361：手写一个流程图数据模型：节点、边、端口与校验

**答**：
- 需求：实现 FlowModel：节点 {id, type, ports: [{id, dir: in|out, type}]}；边 {id, from: {node, port}, to: {node, port}}；校验：端口方向匹配（out→in）、类型兼容、无环（拓扑检测）、无重复边；addNode/addEdge 前置校验。
- 边界：端口类型系统（string/number/any，any 兼容一切）；删除节点级联删边；环检测允许用户强制创建（标记 cycle 警告）。
- 评分点：端口方向/类型校验；级联删除；环检测可配置。

```javascript
class FlowModel {
  constructor({ allowCycle = false } = {}) { this.nodes = new Map(); this.edges = new Map();
    this.allowCycle = allowCycle; this.listeners = new Set(); }
  addNode(node) { this.nodes.set(node.id, node); this.emit(); return node.id; }
  addEdge(fromNode, fromPort, toNode, toPort) {
    const fn = this.nodes.get(fromNode), tn = this.nodes.get(toNode);
    const fp = fn?.ports.find(p => p.id === fromPort), tp = tn?.ports.find(p => p.id === toPort);
    if (!fp || !tp) return { ok: false, reason: 'port-not-found' };
    if (fp.dir !== 'out' || tp.dir !== 'in') return { ok: false, reason: 'direction-mismatch' };   // 方向校验
    if (!typeCompat(fp.type, tp.type)) return { ok: false, reason: 'type-incompatible' };           // 类型校验
    const dup = [...this.edges.values()].some(e =>
      e.from.node === fromNode && e.to.node === toNode);
    if (dup) return { ok: false, reason: 'duplicate-edge' };
    const edge = { id: genId(), from: { node: fromNode, port: fromPort }, to: { node: toNode, port: toPort } };
    this.edges.set(edge.id, edge);
    if (!this.allowCycle && this.hasCycle()) {               // 环检测
      this.edges.delete(edge.id);
      return { ok: false, reason: 'cycle' };
    }
    this.emit(); return { ok: true, id: edge.id };
  }
  removeNode(id) {                                          // 级联删边
    this.nodes.delete(id);
    for (const [eid, e] of [...this.edges])
      if (e.from.node === id || e.to.node === id) this.edges.delete(eid);
    this.emit();
  }
  hasCycle() {                                              // 拓扑环检测
    const indeg = new Map([...this.nodes.keys()].map(k => [k, 0]));
    for (const e of this.edges.values()) indeg.set(e.to.node, indeg.get(e.to.node) + 1);
    const q = [...indeg].filter(([, d]) => d === 0).map(([k]) => k);
    let n = 0;
    while (q.length) { const cur = q.pop(); n++;
      for (const e of this.edges.values())
        if (e.from.node === cur && --indeg.get(e.to.node) === 0) q.push(e.to.node); }
    return n !== this.nodes.size;
  }
  emit() { this.listeners.forEach(fn => fn(this)); }
}
const typeCompat = (a, b) => a === 'any' || b === 'any' || a === b;
```
