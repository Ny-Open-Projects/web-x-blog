## ALG-488：手写跳跃游戏：分片加载可行性

**答**：
- 题目：资源分片 [i] 可向前跳 maxJump[i] 步（加载完第 i 片后最多预取 maxJump 片），判断能否加载到最后一片，及最少跳数。
- 思路：贪心 O(n)：维护当前可达最远 end；逐片扩展；end < i 时卡死。最少跳数：每步覆盖范围内选延伸最远的（同 ALG-482 结构）。
- 评分点：可达性贪心；最少跳数的分层贪心；卡死判定。

```javascript
function canFinish(maxJump) {                              // 可达性
  let end = 0;
  for (let i = 0; i < maxJump.length; i++) {
    if (i > end) return false;                             // 卡死：当前片不可达
    end = Math.max(end, i + maxJump[i]);                   // 扩展可达
    if (end >= maxJump.length - 1) return true;
  }
  return true;
}
function minJumps(maxJump) {                               // 最少跳数
  let jumps = 0, curEnd = 0, farthest = 0;
  for (let i = 0; i < maxJump.length - 1; i++) {
    farthest = Math.max(farthest, i + maxJump[i]);
    if (i === curEnd) {                                    // 本层走完：跳一次
      jumps++;
      curEnd = farthest;
      if (curEnd >= maxJump.length - 1) return jumps;
      if (curEnd === i) return -1;                         // 卡死
    }
  }
  return jumps;
}
```
