## HW-323：手写一个灰度引擎：规则判定、百分比分桶与缓存

**答**：
- 需求：实现 GrayEngine：实验 {name, rules: [{type: user|channel|percent, value}] }；命中判定：user 白名单 → channel 匹配 → percent 哈希分桶（uid 哈希 % 100 < pct）；结果缓存（Map，实验配置版本失效）；上报曝光（防重复曝光去重）。
- 边界：哈希分桶稳定（同 uid 恒同组）；白名单优先级最高；配置热更新 bump 版本清缓存；曝光去重按 (uid, experiment) 会话级。
- 评分点：稳定哈希分桶；优先级；曝光去重。

```javascript
class GrayEngine {
  constructor(experiments) { this.exps = new Map(experiments.map(e => [e.name, e]));
    this.version = 0; this.cache = new Map(); this.exposed = new Set(); }
  update(exps) { exps.forEach(e => this.exps.set(e.name, e)); this.version++; this.cache.clear(); }
  hit(name, { uid, channel }) {
    const key = `${name}|${uid}`;
    const c = this.cache.get(key);
    if (c && c.v === this.version) return c.r;               // 缓存
    const exp = this.exps.get(name);
    if (!exp) return false;
    let r = false;
    for (const rule of exp.rules ?? []) {                    // 优先级：user > channel > percent
      if (rule.type === 'user') { r = rule.value.includes(uid); if (r) break; }
      if (rule.type === 'channel') { r = rule.value.includes(channel); if (r) break; }
      if (rule.type === 'percent') { r = hashStr(uid + name) % 100 < rule.value; break; }   // 稳定分桶
    }
    this.cache.set(key, { v: this.version, r });
    return r;
  }
  expose(name, uid) {                                        // 曝光去重
    const k = name + '|' + uid;
    if (this.exposed.has(k)) return false;
    this.exposed.add(k);
    track('gray-expose', { exp: name, uid });
    return true;
  }
}
function hashStr(s) { let h = 0; for (const c of s) h = (h * 131 + c.charCodeAt(0)) >>> 0; return h; }
```
