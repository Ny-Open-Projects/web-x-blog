## HW-215：手写一个文本插入与删除核心算法：光标处 insert 与跨 block 的 delete

**答**：
- 需求：实现编辑核心操作：insertAt(pos, str)（pos: {blockId, offset}）；deleteRange(range)——同 block 直接删，跨 block 删除需把 tail block 剩余文本并入 head block 并移除中间 blocks；返回新光标位置。
- 边界：删除范围为空返回原位；跨 block 删除后 tail block 的 marks 平移并入 head；blocks 索引重算。
- 评分点：跨 block 合并的 marks 处理；光标返回。

```javascript
function insertAt(doc, pos, str) {
  const b = doc.byId(pos.blockId);
  b.text = b.text.slice(0, pos.offset) + str + b.text.slice(pos.offset);
  return { blockId: pos.blockId, offset: pos.offset + [...str].length };
}
function deleteRange(doc, range) {
  const { head, tail } = range.normalized();
  const hb = doc.byId(head.blockId), tb = doc.byId(tail.blockId);
  if (hb.id === tb.id) {                           // 同 block
    hb.text = hb.text.slice(0, head.offset) + hb.text.slice(tail.offset);
    return { blockId: hb.id, offset: head.offset };
  }
  // 跨 block：head 留前缀，tail 留后缀并入 head
  const tailSuffix = tb.text.slice(tail.offset);
  const removed = doc.blocks.filter((b, i) =>
    i > doc.indexOf(hb) && i < doc.indexOf(tb));
  removed.forEach(b => doc.remove(b.id));
  hb.text = hb.text.slice(0, head.offset) + tailSuffix;
  tb && doc.remove(tb.id);
  return { blockId: hb.id, offset: head.offset };  // 光标落合并点
}
```
