## HW-301：手写一个模板引擎：插值与条件、循环指令的编译执行

**答**：
- 需求：实现 Template：模板语法 {{expr}} 插值、{{#if cond}}...{{/if}} 条件、{{#each list}}...{{/each}} 循环（@item/@index）；编译为 render(data) 函数（正则分段解析，不重复解析）；表达式取值用安全路径求值（禁止函数调用，防注入）。
- 边界：嵌套 each/if；未定义变量输出空串（不抛错）；HTML 转义插值（三括号 {{{}}} 才不转义）。
- 评分点：编译缓存；安全求值（不 eval 用户表达式）；转义默认开。

```javascript
const ESC = s => String(s ?? '').replace(/[&<>"]/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));
function compile(tpl) {
  const tokens = tpl.split(/({{#if [^}]+}}|{{\/if}}|{{#each [^}]+}}|{{\/each}}|{{{?[^}]+}?}})/)
    .filter(Boolean);
  const fn = new Function('d', 'get', 'esc', 'each', 'ifs', tokens.map(t => {
    if (/^{{#if /.test(t)) return `if(ifs(get(d,'${t.slice(6, -3)}'))) {`;
    if (t === '{{/if}}') return '}';
    if (/^{{#each /.test(t)) return `for(const [i,item] of each(get(d,'${t.slice(7, -3)}'))) { const d={...d,item,index:i};`;
    if (t === '{{/each}}') return '}';
    if (/^{{{/.test(t)) return `_o+=get(d,'${t.slice(3, -3)}')??'';`;   // 三括号不转义
    return `_o+=esc(get(d,'${t.slice(2, -2)}'));`;
  }).join('\n'));
  return data => { let _o = ''; fn({ ...data, _o }, getPath, ESC, eachOf, condOf); return _o; };
}
function getPath(obj, path) {                               // 安全路径求值（无函数调用）
  return path.split('.').reduce((o, k) => o?.[k], obj);
}
function eachOf(v) { return Array.isArray(v) ? v.map((item, index) => [index, item]) : []; }
function condOf(v) { return !!v; }
```
