**场景**：筛选面板显示时间区间：同日"今天 09:00-18:00"、跨日"10月1日 - 10月3日"、相对"3 天前起"——按区间语义选择格式（Intl 组合），全 locale 正确。

**思路**：区间分类（同日/同月/同年/跨年）→ 各类用最小充分格式（同月省略月、同年省略年）→ 相对表述优先（"3 天前"比"10月16日"更友好——阈值内用相对）。

```javascript
function formatRange(start, end, locale = navigator.language, now = Date.now()) {
  const rt = new Intl.RelativeTimeFormat(locale);
  const dtf = (opts) => new Intl.DateTimeFormat(locale, opts);
  const daysAgo = Math.floor((now - end) / 86400e3);
  if (daysAgo >= 0 && daysAgo < 7 && start >= now - 7 * 86400e3) {
    return rt.format(-daysAgo, 'day') + (daysAgo === 0
      ? ' ' + dtf({ hour: '2-digit', minute: '2-digit' }).formatRange(start, end).replace(/–|-/, '- ')
      : '');                                        // 7 天内：相对表述（"3 天前"）
  }
  const s = new Date(start), e = new Date(end);
  const sameDay = s.toDateString() === e.toDateString();
  if (sameDay) return dtf({ dateStyle: 'medium' }).format(s) + ' ' +
    dtf({ hour: '2-digit', minute: '2-digit' }).formatRange(start, end);            // 同日：日期+时刻区间
  const sameYear = s.getFullYear() === e.getFullYear();
  const fmt = sameYear ? { month: 'short', day: 'numeric' } : { year: 'numeric', month: 'short', day: 'numeric' };
  return dtf(fmt).formatRange(s, e);                // Intl formatRange 自动去重（"10月1-3日"）
}
// Intl.formatRange 的 locale 智能：英文 "Oct 1 – 3"（月共享）、中文 "10月1-3日" —— 手写拼接做不到
// 相对优先阈值 7 天：更友好（"3 天前"比"10月16日"可读）；超出用绝对（语义精确）
// 同日合并：日期一次 + 时刻区间（不重复日期）
```

**评分点**：
- Intl.DateTimeFormat.formatRange 的 locale 智能去重——本题核心。
- 区间分类格式选择（同日/同月/同年/跨年最小充分格式）。
- 相对表述优先阈值（7 天内 RelativeTimeFormat）。
