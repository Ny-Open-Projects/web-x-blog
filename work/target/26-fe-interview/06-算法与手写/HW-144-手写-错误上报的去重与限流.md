**答**：错误风暴防护三件套：① 指纹去重——同错误（消息+堆栈首行）聚合计数，只报一次 + count（HW-144 的价值：一个循环 bug 每秒抛 1000 次，报 1000 条不如报 1 条 count=1000）；② 窗口限流——每分钟同类最多 N 条、全局限 M 条；③ 熔断——总量超阈值停报（服务端被打挂比丢错误数据严重）。

```javascript
class ErrorReporter {
  constructor({ perMinute = 60, circuitBreak = 2000 } = {}) {
    Object.assign(this, { perMinute, circuitBreak });
    this.fingerprints = new Map();      // 指纹 -> {count, firstAt, reported}
    this.windowCount = 0; this.windowStart = Date.now();
    this.total = 0; this.broken = false;
  }
  #fingerprint(err) {
    const msg = String(err?.message ?? err).slice(0, 200);
    const stackTop = err?.stack?.split('\n')[1]?.trim().slice(0, 120) ?? '';
    return `${msg}|${stackTop}`;
  }
  capture(err) {
    if (this.broken) return;
    this.total++;
    if (this.total > this.circuitBreak) { this.broken = true; this.flush(); return; }   // 熔断
    const now = Date.now();
    if (now - this.windowStart > 60000) { this.windowCount = 0; this.windowStart = now; }
    const fp = this.#fingerprint(err);
    const f = this.fingerprints.get(fp) ?? { count: 0, firstAt: now, reported: false, err };
    f.count++;
    this.fingerprints.set(fp, f);
    if (!f.reported && this.windowCount < this.perMinute) {                 // 指纹首报 + 限流
      f.reported = true; this.windowCount++;
      this.send({ msg: String(err?.message ?? err).slice(0, 500), stack: err?.stack?.slice(0, 2000), count: f.count });
    }
  }
  flush() {                                               // hidden 时把聚合计数补报
    for (const [fp, f] of this.fingerprints) if (f.count > 1 && f.reported)
      this.send({ fingerprint: fp, count: f.count, aggregated: true });
    this.fingerprints.clear();
  }
  send(body) { navigator.sendBeacon?.('/rum/error', JSON.stringify(body)); }
}
// window.addEventListener('error', (e) => reporter.capture(e.error ?? e.message));
// 语义：首条全量（含堆栈）→ 后续聚合计数 → hidden 补报最终 count —— 错误风暴只花一条报文
```

**评分点**：
- 指纹去重（消息+堆栈首行）与聚合计数补报——本题核心。
- 窗口限流 + 全局熔断（错误风暴不打垮服务端）。
- 首条带堆栈、后续聚合的语义分层。
- flush 绑 hidden（卸载前把计数送出去）。
