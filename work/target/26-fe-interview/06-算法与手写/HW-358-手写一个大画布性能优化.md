## HW-358：手写一个大画布性能优化：视口剔除与细节分级

**答**：
- 需求：实现 Culling：视口剔除（只渲染包围盒与视口相交的节点——空间网格加速：节点按网格桶登记，查询只查视口覆盖的桶）；LOD（缩放小的时候略过文字/细边框渲染）；渲染统计（剔除率）。
- 边界：节点跨桶（登记到所有覆盖桶）；节点移动时桶更新；LOD 阈值按 scale 分级（< 0.3 略文字，< 0.15 略描边）。
- 评分点：空间网格索引；跨桶登记；LOD 分级。

```javascript
class SpatialGrid {
  constructor(cell = 200) { this.cell = cell; this.buckets = new Map(); this.nodeCell = new Map(); }
  key(cx, cy) { return cx + ',' + cy; }
  insert(node) {
    const b = bounds(node);
    const cells = [];
    for (let cx = Math.floor(b.x / this.cell); cx <= Math.floor((b.x + b.w) / this.cell); cx++)
      for (let cy = Math.floor(b.y / this.cell); cy <= Math.floor((b.y + b.h) / this.cell); cy++) {
        const k = this.key(cx, cy);
        (this.buckets.get(k) ?? this.buckets.set(k, new Set()).get(k)).add(node.id);   // 跨桶登记
        cells.push(k);
      }
    this.nodeCell.set(node.id, cells);
  }
  move(node) { this.remove(node.id); this.insert(node); }    // 移动时桶更新
  remove(id) {
    for (const k of this.nodeCell.get(id) ?? []) this.buckets.get(k)?.delete(id);
    this.nodeCell.delete(id);
  }
  query(rect) {                                              // 视口查询
    const out = new Set();
    for (let cx = Math.floor(rect.x / this.cell); cx <= Math.floor((rect.x + rect.w) / this.cell); cx++)
      for (let cy = Math.floor(rect.y / this.cell); cy <= Math.floor((rect.y + rect.h) / this.cell); cy++)
        for (const id of this.buckets.get(this.key(cx, cy)) ?? []) out.add(id);
    return out;                                              // 粗筛后再精确相交判断
  }
}
class CullingRenderer {
  constructor(grid) { this.grid = grid; this.stats = { total: 0, drawn: 0 }; }
  frame(nodes, viewport, ctx) {
    const vis = this.grid.query(viewportRect(viewport));
    let drawn = 0;
    for (const n of nodes) {
      if (!vis.has(n.id) || !intersects(bounds(n), viewportRect(viewport))) continue;   // 精确剔除
      const lod = viewport.scale < 0.15 ? 0 : viewport.scale < 0.3 ? 1 : 2;   // LOD 分级
      n.draw(ctx, { lod });                                  // 节点按 lod 决定细节
      drawn++;
    }
    this.stats = { total: nodes.length, drawn,
      cullRate: +((1 - drawn / nodes.length) * 100).toFixed(1) + '%' };
  }
}
```
