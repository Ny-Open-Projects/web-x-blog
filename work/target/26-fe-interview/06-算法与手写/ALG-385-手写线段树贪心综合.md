## ALG-385：手写线段树+贪心综合：区间调度实时统计

**答**：
- 题目：会议室按时间轴占用（每次预约占 [s, e]），调度器要快速查「某时段内已占用的会议室数」并决定能否分配。线段树区间加（预约 +1、取消 -1）+ 查询区间最大并发（瓶颈判断）。
- 思路：max 线段树 + 区间加懒标记（max 无需 len）；预约前查区间 max >= 容量则拒绝。
- 评分点：区间加对 max 的懒标记；预约原子判断（查→改）；取消即负增量。

```javascript
class RoomScheduler {
  constructor(rooms) {
    this.cap = rooms;
    this.n = 1440;                                          // 一天按分钟
    this.max = new Array(4 * this.n).fill(0);
    this.lazy = new Array(4 * this.n).fill(0);
  }
  book(start, end) {                                       // 预约 [start, end)
    if (this.maxInRange(start, end - 1) >= this.cap) return false;   // 瓶颈判断
    this.rangeAdd(start, end - 1, 1);
    return true;
  }
  cancel(start, end) { this.rangeAdd(start, end - 1, -1); }
  rangeAdd(ql, qr, v, node = 1, l = 0, r = this.n - 1) {
    if (qr < l || r < ql) return;
    if (ql <= l && r <= qr) { this.max[node] += v; this.lazy[node] += v; return; }
    this.pushdown(node);
    const m = (l + r) >> 1;
    this.rangeAdd(ql, qr, v, node * 2, l, m); this.rangeAdd(ql, qr, v, node * 2 + 1, m + 1, r);
    this.max[node] = Math.max(this.max[node * 2], this.max[node * 2 + 1]);
  }
  maxInRange(ql, qr, node = 1, l = 0, r = this.n - 1) {
    if (qr < l || r < ql) return 0;
    if (ql <= l && r <= qr) return this.max[node];
    this.pushdown(node);
    const m = (l + r) >> 1;
    return Math.max(this.maxInRange(ql, qr, node * 2, l, m),
      this.maxInRange(ql, qr, node * 2 + 1, m + 1, r));
  }
  pushdown(node) {
    if (!this.lazy[node]) return;
    for (const c of [node * 2, node * 2 + 1]) { this.max[c] += this.lazy[node]; this.lazy[c] += this.lazy[node]; }
    this.lazy[node] = 0;
  }
}
```
