## ALG-358：手写扫描线+线段树求矩形面积并：弹窗遮挡面积统计

**答**：
- 题目：屏幕上有 N 个弹窗矩形（可能重叠），求总遮挡面积（面积并）。经典扫描线。
- 思路：事件按 y 排序（每个矩形产生入边/出边两个事件）；x 方向用线段树维护「当前被覆盖的 x 区间总长度」；扫描线从下往上，每两事件之间累加 覆盖长度 × dy。线段树需要「完全覆盖计数」节点（cover 次数 + 覆盖长度），本题不需下推（cover>0 时长度即区间长）。
- 评分点：离散化 x 坐标；入边 +1 出边 -1；cover 计数不下推的特例。

```javascript
function rectUnionArea(rects) {
  const xs = [...new Set(rects.flatMap(r => [r.x1, r.x2]))].sort((a, b) => a - b);
  const xi = new Map(xs.map((x, i) => [x, i]));
  const events = rects.flatMap(r => [
    { y: r.y1, d: +1, l: xi.get(r.x1), r: xi.get(r.x2) - 1 },
    { y: r.y2, d: -1, l: xi.get(r.x1), r: xi.get(r.x2) - 1 },
  ]).sort((a, b) => a.y - b.y);
  const st = { cover: new Array(xs.length * 4).fill(0), len: new Array(xs.length * 4).fill(0) };
  function update(node, l, r, ql, qr, d) {
    if (qr < l || r < ql) return;
    if (ql <= l && r <= qr) return void (st.cover[node] += d);
    const m = (l + r) >> 1;
    update(node * 2, l, m, ql, qr, d); update(node * 2 + 1, m + 1, r, ql, qr, d);
  }
  function pull(node, l, r) {                              // 自底向上重算覆盖长度
    if (st.cover[node] > 0) st.len[node] = xs[r + 1] - xs[l];   // 完全覆盖：直接区间长
    else if (l === r) st.len[node] = 0;
    else st.len[node] = st.len[node * 2] + st.len[node * 2 + 1];   // 不下推特例
  }
  let area = 0, prevY = events[0].y;
  for (const ev of events) {
    area += st.len[1] * (ev.y - prevY);                    // 覆盖长度 × dy
    update(1, 0, xs.length - 2, ev.l, ev.r, ev.d);
    pull(1, 0, xs.length - 2);
    prevY = ev.y;
  }
  return area;
}
```
