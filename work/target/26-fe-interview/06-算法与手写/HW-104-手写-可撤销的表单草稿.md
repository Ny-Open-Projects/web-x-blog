**答**：草稿系统：① 防抖自动保存到 localStorage（1s 内多次修改只存最后一次）；② 脏检查——保存时 diff 当前值与上次保存值，无变化不写（省配额）；③ 版本栈——最近 N 份草稿可手动恢复；④ 冲突提示——恢复的草稿与服务器版本（updated_at）比对，服务器更新过则提示二选一；⑤ 生命周期——提交成功清草稿，beforeunload 兜底保存。

```javascript
function createDraft(formId, formEl, { debounceMs = 1000, versions = 5 } = {}) {
  const key = `draft:${formId}`;
  const versionKey = `draft-hist:${formId}`;
  let timer = null;

  function serialize() {
    return JSON.stringify(Object.fromEntries(new FormData(formEl).entries()));
  }
  function save({ history = false } = {}) {
    const cur = serialize();
    const last = localStorage.getItem(key);
    if (cur === last) return;                                  // 脏检查：无变化不写
    if (history) {                                             // 版本栈
      const hist = JSON.parse(localStorage.getItem(versionKey) || '[]');
      hist.push({ at: Date.now(), data: cur });
      localStorage.setItem(versionKey, JSON.stringify(hist.slice(-versions)));
    }
    localStorage.setItem(key, cur);
  }
  formEl.addEventListener('input', () => {
    clearTimeout(timer);
    timer = setTimeout(() => save({ history: false }), debounceMs);
  });
  addEventListener('beforeunload', () => save());              // 关页兜底
  return {
    async restore({ serverUpdatedAt } = {}) {
      const draft = JSON.parse(localStorage.getItem(key) || 'null');
      if (!draft) return null;
      const meta = JSON.parse(localStorage.getItem(`${key}:meta`) || '{}');
      if (serverUpdatedAt && meta.savedAt < serverUpdatedAt) return { conflict: true, draft };  // 服务器更新过
      Object.entries(draft).forEach(([name, v]) => { const el = formEl.elements[name]; if (el) el.value = v; });
      return { restored: true };
    },
    history: () => JSON.parse(localStorage.getItem(versionKey) || '[]'),
    clear() { localStorage.removeItem(key); localStorage.removeItem(versionKey); },
    saveNow: () => save({ history: true }),
  };
}
// 提交成功回调里 draft.clear()；冲突时 UI 提示"恢复草稿 / 使用服务器版本"
```

**评分点**：
- 防抖保存 + 脏检查（相同内容不写，省配额）。
- 版本栈（上限 N 份）支持手动恢复；beforeunload 兜底。
- 冲突检测（草稿时间 vs 服务器 updated_at）并返回 conflict 让 UI 决策。
- 提交成功清草稿的生命周期完整性。
