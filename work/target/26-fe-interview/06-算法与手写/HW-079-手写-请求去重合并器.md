**答**：两层去重：① 同 key 在途请求共享同一个 Promise（并发去重）；② 批量接口合并——多个 key 的请求攒在微任务窗口里合并成一次批量请求，结果按 key 分发回去。错误也要分发（reject 所有等待者），并且失败后缓存要清（允许重试）。

```javascript
class RequestMerger {
  constructor(batchFn, { window = 10 } = {}) {   // batchFn(keys) -> Map(key, result)
    this.batchFn = batchFn; this.window = window;
    this.inflight = new Map();                   // key -> {promise, waiters: [{resolve, reject}]}
    this.pending = new Map();                    // key -> {resolve, reject} 批量窗口
    this.timer = null;
  }
  get(key) {
    if (this.inflight.has(key)) return this.inflight.get(key).promise;   // 并发去重
    const entry = { promise: null, waiters: [] };
    entry.promise = new Promise((resolve, reject) => entry.waiters.push({ resolve, reject }));
    this.inflight.set(key, entry);

    this.pending.set(key, { resolve: null, reject: null, entry });
    if (!this.timer) this.timer = queueMicrotask(() => this.#flush());   // 微任务窗口合并
    return entry.promise;
  }
  async #flush() {
    this.timer = null;
    const keys = [...this.pending.keys()];
    const pend = this.pending; this.pending = new Map();
    try {
      const results = await this.batchFn(keys);          // 一次批量请求
      for (const k of keys) {
        const p = pend.get(k);
        this.inflight.delete(k);                          // 失败可重试：先清在途
        if (results?.has(k)) p.entry.waiters.forEach(w => w.resolve(results.get(k)));
        else p.entry.waiters.forEach(w => w.reject(new Error(`no result: ${k}`)));
      }
    } catch (e) {
      for (const k of keys) { this.inflight.delete(k);
        pend.get(k).entry.waiters.forEach(w => w.reject(e)); }
    }
  }
}
// 用法：商品详情批量接口
const merger = new RequestMerger(async (keys) => {
  const r = await fetch('/api/goods/batch', { body: JSON.stringify({ ids: keys }) }).then(x => x.json());
  return new Map(r.map(g => [g.id, g]));
});
// 同一微任务窗口内的 10 个 get(1..10) 合并成一次请求；
// 组件 A、B 同时 get(5) 共享同一个 Promise（并发去重）
```

**评分点**：
- 并发去重：在途 Map 共享 Promise，重复 get 不发新请求。
- 微任务窗口合并（queueMicrotask/setTimeout 0），批量接口一次请求。
- 失败路径完整：所有等待者 reject 且清在途缓存（允许后续重试）。
- 批量窗口 flush 时交换 Map（防 flush 期间新请求进入同一批）。
