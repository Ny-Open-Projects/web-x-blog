## HW-397：手写一个版本对比选择器：三向合并 UI 数据

**答**：
- 需求：实现 MergeView：三方输入（base/ours/theirs 文本）；逐块 diff（base↔ours、base↔theirs 双 diff 对齐）；冲突分类：双方都改且不同=冲突，单方改=自动取，双方同改=自动；输出 UI 数据（块列表 {type: auto-ours|auto-theirs|auto-same|conflict, 三方内容}）；用户决策（冲突块选 ours/theirs/手写）后合成结果。
- 边界：diff 对齐算法（先行级 diff，再块归组）；冲突块相邻合并；合成顺序按块序。
- 评分点：双 diff 对齐；冲突分类正确性；决策合成。

```javascript
function mergeView(base, ours, theirs) {
  const bo = lineDiff(base, ours);                           // base↔ours 行 diff
  const bt = lineDiff(base, theirs);                         // base↔theirs
  const blocks = alignChanges(bo, bt);                       // 块归组对齐
  return blocks.map(b => {
    if (!b.oursChanged && !b.theirsChanged)
      return { type: 'auto-same', content: b.base, resolved: b.base };       // 双方未改
    if (b.oursChanged && !b.theirsChanged)
      return { type: 'auto-ours', content: b.ours, resolved: b.ours };       // 单方改
    if (!b.oursChanged && b.theirsChanged)
      return { type: 'auto-theirs', content: b.theirs, resolved: b.theirs };
    if (b.ours === b.theirs)
      return { type: 'auto-same', content: b.ours, resolved: b.ours };       // 双方同改
    return { type: 'conflict', ours: b.ours, theirs: b.theirs, base: b.base,
      resolved: null };                                      // 真冲突：待决策
  });
}
function resolveConflicts(blocks, decisions) {               // 决策合成
  return blocks.map((b, i) => {
    if (b.type !== 'conflict') return b.resolved;
    const d = decisions[i];
    return d === 'ours' ? b.ours : d === 'theirs' ? b.theirs : d ?? b.base;   // 手写或回 base
  }).join('\n');
}
function alignChanges(bo, bt) {                              // 块归组（简化）
  const blocks = [];
  for (let i = 0; i < bo.length; i++) {
    blocks.push({ base: bo[i].old ?? bo[i].new,
      ours: bo[i].new ?? bo[i].old, oursChanged: bo[i].type !== 'same',
      theirs: bt[i]?.new ?? bt[i]?.old, theirsChanged: bt[i]?.type !== 'same' });
  }
  return blocks;
}
```
