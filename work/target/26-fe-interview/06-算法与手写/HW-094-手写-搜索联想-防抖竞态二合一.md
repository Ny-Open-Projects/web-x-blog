**答**：搜索联想四要素：防抖（300ms）、竞态丢弃（只认最后一次请求）、minChars（1 个字符不触发）、组件卸载清理（timer + abort）。与 HW-050 的差异：hook 形态带完整生命周期。

```javascript
function useSuggest({ minChars = 2, debounceMs = 300 } = {}) {
  const [suggestions, setSuggestions] = useState([]);
  const [loading, setLoading] = useState(false);
  const seq = useRef(0), timer = useRef(null), ctrl = useRef(null);

  const search = useCallback((keyword) => {
    clearTimeout(timer.current);
    if (!keyword || keyword.length < minChars) { setSuggestions([]); setLoading(false); return; }
    setLoading(true);
    timer.current = setTimeout(async () => {
      const mySeq = ++seq.current;
      ctrl.current?.abort();                      // 取消上一个在途请求
      ctrl.current = new AbortController();
      try {
        const r = await fetch(`/api/suggest?q=${encodeURIComponent(keyword)}`, { signal: ctrl.current.signal });
        const list = await r.json();
        if (seq.current === mySeq) { setSuggestions(list); setLoading(false); }  // 竞态丢弃
      } catch (e) {
        if (e.name !== 'AbortError' && seq.current === mySeq) { setLoading(false); setSuggestions([]); }
      }
    }, debounceMs);
  }, [minChars, debounceMs]);

  useEffect(() => () => {                        // 卸载清理：timer + 在途请求
    clearTimeout(timer.current);
    ctrl.current?.abort();
  }, []);

  return { suggestions, loading, search };
}
// <input onInput={(e) => search(e.target.value)} />
// 快速输入 "react hooks"：只发 1 次请求（防抖），旧请求被 abort（竞态双保险）
```

**评分点**：
- 防抖 + 竞态双重保险（防抖减少请求，abort + seq 丢弃过期响应）。
- minChars 下限直接清空不请求。
- 卸载清理完整（clearTimeout + abort），不会对已卸载组件 setState。
- AbortError 与真错误区分处理。
