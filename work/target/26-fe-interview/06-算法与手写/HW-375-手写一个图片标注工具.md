## HW-375：手写一个图片标注工具：矩形、箭头、文字标注与序列化

**答**：
- 需求：实现 Annotator：标注类型（rect/arrow/text/freehand），叠加在图片上层 canvas；绘制交互（拖拽生成 rect/arrow，点击放置 text，freehand 采集路径点）；标注可选中/删除/编辑；序列化（归一化坐标 0~1，图片尺寸无关）；渲染回放。
- 边界：归一化坐标（图片缩放后标注跟随）；freehand 点抽稀（距离阈值采样）；文字标注自动换行与框自适应。
- 评分点：归一化序列化；路径抽稀；标注 CRUD。

```javascript
class Annotator {
  constructor(canvas, img) { this.c = canvas; this.img = img; this.items = []; this.sel = null; }
  norm(x, y) { return { x: x / this.c.width, y: y / this.c.height }; }   // 归一化
  denorm(p) { return { x: p.x * this.c.width, y: p.y * this.c.height }; }
  addRect(x1, y1, x2, y2) {
    const a = this.norm(x1, y1), b = this.norm(x2, y2);
    const item = { id: genId(), type: 'rect', x: Math.min(a.x, b.x), y: Math.min(a.y, b.y),
      w: Math.abs(a.x - b.x), h: Math.abs(a.y - b.y), color: '#f43' };
    this.items.push(item); this.sel = item; return item;
  }
  addArrow(x1, y1, x2, y2) {
    const a = this.norm(x1, y1), b = this.norm(x2, y2);
    return this.items[this.items.push({ id: genId(), type: 'arrow', from: a, to: b,
      color: '#3af' }) - 1];
  }
  addFreehand(points) {                                      // 路径抽稀
    const th = 3 / this.c.width;                             // 归一化距离阈值
    const sparse = points.filter((p, i) => i === 0
      || Math.hypot(p.x - points[i - 1].x, p.y - points[i - 1].y) > th);
    return this.items[this.items.push({ id: genId(), type: 'freehand', pts: sparse,
      color: '#3a3' }) - 1];
  }
  render(ctx) {                                              // 回放渲染
    for (const it of this.items) {
      ctx.strokeStyle = it.color;
      ctx.lineWidth = it === this.sel ? 3 : 2;
      if (it.type === 'rect') { const p = this.denorm({ x: it.x, y: it.y });
        ctx.strokeRect(p.x, p.y, it.w * this.c.width, it.h * this.c.height); }
      if (it.type === 'arrow') drawArrow(ctx, this.denorm(it.from), this.denorm(it.to));
      if (it.type === 'freehand') { ctx.beginPath();
        it.pts.forEach((p, i) => { const d = this.denorm(p);
          i ? ctx.lineTo(d.x, d.y) : ctx.moveTo(d.x, d.y); });
        ctx.stroke(); }
    }
  }
  serialize() { return JSON.stringify({ w: this.img.naturalWidth, h: this.img.naturalHeight,
    items: this.items }); }                                  // 尺寸无关
}
function drawArrow(ctx, a, b) {
  ctx.beginPath(); ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y); ctx.stroke();
  const ang = Math.atan2(b.y - a.y, b.x - a.x);
  ctx.beginPath();
  ctx.moveTo(b.x, b.y);
  ctx.lineTo(b.x - 10 * Math.cos(ang - 0.4), b.y - 10 * Math.sin(ang - 0.4));
  ctx.lineTo(b.x - 10 * Math.cos(ang + 0.4), b.y - 10 * Math.sin(ang + 0.4));
  ctx.fill();
}
```
