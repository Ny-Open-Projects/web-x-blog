## HW-101：手写表单异步联动——城市选择后异步拉取区县、级联禁用与缓存
**答**：级联表单要点：① 父字段变化触发子选项异步加载（带缓存：同城市第二次秒开）；② 加载中子字段禁用+loading 态；③ 父变更清空子值（旧选择失效）；④ 竞态丢弃（快速切换城市只认最后）；⑤ 子选项依赖父值参数化。

```javascript
function createCascadeForm(config, { onChange } = {}) {
  // config: { city: { load: (parentVal) => Promise<{value,label}[]> } } 按字段声明依赖
  const state = { values: {}, options: {}, loading: {}, seq: {} };
  const cache = new Map();                            // 父值 -> 选项（缓存）

  async function load(field, parentVal) {
    const cfg = config[field];
    if (!cfg) return;
    state.seq[field] = (state.seq[field] || 0) + 1;
    const mySeq = state.seq[field];
    if (cache.has(parentVal)) { state.options[field] = cache.get(parentVal); return notify(); }
    state.loading[field] = true; notify();
    try {
      const opts = await cfg.load(parentVal);
      cache.set(parentVal, opts);
      if (state.seq[field] === mySeq) {               // 竞态丢弃
        state.options[field] = opts;
        if (!opts.some((o) => o.value === state.values[field])) state.values[field] = ''; // 旧值失效清空
      }
    } finally {
      if (state.seq[field] === mySeq) state.loading[field] = false;
      notify();
    }
  }
  function setValue(field, val) {
    state.values[field] = val;
    for (const [child, cfg] of Object.entries(config)) {
      if (cfg.dependsOn === field) {                  // 父变化：清空子值并重载子选项
        state.values[child] = '';
        load(child, val);
      }
    }
    notify();
  }
  function notify() { onChange({ ...state.values },
    Object.fromEntries(Object.entries(state.loading).map(([k, v]) => [k, v])),
    state.options); }
  return { setValue, get values() { return state.values; }, get loading() { return state.loading; } };
}
// 用法：城市->区县级联
const form = createCascadeForm({ district: { dependsOn: 'city', load: (city) => fetch(`/api/districts?city=${city}`).then(r => r.json()) } });
form.setValue('city', 'hangzhou');   // district 清空 + loading + 拉取（缓存后二次秒开）
```

**评分点**：
- 父变更联动：清空子值 + 重载选项（旧子值失效语义正确）。
- 选项缓存（按父值键），二次选择零延迟；竞态丢弃（seq）。
- loading 态驱动子字段禁用；finally 里条件复位 loading。
- onChange 统一出口（values/loading/options 三元组）。
