## HW-398：手写一个编辑器性能预算门禁：大文档指标 CI 断言

**答**：
- 需求：实现 PerfGate：预算定义（大文档 10 万字：打开 < 800ms、输入延迟 p95 < 50ms、内存 < 150MB、滚动 FPS > 50）；测试执行（headless 环境跑脚本生成指标）；断言（超预算列出明细并 exit 1）；输出报告（Markdown 表格供 PR 评论）。
- 边界：环境噪声（多轮取中位数，首轮预热丢弃）；机器差异（基准机规格固定或按算力系数校准）；门禁豁免（标 perf-exempt 的 PR 跳过并要求后续补）。
- 评分点：中位数去噪；算力校准；豁免流程。

```javascript
async function perfGate(scenarios, { rounds = 3 } = {}) {
  const results = [];
  for (const sc of scenarios) {                              // 每场景
    const runs = [];
    for (let i = 0; i < rounds + 1; i++) {                   // 首轮预热丢弃
      const r = await runScenario(sc);                       // headless 跑
      if (i > 0) runs.push(r);
    }
    results.push({ scenario: sc.name, budget: sc.budget,
      metrics: medianOf(runs) });                            // 中位数去噪
  }
  const violations = [];
  for (const r of results) {
    for (const [metric, budget] of Object.entries(r.budget)) {
      const actual = r.metrics[metric];
      const ok = metric.includes('fps') ? actual >= budget : actual <= budget;
      if (!ok) violations.push({ scenario: r.scenario, metric, actual, budget });   // 明细
    }
  }
  if (violations.length && !process.env.PERF_EXEMPT) {       // 豁免检查
    console.error(formatReport(results, violations));
    process.exit(1);                                         // CI 断言失败
  }
  return { pass: !violations.length, results, violations,
    report: formatReport(results, violations) };             // Markdown 报告
}
function medianOf(runs) {
  const keys = Object.keys(runs[0]);
  return Object.fromEntries(keys.map(k => {
    const arr = runs.map(r => r[k]).sort((a, b) => a - b);
    return [k, arr[Math.floor(arr.length / 2)]];
  }));
}
function formatReport(results, violations) {
  const rows = results.map(r => `| ${r.scenario} | ${JSON.stringify(r.metrics)} |`).join('\n');
  return `## 性能门禁报告\n\n| 场景 | 指标 |\n|---|---|\n${rows}\n\n超标项：${JSON.stringify(violations)}`;
}
```
