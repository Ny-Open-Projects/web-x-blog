## ALG-483：手写反悔贪心入门：带退出的调度

**答**：
- 题目：任务流按时间到达，每任务有收益，同一时刻只能做一个；发现更优时可「反悔」换掉已选的。反悔贪心：堆里保留已选，更优时弹出最小换入。
- 思路：按 deadline 排序；逐个入堆（当前已选数）；若超时（数量超过时间容量），弹出收益最小的；最终堆内即最优。
- 评分点：反悔 = 弹出历史最小；堆维护当前方案；正确性直觉（任何被换出者不可能在更优解中）。

```javascript
function regretSchedule(tasks) {                           // tasks: [{time, profit}]
  const sorted = [...tasks].sort((a, b) => a.time - b.time);   // 按截止时间
  const heap = new MinHeap();                              // 已选任务（按收益小顶）
  let total = 0;
  for (const t of sorted) {
    heap.push(t.profit); total += t.profit;
    if (heap.size > t.time) {                              // 超容量：反悔
      const removed = heap.pop();
      total -= removed;                                    // 换出收益最小者
    }
  }
  return { total, chosen: heap.toArray() };
}
class MinHeap {
  constructor() { this.a = []; }
  get size() { return this.a.length; }
  push(v) { this.a.push(v); this.up(this.a.length - 1); }
  pop() { const top = this.a[0], last = this.a.pop();
    if (this.a.length) { this.a[0] = last; this.down(0); } return top; }
  up(i) { while (i > 0) { const p = (i - 1) >> 1;
    if (this.a[p] <= this.a[i]) break;
    [this.a[p], this.a[i]] = [this.a[i], this.a[p]]; i = p; } }
  down(i) { while (true) { let m = i;
    for (const c of [2 * i + 1, 2 * i + 2])
      if (c < this.a.length && this.a[c] < this.a[m]) m = c;
    if (m === i) break;
    [this.a[m], this.a[i]] = [this.a[i], this.a[m]]; i = m; } }
  toArray() { return [...this.a]; }
}
```
