## HW-075：手写状态机驱动的多步表单向导（分步校验、可回退、草稿保存）
**答**：分步表单向导（注册三步：基本信息 → 验证码 → 确认）用 FSM 建模：每步是一个 state，next 事件的 guard 是"当前步校验通过"。支持 back 回退（校验过的步骤保留数据）、任意跳步需 guard、草稿自动保存到 sessionStorage（每步 next 时持久化）。

```javascript
const steps = ['info', 'code', 'confirm'];
const validators = {
  info: (d) => d.name && /^\S+@\S+$/.test(d.email) || '请填写姓名与合法邮箱',
  code: (d) => /^\d{6}$/.test(d.code || '') || '请输入 6 位验证码',
  confirm: () => true,
};

function createWizard({ onStep, draftKey = 'wizard-draft' }) {
  const data = JSON.parse(sessionStorage.getItem(draftKey) || '{}');   // 草稿恢复
  let idx = 0;
  const save = () => sessionStorage.setItem(draftKey, JSON.stringify(data));

  async function next() {
    const check = validators[steps[idx]](data);
    if (check !== true) return { ok: false, msg: check };
    save();                                            // 每步通过即存草稿
    if (idx < steps.length - 1) {
      idx++;
      onStep(steps[idx], data, { canBack: true });
    } else {
      sessionStorage.removeItem(draftKey);             // 提交成功清草稿
      onStep('done', data, { canBack: false });
    }
    return { ok: true };
  }
  function back() {
    if (idx === 0) return { ok: false, msg: '已是第一步' };
    idx--; onStep(steps[idx], data, { canBack: idx > 0 });
    return { ok: true };
  }
  function jump(step) {                                // 侧边栏点步骤名跳转：只能回到已过步骤
    const target = steps.indexOf(step);
    if (target > idx) return next();                   // 前进走校验
    if (target >= 0) { idx = target; onStep(step, data, { canBack: idx > 0 }); return { ok: true }; }
    return { ok: false, msg: '未知步骤' };
  }
  return { data, next, back, jump, get step() { return steps[idx]; } };
}
// 用法：向导刷新后从草稿恢复当前步骤（草稿里记 idx 即可扩展）
```

**评分点**：
- 分步 guard 校验失败返回结构化结果（不静默），数据保留在 data 上。
- back 不重新校验（已过步骤），jump 前进走校验、后退直接跳。
- 草稿持久化时机正确：每步通过时保存、done 时清除。
- 能说清"为什么用状态机而不是 if-else"：步骤间转移与回退规则集中在一张表里。
