## HW-381：手写一个大文件导入管线：CSV/Excel 分片解析与流式入库

**答**：
- 需求：实现 ImportPipeline：CSV 流式解析（File.stream() 逐块 + 行边界处理，半行缓存）；类型推断（采样 100 行推断列类型：数字/日期/字符串）；入库（IndexedDB 批量事务，每 1000 行一事务）；进度与取消；Excel 走 SheetJS 按行流式（此处 CSV 全手写）。
- 边界：引号内换行/逗号（CSV 状态机解析，非简单 split）；半行跨块缓存；取消中断（AbortSignal + 清理 IDB 临时库）。
- 评分点：CSV 状态机；半行缓存；批量事务入库。

```javascript
class ImportPipeline {
  constructor({ batchSize = 1000 } = {}) { this.batch = batchSize; }
  async import(file, { signal } = {}) {
    const reader = file.stream().getReader();
    const decoder = new TextDecoder();
    let buf = '', header = null, rows = [], n = 0;
    const store = idbOpen('import-' + Date.now());
    while (true) {
      if (signal?.aborted) { await idbDelete(store); throw new Error('cancelled'); }   // 取消清理
      const { done, value } = await reader.read();
      if (done) break;
      buf += decoder.decode(value, { stream: true });
      const lines = splitCsvLines(buf);                      // 状态机切行（引号内换行安全）
      buf = lines.pop();                                     // 半行缓存
      for (const line of lines) {
        const cells = parseCsvLine(line);
        if (!header) { header = cells; continue; }
        rows.push(coerceRow(header, cells));
        if (++n % this.batch === 0) {                        // 批量事务入库
          await idbBulkPut(store, rows);
          this.onProgress?.(n / estimateRows(file));
          rows = [];
        }
      }
    }
    if (rows.length) await idbBulkPut(store, rows);
    return { header, rows: n };
  }
}
function splitCsvLines(buf) {                               // 引号内换行安全切行
  const out = []; let inQ = false, cur = '';
  for (let i = 0; i < buf.length; i++) {
    const c = buf[i];
    if (c === '"') inQ = !inQ;
    if (c === '\n' && !inQ) { out.push(cur); cur = ''; continue; }
    cur += c;
  }
  out.push(cur);
  return out;
}
function parseCsvLine(line) {                               // 引号内逗号安全
  const out = []; let cur = '', inQ = false;
  for (let i = 0; i < line.length; i++) {
    const c = line[i];
    if (c === '"') { if (inQ && line[i + 1] === '"') { cur += '"'; i++; } else inQ = !inQ; }
    else if (c === ',' && !inQ) { out.push(cur); cur = ''; }
    else cur += c;
  }
  out.push(cur);
  return out;
}
function coerceRow(header, cells) {                        // 类型推断转换
  return Object.fromEntries(header.map((h, i) => {
    const v = cells[i] ?? '';
    const num = Number(v);
    return [h, v !== '' && !isNaN(num) && /^\d/.test(v) ? num : v];
  }));
}
```
