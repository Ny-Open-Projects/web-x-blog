**答**：把"下单流程"（校验→并行[库存/风控]→条件[需支付?]→创建订单）写成声明式流程：节点 { run, next, branch }。引擎递归执行：next 是下一个节点 key，branch(ctx) 返回 key 实现条件跳转。错误处理：节点抛错走 onError 节点或中止。

```javascript
async function runFlow(nodes, entry, ctx = {}) {
  const visited = new Set();
  let key = entry;
  while (key && !visited.has(key)) {              // visited 防流程环
    visited.add(key);
    const node = nodes[key];
    if (!node) throw new Error(`未知节点: ${key}`);
    try {
      ctx = await node.run(ctx) ?? ctx;           // 节点可返回更新后的 ctx
    } catch (e) {
      if (node.onError) { key = node.onError; continue; }   // 错误处理节点
      throw e;
    }
    key = node.branch ? node.branch(ctx) : node.next;       // 条件分支或顺序
  }
  return ctx;
}
// 用法：下单流程
const flow = await runFlow({
  validate: { run: (ctx) => validateOrder(ctx.form), next: 'parallel' },
  parallel: {
    run: (ctx) => Promise.all([checkStock(ctx.form), riskCheck(ctx.user)])
      .then(([stock, risk]) => ({ ...ctx, stock, risk })),
    next: 'payGate',
  },
  payGate: { branch: (ctx) => (ctx.stock.ok && ctx.risk.pass ? (ctx.amount > 0 ? 'pay' : 'done') : 'reject') },
  pay:     { run: (ctx) => pay(ctx), next: 'create', onError: 'reject' },
  create:  { run: (ctx) => createOrder(ctx), next: 'done' },
  reject:  { run: (ctx) => ({ ...ctx, rejected: true }) },
  done:    { run: (ctx) => ctx },
}, 'validate', { form, user });
// 声明式价值：流程图一眼可读，改流程不动引擎；节点复用（payGate 可被别的流程引用）
```

**评分点**：
- 引擎与流程声明分离（引擎通用于任何流程图）。
- branch 条件跳转与 next 顺序两种连接都支持。
- 节点 ctx 传递与更新语义清晰（返回值合并）。
- 防环（visited）与 onError 错误转移节点。
