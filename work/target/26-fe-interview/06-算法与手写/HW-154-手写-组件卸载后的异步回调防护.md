**答**：卸载后 setState 警告/泄漏的三合一防护 hook：① alive 标志（卸载后回调直接短路）；② 自动 Abort（组件卸载取消在途请求）；③ 取消令牌（长任务循环检查 token 判断是否继续）。

```jsx
import { useEffect, useRef, useCallback } from 'react';

function useSafeAsync() {
  const alive = useRef(true);
  const ctrl = useRef(new AbortController());
  useEffect(() => {
    alive.current = true;
    return () => { alive.current = false; ctrl.current.abort(); };   // 卸载：短路 + 取消
  }, []);
  const safeSetState = useCallback((setter) => {
    if (!alive.current) return;                             // 卸载后：直接短路
    setter();
  }, []);
  const signal = useCallback(() => ctrl.current.signal, []);
  const token = useRef({ cancelled: false });
  const safeLoop = useCallback(async (task) => {            // 长任务循环：每步检查 token
    token.current = { cancelled: false };
    for (let i = 0; token.current && !token.current.cancelled && alive.current; i++) {
      await task(i);
    }
  }, []);
  return { safeSetState, signal, safeLoop, isAlive: () => alive.current };
}
// 用法：
// const { safeSetState, signal, safeLoop } = useSafeAsync();
// fetch(url, { signal: signal() }).then((d) => safeSetState(() => setData(d)));   // 卸载不炸
// await safeLoop(async (i) => { await processBatch(i); });                         // 卸载自动停
// 三合一语义：alive 短路（防 setState 警告）/ Abort（省资源）/ token（长任务可中断）
```

**评分点**：
- 三合一防护语义分别防什么（警告/资源/长任务）说清。
- 卸载清理顺序（alive 置 false 先于 abort）。
- safeLoop 每步检查（不是只在开始检查）。
- 能对照 React 18 的"卸载后 setState 警告已移除"但内存泄漏风险仍在的认知。
