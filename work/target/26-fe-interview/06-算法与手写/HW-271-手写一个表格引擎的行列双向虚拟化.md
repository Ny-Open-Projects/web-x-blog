## HW-271：手写一个表格引擎的行列双向虚拟化

**答**：
- 需求：实现 TableVirtualizer：行高 rowH、列宽数组 colW（可变）；onScroll(top, left) 输出可视行区间与可视列区间及各列偏移；渲染网格容器 translate 定位；水平方向按前缀和定位（列宽不等）。
- 边界：行缓冲 10 行、列缓冲 3 列；列宽变化时前缀和重建；总宽/总高由前缀和末值计算（滚动条尺寸正确）。
- 评分点：可变列宽前缀和；双向缓冲；translate 定位。

```javascript
class TableVirtualizer {
  constructor(rowH, colW, viewH, viewW, { rowBuf = 10, colBuf = 3 } = {}) {
    this.rowH = rowH; this.colW = colW; this.vh = viewH; this.vw = viewW;
    this.rb = rowBuf; this.cb = colBuf;
    this.prefix = prefixSums(colW);                          // 列宽前缀和
    this.totalW = this.prefix.at(-1);
  }
  onScroll(top, left, totalRows) {
    const r0 = Math.max(0, Math.floor(top / this.rowH) - this.rb);
    const r1 = Math.min(totalRows, Math.ceil((top + this.vh) / this.rowH) + this.rb);
    // 列：前缀和二分找可视列
    const c0 = Math.max(0, lowerBound(this.prefix, left) - this.cb);
    const c1 = Math.min(this.colW.length, upperBound(this.prefix, left + this.vw) + this.cb);
    return { rows: [r0, r1], cols: [c0, c1],
      x: c0 > 0 ? this.prefix[c0 - 1] : 0,                   // 可视块 x 偏移
      y: r0 * this.rowH,
      totalW: this.totalW, totalH: totalRows * this.rowH };
  }
  setColWidths(colW) { this.colW = colW; this.prefix = prefixSums(colW); this.totalW = this.prefix.at(-1); }
}
function prefixSums(a) { let s = 0; return a.map(v => (s += v)); }
function lowerBound(pre, x) { let lo = 0, hi = pre.length;
  while (lo < hi) { const m = (lo + hi) >> 1; pre[m] < x ? lo = m + 1 : hi = m; } return lo; }
function upperBound(pre, x) { let lo = 0, hi = pre.length;
  while (lo < hi) { const m = (lo + hi) >> 1; pre[m] <= x ? lo = m + 1 : hi = m; } return lo; }
```
