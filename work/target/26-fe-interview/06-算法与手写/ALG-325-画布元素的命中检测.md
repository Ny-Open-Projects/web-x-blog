**场景**：白板点击选中元素：全元素遍历 O(n) → 空间网格（ALG-245）粗筛视口附近 → 精确几何（点在旋转矩形内/贝塞尔距离）→ z 序最上者。

**思路**：网格查询候选 → 逆 z 序遍历（最上层优先命中即返回）→ 点在旋转矩形内（反旋转到局部坐标判定）。

```javascript
function hitTest(grid, point, elements) {
  const candidates = grid.query({ x: point.x - 5, y: point.y - 5, w: 10, h: 10 });   // 网格粗筛（含容差 5px）
  const sorted = [...candidates].sort((a, b) => b.z - a.z);   // 逆 z 序：最上层优先
  for (const el of sorted) {
    if (hitElement(el, point)) return el;                 // 命中即返回（最上层）
  }
  return null;
}
function hitElement(el, p) {
  if (el.type === 'rect') {                               // 旋转矩形：反变换到局部坐标
    const dx = p.x - el.x - el.w / 2, dy = p.y - el.y - el.h / 2;
    const cos = Math.cos(-el.rotation), sin = Math.sin(-el.rotation);
    const lx = dx * cos - dy * sin + el.w / 2, ly = dx * sin + dy * cos + el.h / 2;
    return lx >= 0 && lx <= el.w && ly >= 0 && ly <= el.h;   // 局部坐标判定
  }
  if (el.type === 'ellipse') {                            // 椭圆：归一化距离
    const dx = (p.x - el.x) / (el.rx || 1), dy = (p.y - el.y) / (el.ry || 1);
    return dx * dx + dy * dy <= 1;
  }
  return false;                                           // 其他类型：包围盒近似
}
// 两级检测：网格粗筛（ALG-245 复用）→ 精确几何（旋转矩形反变换/椭圆归一化）
// z 序语义：逆序遍历命中即返回（用户点击的就是看到的最高层）
// 容差 5px：细线条（1px 线宽）点击容差——交互可用性（几何精确 ≠ 交互精确）
```

**评分点**：
- 网格粗筛 + 精确几何两级检测（复用 ALG-245）——本题核心。
- 旋转矩形反变换到局部坐标的判定。
- 逆 z 序命中即返回（所见即所选）。
- 容差设计（几何精确 ≠ 交互精确）。
