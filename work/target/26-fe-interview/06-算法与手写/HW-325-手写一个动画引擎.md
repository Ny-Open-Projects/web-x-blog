## HW-325：手写一个动画引擎：时间线、缓动与顺序并行动画

**答**：
- 需求：实现 AnimEngine：animate(target, props, {duration, ease, delay}) 返回控制句柄（pause/resume/cancel）；timeline（顺序 then / 并行 parallel 组合）；每帧插值写入 target.style（transform 合成）。
- 边界：delay 后才开始；pause 冻结进度（记录 elapsed），resume 续播；cancel 停止并跳到终态或保留当前（可配）；多个引擎实例共享 rAF 心跳。
- 评分点：pause/resume 进度冻结；组合时间线；共享心跳。

```javascript
const EASE = { linear: t => t, outCubic: t => 1 - (1 - t) ** 3, inOut: t => t < .5 ? 2 * t * t : 1 - (-2 * t + 2) ** 2 / 2 };
let heartbeatSubs = new Set();
function raf(fn) { heartbeatSubs.add(fn); if (heartbeatSubs.size === 1) tick(); }   // 共享心跳
function tick() { requestAnimationFrame(() => {
  const fns = [...heartbeatSubs];
  fns.forEach(f => f(performance.now()));
  heartbeatSubs = new Set([...heartbeatSubs].filter(f => fns.includes(f) && !f.__done));
  if (heartbeatSubs.size) tick(); }); }
function animate(target, props, { duration = 400, ease = 'outCubic', delay = 0, fill = 'forwards' } = {}) {
  const from = {}, to = {};
  for (const [k, v] of Object.entries(props)) { from[k] = parseFloat(getComputedStyle(target)[k]) || 0; to[k] = v; }
  let elapsed = 0, paused = false, last = null, done = false;
  const fn = now => {
    if (paused) { last = now; return; }
    if (last == null) last = now;
    if (elapsed < delay) { elapsed += now - last; last = now; return; }
    elapsed += now - last; last = now;
    const p = Math.min(elapsed / duration, 1);
    const e = EASE[ease](p);
    for (const [k, v] of Object.entries(to))
      target.style[k] = from[k] + (v - from[k]) * e + (k.includes('x') || k.includes('y') ? 'px' : '');
    if (p >= 1) { done = true; fn.__done = true; }
  };
  raf(fn);
  return { pause: () => paused = true, resume: () => { paused = false; last = null; },
    cancel: () => { done = true; fn.__done = true; if (fill === 'forwards') for (const [k, v] of Object.entries(to)) target.style[k] = v; },
    get done() { return done; } };
}
function timeline(steps) {                                   // 顺序组合
  return steps.reduce((p, s) => p.then(() => s()), Promise.resolve());
}
```
