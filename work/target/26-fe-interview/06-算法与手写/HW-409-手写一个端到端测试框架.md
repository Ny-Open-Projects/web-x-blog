## HW-409：手写一个端到端测试框架：行为驱动断言

**答**：
- 需求：实现 E2EEditor：测试用例 DSL（type/select/copy/paste/undo/expectDoc 链式）；执行引擎（模拟输入序列走真实编辑器管线——非直接调 API）；断言失败输出 diff（期望 vs 实际文档 JSON diff）；快照模式（golden file 首跑生成，后续比对）。
- 边界：异步操作等待（粘贴净化、保存完成需轮询/事件等待）；时序稳定性（防抖操作等待静默期）；失败截图（DOM 快照）辅助定位。
- 评分点：走真实管线；异步等待静默期；golden diff。

```javascript
class E2EEditor {
  constructor(editor) { this.ed = editor; this.steps = []; }
  type(text) { this.steps.push(['type', text]); return this; }
  key(combo) { this.steps.push(['key', combo]); return this; }
  paste(text, html) { this.steps.push(['paste', text, html]); return this; }
  select(range) { this.steps.push(['select', range]); return this; }
  expectDoc(matcher) { this.steps.push(['expect', matcher]); return this; }
  async run({ settleMs = 100 } = {}) {
    for (const [op, ...args] of this.steps) {
      switch (op) {
        case 'type': for (const ch of args[0]) this.ed.dispatchKey(ch); break;   // 走真实管线
        case 'key': this.ed.dispatchCombo(args[0]); break;
        case 'paste': this.ed.dispatchPaste(args[0], args[1]); break;
        case 'select': this.ed.setSelection(args[0]); break;
        case 'expect': {
          await settle(this.ed, settleMs);                   // 等待防抖静默期
          const actual = this.ed.getDoc();
          const diffs = diffJson(args[1], actual);
          if (diffs.length) {
            await snapshotDom(this.ed.el, 'failure');        // 失败 DOM 快照
            throw new Error('doc-mismatch:\n' + diffs
              .map(d => `${d.path}: want ${JSON.stringify(d.want)} got ${JSON.stringify(d.got)}`)
              .join('\n'));
          }
          break;
        }
      }
    }
    return 'pass';
  }
  golden() { return JSON.stringify(this.ed.getDoc(), null, 2); }   // golden file
}
async function settle(ed, ms) {                              // 静默期等待
  let stable = 0, last = JSON.stringify(ed.getDoc());
  while (stable < 3) {
    await new Promise(r => setTimeout(r, ms));
    const cur = JSON.stringify(ed.getDoc());
    if (cur === last) stable++; else { stable = 0; last = cur; }
  }
}
```
