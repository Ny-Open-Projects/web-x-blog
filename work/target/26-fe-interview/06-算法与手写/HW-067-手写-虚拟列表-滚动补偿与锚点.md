## HW-067：手写虚拟列表进阶——快速滚动白屏修复（锚点定位 + 双向缓冲 + rAF 合帧）
**答**：定高虚拟列表（HW-037）在快速滚动时白屏，根因是"渲染追不上滚动"。三招：① 加大上下缓冲区（overscan）；② rAF 合帧滚动事件；③ 滚动超过一屏时用"锚点渲染"——先渲染目标区间为占位骨架，数据可用后替换，绝不渲染空区间。

```javascript
function createVirtualList(container, { itemHeight, overscan = 10, total }) {
  let range = { start: 0, end: 0 };
  let ticking = false;
  const pool = new Map();                     // key=index -> 节点（复用池）

  function onScroll() {
    if (ticking) return;
    ticking = true;
    requestAnimationFrame(() => {             // rAF 合帧：每帧最多计算一次
      ticking = false;
      const scrollTop = container.scrollTop;
      const visible = Math.ceil(container.clientHeight / itemHeight);
      const start = Math.max(0, Math.floor(scrollTop / itemHeight) - overscan);
      const end = Math.min(total, start + visible + overscan * 2);
      if (start !== range.start || end !== range.end) render(start, end);
    });
  }
  function render(start, end) {
    const gap = start - range.end;            // 跳变超过缓冲区 => 快速滚动
    if (range.end && gap > overscan * 2) {
      renderSkeleton(start, end);             // 先占位（同高骨架），再异步填充
      requestAnimationFrame(() => renderItems(start, end));
    } else renderItems(start, end);
    range = { start, end };
    container.style.setProperty('--pad-top', start * itemHeight + 'px');
    container.style.setProperty('--pad-bottom', (total - end) * itemHeight + 'px');
  }
  function renderItems(start, end) {
    const used = new Set();
    for (let i = start; i < end; i++) {       // 节点复用：不销毁不重建
      used.add(i);
      if (!pool.has(i)) { const n = renderItem(i); pool.set(i, n); }
      place(pool.get(i), i * itemHeight);
    }
    for (const [i, n] of pool) if (!used.has(i)) { recycle(n); pool.delete(i); }
  }
  container.addEventListener('scroll', onScroll, { passive: true });
  return { render, pool };
}
```

**评分点**：
- rAF 合帧滚动事件（不每事件重算）。
- overscan 缓冲区 + 快滚跳变检测（gap 判断）触发骨架占位，消除白屏。
- 节点复用池：不重复创建 DOM，recycle 处理正确。
- 上下 padding 占位计算正确（start / total-end 两个方向）。
