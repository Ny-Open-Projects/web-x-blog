## HW-400：手写一个跨端编辑器内核抽象

**答**：
- 需求：实现 EditorKernel：平台无关核心（文档模型/命令/undo，纯 JS 无 DOM）；平台适配层接口（渲染 render(blocks, cursor)、输入 input(events)、剪贴板 paste(text, html)）；H5/小程序/RN 三个适配器实现该接口；内核通过事件与适配层通信（不直接操作 DOM）。
- 边界：核心不 import DOM API（构建期断言）；事件协议统一（内核发 render 指令，适配层回 input 事件）；平台差异隔离（IME 处理在适配层）。
- 评分点：核心无 DOM；事件协议；平台差异下沉。

```javascript
// editor-kernel.js —— 平台无关核心（无 DOM）
class EditorKernel {
  constructor() { this.doc = createDoc(); this.cursor = null; this.history = new EditorHistory();
    this.listeners = new Map(); }
  on(event, fn) { this.listeners.set(event, fn); }           // render/input 事件协议
  emit(event, payload) { this.listeners.get(event)?.(payload); }
  insertText(str) {
    const inverse = captureInverse(this, 'insert', str);
    applyInsert(this.doc, this.cursor, str);
    this.history.push(inverse);
    this.emit('render', { doc: this.doc, cursor: this.cursor });   // 通知适配层渲染
  }
  deleteBackward() {
    if (!this.cursor?.offset) return;
    const inverse = captureInverse(this, 'delete', 1);
    applyDelete(this.doc, this.cursor, 1);
    this.history.push(inverse);
    this.emit('render', { doc: this.doc, cursor: this.cursor });
  }
}
// h5-adapter.js —— H5 适配层（DOM + IME）
class H5Adapter {
  constructor(kernel, el) {
    this.kernel = kernel; this.el = el;
    kernel.on('render', ({ doc, cursor }) => this.render(doc, cursor));
    el.addEventListener('beforeinput', e => {                // 输入协议：适配层→内核
      if (e.inputType === 'insertText') kernel.insertText(e.data);
      if (e.inputType === 'deleteContentBackward') kernel.deleteBackward();
      e.preventDefault();                                    // 内核全权管理文档
    });
  }
  render(doc, cursor) { /* DOM 渲染 + 光标定位 */ }
}
// mp-adapter.js —— 小程序适配层（setData 渲染，无 composition 原生事件）
class MPAdapter {
  constructor(kernel, page) { this.k = kernel; this.page = page;
    kernel.on('render', ({ doc }) => page.setData({ blocks: doc.blocks })); }
  onInput(e) { this.k.insertText(e.detail.value.slice(-1)); }   // 差量输入
}
// rn-adapter.js —— RN 适配层（TextInput 受控 + onKeyPress）
class RNAdapter {
  constructor(kernel) { this.k = kernel;
    kernel.on('render', ({ doc }) => this.setState({ value: docToText(doc) })); }
  onKeyPress(e) { if (e.nativeEvent.key === 'Backspace') this.k.deleteBackward(); }
}
```
