## ALG-383：手写线段树维护最大子段和：连续峰值时段

**答**：
- 题目：按秒的流量序列，查「区间内连续子段的流量和最大值」（定位最佳投放时段），支持单点修正。经典四字段线段树。
- 思路：节点 {sum, pre, suf, best}——sum 总和、pre 前缀最大、suf 后缀最大、best 区间最大子段。合并：pre = max(L.pre, L.sum + R.pre)；best = max(L.best, R.best, L.suf + R.pre)。
- 评分点：四字段定义；合并公式（含跨中点）；与 ALG-357 同构（连续 1 → 连续和）。

```javascript
class MaxSubarrayTree {
  constructor(a) { this.n = a.length; this.t = new Array(4 * this.n); this.build(a); }
  leaf(v) { return { sum: v, pre: v, suf: v, best: Math.max(0, v) }; }
  merge(L, R) {
    return {
      sum: L.sum + R.sum,
      pre: Math.max(L.pre, L.sum + R.pre),                 // 前缀可跨
      suf: Math.max(R.suf, R.sum + L.suf),
      best: Math.max(L.best, R.best, L.suf + R.pre),       // 跨中点拼接
    };
  }
  build(a, node = 1, l = 0, r = this.n - 1) {
    if (l === r) return this.t[node] = this.leaf(a[l]);
    const m = (l + r) >> 1;
    this.build(a, node * 2, l, m); this.build(a, node * 2 + 1, m + 1, r);
    this.t[node] = this.merge(this.t[node * 2], this.t[node * 2 + 1]);
  }
  set(i, v, node = 1, l = 0, r = this.n - 1) {
    if (l === r) return this.t[node] = this.leaf(v);
    const m = (l + r) >> 1;
    i <= m ? this.set(i, v, node * 2, l, m) : this.set(i, v, node * 2 + 1, m + 1, r);
    this.t[node] = this.merge(this.t[node * 2], this.t[node * 2 + 1]);
  }
  best(ql, qr, node = 1, l = 0, r = this.n - 1) {
    if (ql <= l && r <= qr) return this.t[node].best;
    const m = (l + r) >> 1;
    if (qr <= m) return this.best(ql, qr, node * 2, l, m);
    if (ql > m) return this.best(ql, qr, node * 2 + 1, m + 1, r);
    const L = this.queryNode(ql, m, node * 2, l, m), R = this.queryNode(m + 1, qr, node * 2 + 1, m + 1, r);
    return Math.max(L.best, R.best, L.suf + R.pre);        // 查询也需跨中点拼接
  }
  queryNode(ql, qr, node, l, r) {
    if (ql <= l && r <= qr) return this.t[node];
    const m = (l + r) >> 1;
    if (qr <= m) return this.queryNode(ql, qr, node * 2, l, m);
    if (ql > m) return this.queryNode(ql, qr, node * 2 + 1, m + 1, r);
    return this.merge(this.queryNode(ql, m, node * 2, l, m),
      this.queryNode(m + 1, qr, node * 2 + 1, m + 1, r));
  }
}
```
