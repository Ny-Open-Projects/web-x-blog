## HW-276：手写一个甘特图布局引擎：任务时间轴计算与依赖连线

**答**：
- 需求：实现 GanttLayout：输入任务 [{id, start, end, deps: [id]}]，输出：时间轴范围（min~max 天）与像素映射（dayWidth）；每任务 x/w 坐标；依赖连线（FS 关系：前置结束→后置开始）的折线路径（先竖后横）；行高布局按依赖层级分层防交叉（简化：按拓扑层级）。
- 边界：循环依赖检测（拓扑排序失败报错）；同层任务重叠时行错开；时间轴刻度自适应（天/周/月）。
- 评分点：拓扑分层；折线路径；循环依赖检测。

```javascript
class GanttLayout {
  constructor(tasks, { dayWidth = 30 } = {}) { this.tasks = tasks; this.dw = dayWidth; }
  layout() {
    const { order, levels, cyclic } = topoSort(this.tasks);
    if (cyclic) throw new Error('cyclic-deps');
    const min = Math.min(...this.tasks.map(t => t.start.getTime()));
    const max = Math.max(...this.tasks.map(t => t.end.getTime()));
    const x = t => (t.start.getTime() - min) / 86400000 * this.dw;
    const w = t => (t.end.getTime() - t.start.getTime()) / 86400000 * this.dw;
    const rows = this.tasks.map(t => ({ ...t, x: x(t), w: w(t), row: levels.get(t.id) }));   // 分层防交叉
    const links = this.tasks.flatMap(t => t.deps.map(d => ({
      from: d, to: t.id,
      path: this.linkPath(this.tasks.find(x => x.id === d), t, x, w),
    })));
    return { rows, links, axis: { min, max, unit: this.pickUnit(max - min) } };
  }
  linkPath(from, to, x, w) {                                 // FS 连线：先竖后横
    const x1 = x(from) + w(from), y1 = from.row * ROW_H + ROW_H / 2;
    const x2 = x(to), y2 = to.row * ROW_H + ROW_H / 2;
    const midX = x1 + (x2 > x1 ? 12 : -12);
    return `M${x1},${y1} L${midX},${y1} L${midX},${y2} L${x2},${y2}`;
  }
  pickUnit(ms) { const days = ms / 86400000;
    return days > 365 ? 'month' : days > 60 ? 'week' : 'day'; }
}
function topoSort(tasks) {
  const indeg = new Map(tasks.map(t => [t.id, 0]));
  tasks.forEach(t => t.deps.forEach(d => indeg.set(t.id, indeg.get(t.id) + 1)));
  const levels = new Map(); const q = tasks.filter(t => !indeg.get(t.id));
  q.forEach(t => levels.set(t.id, 0));
  let order = [], cyclic = false;
  while (q.length) {
    const t = q.shift(); order.push(t.id);
    tasks.forEach(x => x.deps.includes(t.id) && indeg.set(x.id, indeg.get(x.id) - 1));
    tasks.forEach(x => { if (indeg.get(x.id) === 0 && !levels.has(x.id)) {
      levels.set(x.id, Math.max(...x.deps.map(d => levels.get(d) ?? 0)) + 1); q.push(x); } });
  }
  if (order.length !== tasks.length) cyclic = true;
  return { order, levels, cyclic };
}
```
