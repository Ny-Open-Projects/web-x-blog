## HW-320：手写一个完整序列化引擎：循环引用、Map、Set 与 Date 还原

**答**：
- 需求：实现 SuperJSON：stringify(value) 处理循环引用（引用表+路径占位）、Map/Set/Date/RegExp/BigInt 特殊类型（type+payload 标注）；parse 还原（按类型标注重建）；undefined 与 NaN 支持（JSON 原生不支持）。
- 边界：循环引用还原后身份一致（同一对象只建一次）；嵌套特殊类型（Map 内含 Date）；key 为数字的 Map。
- 评分点：循环引用身份还原；类型标注协议；往返一致性。

```javascript
function superStringify(v) {
  const seen = new Map();                                    // obj -> refId
  const refs = [];
  function walk(v, path) {
    if (v === undefined) return { __t: 'undef' };
    if (typeof v === 'number' && Number.isNaN(v)) return { __t: 'nan' };
    if (typeof v === 'bigint') return { __t: 'bigint', v: String(v) };
    if (v instanceof Date) return { __t: 'date', v: v.toISOString() };
    if (v instanceof RegExp) return { __t: 'regexp', v: [v.source, v.flags] };
    if (v instanceof Map) return { __t: 'map', v: [...v].map(([k, x]) => [walk(k, path), walk(x, path)]) };
    if (v instanceof Set) return { __t: 'set', v: [...v].map(x => walk(x, path)) };
    if (v && typeof v === 'object') {
      if (seen.has(v)) return { __ref: seen.get(v) };        // 循环/共享引用
      const id = seen.size;
      seen.set(v, id);
      const out = Array.isArray(v) ? [] : {};
      refs.push({ id, out, path });
      for (const [k, x] of Object.entries(v)) out[k] = walk(x, path + '.' + k);
      return out;
    }
    return v;
  }
  const out = walk(v, '$');
  return JSON.stringify({ root: out });
}
function superParse(s) {
  const { root } = JSON.parse(s);
  const built = new Map();                                   // refId -> 实例
  function rebuild(v) {
    if (v?.__t === 'undef') return undefined;
    if (v?.__t === 'nan') return NaN;
    if (v?.__t === 'bigint') return BigInt(v.v);
    if (v?.__t === 'date') return new Date(v.v);
    if (v?.__t === 'regexp') return new RegExp(v.v[0], v.v[1]);
    if (v?.__t === 'map') return new Map(v.v.map(([k, x]) => [rebuild(k), rebuild(x)]));
    if (v?.__t === 'set') return new Set(v.v.map(rebuild));
    if (v && typeof v === 'object') {
      if (v.__ref != null) return built.get(v.__ref);        // 引用还原（身份一致）
      const obj = Array.isArray(v) ? [] : {};
      built.set(obj.__id ?? (obj.__id = built.size), obj);
      for (const [k, x] of Object.entries(v)) obj[k] = rebuild(x);
      return obj;
    }
    return v;
  }
  return rebuild(root);
}
```
