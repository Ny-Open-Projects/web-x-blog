## HW-331：手写一个日志引擎：分级、上下文注入、环形缓冲与导出

**答**：
- 需求：实现 Logger：level 分级（debug/info/warn/error，运行时可调）；上下文注入（requestId/uid/路由自动附加）；环形缓冲（1000 条，error 常驻不淘汰）；导出（全部/按级别/按时间窗）；日志输出到 console 按级别着色 + 批量上报 error 级。
- 边界：环形淘汰保 error（分环）；导出脱敏选项；时间窗导出用于 bug 报告附带。
- 评分点：双环结构；上下文自动注入；脱敏导出。

```javascript
class Logger {
  constructor({ cap = 1000 } = {}) {
    this.level = 'info';
    this.ring = []; this.errorRing = []; this.cap = cap;     // 双环：普通 + error 常驻
    this.ctx = {};
  }
  setContext(ctx) { this.ctx = { ...this.ctx, ...ctx }; }    // 上下文注入
  child(ctx) { const l = Object.create(this); l.ctx = { ...this.ctx, ...ctx }; return l; }   // 子 logger
  log(level, msg, data) {
    if (LEVELS[level] < LEVELS[this.level]) return;          // 级别过滤
    const entry = { t: Date.now(), level, msg, data, ...this.ctx };
    this.ring.push(entry);
    if (this.ring.length > this.cap) this.ring.shift();      // 普通环淘汰
    if (level === 'error') {
      this.errorRing.push(entry);
      if (this.errorRing.length > this.cap) this.errorRing.shift();
      reportError(entry);                                    // error 级上报
    }
    (console[level] ?? console.log)(`[${level}]`, msg, data ?? '');
  }
  debug(m, d) { this.log('debug', m, d); } info(m, d) { this.log('info', m, d); }
  warn(m, d) { this.log('warn', m, d); } error(m, d) { this.log('error', m, d); }
  export({ since, levels, redact = true } = {}) {            // 时间窗导出
    const all = [...this.errorRing, ...this.ring].sort((a, b) => a.t - b.t);
    return all.filter(e => (!since || e.t >= since)
      && (!levels || levels.includes(e.level)))
      .map(e => redact ? { ...e, data: redactPII(e.data) } : e);
  }
}
const LEVELS = { debug: 0, info: 1, warn: 2, error: 3 };
function redactPII(d) { return JSON.parse(JSON.stringify(d ?? {}, (k, v) =>
  /phone|idcard|token/i.test(k) ? '***' : v)); }
```
