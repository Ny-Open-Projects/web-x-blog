**场景**：用户权限点 128 个（bool 数组或 Set 都太重），JWT 里要塞权限——位图 16 字节搞定；判断"是否拥有某权限"O(1)；"两个角色权限的交集/并集"用位运算批量算。

**思路**：权限码 → 位号映射表；Uint8Array 存位；has/and/or 实现；序列化成 base64 塞 JWT。

```javascript
class PermBitmap {
  #bytes;
  constructor(sizeBits = 128, bytes = null) {
    this.size = sizeBits;
    this.#bytes = bytes ?? new Uint8Array(Math.ceil(sizeBits / 8));   // 128 位 = 16 字节
  }
  static #map = new Map();                            // 权限码 -> 位号（注册表）
  static register(code, bit) { this.#map.set(code, bit); }
  grant(code) { const b = PermBitmap.#map.get(code); if (b == null) return false;
    this.#bytes[b >> 3] |= 1 << (b & 7); return true; }
  revoke(code) { const b = PermBitmap.#map.get(code); if (b == null) return false;
    this.#bytes[b >> 3] &= ~(1 << (b & 7)); return true; }
  has(code) { const b = PermBitmap.#map.get(code);
    return b != null && !!(this.#bytes[b >> 3] & (1 << (b & 7))); }   // O(1)
  and(other) {                                        // 交集：两个角色都有的权限
    const r = new PermBitmap(this.size);
    for (let i = 0; i < this.#bytes.length; i++) r.#bytes[i] = this.#bytes[i] & other.#bytes[i];
    return r;
  }
  or(other) { const r = new PermBitmap(this.size);    // 并集
    for (let i = 0; i < this.#bytes.length; i++) r.#bytes[i] = this.#bytes[i] | other.#bytes[i];
    return r; }
  toJWT() { return btoa(String.fromCharCode(...this.#bytes)); }      // 16 字节 → 24 字符 base64
  static fromJWT(s) { const bytes = Uint8Array.from(atob(s), (c) => c.charCodeAt(0));
    return new PermBitmap(bytes.length * 8, bytes); }
}
// 128 权限点：Set ≈ 1KB+ → 位图 16 字节（JWT 友好）；has O(1)；and/or 批量 O(size/8)
// 位号注册表由权限系统统一下发（前后端一致才有意义）
```

**评分点**：
- 位图压缩量级（128 点 16 字节 vs Set 1KB+）与 JWT 场景适配。
- 位运算正确：`b >> 3` 定字节、`b & 7` 定位、置 1/清 0/读。
- and/or 批量运算（角色合并/交集）O(size/8)。
- 位号注册表统一管理（前后端映射一致）的前提说明。
