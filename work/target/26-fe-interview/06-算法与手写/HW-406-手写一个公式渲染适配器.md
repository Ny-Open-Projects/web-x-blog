## HW-406：手写一个公式渲染适配器：LaTeX 子集解析与 DOM 渲染

**答**：
- 需求：实现 MiniLatex：解析子集（上下标 ^ _、分数 \frac{a}{b}、根号 \sqrt{x}、希腊字母 \alpha 等、运算符）；解析为 AST → 渲染为嵌套 span（上下标用 vertical-align，分数用 flex 布局）；光标定位与编辑（渲染结构到文本偏移的映射）。
- 边界：嵌套分数/根号（递归解析）；未知命令原样输出（不崩）；渲染结构与源文本位置映射（编辑回填）；转义（\{ 输出字面花括号）。
- 评分点：递归 AST；分数 flex 渲染；位置映射。

```javascript
function parseLatex(src) {                                   // 子集 AST
  let i = 0;
  function parse() {
    const items = [];
    while (i < src.length) {
      const c = src[i];
      if (c === '\\') {                                     // 命令
        const m = /^\\([a-zA]+|.)/.exec(src.slice(i));
        const name = m[1];
        i += m[0].length;
        if (name === 'frac') { const a = group(), b = group();
          items.push({ t: 'frac', a, b }); continue; }
        if (name === 'sqrt') { items.push({ t: 'sqrt', x: group() }); continue; }
        items.push({ t: 'sym', name }); continue;
      }
      if (c === '^' || c === '_') { i++;
        items.push({ t: c === '^' ? 'sup' : 'sub', x: atom() }); continue; }
      if (c === '{') { items.push({ t: 'group', x: parse() }); continue; }
      if (c === '}') { i++; return items; }
      items.push({ t: 'char', c }); i++;
    }
    return items;
  }
  function atom() { return src[i] === '{' ? (i++, parse()) : [{ t: 'char', c: src[i++] }]; }
  function group() { i++; const x = parse(); return x; }
  return parse();
}
function renderLatex(items, el) {                            // DOM 渲染
  for (const it of items) {
    if (it.t === 'char') el.append(it.c);
    else if (it.t === 'sym') el.append(GREEK[it.name] ?? '\\' + it.name);   // 未知原样输出
    else if (it.t === 'sup' || it.t === 'sub') {
      const span = document.createElement('sup');
      if (it.t === 'sub') span.style.verticalAlign = 'sub';
      renderLatex(it.x, span); el.append(span);
    } else if (it.t === 'frac') {                            // flex 分数
      const frac = document.createElement('span');
      frac.style.cssText = 'display:inline-flex;flex-direction:column;text-align:center;vertical-align:middle;';
      const [a, b] = [document.createElement('span'), document.createElement('span')];
      a.style.borderBottom = '1px solid currentColor';
      renderLatex(it.a, a); renderLatex(it.b, b);
      frac.append(a, b); el.append(frac);
    } else if (it.t === 'sqrt') {
      const s = document.createElement('span');
      s.textContent = '√'; s.style.borderTop = '1px solid currentColor';
      renderLatex(it.x, s); el.append(s);
    } else if (it.t === 'group') renderLatex(it.x, el);
  }
}
const GREEK = { alpha: 'α', beta: 'β', gamma: 'γ', pi: 'π', theta: 'θ', sum: '∑' };
```
