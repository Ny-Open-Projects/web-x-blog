## HW-337：手写一个多语言文案热更新引擎：远程文案包加载与回退

**答**：
- 需求：实现 I18nHot：启动加载当前语言文案包（版本号）；后台轮询版本（变更即拉新包热替换，订阅回调通知组件刷新）；语言切换懒加载对应包；缺 key 回退默认语言包；包加载失败用缓存版本。
- 边界：热替换不刷新页面（引用替换 + 订阅通知）；包体积按需（按命名空间拆分懒加载）；版本比较（semver 或递增号）。
- 评分点：热替换订阅；懒加载命名空间；失败回退缓存。

```javascript
class I18nHot {
  constructor({ fallbackLang = 'en' } = {}) {
    this.lang = localStorage.getItem('lang') ?? navigator.language.slice(0, 2);
    this.packs = new Map(); this.subs = new Set(); this.version = 0;
    this.load(this.lang);
  }
  async load(lang) {
    try {
      const res = await fetch(`/i18n/${lang}.json?v=${this.version}`);
      if (!res.ok) throw new Error('pack-fetch-fail');
      const pack = await res.json();
      if (pack.version < this.version) return;               // 防旧覆盖新
      this.version = pack.version;
      this.packs.set(lang, pack);
      localStorage.setItem(`i18n-${lang}`, JSON.stringify(pack));   // 缓存
      this.subs.forEach(fn => fn(lang));                     // 热替换通知
    } catch {
      const cached = localStorage.getItem(`i18n-${lang}`);
      if (cached) this.packs.set(lang, JSON.parse(cached));  // 失败回退缓存
    }
  }
  async setLang(lang) {
    this.lang = lang; localStorage.setItem('lang', lang);
    if (!this.packs.has(lang)) await this.load(lang);        // 懒加载语言包
    this.subs.forEach(fn => fn(lang));
  }
  t(key, params = {}) {
    let str = getPath(this.packs.get(this.lang), key)
      ?? getPath(this.packs.get(this.fallbackLang), key);    // 回退默认语言
    if (!str) return key;
    return str.replace(/\{(\w+)\}/g, (_, k) => params[k] ?? `{${k}}`);
  }
  onUpdate(fn) { this.subs.add(fn); return () => this.subs.delete(fn); }
  poll() { setInterval(async () => {                         // 版本轮询热更新
    const meta = await fetch('/i18n/version').then(r => r.json());
    if (meta.version > this.version) this.load(this.lang);
  }, 60000); }
}
```
