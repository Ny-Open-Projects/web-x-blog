## HW-350：手写一个连线路由：正交路由与避障简化

**答**：
- 需求：实现 EdgeRouter：节点间连线正交路由（H-V-H / V-H-V 按起终点方位选择）；简单避障（连线穿过的中间节点时绕行——检测段与节点盒相交，加偏移绕过）；锚点计算（两节点相对位置选边缘中点）。
- 边界：同水平/垂直对齐的直线连接；自连线（出右边回右边绕行）；避障偏移步长（16px）可配，最多绕 3 次防死循环。
- 评分点：正交模式选择；段相交检测与绕行；自连线。

```javascript
function segmentsIntersect(a1, a2, b1, b2) {               // 线段相交
  const d = (a2.x - a1.x) * (b2.y - b1.y) - (a2.y - a1.y) * (b2.x - b1.x);
  if (!d) return false;
  const t = ((b1.x - a1.x) * (b2.y - b1.y) - (b1.y - a1.y) * (b2.x - b1.x)) / d;
  const u = ((b1.x - a1.x) * (a2.y - a1.y) - (b1.y - a1.y) * (a2.x - a1.x)) / d;
  return t >= 0 && t <= 1 && u >= 0 && u <= 1;
}
function rectSegIntersect(rect, a, b) {                    // 段与节点盒相交
  const c = [[rect.x, rect.y], [rect.x + rect.w, rect.y], [rect.x + rect.w, rect.y + rect.h], [rect.x, rect.y + rect.h]];
  for (let i = 0; i < 4; i++)
    if (segmentsIntersect(a, b, c[i], c[(i + 1) % 4])) return true;
  return false;
}
class EdgeRouter {
  constructor(nodes, { step = 16 } = {}) { this.nodes = nodes; this.step = step; }
  route(from, to) {
    const a = anchorOf(from, to), b = anchorOf(to, from);   // 锚点：相对边缘中点
    let path = a.x < b.x ? [[a.x, a.y], [(a.x + b.x) / 2, a.y], [(a.x + b.x) / 2, b.y], [b.x, b.y]]
      : [[a.x, a.y], [a.x - this.step, a.y], [a.x - this.step, b.y], [b.x, b.y]];   // H-V-H
    for (let round = 0; round < 3; round++) {               // 最多绕 3 次
      const blockers = this.nodes.filter(n => n !== from && n !== to
        && path.some((p, i) => i && rectSegIntersect(bounds(n), path[i - 1], p)));
      if (!blockers.length) break;
      const b0 = bounds(blockers[0]);
      path = path.map(p => p[0] > b0.x && p[0] < b0.x + b0.w
        ? [p[0], p[1] < b0.y ? b0.y - this.step : b0.y + b0.h + this.step] : p);   // 绕行偏移
    }
    return path;
  }
  selfLoop(node) {                                          // 自连线
    const b = bounds(node);
    const x = b.x + b.w, y = b.y + b.h / 3;
    return [[x, y], [x + 30, y], [x + 30, y + 30], [x, y + 30]];
  }
}
function anchorOf(n, toward) {
  const b = bounds(n);
  const cx = b.x + b.w / 2, cy = b.y + b.h / 2;
  if (Math.abs(cx - toward.x) > Math.abs(cy - toward.y))    // 水平主导：左右边中点
    return { x: cx < toward.x ? b.x + b.w : b.x, y: cy };
  return { x: cx, y: cy < toward.y ? b.y + b.h : b.y };     // 垂直主导
}
```
