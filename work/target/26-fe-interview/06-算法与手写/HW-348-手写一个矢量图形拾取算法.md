## HW-348：手写一个矢量图形拾取算法：点在多边形内与路径包含

**答**：
- 需求：实现 HitTest：矩形/圆/多边形/贝塞尔路径的包含判断——多边形用射线法（交叉计数），圆用距离，路径用 ctx.isPointInPath 兜底；返回是否命中与命中深度（供层级选择）。
- 边界：射线法处理顶点重复与水平边（standard even-odd 规则）；旋转图形先逆变换到局部坐标再测；线宽命中（描边区域也算，isPointInStroke）。
- 评分点：射线法 even-odd；逆变换前置；isPointInPath 兜底。

```javascript
function pointInPolygon(pt, poly) {                        // 射线法 even-odd
  const [x, y] = pt;
  let inside = false;
  for (let i = 0, j = poly.length - 1; i < poly.length; j = i++) {
    const [xi, yi] = poly[i], [xj, yj] = poly[j];
    if ((yi > y) !== (yj > y)
      && x < (xj - xi) * (y - yi) / (yj - yi) + xi) inside = !inside;   // 交叉计数
  }
  return inside;
}
function hitShape(shape, px, py) {                         // 分类型判定
  if (shape.type === 'rect')
    return px >= shape.x && px <= shape.x + shape.w && py >= shape.y && py <= shape.y + shape.h;
  if (shape.type === 'circle') {
    const dx = px - shape.cx, dy = py - shape.cy;
    return dx * dx + dy * dy <= shape.r * shape.r;         // 距离判定
  }
  if (shape.type === 'poly') return pointInPolygon([px, py], shape.points);
  return null;                                             // 兜底走 isPointInPath
}
function hitRotated(shape, px, py) {                       // 旋转图形：先逆变换
  const cx = shape.x + shape.w / 2, cy = shape.y + shape.h / 2;
  const cos = Math.cos(-shape.rotation), sin = Math.sin(-shape.rotation);
  const dx = px - cx, dy = py - cy;
  const local = { x: cx + dx * cos - dy * sin, y: cy + dx * sin + dy * cos };
  return hitShape({ ...shape, rotation: 0 }, local.x, local.y);
}
function hitPath(ctx, path2d, px, py, { strokeWidth = 0 } = {}) {
  return ctx.isPointInPath(path2d, px, py)                 // 路径包含兜底
    || (strokeWidth > 0 && ctx.isPointInStroke(path2d, px, py));   // 描边区域
}
```
