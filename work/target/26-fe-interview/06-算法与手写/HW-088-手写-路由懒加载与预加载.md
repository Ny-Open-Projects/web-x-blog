**答**：懒加载路由：handler 返回 `() => import('./page')` 动态加载。增强：① 加载结果缓存（重复进入不重复请求）；② 空闲预取（当前路由稳定后 idle 时 import 下一路由 chunk，预热 HTTP 缓存）；③ 失败重试一次（弱网 chunk 加载失败重载）。

```javascript
class LazyRouter {
  constructor() { this.routes = new Map(); this.cache = new Map(); }
  register(path, loader) { this.routes.set(path, loader); }
  load(path) {
    if (this.cache.has(path)) return this.cache.get(path);          // 已加载复用
    const promise = this.#withRetry(this.routes.get(path), 1)
      .then((mod) => { this.cache.set(path, Promise.resolve(mod)); return mod; });
    this.cache.set(path, promise);                                   // 缓存在途 Promise
    return promise;
  }
  async #withRetry(loader, retries) {
    try { return await loader(); }
    catch (e) {
      if (retries <= 0) throw e;
      await new Promise((r) => setTimeout(r, 300));                 // 弱网 chunk 重试
      return this.#withRetry(loader, retries - 1);
    }
  }
  prefetch(path) {                                                  // 空闲预取
    (window.requestIdleCallback || ((f) => setTimeout(f, 200)))(() => {
      this.load(path).catch(() => {});                              // 预取失败静默
    }, { timeout: 5000 });
  }
  async navigate(path) {
    const hit = [...this.routes.keys()].find((r) => path.startsWith(r));
    if (!hit) throw new Error('no route');
    const mod = await this.load(hit);
    mod.default.render();                                            // chunk 就绪后渲染
  }
}
// 用法：进入 /goods 后空闲预取 /pay
router.register('/goods', () => import('./pages/goods.js'));
router.register('/pay', () => import('./pages/pay.js'));
await router.navigate('/goods');
router.prefetch('/pay');
// 关键点：cache 缓存的是 Promise（在途也能命中），chunk 网络失败自动重试一次
```

**评分点**：
- 缓存的是 Promise 而非模块（在途重复导航不重复发请求）。
- 失败重试有限次（1 次），避免无限循环。
- 空闲预取（rIC + timeout）提前拉 chunk，点进时几乎零等待。
- 预取失败静默不弹错。
