## HW-340：手写一个插件系统：生命周期钩子、隔离与依赖声明

**答**：
- 需求：实现 PluginSystem：register(plugin {name, version, depends: [name], hooks: {init/ready/destroy}})；依赖拓扑排序决定加载顺序（缺失依赖报错）；钩子调用（生命周期：init→ready→destroy，销毁逆序）；插件间通信走受控 API（沙箱：插件只能调 host 提供的接口）。
- 边界：版本兼容检查（semver 区间）；钩子异常隔离（单插件失败不影响 host 与其他插件）；destroy 逆序释放资源。
- 评分点：拓扑加载序；异常隔离；逆序销毁。

```javascript
class PluginSystem {
  constructor(api) { this.plugins = new Map(); this.active = []; this.api = api; }   // api: 沙箱接口
  register(plugin) {
    for (const dep of plugin.depends ?? []) {
      const d = this.plugins.get(dep);
      if (!d) throw new Error('missing-dep: ' + dep);
      if (plugin.semver && !satisfies(d.version, plugin.semver))
        throw new Error('version-conflict: ' + dep);         // 版本兼容
    }
    this.plugins.set(plugin.name, plugin);
  }
  async boot() {
    const order = topoSort([...this.plugins.values()]);      // 依赖拓扑序
    for (const p of order) {
      try {
        await p.hooks?.init?.(this.api);                     // 沙箱 API
        await p.hooks?.ready?.(this.api);
        this.active.push(p);
      } catch (e) { report('plugin-fail', { name: p.name, error: String(e) }); }   // 异常隔离
    }
  }
  async shutdown() {
    for (const p of [...this.active].reverse()) {            // 逆序销毁
      try { await p.hooks?.destroy?.(); } catch { /* 隔离 */ }
    }
    this.active = [];
  }
}
function topoSort(plugins) {
  const done = [], seen = new Set();
  const visit = p => {
    if (seen.has(p.name)) return;
    seen.add(p.name);
    (p.depends ?? []).forEach(d => { const dp = plugins.find(x => x.name === d); dp && visit(dp); });
    done.push(p);
  };
  plugins.forEach(visit);
  return done;
}
function satisfies(version, range) { return range === '*' || version.startsWith(range.replace(/[~^]/, '').split('.').slice(0, 2).join('.')); }
```
