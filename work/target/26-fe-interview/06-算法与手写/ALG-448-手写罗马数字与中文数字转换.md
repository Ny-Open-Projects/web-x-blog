## ALG-448：手写罗马数字与中文数字转换：账单文案

**答**：
- 题目：账单大写金额（壹佰贰拾叁元）与数字互转。中文数字：四位一组（万/亿进制），组内按十百千读。
- 思路：数字→中文：按 4 位分节（亿/万/元），节内按千/百/十/个转换，处理零的读法（连续零读一个、节间补零）；罗马数字：贪心查表（1000→M, 900→CM...）。
- 评分点：分节进位；零的读法规则；罗马数字贪心查表。

```javascript
function toChineseAmount(n) {                              // 数字 → 中文大写
  const digits = '零壹贰叁肆伍陆柒捌玖';
  const units = ['', '拾', '佰', '仟'];
  const bigUnits = ['', '万', '亿'];
  if (n === 0) return '零元';
  let s = '', section = 0;
  while (n > 0) {
    const sec = n % 10000;                                 // 四位分节
    if (sec) {
      let part = '', v = sec, zero = false;
      for (let u = 0; v > 0; u++, v = Math.floor(v / 10)) {
        const d = v % 10;
        if (d === 0) zero = true;
        else { if (zero && part) part = digits[0] + part; zero = false;
          part = digits[d] + units[u] + part; }
      }
      s = part + (bigUnits[section] ?? '') + (section && sec < 1000 ? '零' : '') + s;
    } else if (s && !s.startsWith('零')) s = '零' + s;
    n = Math.floor(n / 10000); section++;
  }
  return s + '元';
}
function toRoman(num) {                                    // 罗马数字贪心
  const map = [[1000, 'M'], [900, 'CM'], [500, 'D'], [400, 'CD'], [100, 'C'],
    [90, 'XC'], [50, 'L'], [40, 'XL'], [10, 'X'], [9, 'IX'], [5, 'V'], [4, 'IV'], [1, 'I']];
  let out = '';
  for (const [v, sym] of map) while (num >= v) { out += sym; num -= v; }
  return out;
}
```
