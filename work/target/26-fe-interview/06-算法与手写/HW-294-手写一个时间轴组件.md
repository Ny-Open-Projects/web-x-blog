## HW-294：手写一个时间轴组件：刻度生成与缩放自适应

**答**：
- 需求：实现 Timeline：时间范围与像素宽度映射（线性）；刻度生成（按当前缩放选择单位：分钟/小时/天/月，刻度间距 >= 60px 自适应）；拖拽缩放（以鼠标为中心缩放，时间锚点保持）；当前位置指示线。
- 边界：缩放锚点计算（鼠标处时间不变）；刻度密度上限（过密时升单位）；非线性时间轴（工作日时间轴）不做（标注取舍）。
- 评分点：锚点缩放；密度自适应；线性映射。

```javascript
class Timeline {
  constructor({ range, width, minPxPerTick = 60 } = {}) {
    this.t0 = range[0].getTime(); this.t1 = range[1].getTime();
    this.width = width; this.minPx = minPxPerTick;
  }
  x(time) { return (time.getTime() - this.t0) / (this.t1 - this.t0) * this.width; }   // 线性映射
  timeAt(x) { return new Date(this.t0 + x / this.width * (this.t1 - this.t0)); }
  ticks() {                                                  // 自适应单位
    const spanMs = this.t1 - this.t0;
    const pxPerMs = this.width / spanMs;
    const units = [
      ['minute', 60000], ['hour', 3600000], ['day', 86400000],
      ['week', 604800000], ['month', 2629800000],
    ];
    const unit = units.find(([_, ms]) => ms * pxPerMs >= this.minPx) ?? units.at(-1);   // 密度上限
    const out = [];
    let t = new Date(this.t0); t = floorTo(t, unit[0]);
    while (t.getTime() <= this.t1) {
      out.push({ time: t, x: this.x(t), label: fmt(t, unit[0]) });
      t = advance(t, unit[0]);
    }
    return out;
  }
  zoomAt(px, factor) {                                       // 以鼠标为中心缩放
    const anchor = this.timeAt(px);                          // 锚点时间不变
    const center = anchor.getTime();
    const half = (this.t1 - this.t0) / 2 / factor;
    this.t0 = center - half * (px / this.width * 2 - 1);
    this.t1 = center + half * (1 - px / this.width * 2);
    return [new Date(this.t0), new Date(this.t1)];
  }
}
function floorTo(d, unit) { const x = new Date(d);
  if (unit === 'hour') x.setMinutes(0, 0, 0);
  else if (unit === 'day') x.setHours(0, 0, 0, 0);
  else if (unit === 'week') { x.setHours(0, 0, 0, 0); x.setDate(x.getDate() - x.getDay()); }
  else if (unit === 'month') { x.setHours(0, 0, 0, 0); x.setDate(1); }
  return x; }
function advance(d, unit) { const x = new Date(d);
  if (unit === 'minute') x.setMinutes(x.getMinutes() + 1);
  else if (unit === 'hour') x.setHours(x.getHours() + 1);
  else if (unit === 'day') x.setDate(x.getDate() + 1);
  else if (unit === 'week') x.setDate(x.getDate() + 7);
  else x.setMonth(x.getMonth() + 1);
  return x; }
function fmt(d, unit) { return unit === 'minute' ? d.toISOString().slice(11, 16)
  : unit === 'hour' ? d.toISOString().slice(11, 13) + ':00' : d.toISOString().slice(0, 10); }
```
