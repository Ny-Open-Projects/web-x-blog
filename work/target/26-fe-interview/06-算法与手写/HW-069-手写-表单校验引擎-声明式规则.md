**答**：校验引擎三要素：① 声明式规则（每个字段一组 validators，返回 true 或错误消息）；② 联动（依赖字段变化时重校验相关字段，用 deps 图）；③ 异步校验（唯一性检查）带防抖 + 竞态丢弃；④ 提交时全量校验聚合错误，单字段 blur 时增量校验。

```javascript
class FormEngine {
  constructor(rules) {
    this.rules = rules;                    // { field: [{ test, msg, deps, async }] }
    this.values = {}; this.errors = {}; this.asyncSeq = new Map();
  }
  validateField(field, trigger = 'change') {
    const rs = this.rules[field] || [];
    for (const r of rs) {
      if (r.when && !r.when(this.values)) continue;          // 条件规则
      const v = r.test(this.values[field], this.values);     // 联动：拿到全量 values
      if (v !== true) return (this.errors[field] = typeof v === 'string' ? v : r.msg), false;
      if (r.async) return this.#validateAsync(field, r);     // 同步规则全过才跑异步
    }
    delete this.errors[field]; return true;
  }
  #validateAsync(field, rule) {
    const seq = (this.asyncSeq.get(field) || 0) + 1;         // 竞态丢弃
    this.asyncSeq.set(field, seq);
    this.errors[field] = 'checking...';
    return rule.test(this.values[field], this.values).then((ok) => {
      if (this.asyncSeq.get(field) !== seq) return;          // 过期响应丢弃
      ok ? delete this.errors[field] : (this.errors[field] = rule.msg);
      return ok;
    });
  }
  validateAll() {                                            // 提交：全量并行
    const results = Object.keys(this.rules).map((f) => this.validateField(f, 'submit'));
    return Promise.all(results).then((rs) => {
      const firstError = Object.keys(this.rules).find((f) => this.errors[f] && this.errors[f] !== 'checking...');
      return { valid: rs.every(Boolean), errors: { ...this.errors }, focusField: firstError };
    });
  }
  setValue(field, value) {
    this.values[field] = value;
    this.validateField(field);
    for (const f of Object.keys(this.rules))                 // 联动：deps 反查
      if ((this.rules[f] || []).some((r) => r.deps?.includes(field))) this.validateField(f);
  }
}
// 用法
const form = new FormEngine({
  password: [{ test: (v) => v.length >= 8 || '至少 8 位' }],
  confirm:  [{ test: (v, all) => v === all.password || '两次不一致', deps: ['password'] }],
  email:    [{ test: (v) => /^\S+@\S+$/.test(v) || '邮箱格式错误' },
             { msg: '邮箱已被注册', async: true, test: (v) => fetch(`/api/check?email=${v}`).then(r => r.json()).then(d => d.ok) }],
});
```

**评分点**：
- 规则返回 true 或错误消息的协议设计清晰；条件规则（when）支持。
- 联动正确：deps 反查 + validateField 拿全量 values（confirm 依赖 password 场景能跑通）。
- 异步校验竞态丢弃（seq 比对），"checking" 中间态不判失败。
- validateAll 返回 focusField（第一个错误字段）便于提交后聚焦。
