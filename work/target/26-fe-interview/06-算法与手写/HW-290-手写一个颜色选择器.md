## HW-290：手写一个颜色选择器：HSV 面板交互与格式转换

**答**：
- 需求：实现 ColorPicker：HSV 面板（x=饱和度，y=明度）+ 色相条；面板点击/拖拽更新 SV（指针位置→比例）；HSV↔RGB↔HEX 转换函数；输出多格式值（hex/rgb/hsl）与历史色板。
- 边界：拖拽出边界钳制到 0~1；色相 0 与 360 同色（归一到 359）；hex 解析容错（#abc 缩写）。
- 评分点：HSV 交互映射；三格式转换正确性；边界钳制。

```javascript
class ColorPicker {
  constructor(el) {
    this.h = 0; this.s = 1; this.v = 1;
    el.addEventListener('pointerdown', e => this.pick(e));
  }
  pick(e) {                                                  // 指针位置 → SV
    const r = e.currentTarget.getBoundingClientRect();
    this.s = clamp01((e.clientX - r.left) / r.width);
    this.v = 1 - clamp01((e.clientY - r.top) / r.height);    // y 反向：上亮下暗
    this.onChange?.(this.value());
  }
  hue(h) { this.h = ((h % 360) + 360) % 360; this.onChange?.(this.value()); }
  value() {
    const rgb = hsv2rgb(this.h, this.s, this.v);
    return { rgb, hex: rgb2hex(rgb),
      hsl: rgb2hsl(rgb.r, rgb.g, rgb.b) };
  }
}
function hsv2rgb(h, s, v) {
  const c = v * s, x = c * (1 - Math.abs((h / 60) % 2 - 1)), m = v - c;
  const [r, g, b] = [[c, x, 0], [x, c, 0], [0, c, x], [0, x, c], [x, 0, c], [c, 0, x]][Math.floor(h / 60) % 6];
  return { r: Math.round((r + m) * 255), g: Math.round((g + m) * 255), b: Math.round((b + m) * 255) };
}
function rgb2hex({ r, g, b }) { return '#' + [r, g, b].map(x => x.toString(16).padStart(2, '0')).join(''); }
function rgb2hsl(r, g, b) {
  r /= 255; g /= 255; b /= 255;
  const max = Math.max(r, g, b), min = Math.min(r, g, b), l = (max + min) / 2;
  const d = max - min;
  const s = d === 0 ? 0 : d / (1 - Math.abs(2 * l - 1));
  const h = d === 0 ? 0 : max === r ? 60 * (((g - b) / d) % 6)
    : max === g ? 60 * ((b - r) / d + 2) : 60 * ((r - g) / d + 4);
  return { h: (h + 360) % 360, s, l };
}
const clamp01 = v => Math.max(0, Math.min(1, v));
```
