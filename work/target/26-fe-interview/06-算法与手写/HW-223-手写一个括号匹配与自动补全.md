## HW-223：手写一个括号匹配与自动补全：插入配对与跳转

**答**：
- 需求：实现 BracketHelper：onKey(key)——输入左括号自动补右括号（光标居中）；输入右括号且右侧已是同款则跳过（防止重复）；Backspace 删除左括号时若右侧紧邻配对右括号则一起删；Ctrl+\ 在括号间跳转。
- 边界：引号补全需判断前一字符是否为词字符（防补进单词）；嵌套括号跳转用计数配对。
- 评分点：跳过式补全；配对删除；计数跳转。

```javascript
const PAIRS = { '(': ')', '[': ']', '{': '}', '"': '"', "'": "'" };
class BracketHelper {
  constructor(editor) { this.ed = editor; }
  onKey(key) {
    const { text, pos } = this.ed;
    if (PAIRS[key]) {
      const prev = text[pos - 1] ?? '';
      if ((key === '"' || key === "'") && /\w/.test(prev)) return 'insert-raw';   // 词内引号不补
      if (text[pos] === PAIRS[key]) { this.ed.move(1); return 'skip'; }            // 已有：跳过
      this.ed.insert(key + PAIRS[key]); this.ed.move(-1); return 'auto-pair';
    }
    if (Object.values(PAIRS).includes(key) && text[pos] === key) { this.ed.move(1); return 'skip'; }
    if (key === 'Backspace') {
      if (PAIRS[text[pos - 1]] === text[pos]) { this.ed.delete(2); return 'pair-delete'; }
      return 'delete';
    }
    if (key === 'Ctrl-\\') return this.jump();
    return 'default';
  }
  jump() {                                                     // 计数配对跳转
    const { text, pos } = this.ed;
    const open = text[pos - 1], close = text[pos];
    if (PAIRS[open]) {                                         // 向前找匹配右
      let d = 0;
      for (let i = pos; i < text.length; i++) {
        if (text[i] === open) d++;
        else if (text[i] === PAIRS[open] && --d === 0) { this.ed.move(i - pos + 1); return 'jump'; }
      }
    } else if (Object.values(PAIRS).includes(close)) {         // 向后找匹配左
      let d = 0;
      for (let i = pos - 1; i >= 0; i--) {
        if (text[i] === close) d++;
        else if (Object.entries(PAIRS).some(([k, v]) => v === close && text[i] === k) && --d === 0)
          { this.ed.move(i - pos); return 'jump'; }
      }
    }
    return 'none';
  }
}
```
