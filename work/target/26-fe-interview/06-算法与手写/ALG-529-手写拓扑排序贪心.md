## ALG-529：手写拓扑排序+贪心：构建依赖调度

**答**：
- 题目：构建任务有依赖（DAG），相同优先级下「先做解锁更多下游的」（贪心：下游数排序）——拓扑排序 + 优先队列（下游计数为堆权重）。
- 思路：入度拓扑；堆按「解锁下游数」降序出队；完成时解锁下游入堆。
- 评分点：拓扑 + 优先级堆；下游计数预计算；与纯拓扑顺序的差异。

```javascript
function scheduleBuild(tasks) {                            // tasks: [{id, deps}]
  const indeg = new Map(), downstream = new Map();
  for (const t of tasks) {
    indeg.set(t.id, indeg.get(t.id) ?? 0);
    for (const d of t.deps ?? []) {
      indeg.set(t.id, indeg.get(t.id) + 1);
      downstream.set(d, [...(downstream.get(d) ?? []), t.id]);
    }
  }
  const unlockCount = new Map();                           // 下游计数（贪心权重）
  for (const t of tasks)
    unlockCount.set(t.id, (downstream.get(t.id) ?? []).length);
  const heap = new MaxHeapKey([...tasks.filter(t => !indeg.get(t.id))]);
  const order = [];
  while (heap.size) {
    const t = heap.pop();                                  // 解锁下游最多者优先
    order.push(t.id);
    for (const d of downstream.get(t.id) ?? []) {
      indeg.set(d, indeg.get(d) - 1);
      if (indeg.get(d) === 0) {
        const task = tasks.find(x => x.id === d);
        heap.push(task);
      }
    }
  }
  return order.length === tasks.length ? order : null;     // 环检测
}
class MaxHeapKey {
  constructor(arr) { this.a = [...arr]; this.key = t => unlockCount.get(t.id) ?? 0;
    for (let i = this.a.length >> 1; i >= 0; i--) this.down(i); }
  get size() { return this.a.length; }
  pop() { const t = this.a[0], l = this.a.pop();
    if (this.a.length) { this.a[0] = l; this.down(0); } return t; }
  push(v) { this.a.push(v); let i = this.a.length - 1;
    while (i > 0) { const p = (i - 1) >> 1;
      if (this.key(this.a[p]) >= this.key(this.a[i])) break;
      [this.a[p], this.a[i]] = [this.a[i], this.a[p]]; i = p; } }
  down(i) { while (true) { let m = i;
    for (const c of [2 * i + 1, 2 * i + 2])
      if (c < this.a.length && this.key(this.a[c]) > this.key(this.a[m])) m = c;
    if (m === i) break;
    [this.a[m], this.a[i]] = [this.a[i], this.a[m]]; i = m; } }
}
```
