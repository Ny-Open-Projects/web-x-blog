## HW-302：手写一个表达式求值引擎：四则运算、变量与函数调用

**答**：
- 需求：实现 ExprEval：支持 + - * / % 括号、一元负号、变量引用、白名单函数调用（max/min/round）；实现：分词器 → 递归下降解析（expr→term→factor 层级）→ 求值；变量从 context 取（未定义报错）。
- 边界：运算优先级正确（乘除高于加减）；除零返回 Infinity（JS 语义）；函数仅白名单（防任意调用）；表达式长度上限防 ReDoS/深递归。
- 评分点：递归下降层级；白名单函数；深度上限。

```javascript
const FUNCS = { max: Math.max, min: Math.min, round: Math.round };
function tokenize(s) {
  const re = /\s*(\d+\.?\d*|[A-Za-z_]\w*|[()+\-*/%&,])/y;
  const out = []; let m, i = 0;
  while (i < s.length && (m = re.exec(s.slice(i)))) { out.push(m[1]); i += m[0].length; }
  return out;
}
function exprEval(src, ctx = {}) {
  const t = tokenize(src); let p = 0;
  const MAXD = 100; let depth = 0;
  const peek = () => t[p];
  const eat = x => { if (t[p] === x) { p++; return true; } return false; };
  function expr() {                                          // 加减层
    if (++depth > MAXD) throw new Error('too-deep');
    let v = term();
    while (peek() === '+' || peek() === '-') v = peek() === '+' ? v + term() : v - term();
    depth--; return v;
  }
  function term() {                                          // 乘除层
    let v = factor();
    while (peek() === '*' || peek() === '/' || peek() === '%') {
      const op = t[p++]; const r = factor();
      v = op === '*' ? v * r : op === '/' ? v / r : v % r;
    }
    return v;
  }
  function factor() {
    if (eat('-')) return -factor();                          // 一元负号
    if (eat('(')) { const v = expr(); eat(')'); return v; }
    if (/^\d/.test(peek() ?? '')) return parseFloat(t[p++]);
    const name = t[p++];                                     // 变量或函数
    if (eat('(')) {                                          // 函数调用
      if (!FUNCS[name]) throw new Error('func-not-allowed: ' + name);
      const args = [];
      if (!eat(')')) { do { args.push(expr()); } while (eat(',')); eat(')'); }
      return FUNCS[name](...args);
    }
    if (!(name in ctx)) throw new Error('undefined-var: ' + name);
    return ctx[name];
  }
  const v = expr();
  if (p < t.length) throw new Error('unexpected: ' + t[p]);
  return v;
}
```
