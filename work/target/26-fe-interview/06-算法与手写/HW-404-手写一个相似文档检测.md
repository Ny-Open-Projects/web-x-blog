## HW-404：手写一个相似文档检测：SimHash 指纹与去重

**答**：
- 需求：实现 SimHash：文档指纹（分词 → 每词哈希 → 加权累加（正负位）→ 生成 64 位指纹）；相似判定（汉明距离 <= 3 视为近似重复）；批量去重（新文档与已有指纹比对，距离阈值内标记重复）；指纹索引（按位分段倒排加速近邻查找）。
- 边界：短文本不稳定（< 50 字加长指纹容差或跳过）；汉明距离计算（异或+popcount）；分段索引（16 位 × 4 段，任一段完全匹配才细比——鸽笼原理）。
- 评分点：SimHash 位累加；汉明距离；分段近邻加速。

```javascript
function simhash(text, bits = 64) {
  const terms = tokenize(text);
  if (terms.length < 5) return null;                         // 短文本不稳定
  const v = new Array(bits).fill(0);
  const tf = new Map();
  for (const t of terms) tf.set(t, (tf.get(t) ?? 0) + 1);
  for (const [term, weight] of tf) {
    const h = BigInt('0x' + md5Hex(term).slice(0, 16));      // 词哈希
    for (let i = 0; i < bits; i++)
      v[i] += (h >> BigInt(i)) & 1n ? weight : -weight;     // 加权位累加
  }
  let fp = 0n;
  for (let i = 0; i < bits; i++) if (v[i] > 0) fp |= 1n << BigInt(i);
  return fp;                                                // 64 位指纹
}
function hamming(a, b) {                                     // 汉明距离
  let x = a ^ b, d = 0;
  while (x) { d += Number(x & 1n); x >>= 1n; }
  return d;
}
class DupDetector {
  constructor({ threshold = 3 } = {}) { this.fpIndex = new Map(); this.threshold = threshold; }
  index(id, text) {
    const fp = simhash(text);
    if (fp == null) return { indexed: false, reason: 'too-short' };
    for (let seg = 0; seg < 4; seg++) {                      // 16 位 × 4 段索引
      const part = (fp >> BigInt(seg * 16)) & 0xffffn;
      const k = seg + ':' + part;
      (this.fpIndex.get(k) ?? this.fpIndex.set(k, []).get(k)).push({ id, fp });
    }
    return { indexed: true };
  }
  check(text) {                                              // 近邻查找：分段粗筛+细比
    const fp = simhash(text);
    if (fp == null) return { dup: false };
    const candidates = new Set();
    for (let seg = 0; seg < 4; seg++) {
      const part = (fp >> BigInt(seg * 16)) & 0xffffn;
      for (const { id } of this.fpIndex.get(seg + ':' + part) ?? []) candidates.add(id);   // 鸽笼：至少一段全匹配
    }
    for (const id of candidates)
      if (hamming(fp, this.fpOf(id)) <= this.threshold) return { dup: true, of: id };   // 细比
    return { dup: false };
  }
  fpOf(id) { for (const list of this.fpIndex.values())
    for (const e of list) if (e.id === id) return e.fp; return null; }
}
```
