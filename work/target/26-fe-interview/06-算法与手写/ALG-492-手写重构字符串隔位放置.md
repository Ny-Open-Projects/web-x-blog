## ALG-492：手写重构字符串（隔位放置）：验证码防连击

**答**：
- 题目：验证码字符频次已知，重排使「相邻字符不同」（防连击模式），不可行返回空。贪心：按频次降序隔位放置。
- 思路：可行性：最大频次 <= (n+1)/2；放置：偶位（0,2,4...）先放最高频，奇位放次频。
- 评分点：可行性判定；隔位放置顺序；堆实现（每次取频次最高两个）。

```javascript
function reorganizeString(counts) {                        // counts: {char: freq}
  const total = Object.values(counts).reduce((s, v) => s + v, 0);
  const maxFreq = Math.max(...Object.values(counts));
  if (maxFreq > Math.ceil(total / 2)) return '';           // 不可行
  const heap = new MaxHeap(Object.entries(counts).map(([c, f]) => ({ c, f })));
  const out = [];
  let prev = null;
  while (heap.size) {
    const top = heap.pop();
    out.push(top.c);
    if (prev) heap.push(prev);                             // 上一个用完回堆
    top.f--;
    prev = top.f > 0 ? top : null;
  }
  return out.join('');
}
class MaxHeap {
  constructor(arr) { this.a = [...arr];
    for (let i = this.a.length >> 1; i >= 0; i--) this.down(i); }
  get size() { return this.a.length; }
  pop() { const t = this.a[0], l = this.a.pop();
    if (this.a.length) { this.a[0] = l; this.down(0); } return t; }
  push(v) { this.a.push(v); let i = this.a.length - 1;
    while (i > 0) { const p = (i - 1) >> 1;
      if (this.a[p].f >= this.a[i].f) break;
      [this.a[p], this.a[i]] = [this.a[i], this.a[p]]; i = p; } }
  down(i) { while (true) { let m = i;
    for (const c of [2 * i + 1, 2 * i + 2])
      if (c < this.a.length && this.a[c].f > this.a[m].f) m = c;
    if (m === i) break;
    [this.a[m], this.a[i]] = [this.a[i], this.a[m]]; i = m; } }
}
```
