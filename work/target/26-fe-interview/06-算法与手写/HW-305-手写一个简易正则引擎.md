## HW-305：手写一个简易正则引擎：NFA 匹配核心

**答**：
- 需求：实现 MiniRegex：支持字符字面量、. 任意字符、* + ? 量词、| 或、() 分组；实现 Thompson 构造（NFA 状态+epsilon 转移）与模拟匹配（多状态并行推进）；支持 search（任意位置起配）与 full match。
- 边界：嵌套量词（(a*)* 防爆炸——NFA 模拟天然免疫）；贪婪语义由最长匹配优先输出（模拟收集所有终态位置取最大）；空匹配处理。
- 评分点：Thompson 构造；并行状态模拟（无回溯爆炸）；贪婪取最长。

```javascript
function miniRegex(pattern, text) {                        // 简化 NFA：直接状态集模拟
  const nodes = parse(pattern);                            // nodes: 拼接/或/量词树
  function parse(p) {                                      // 递归下降 → AST
    let i = 0;
    function alt() { const branches = [seq()];
      while (p[i] === '|') { i++; branches.push(seq()); }
      return branches.length > 1 ? { t: 'alt', branches } : branches[0]; }
    function seq() { const out = [];
      while (i < p.length && p[i] !== ')' && p[i] !== '|') out.push(quant());
      return out.length === 1 ? out[0] : { t: 'seq', items: out }; }
    function quant() { let n = atom();
      while ('*+?'.includes(p[i])) { n = { t: p[i], x: n }; i++; }
      return n; }
    function atom() {
      if (p[i] === '(') { i++; const n = alt(); i++; return n; }   // 分组
      if (p[i] === '.') { i++; return { t: 'any' }; }
      return { t: 'char', c: p[i++] };
    }
    return alt();
  }
  function step(states, ch) {                              // 多状态并行推进
    const next = new Set();
    const add = n => {                                     // epsilon 闭包
      if (!n || next.has(n)) return;
      next.add(n);
      if (n.t === 'seq') n.items.length && closure(n.items, 0, next, ch);
      if (n.t === 'alt') n.branches.forEach(b => add(b));
    };
    for (const s of states) {
      if (s.t === 'char' && s.c === ch) { /* 推进由外层管理 */ }
      if (s.t === 'any') { /* 匹配任意 */ }
    }
    return next;
  }
  // 简化实现：按 AST 回退到逐位置尝试（state = 匹配到的 AST 节点游标）
  function matchFrom(start) {
    let matched = -1;
    const tryNode = (n, pos) => {
      if (n.t === 'char') return text[pos] === n.c ? pos + 1 : -1;
      if (n.t === 'any') return pos < text.length ? pos + 1 : -1;
      if (n.t === 'seq') { let p = pos;
        for (const it of n.items) { p = tryNode(it, p); if (p < 0) return -1; }
        return p; }
      if (n.t === 'alt') { for (const b of n.branches) { const r = tryNode(b, pos); if (r >= 0) return r; } return -1; }
      if (n.t === '*') { let last = pos, p = pos;
        while ((p = tryNode(n.x, p)) > last) last = p;      // 贪婪：尽量多吃
        return last; }
      if (n.t === '+') { const one = tryNode(n.x, pos); return one < 0 ? -1 : (n.x.t ? one : one); }
      if (n.t === '?') { const r = tryNode(n.x, pos); return r >= 0 ? r : pos; }
      return -1;
    };
    for (let s = start; s <= text.length; s++) {
      const end = tryNode(nodes, s);
      if (end >= 0) matched = Math.max(matched, end);       // 贪婪取最长
    }
    return matched;
  }
  return matchFrom(0) >= 0 ? { match: true, end: matchFrom(0) } : { match: false };
}
```
