**场景**：zh-TW 缺某键 → 回退 zh → 还缺 → 回退 en → 还缺 → 显示 key 本身（调试可见）。回退链可配置（每语言自己的链）+ 缺失统计（哪些键在哪些语言缺失——翻译补齐清单）。

**思路**：t(key) 沿链查找 → 命中返回；全链 miss → 返回 key + 记缺失（按语言×键统计，CI 报表 ALG-242 联动）。

```javascript
function createT(bundles, { fallbacks = { 'zh-TW': ['zh-TW', 'zh', 'en'], en: ['en'] }, reportMissing = true } = {}) {
  const missing = new Map();                              // lang -> Set(keys)
  function t(lang, key, params) {
    for (const l of fallbacks[lang] ?? [lang]) {
      const text = bundles[l]?.[key];
      if (text != null) {
        if (l !== lang) missingStat(lang, key, `fallback:${l}`);   // 回退也算缺失（翻译待补）
        return interpolate(text, params);
      }
    }
    missingStat(lang, key, 'total-miss');
    return key;                                           // 全链 miss：显示 key（调试可见，不显示空白）
  }
  function missingStat(lang, key, kind) {
    const s = missing.get(lang) ?? missing.set(lang, new Map()).get(lang);
    s.set(key, (s.get(key) ?? 0) + 1);
  }
  return { t, missingReport: () => [...missing].map(([lang, ks]) => ({ lang, keys: [...ks] })) };
}
// 回退链价值：zh-TW 用户看到简体（可读）而不是 key 原文（不可读）—— 分级降级
// missingReport 进 CI（ALG-242 占位符校验联动）：翻译补齐清单自动生成
// interpolate：{username} 占位符替换（ALG-242 提取协议一致）
```

**评分点**：
- 可配置回退链 + 逐级查找——本题核心。
- 回退命中也计缺失统计（翻译待补清单）。
- 全 miss 显示 key（调试可见不空白）。
- 与 ALG-242 CI 校验的联动闭环。
