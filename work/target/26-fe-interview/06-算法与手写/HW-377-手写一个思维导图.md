## HW-377：手写一个思维导图：节点布局与折叠

**答**：
- 需求：实现 MindMap：树数据（root + children 双向延伸：左右分支）；布局：横向树形排布（Tidy tree 简化——子树高度累计，父节点垂直居中于子节点范围）；折叠（折叠节点只渲染自身，子树高度不计）；节点连线（贝塞尔）；拖拽改父。
- 边界：左右分支方向分配（奇偶交替）；折叠状态持久化；子树高度缓存（折叠/展开增量更新）；根节点两侧平衡（子树重的放多的一侧）。
- 评分点：子树高度累计；折叠增量更新；左右分支分配。

```javascript
class MindMap {
  constructor(root) { this.root = root; this.collapsed = new Set(); this.side = new Map(); }
  subtreeHeight(node, nodeH = 36, gap = 12) {                // 子树高度累计
    if (this.collapsed.has(node.id) || !node.children?.length) return nodeH;
    let h = 0;
    for (const c of node.children) h += this.subtreeHeight(c, nodeH, gap) + gap;
    return Math.max(nodeH, h - gap);
  }
  layout(node, x, y, dir, depth = 0) {                       // 横向树形排布
    const positions = new Map([[node.id, { x, y, depth, dir }]]);
    if (this.collapsed.has(node.id) || !node.children?.length) return positions;
    const total = this.subtreeHeight(node, 36, 12);
    let cy = y - total / 2;
    const childX = x + dir * (NODE_W + 60);                  // dir: 1 右 / -1 左
    for (const c of node.children) {
      const ch = this.subtreeHeight(c, 36, 12);
      const cyy = cy + ch / 2;
      const cdir = depth === 0 ? (node.children.indexOf(c) % 2 ? -1 : 1)   // 根：奇偶分配左右
        : dir;
      positions.set(c.id, { x: childX, y: cyy, depth: depth + 1, dir: cdir });
      for (const [id, p] of this.layout(c, childX, cyy, cdir, depth + 1)) positions.set(id, p);
      cy += ch + 12;
    }
    return positions;
  }
  toggle(id) {                                               // 折叠：增量更新布局
    this.collapsed.has(id) ? this.collapsed.delete(id) : this.collapsed.add(id);
    localStorage.setItem('mm-collapsed', JSON.stringify([...this.collapsed]));
    return this.layout(this.root, 400, 300, 1);
  }
  linkPath(parent, child) {                                  // 贝塞尔连线
    const dir = child.dir;
    const mx = (parent.x + child.x) / 2;
    return `M${parent.x},${parent.y} C${mx},${parent.y} ${mx},${child.y} ${child.x},${child.y}`;
  }
}
```
