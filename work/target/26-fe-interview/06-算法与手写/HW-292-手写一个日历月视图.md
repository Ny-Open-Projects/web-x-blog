## HW-292：手写一个日历月视图：周首日、跨月溢出与事件布局

**答**：
- 需求：实现 MonthView：给定年月输出 6×7 网格（含前后月溢出日期，周首日可配置周一开始）；事件按天分组渲染（每天最多显示 3 条 + "更多"）；多天事件的横向条布局（同周内合并为一条，跨周拆分）。
- 边界：溢出日期标记 other-month；闰年 2 月；多天事件跨周拆分时每段独立渲染并延续样式。
- 评分点：溢出网格；多天事件分段；事件折叠。

```javascript
function monthGrid(year, month, { weekStart = 1 } = {}) {   // weekStart: 0 周日 / 1 周一
  const first = new Date(year, month, 1);
  const offset = (first.getDay() - weekStart + 7) % 7;
  const cells = [];
  const start = new Date(year, month, 1 - offset);
  for (let i = 0; i < 42; i++) {                            // 固定 6 行
    const d = new Date(start.getFullYear(), start.getMonth(), start.getDate() + i);
    cells.push({ date: d, otherMonth: d.getMonth() !== month,
      isToday: sameDay(d, new Date()) });
  }
  return cells;
}
function layoutEvents(events, cells) {                      // 多天事件分段
  const byDay = new Map();
  for (const ev of events) {
    const weeks = splitByWeek(ev, cells);                   // 跨周拆分
    weeks.forEach(w => {
      const key = dateKey(w.start);
      const arr = byDay.get(key) ?? [];
      arr.push({ ...ev, segStart: w.start, segEnd: w.end, continues: !sameDay(w.start, ev.start) });
      byDay.set(key, arr);
    });
  }
  return byDay;                                             // 渲染时每天取前 3 + 更多
}
function splitByWeek(ev, cells) {
  const out = []; let s = new Date(ev.start);
  const end = new Date(ev.end);
  while (s <= end) {                                        // 按周切
    const weekEnd = endOfWeek(s);
    const e = end < weekEnd ? end : weekEnd;
    out.push({ start: s, end: e });
    s = new Date(e.getTime() + 86400000);
  }
  return out;
}
const dateKey = d => d.toISOString().slice(0, 10);
const sameDay = (a, b) => dateKey(a) === dateKey(b);
const endOfWeek = d => { const x = new Date(d);
  x.setDate(x.getDate() + (7 - x.getDay()) % 7); return x; };
```
