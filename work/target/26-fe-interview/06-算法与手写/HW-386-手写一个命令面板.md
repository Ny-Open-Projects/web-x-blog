## HW-386：手写一个命令面板：模糊搜索与最近使用

**答**：
- 需求：实现 CommandPalette：Ctrl+K 唤起；命令注册（{id, title, section, run, keywords}）；搜索（空 query 显示最近使用 + 分组；有 query 模糊匹配 title/keywords 打分）；键盘导航（上下/Enter/Esc）；执行后关闭并记录 MRU。
- 边界：MRU 持久化（localStorage 前 5 条）；分组渲染（最近使用/各 section）；fuzzy 连续命中加分；长列表虚拟化（> 50 条）。
- 评分点：MRU 持久化；分组渲染；fuzzy 打分。

```javascript
class CommandPalette {
  constructor(commands) {
    this.cmds = commands;
    this.mru = JSON.parse(localStorage.getItem('cmd-mru') ?? '[]');
  }
  search(query) {
    if (!query) {                                            // 空 query：MRU + 分组
      const recent = this.mru.map(id => this.cmds.find(c => c.id === id)).filter(Boolean);
      const grouped = groupBy(this.cmds.filter(c => !recent.includes(c)), 'section');
      return { groups: [{ title: '最近使用', items: recent.slice(0, 5) },
        ...Object.entries(grouped).map(([title, items]) => ({ title, items }))] };
    }
    const scored = this.cmds.map(c => ({ c,
      score: Math.max(fuzzyScore(query, c.title),
        ...(c.keywords ?? []).map(k => fuzzyScore(query, k))) }))
      .filter(x => x.score > 0)
      .sort((a, b) => b.score - a.score);
    return { groups: [{ title: '结果', items: scored.map(x => x.c) }] };
  }
  async exec(cmd) {                                          // 执行 + MRU 记录
    await cmd.run();
    this.mru = [cmd.id, ...this.mru.filter(id => id !== cmd.id)].slice(0, 5);
    localStorage.setItem('cmd-mru', JSON.stringify(this.mru));   // 持久化
    this.close();
  }
  onKey(e, flat) {                                           // 键盘导航（扁平列表索引）
    if (e.key === 'ArrowDown') { this.idx = Math.min(this.idx + 1, flat.length - 1); return 'handled'; }
    if (e.key === 'ArrowUp') { this.idx = Math.max(this.idx - 1, 0); return 'handled'; }
    if (e.key === 'Enter') { this.exec(flat[this.idx]); return 'handled'; }
    if (e.key === 'Escape') { this.close(); return 'handled'; }
    return 'none';
  }
}
function groupBy(list, key) { return list.reduce((m, x) => {
  (m[x[key]] ??= []).push(x); return m; }, {}); }
function fuzzyScore(p, t) { t = t.toLowerCase(); p = p.toLowerCase();
  let s = 0, run = 0, i = 0;
  for (const c of t) { if (c === p[i]) { run++; s += 2 + run; i++; } else run = 0; }
  return i === p.length ? s : 0; }
```
