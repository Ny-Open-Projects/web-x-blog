## ALG-363：手写离散化+线段树：大坐标广告位统计

**答**：
- 题目：广告位坐标在 [0, 1e9]（像素），只有最多 1e4 个关注坐标点，要统计落在给定坐标区间的广告数。离散化：把出现的坐标映射到 1..k，线段树建在 k 上。
- 思路：收集所有出现的坐标（含查询端点）排序去重 → rank 映射 → 线段树按 rank 建树；区间查询时端点也做 rank 转换。
- 评分点：离散化三步（收集/排序去重/二分映射）；查询端点也要离散化；值域压缩后的空间收益。

```javascript
function discretize(points) {
  const sorted = [...new Set(points)].sort((a, b) => a - b);
  const rank = new Map(sorted.map((v, i) => [v, i]));      // 值 → rank
  return { sorted, rank, k: sorted.length };
}
class SparseStats {
  constructor(coords) {
    const { sorted, rank, k } = discretize(coords);
    this.rank = rank; this.sorted = sorted;
    this.t = new Array(4 * k).fill(0);
  }
  add(x) { this.update(this.rank.get(x), 1); }
  update(i, v, node = 1, l = 0, r = this.sorted.length - 1) {
    if (l === r) return this.t[node] += v;
    const m = (l + r) >> 1;
    i <= m ? this.update(i, v, node * 2, l, m) : this.update(i, v, node * 2 + 1, m + 1, r);
    this.t[node] = this.t[node * 2] + this.t[node * 2 + 1];
  }
  count(lo, hi) {                                          // 坐标区间 → rank 区间
    const li = lowerBound(this.sorted, lo), ri = upperBound(this.sorted, hi) - 1;
    if (li > ri) return 0;
    return this.query(li, ri);
  }
  query(ql, qr, node = 1, l = 0, r = this.sorted.length - 1) {
    if (qr < l || r < ql) return 0;
    if (ql <= l && r <= qr) return this.t[node];
    const m = (l + r) >> 1;
    return this.query(ql, qr, node * 2, l, m) + this.query(ql, qr, node * 2 + 1, m + 1, r);
  }
}
const lowerBound = (a, x) => { let lo = 0, hi = a.length;
  while (lo < hi) { const m = (lo + hi) >> 1; a[m] < x ? lo = m + 1 : hi = m; } return lo; };
const upperBound = (a, x) => { let lo = 0, hi = a.length;
  while (lo < hi) { const m = (lo + hi) >> 1; a[m] <= x ? lo = m + 1 : hi = m; } return lo; };
```
