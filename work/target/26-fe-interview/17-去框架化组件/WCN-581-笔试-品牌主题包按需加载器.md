## WCN-581：笔试：实现「品牌主题包」的按需加载器（品牌 id → CSS 变量覆盖 + 主题资源 chunk）。

**答**：
- 结论：加载器按品牌 id 懒加载主题 CSS chunk，加载后把 token 覆盖注入根节点，并缓存已加载品牌避免重复请求。
- 支持 preload（预测品牌提前加载）与切换竞态（快速切换只应用最后一次）。
- 主题资源（如品牌字体/图标）作为额外 chunk 一并加载。
- 追问：加载失败如何降级？（答：回退到默认品牌 token + 告警）

```javascript
function createThemeLoader(manifest, defaultBrand = 'default') {
  const loaded = new Set(); let token = 0;
  async function apply(id) {
    if (!manifest[id]) id = defaultBrand;
    const mine = ++token;                                  // 竞态：只应用最后一次
    if (!loaded.has(id)) { await loadCSS(manifest[id].css); await loadAssets(manifest[id].assets); loaded.add(id); }
    if (mine !== token) return;
    const root = document.documentElement;
    for (const [k, v] of Object.entries(manifest[id].vars)) root.style.setProperty(k, v);
  }
  function loadCSS(href) { return new Promise((res, rej) => { const l = document.createElement('link'); l.rel = 'stylesheet'; l.href = href; l.onload = res; l.onerror = () => { if (id !== defaultBrand) apply(defaultBrand); res(); }; document.head.append(l); }); }
  function loadAssets(list = []) { return Promise.all(list.map(u => fetch(u).catch(() => {}))); }
  function preload(id) { const l = document.createElement('link'); l.rel = 'preload'; l.as = 'style'; l.href = manifest[id].css; document.head.append(l); }
  return { apply, preload };
}
```

**评分点**：
- 懒加载 + 已加载缓存
- 竞态 token 只应用最后一次
- 主题资源一并加载
- 失败降级默认品牌
