## MP-396：笔试：手写栈深监控

**答**：实现页面栈观测器：① 全局路由拦截（包装 wx.navigateTo/redirectTo/switchTab/reLaunch——统一记录栈变化事件）；② 栈深水位线（8 层预警/10 层告警——接近上限的行为模式分析：哪些路径堆积）；③ 自动策略：水位线触发时「后续 navigateTo 自动降级 redirectTo」——可配置开关；④ 栈快照上报（getCurrentPages 全路由链——堆积路径还原）；⑤ 热点堆积路径统计（同路径链高频出现 → 产品层优化入口）。写出实现。

```javascript
let waterMark = 8, autoRedirect = true;
const stackEvents = [];

export function installStackMonitor({ warning = 8, auto = true } = {}) {
  waterMark = warning; autoRedirect = auto;
  for (const method of ['navigateTo', 'redirectTo']) {
    const orig = wx[method].bind(wx);
    wx[method] = (opts) => {
      const depth = getCurrentPages().length;
      stackEvents.push({ method, to: opts.url, depth, t: Date.now() });
      if (stackEvents.length > 200) stackEvents.shift();
      if (depth >= 10 && method === 'navigateTo') {
        beacon({ type: 'stack-full', to: opts.url, chain: chainSnapshot() });
        if (autoRedirect) return orig({ ...opts, url: opts.url });   // 自动降级
        return orig(opts);
      }
      if (depth >= waterMark && method === 'navigateTo')
        beacon({ type: 'stack-warn', depth, to: opts.url });
      return orig(opts);
    };
  }
}
function chainSnapshot() {
  return getCurrentPages().map((p) => p.route).join(' → ');
}
export function hotPaths(minCount = 5) {             // 堆积路径统计
  const counter = new Map();
  for (const e of stackEvents) if (e.depth >= waterMark)
    counter.set(e.to, (counter.get(e.to) ?? 0) + 1);
  return [...counter.entries()].filter(([, c]) => c >= minCount).sort((a, b) => b[1] - a[1]);
}
// 价值: hotPaths 输出「哪些页面反复压栈」→ 产品层改 redirectTo/合并流程
```

**评分点**：
- 统一路由拦截（四 API 包装的覆盖面）
- 双水位线（预警/告警）与自动降级开关
- 栈快照链上报（堆积路径还原）
- hotPaths 的产品化输出（优化入口）
