## FRP-288：笔试：手写 Signals 依赖图
**答**：实现 createSignal/createComputed：computed 自动收集依赖（读取时追踪）；上游变更沿图传播脏标记；检测到循环依赖抛错；读取 computed 时惰性重算。

```javascript
let active = null
let inCompute = false
function createSignal(value) {
  const subs = new Set()
  return {
    get() { if (active) { subs.add(active); active.deps.push(this) } return value },
    set(nv) {
      if (Object.is(nv, value)) return
      value = nv
      subs.forEach(s => s.markDirty())
    }
  }
}
function createComputed(fn) {
  const node = { deps: [], dirty: true, value: undefined,
    markDirty() { if (!node.dirty) { node.dirty = true; node.subs?.forEach?.(s => s.markDirty()) } },
    subs: null }
  node.read = () => {
    if (!node.dirty) return node.value
    if (inCompute) throw new Error('循环依赖') // 简化检测：计算中再读同图节点
    const prev = active; active = node; inCompute = true
    node.deps.forEach(d => d.subs?.delete?.(node)) // 清旧依赖
    node.deps = []
    try { node.value = fn(); node.dirty = false }
    finally { active = prev; inCompute = false }
    return node.value
  }
  return node
}
```
**评分点**：
- 读取时依赖追踪（active 指针）。
- 脏标记沿图传播 + 惰性重算。
- 旧依赖清理（条件分支换依赖）。
- 环检测（至少说出「计算栈内重入即环」的简化方案）。
