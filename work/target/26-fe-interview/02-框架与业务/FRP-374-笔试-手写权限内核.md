## FRP-374：笔试：手写权限内核

**答**：实现 `createAuth`：① has/hasAny/hasAll（权限点 + 通配符 'order:*'）；② 角色映射：roles → 权限点展开（角色变更重算集合）；③ 动态更新（updateRoles/mergePerms）通知订阅；④ 判定缓存（has 结果缓存，更新时清缓存——高频判定的性能）；⑤ deny 优先（显式拒绝集合优先于允许）。写出实现。

```javascript
export function createAuth({ roles = {}, initialRoles = [], deny = [] } = {}) {
  let userRoles = new Set(initialRoles);
  let denySet = new Set(deny);
  let allowSet = new Set();
  const cache = new Map();
  const subs = new Set();

  function rebuild() {
    allowSet = new Set();
    for (const r of userRoles)
      for (const p of roles[r] ?? []) allowSet.add(p);
    cache.clear();                                    // 变更清缓存
  }
  rebuild();

  function match(perm) {
    if (denySet.has(perm)) return false;              // deny 优先
    if (allowSet.has(perm)) return true;
    return [...allowSet].some((p) => p.endsWith('*') && perm.startsWith(p.slice(0, -1)));
  }
  function has(perm) {
    if (!cache.has(perm)) cache.set(perm, match(perm));   // 缓存
    return cache.get(perm);
  }
  return {
    has,
    hasAny: (list) => list.some((p) => has(p)),
    hasAll: (list) => list.every((p) => has(p)),
    setRoles(list) { userRoles = new Set(list); rebuild(); subs.forEach((f) => f()); },
    setDeny(list) { denySet = new Set(list); cache.clear(); subs.forEach((f) => f()); },
    sub(fn) { subs.add(fn); return () => subs.delete(fn); },
  };
}
// 用法: roles = { admin: ['order:*'], cs: ['order:read'] }
// has('order:delete') → admin true / cs false
```

**评分点**：
- deny 优先语义（显式拒绝覆盖允许）
- 通配符匹配（前缀 *）与角色展开
- 判定缓存 + 变更清缓存（性能与正确性平衡）
- 动态更新的订阅通知
