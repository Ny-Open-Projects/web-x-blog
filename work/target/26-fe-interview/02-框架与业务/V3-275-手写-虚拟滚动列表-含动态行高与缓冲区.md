## V3-275：手写：虚拟滚动列表（动态行高 + 二分定位 + 双向缓冲）
**答**：实现 `useVirtualList(items, estimateHeight)` 组合式函数：返回 `{ containerProps, wrapperProps, list, scrollToIndex }`；行高动态（渲染后实测回填缓存），支持滚动定位用**前缀和 + 二分查找**，上下各保留 OVERSCAN 缓冲行；数据变化时正确重算。
**评分点**：
- 前缀和偏移数组与二分查找定位（不能线性扫）
- 测量后缓存更新的同步逻辑（回填导致偏移变化）
- OVERSCAN 与越界保护
- 数据长度变化时缓存重建/裁剪策略
- scrollToIndex 支持 align: start/center

```javascript
import { ref, computed } from 'vue';
const OVERSCAN = 5;
export function useVirtualList(items, estimateHeight = 40) {
  const container = ref(null);
  const scrollTop = ref(0), viewH = ref(0);
  const offsets = ref([0]);          // 前缀和：offsets[i] = 前 i 行总高
  const measured = ref(new Map());   // index -> 实测高
  function rebuildOffsets() {
    const arr = [0];
    for (let i = 0; i < items.value.length; i++)
      arr.push(arr[i] + (measured.value.get(i) ?? estimateHeight));
    offsets.value = arr;
  }
  rebuildOffsets();
  function findIndex(scrollY) {       // 二分：找最后一个 offsets[i] <= scrollY
    let lo = 0, hi = offsets.value.length - 1;
    while (lo < hi) { const mid = (lo + hi + 1) >> 1;
      offsets.value[mid] <= scrollY ? (lo = mid) : (hi = mid - 1); }
    return lo;
  }
  const start = computed(() => Math.max(0, findIndex(scrollTop.value) - OVERSCAN));
  const end = computed(() => {
    const bottom = scrollTop.value + viewH.value;
    let i = findIndex(bottom);
    return Math.min(items.value.length, i + 1 + OVERSCAN);
  });
  const list = computed(() => items.value.slice(start.value, end.value).map((data, k) => ({
    data, index: start.value + k,
    top: offsets.value[start.value + k],
  })));
  const total = computed(() => offsets.value[offsets.value.length - 1]);
  function onItemMounted(el, index) { // 渲染后实测回填
    const h = el.getBoundingClientRect().height;
    if (h && h !== (measured.value.get(index) ?? estimateHeight)) {
      measured.value.set(index, h); rebuildOffsets();
    }
  }
  function scrollToIndex(i, align = 'start') {
    let y = offsets.value[i];
    if (align === 'center') y -= (viewH.value - (offsets.value[i + 1] - offsets.value[i])) / 2;
    container.value.scrollTop = Math.max(0, y);
  }
  const onScroll = () => { scrollTop.value = container.value.scrollTop; viewH.value = container.value.clientHeight; };
  return {
    list, total,
    containerProps: { ref: container, onScroll: { scroll: onScroll } },
    wrapperProps: { style: { height: total.value + 'px', position: 'relative' } },
    onItemMounted, scrollToIndex,
  };
}
```
