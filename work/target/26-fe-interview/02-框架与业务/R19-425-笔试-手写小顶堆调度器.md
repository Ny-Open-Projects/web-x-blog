## R19-425：笔试：手写小顶堆调度器

**答**：实现 `scheduleCallback(priority, cb)`：① 小顶堆按 expiration 排序（priority → timeout 映射）；② MessageChannel 宏任务驱动，堆顶任务执行；③ shouldYield：当前时间超过「帧 deadline」（16ms 窗口）让出，剩余任务下次宏任务继续；④ 高优先级任务插入能「抢占」（堆序保证）；⑤ 任务取消（cancelCallback）。写出实现。

```javascript
const PRIORITY_TIMEOUT = { immediate: -1, userBlocking: 250, normal: 5000, idle: 1073741823 };
const heap = [];
const chan = new MessageChannel();
let isRunning = false, frameStart = 0;

chan.port1.onmessage = performWork;

function scheduleCallback(priority, cb) {
  const entry = { cb, expiration: currentTime() + PRIORITY_TIMEOUT[priority], id: ++id };
  heapPush(entry);
  if (!isRunning) { isRunning = true; frameStart = currentTime(); chan.port2.postMessage(null); }
  return () => { entry.cb = null; };                 // cancel: 置空惰性跳过
}

function performWork() {
  frameStart = currentTime();
  while (heap.length) {
    if (shouldYield()) { chan.port2.postMessage(null); return; }   // 让出, 下次继续
    const entry = heapPop();
    entry.cb?.();                                    // cancel 后 cb 为 null
  }
  isRunning = false;
}

function shouldYield() {
  return currentTime() - frameStart >= 5;            // 5ms 时间片
}
const currentTime = () => performance.now();

// 堆操作
function heapPush(e) { heap.push(e); let i = heap.length - 1;
  while (i > 0) { const p = (i - 1) >> 1;
    if (heap[p].expiration <= heap[i].expiration) break;
    [heap[p], heap[i]] = [heap[i], heap[p]]; i = p; } }
function heapPop() { const top = heap[0], last = heap.pop();
  if (heap.length) { heap[0] = last; let i = 0;
    while (true) { const l = 2*i+1, r = 2*i+2; let m = i;
      if (l < heap.length && heap[l].expiration < heap[m].expiration) m = l;
      if (r < heap.length && heap[r].expiration < heap[m].expiration) m = r;
      if (m === i) break; [heap[m], heap[i]] = [heap[i], heap[m]]; i = m; } }
  return top; }
```

**评分点**：
- expiration 排序的小顶堆（push/pop 完整）
- MessageChannel 宏任务驱动 + 让出续跑
- shouldYield 时间片判定（5ms）
- cancel 的惰性实现（cb 置空）
