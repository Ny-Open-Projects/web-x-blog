## MP-397：笔试：手写发布检查

**答**：实现提审前检查脚本 `preReleaseCheck.js`：① 环境一致性：baseUrl 无测试域名残留（grep 构建产物）、体验版资源链接检查；② 域名白名单核对：代码中 wx.request/uploadFile/downloadFile 的 URL 与后台合法域名清单比对（未配置域名直接阻断）；③ 隐私接口核对：代码 wx.* 调用清单 vs 后台隐私声明清单 diff（多声明/少声明都报）；④ 遗留调试开关（__debug/console.log 高频/测试账号硬编码）；⑤ 输出检查报告（阻断项/警告项分级）。写出实现。

```javascript
#!/usr/bin/env node
const fs = require('fs'), path = require('path');
const API_DOMAINS = ['api.example.com', 'cdn.example.com'];    // 后台合法域名清单
const PRIVACY_DECLARED = ['getLocation', 'chooseImage', 'getPhoneNumber'];   // 后台已声明隐私接口

function walkFiles(dir, ext) {
  const out = [];
  (function rec(d) {
    for (const f of fs.readdirSync(d, { withFileTypes: true })) {
      const p = path.join(d, f.name);
      f.isDirectory() ? rec(p) : (p.endsWith(ext) && out.push(p));
    }
  })(dir);
  return out;
}

function preReleaseCheck({ dist }) {
  const blockers = [], warnings = [];
  const files = [...walkFiles(dist, '.js'), ...walkFiles(dist, '.json'), ...walkFiles(dist, '.wxml')];

  // 1) 测试域名残留
  for (const f of files) {
    const src = fs.readFileSync(f, 'utf8');
    if (/test\.example\.com|localhost|127\.0\.0\.1/.test(src))
      blockers.push(`测试域名残留: ${f}`);
    // 2) 域名白名单核对
    for (const m of src.matchAll(/https:\/\/([a-z0-9.-]+)/g)) {
      const host = m[1];
      if (!API_DOMAINS.some((d) => host === d || host.endsWith('.' + d)))
        blockers.push(`未配置域名: ${host} in ${f}`);
    }
    // 3) 隐私接口核对
    for (const api of PRIVACY_DECLARED) if (!src.includes(api)) continue;
  }
  const usedPrivacy = new Set();
  for (const f of files) {
    const src = fs.readFileSync(f, 'utf8');
    for (const api of PRIVACY_DECLARED) if (src.includes(`wx.${api}`)) usedPrivacy.add(api);
  }
  for (const api of usedPrivacy) if (!PRIVACY_DECLARED.includes(api)) blockers.push(`隐私接口未声明: wx.${api}`);
  // 4) 调试开关
  for (const f of files) {
    const src = fs.readFileSync(f, 'utf8');
    if ((src.match(/console\.log/g) ?? []).length > 5) warnings.push(`console.log 高频: ${f}`);
    if (/__debug/.test(src)) warnings.push(`调试开关残留: ${f}`);
  }
  console.log(blockers.length ? `BLOCKERS:\n${blockers.join('\n')}` : 'ALL PASS');
  warnings.length && console.log(`WARNINGS:\n${warnings.join('\n')}`);
  process.exit(blockers.length ? 1 : 0);
}
```

**评分点**：
- 域名白名单核对（未配置域名 = 线上必然失败——阻断）
- 隐私声明与实际调用 diff（合规双向对齐）
- 测试残留扫描（域名/调试开关/日志）
- 阻断/警告分级 + exit code CI 语义
