## FRP-377：笔试：手写自定义渲染器

**答**：实现 `createRenderer(hostOps)`：① 宿主操作注入：createElement/insert/remove/patchProp/setElementText——DOM 与自定义宿主（Canvas/终端/SSR 字符串）的抽象边界；② render(vnode, container)：挂载/更新/卸载三分支；③ patch：同类型更新 props/children、异类型替换；④ 用它实现一个「JSON 渲染器」（把 UI 渲染为 JSON 树而非 DOM）；⑤ 宿主协议的最小完备性讨论（哪些 op 不可省）。写出实现。

```javascript
function createRenderer(host) {
  function render(vnode, container) {
    if (vnode) patch(container._vnode ?? null, vnode, container);
    else if (container._vnode) unmount(container._vnode);
    container._vnode = vnode;                        // 记录旧树(更新基准)
  }
  function patch(n1, n2, container) {
    if (n1 && !sameType(n1, n2)) { unmount(n1); n1 = null; }
    n1 ? patchElement(n1, n2) : mountElement(n2, container);
  }
  function mountElement(vnode, container) {
    const el = (vnode.el = host.createElement(vnode.type));
    if (typeof vnode.children === 'string') host.setElementText(el, vnode.children);
    else vnode.children.forEach((c) => mountElement(c, el));
    for (const [k, v] of Object.entries(vnode.props ?? {})) host.patchProp(el, k, null, v);
    host.insert(el, container);
  }
  function patchElement(n1, n2) {
    const el = (n2.el = n1.el);
    for (const [k, v] of Object.entries(n2.props)) host.patchProp(el, k, n1.props[k], v);
    for (const k of Object.keys(n1.props)) if (!(k in n2.props)) host.patchProp(el, k, n1.props[k], null);
    // children diff 简化: 文本直接设 / 数组逐位 patch (完整实现需 key diff)
    if (typeof n2.children === 'string') host.setElementText(el, n2.children);
    else { const oc = n1.children, nc = n2.children;
      const max = Math.max(oc.length, nc.length);
      for (let i = 0; i < max; i++) { oc[i] && nc[i] ? patch(oc[i], nc[i], el) : nc[i] ? mountElement(nc[i], el) : unmount(oc[i]); }
    }
  }
  function unmount(vnode) { host.remove(vnode.el); }
  return { render };
}
// JSON 宿主: createElement → {type}, insert → push, patchProp → 记录
const jsonRenderer = createRenderer({
  createElement: (t) => ({ type: t, props: {}, children: [] }),
  insert: (el, parent) => parent.children.push(el),
  remove: (el) => {/* 从父级 splice */},
  patchProp: (el, k, o, v) => { el.props[k] = v; },
  setElementText: (el, t) => { el.children = [t]; },
});
```

**评分点**：
- 宿主操作抽象边界（六 ops 的最小完备性）
- render 的旧树记录（container._vnode）与三分支
- patch 的同/异类型分派与 props 前后对比
- JSON 宿主验证「渲染器与 DOM 解耦」
