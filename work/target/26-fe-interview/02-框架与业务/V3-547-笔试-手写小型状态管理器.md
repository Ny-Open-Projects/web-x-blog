## V3-547：笔试：手写小型状态管理器

**答**：实现 `createStore(definition)`（Pinia 风格最小实现）：① definition: `{ state: () => ({}), getters: {}, actions: {} }`；② state 用 reactive 包装；getters 转为 computed（惰性缓存）；actions 的 this 绑定整个 store；③ storeToRefs 语义（state 与 getter 转 ref，action 保持函数）；④ 单例：同 id 多次调用返回同一 store；⑤ $reset 恢复初始 state。写出实现。

```javascript
const storeMap = new Map();

function createStore(id, definition) {
  if (storeMap.has(id)) return storeMap.get(id);       // 单例

  const rawState = definition.state?.() ?? {};
  const store = reactive({ ...rawState });

  const computedMap = {};
  for (const [k, fn] of Object.entries(definition.getters ?? {}))
    computedMap[k] = computed(() => fn.call(storeProxy));

  const storeProxy = new Proxy(store, {
    get(t, k) {
      if (k in computedMap) return computedMap[k].value;   // getter 走 computed
      return t[k];
    },
    set(t, k, v) { t[k] = v; return true; },
  });
  for (const [k, fn] of Object.entries(definition.actions ?? {}))
    storeProxy[k] = (...args) => fn.apply(storeProxy, args);  // this 绑定 store

  storeProxy.$reset = () => Object.assign(store, definition.state?.() ?? {});
  storeProxy._computed = computedMap;
  storeMap.set(id, storeProxy);
  return storeProxy;
}

function storeToRefs(store) {
  const out = {};
  for (const k of Object.keys(store)) {
    if (typeof store[k] === 'function') out[k] = store[k];        // action 保持
    else out[k] = toRef(store, k);                                // state/getter → ref
  }
  return out;
}
// 用法
const useCounter = () => createStore('counter', {
  state: () => ({ n: 0 }),
  getters: { double: (s) => s.n * 2 },
  actions: { inc() { this.n++; } },
});
```

**评分点**：
- reactive state + computed getter 的 Proxy 组合
- actions 的 this 绑定与 action 不转 ref（storeToRefs 语义）
- 单例 Map 与 $reset
- getter 惰性缓存（computed 语义）
