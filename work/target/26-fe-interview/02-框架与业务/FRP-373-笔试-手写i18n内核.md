## FRP-373：笔试：手写 i18n 内核

**答**：实现 `createI18n`：① t(key, params, count)：查消息 → 复数分支（Intl.PluralRules）→ 插值（{name} 替换）；② 回退链：locale → 语言前缀（zh-CN→zh）→ fallback → key 本身；③ 嵌套键（'order.status.paid'）点路径查找；④ 缺失键收集（dev 下收集 + exportMissing 供 CI 补齐）；⑤ setLocale 懒加载语言包 + 通知订阅。写出实现。

```javascript
export function createI18n({ fallback = 'en', loader } = {}) {
  let locale = fallback;
  const packs = new Map();          // locale -> messages
  const missing = new Set();
  const subs = new Set();

  function lookup(locale, key) {
    const pack = packs.get(locale);
    if (!pack) return undefined;
    return key.split('.').reduce((o, k) => o?.[k], pack);
  }
  function candidates(l) {
    const out = [l];
    if (l.includes('-')) out.push(l.split('-')[0]);
    if (fallback !== l) out.push(fallback, fallback.split('-')[0]);
    return out;
  }
  function t(key, params = {}, count) {
    const pack = candidates(locale).map((l) => lookup(l, key)).find((v) => v != null);
    if (pack == null) {
      missing.add(`${locale}:${key}`);               // 缺失键收集
      return key;
    }
    const tpl = count != null
      ? (typeof pack === 'object' ? pack[new Intl.PluralRules(locale).select(count)] ?? pack.other : pack)
      : pack;
    return String(tpl).replace(/\{(\w+)\}/g, (_, k) => params[k] ?? `{${k}}`);
  }
  async function setLocale(l) {
    if (!packs.has(l)) packs.set(l, (await loader(l))?.default ?? {});
    locale = l;
    subs.forEach((fn) => fn(l));
  }
  return {
    t, get locale() { return locale; }, setLocale,
    sub(fn) { subs.add(fn); return () => subs.delete(fn); },
    exportMissing: () => [...missing],
  };
}
// 用法: t('cart.items', { n: 3 }, 3) → messages: { cart: { items: { one: '{n} item', other: '{n} items' } } }
```

**评分点**：
- 回退链（locale→前缀→fallback→key）顺序正确
- PluralRules 分支（one/other 等类别）与缺失键收集
- 嵌套键点路径查找 + 插值未命中保底
- setLocale 懒加载与订阅通知
