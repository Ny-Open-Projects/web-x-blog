## MP-390：笔试：手写错误聚合

**答**：实现 `ErrorAggregator`：① App.onError 的 msg 字符串解析（微信格式「page\r\nmsg\r\nstack」三段）→ 结构化；② 指纹聚合（同类错误计数、首条保留——限流防刷爆上报额度）；③ 窗口限流（同指纹 1 分钟一条 + 计数补报）；④ 环形缓冲现场（最近 10 条用户动作轨迹——页面跳转+点击附到错误）；⑤ unhandledrejection 与 onError 的统一入口。写出实现。

```javascript
export function createErrorAggregator({ report } = {}) {
  const seen = new Map();          // fp -> { count, lastSent }
  const trail = new RingBuffer(10);
  let seq = 0;

  function capture(kind, raw, extra = {}) {
    const parsed = kind === 'js' ? parseWxError(raw) : { msg: String(raw).slice(0, 500), stack: '' };
    const fp = hash(`${parsed.msg.slice(0, 100)}:${parsed.route ?? ''}`);
    const item = seen.get(fp) ?? { count: 0, lastSent: 0, first: parsed };
    item.count++;
    seen.set(fp, item);
    if (Date.now() - item.lastSent < 60_000) return;   // 1 分钟限流
    item.lastSent = Date.now();
    report({
      kind, fp, count: item.count, msg: parsed.msg,
      stack: parsed.stack?.slice(0, 1000),
      trail: trail.last(10), seq: seq++, ...extra,
    });
  }
  function parseWxError(msg) {
    const [route, m, stack] = String(msg).split('\n'); // 微信三段格式
    return { route: route?.trim(), msg: m?.trim(), stack };
  }
  function trackAction(action) { trail.push({ action, t: Date.now() }); }   // 行为轨迹
  return { capture, trackAction };
}
// App 接入:
App({
  onError(msg) { agg.capture('js', msg); },
  onUnhandledRejection({ reason }) { agg.capture('promise', reason); },
});
// 页面工厂: trackAction(`nav:${route}`) / trackAction(`tap:${target}`)
```

**评分点**：
- 微信 onError 三段格式解析（route/msg/stack）
- 指纹聚合 + 1 分钟限流 + 计数补报（防刷爆限额）
- 行为轨迹环形缓冲附现场
- 两种错误源统一入口
