## V3-396：笔试：手写虚拟滚动列表（动态行高版核心逻辑）
**答**：实现 `useVirtualList(items, estimateSize = 40)`：返回 `{ containerProps, wrapperProps, visible }`；动态行高：用前缀和估算 + 渲染后校正实际高度；支持滚动到指定 index。

```javascript
function useVirtualList(items, estimateSize = 40) {
  const container = ref(null)
  const scrollTop = ref(0)
  const viewport = ref(600)
  const sizes = ref(items.value.map(() => estimateSize)) // 实测高度缓存
  const offsets = computed(() => {
    const arr = [0]
    for (const s of sizes.value) arr.push(arr[arr.length - 1] + s)
    return arr
  })
  const start = computed(() => {
    const list = offsets.value
    let lo = 0, hi = list.length - 1
    while (lo < hi) { const mid = (lo + hi) >> 1; list[mid + 1] <= scrollTop.value ? lo = mid + 1 : hi = mid }
    return Math.max(0, lo - 5) // overscan
  })
  const end = computed(() => {
    const list = offsets.value, s = start.value
    let e = s
    while (e < list.length - 1 && list[e] - list[start.value] < viewport.value + 200) e++
    return e
  })
  const visible = computed(() => items.value.slice(start.value, end.value).map((item, i) => ({
    item, index: start.value + i, style: {
      position: 'absolute', top: `${offsets.value[start.value + i]}px`,
      width: '100%'
    }
  })))
  function scrollToIndex(i) {
    container.value.scrollTop = offsets.value[i]
  }
  const onItemMounted = (i, el) => { // 渲染后校正真实高度
    const h = el.offsetHeight
    if (h && h !== sizes.value[i]) sizes.value[i] = h
  }
  return { visible, onItemMounted, scrollToIndex, totalHeight: computed(() => offsets.value.at(-1)) }
}
```
**评分点**：
- 前缀和 offsets + 二分查找定位 start。
- overscan 缓冲区避免白屏。
- 实测高度校正形成闭环（估算 → 渲染 → 校正）。
- 滚动到 index 用 offsets 定位。
