## V3-274：可编辑表格的单元格状态与脏数据管理怎么设计？
**答**：
- 编辑态独立：`editingSet: Set<rowId-colKey>` 记录哪些单元格处于编辑态；值写入 `draftMap: Map<rowId-colKey, value>`，**不污染原始行数据**。
- 校验分层：单元格级（失焦校验）+ 行级（依赖多列）+ 表级（提交时全量），错误存 `errorMap`，渲染层按 key 读取。
- 提交策略：只提交脏数据 diff（对比 draftMap），支持单行保存与批量保存；失败行保留 draft 并标记错误行滚动定位。
- 快捷体验：方向键/Tab 切换单元格、Enter 提交当前行、Esc 还原；用单一 focus 管理器统一处理。
- 性能：大表避免 v-model 全表响应式绑定，编辑态才渲染 input，展示态用纯文本节点。

```typescript
const draft = reactive(new Map<string, any>());     // 脏值
const errors = reactive(new Map<string, string>());
const editing = reactive(new Set<string>());
const cellKey = (r: string, c: string) => `${r}__${c}`;
function startEdit(rowId: string, col: string) {
  editing.add(cellKey(rowId, col));
  if (!draft.has(cellKey(rowId, col))) draft.set(cellKey(rowId, col), rowData(rowId)[col]); // 初始草稿
}
function validateCell(rowId: string, col: string) {
  const v = draft.get(cellKey(rowId, col));
  const err = columnRules[col]?.(v, rowData(rowId)); // 单元格校验
  err ? errors.set(cellKey(rowId, col), err) : errors.delete(cellKey(rowId, col));
}
function commitRow(rowId: string) {
  const dirty: Record<string, any> = {};
  for (const [k, v] of draft) if (k.startsWith(rowId + '__')) dirty[k.split('__')[1]] = v;
  if ([...errors].some(([k]) => k.startsWith(rowId + '__'))) return toast('行内校验未通过');
  api.saveRow(rowId, dirty);                          // 只提交脏字段
}
```
