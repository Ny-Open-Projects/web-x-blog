## MP-398：笔试：手写降级开关

**答**：实现 FeatureFlag 中心：① 配置拉取：启动时同步拉（缓存兜底——上次配置先用）+ 推送更新（onShow 时拉一次增量）；② 开关分级：全局开关/用户分群开关（按 openid 尾号灰度）/实验开关（AB）；③ 消费协议：`flag('seckill')` 判定 + `whenReady(fn)`（配置就绪回调——首屏渲染前就绪的保证）；④ 默认值兜底（拉取失败用内置默认——脏配置 schema 校验）；⑤ 操作审计（开关变更记录——谁改的/回滚入口）。写出实现。

```javascript
const DEFAULTS = { seckill: false, recommend: true, newCart: true };
let flags = wx.getStorageSync('flags') ?? { ...DEFAULTS, _v: 0 };
const subs = new Set();

function hashUserId(uid, expId) {                     // 灰度哈希
  let h = 0;
  for (const c of `${expId}:${uid}`) h = (h * 31 + c.charCodeAt(0)) >>> 0;
  return (h % 1000) / 1000;
}

export function flag(key) {
  const cfg = flags[key];
  if (!cfg) return DEFAULTS[key] ?? false;
  if (typeof cfg === 'boolean') return cfg;                        // 全局开关
  if (cfg.rollout != null)                                          // 分群灰度
    return hashUserId(getApp().store.openid, cfg.expId) < cfg.rollout;
  return cfg.value ?? DEFAULTS[key];
}
export function whenReady(fn) { flags._ready ? fn() : subs.add(fn); }

export async function refreshFlags() {
  try {
    const fresh = await api.flags(flags._v);                        // 增量(版本号)
    if (fresh) {
      const merged = { ...flags, ...fresh, _ready: true };
      if (schemaOk(merged)) {                                       // 脏配置防御
        flags = merged;
        wx.setStorageSync('flags', merged);
        subs.forEach((f) => f());
      }
    } else flags._ready = true;
  } catch { flags._ready = true; }                                  // 拉取失败: 缓存兜底
}
function schemaOk(f) { return Object.entries(f).every(([k, v]) => k === '_v' || k === '_ready' || typeof v === 'boolean' || (v && typeof v === 'object' && 'rollout' in v || 'value' in v)); }
App({ onLaunch() { refreshFlags(); }, onShow() { refreshFlags(); } });   // 启动+回前台增量
// 审计: 服务端记录每次变更(操作人/时间/回滚到版本)
```

**评分点**：
- 双拉取时机（启动同步缓存兜底 + onShow 增量）
- 三类开关语义（全局/灰度哈希/实验）
- whenReady 保证首屏前就绪 + 脏配置 schema 防御
- 拉取失败缓存兜底（永不白屏）
