## V3-400：笔试：手写「分页/筛选/排序 + URL 同步」的可复用逻辑
**题**：实现 `useTableQuery(route, router)`：从 URL query 初始化 page/pageSize/keyword/sort；任何字段变化写回 URL（replace 不留历史）；表单筛选变化时 page 重置为 1；返回 requestParams computed。

```javascript
function useTableQuery(route, router) {
  const query = reactive({
    page: Number(route.query.page) || 1,
    pageSize: Number(route.query.pageSize) || 20,
    keyword: route.query.keyword || '',
    sort: route.query.sort || ''
  })
  const syncUrl = () => router.replace({ query: {
    ...route.query,
    page: query.page, pageSize: query.pageSize,
    keyword: query.keyword || undefined, sort: query.sort || undefined
  }})
  watch(() => ({ ...query }), syncUrl, { deep: true })
  function setFilter(k, v) {
    query[k] = v
    query.page = 1          // 筛选变化重置页码
  }
  const requestParams = computed(() => ({
    page: query.page, pageSize: query.pageSize,
    keyword: query.keyword,
    orderBy: query.sort.split(':')[0] || undefined,
    order: query.sort.includes('desc') ? 'desc' : undefined
  }))
  return { query, setFilter, requestParams }
}
```
**评分点**：
- URL 初始化 + 防抖/replace 写回（不留历史栈）。
- 筛选变化重置 page=1（高频考点）。
- 空值从 URL 中剔除（undefined），避免 URL 垃圾参数。
- requestParams 纯派生，排序解析健壮（无冒号时不崩）。
