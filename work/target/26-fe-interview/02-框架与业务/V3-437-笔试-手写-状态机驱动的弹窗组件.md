## V3-437：笔试：手写状态机驱动的弹窗组件（loading/success/error 三态防竞态）
**题**：实现 `useDialogTask()`：状态机 idle → loading → success/error；只有 idle 可发起；loading 中重复点击忽略；success 自动 2s 后回 idle；error 提供 retry；防止上一次未完成的 success 定时器影响新一轮。

```javascript
function useDialogTask(taskFn) {
  const state = ref('idle')
  const error = ref(null)
  let timer = null
  function resetTimer() { clearTimeout(timer) }
  async function run(...args) {
    if (state.value !== 'idle') return        // 非空闲忽略
    resetTimer()
    state.value = 'loading'
    try {
      const res = await taskFn(...args)
      state.value = 'success'
      timer = setTimeout(() => { state.value = 'idle' }, 2000) // 成功自动回 idle
      return res
    } catch (e) {
      error.value = e
      state.value = 'error'
    }
  }
  function retry() {
    if (state.value !== 'error') return
    state.value = 'idle'
    run()
  }
  function forceIdle() { resetTimer(); state.value = 'idle' }
  onBeforeUnmount(resetTimer)
  return { state, error, run, retry, forceIdle }
}
```
**评分点**：
- 非法状态转换被拒绝（loading 中不可再次发起）。
- success 定时器在新一轮 run 前被清（防竞态，核心）。
- retry 只能从 error 态发起。
- 卸载清 timer。
