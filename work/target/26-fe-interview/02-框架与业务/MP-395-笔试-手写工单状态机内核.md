## MP-395：笔试：手写工单状态机

**答**：实现工单状态机 UI 内核：① 迁移表与服务端状态机对齐（open→processing→resolved/closed，processing 可回 open）——「前端只声明合法迁移，非法操作禁用按钮」；② send(action)：乐观迁移 + 服务端确认——失败回滚（状态恢复 + toast）；③ 迁移历史（操作轨迹展示——每步 action/操作人/时间）；④ 状态驱动的 UI 权限（每个状态允许的操作集合——按钮渲染就绪）；⑤ 竞态（两个端同时操作——服务端版本号冲突时刷新状态）。写出实现。

```javascript
export function createTicketMachine(initial, rev) {
  const TRANSITIONS = {
    open: { accept: 'processing', close: 'closed' },
    processing: { resolve: 'resolved', reopen: 'open', close: 'closed' },
    resolved: { close: 'closed', reopen: 'processing' },
    closed: {},
  };
  let state = initial, version = rev;
  const subs = new Set();
  const history = [];

  function allowedActions() { return Object.keys(TRANSITIONS[state] ?? {}); }   // UI 权限就绪
  async function send(action, payload) {
    const target = TRANSITIONS[state]?.[action];
    if (!target) return { ok: false, reason: 'illegal' };       // 前端拦截
    const prev = { state, version };
    state = target; notify();                                    // 乐观迁移
    try {
      const r = await api.ticketTransition(id, action, { baseVersion: version, ...payload });
      version = r.rev; history.push({ action, at: Date.now() });
      return { ok: true };
    } catch (e) {
      state = prev.state; version = prev.version;                // 回滚
      if (e.status === 409) await refreshState();                // 版本冲突: 刷新服务端真实态
      notify();
      return { ok: false, reason: e.status === 409 ? 'conflict' : 'fail' };
    }
  }
  async function refreshState() {
    const s = await api.ticketState(id);
    state = s.state; version = s.rev; notify();
  }
  function notify() { subs.forEach((f) => f({ state, allowed: allowedActions(), history })); }
  return { send, refreshState, allowedActions, sub(fn) { subs.add(fn); return () => subs.delete(fn); } };
}
```

**评分点**：
- 迁移表与服务端对齐（前端只声明合法——非法操作按钮禁用）
- 乐观迁移 + 版本冲突回滚刷新（两端并发操作）
- allowedActions 驱动 UI 权限（按钮渲染就绪）
- 历史轨迹的操作审计
