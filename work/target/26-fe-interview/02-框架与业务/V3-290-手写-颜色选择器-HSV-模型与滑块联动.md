## V3-290：手写：颜色选择器（HSV 模型与滑块联动）
**需求**：实现颜色选择器核心逻辑：HSV ↔ RGB/HEX 互转；饱和度/明度面板（2D 拖动）与色相滑块（1D）、透明度滑块联动；输入面板点击或拖动正确更新 hsv 状态并输出 hex。要求函数式实现（不要求 UI 完整样式）。
**评分点**：
- hsvToRgb 转换公式正确（六段分段）
- 面板点击坐标 → s/v 的映射（注意 y 反向）
- 拖动越界 clamp
- hex 输出带 alpha 处理（#RRGGBBAA）
- 色相 0-360 环形取模

```javascript
function hsvToRgb(h, s, v) { // h:0-360, s/v:0-1
  const c = v * s, x = c * (1 - Math.abs(((h / 60) % 2) - 1)), m = v - c;
  let [r, g, b] = h < 60 ? [c, x, 0] : h < 120 ? [x, c, 0] : h < 180 ? [0, c, x]
    : h < 240 ? [0, x, c] : h < 300 ? [x, 0, c] : [c, 0, x];
  return [Math.round((r + m) * 255), Math.round((g + m) * 255), Math.round((b + m) * 255)];
}
function rgbToHex(r, g, b, a = 1) {
  const to2 = (n) => n.toString(16).padStart(2, '0');
  return `#${to2(r)}${to2(g)}${to2(b)}${a < 1 ? to2(Math.round(a * 255)) : ''}`;
}
function rgbToHsv(r, g, b) {
  r /= 255; g /= 255; b /= 255;
  const max = Math.max(r, g, b), min = Math.min(r, g, b), d = max - min;
  const h = d === 0 ? 0 : max === r ? 60 * (((g - b) / d + 6) % 6)
    : max === g ? 60 * ((b - r) / d + 2) : 60 * ((r - g) / d + 4);
  return [h, max === 0 ? 0 : d / max, max];
}
// 面板交互：点击/拖动 → 坐标转 s/v（y 反向：顶部 v=1，底部 v=0）
function onPanelPointer(e, panelRect, state) {
  const s = Math.min(1, Math.max(0, (e.clientX - panelRect.left) / panelRect.width));
  const v = 1 - Math.min(1, Math.max(0, (e.clientY - panelRect.top) / panelRect.height));
  state.s = s; state.v = v;   // 色相滑块类似：h = (x / width) * 360
}
// 输出：const [r, g, b] = hsvToRgb(state.h, state.s, state.v); hex = rgbToHex(r, g, b, state.a)
```
