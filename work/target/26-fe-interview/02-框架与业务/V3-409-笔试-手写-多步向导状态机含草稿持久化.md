## V3-409：笔试：手写多步向导状态机（分步表单 + 草稿持久化）
**题**：实现 `useWizard(steps)`：state ∈ idle/active/submitting/done；next() 需当前步校验通过才能前进；back() 保留已填数据；草稿自动持久化到 localStorage（防抖），进入时恢复；提交中禁止重复点击。

```javascript
function useWizard(steps) {
  const state = ref('idle')
  const current = ref(0)
  const form = ref(steps[0].initial || {})
  const DRAFT = 'wizard-draft'
  // 恢复草稿
  const saved = localStorage.getItem(DRAFT)
  if (saved) { form.value = JSON.parse(saved); current.value = JSON.parse(saved).__step || 0 }
  const saveDraft = useDebounceFn(() => {
    localStorage.setItem(DRAFT, JSON.stringify({ ...form.value, __step: current.value }))
  }, 500)
  watch(form, saveDraft, { deep: true })
  function next() {
    if (state.value === 'submitting') return
    if (!steps[current.value].validate(form.value)) return
    if (current.value < steps.length - 1) current.value++
    else submit()
  }
  function back() { if (state.value !== 'submitting' && current.value > 0) current.value-- }
  async function submit() {
    state.value = 'submitting'
    try { await steps.at(-1).onSubmit(form.value); state.value = 'done'; localStorage.removeItem(DRAFT) }
    catch { state.value = 'active' }
  }
  return { state, current, form, next, back }
}
```
**评分点**：
- 状态机隔离 submitting（防重复提交，核心）。
- next 前置校验 + back 保留数据。
- 草稿防抖持久化 + 恢复（含步号），提交成功清草稿。
- watch form 深度监听保存。
