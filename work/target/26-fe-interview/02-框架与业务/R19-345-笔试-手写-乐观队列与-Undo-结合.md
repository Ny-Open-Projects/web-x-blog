## R19-345：笔试：手写「删除待办 + Undo」
**题**：删除待办：乐观移除 + 底部 Undo 条（5s 倒计时）；倒计时结束执行删除 action（软删除）；点 Undo 取消删除（不调 action）；同一时间允许多条 pending 删除。

```javascript
function TodoList({ todos }) {
  const [optimistic, addOptimistic, clearOptimistic] = useMyOptimistic(
    todos,
    (list, op) => list.filter(t => t.id !== op.id)
  )
  const [undoQueue, setUndoQueue] = useState([]) // { id, timer }
  function remove(id) {
    addOptimistic({ id })
    const timer = setTimeout(() => {          // 5s 后真正删除
      startTransition(async () => {
        const res = await deleteTodoAction(id)
        if (!res.ok) toast('删除失败，已恢复')
      })
      setUndoQueue(q => q.filter(u => u.id !== id))
    }, 5000)
    setUndoQueue(q => [...q, { id, timer }])
  }
  function undo(id) {
    const entry = undoQueue.find(u => u.id === id)
    if (!entry) return
    clearTimeout(entry.timer)                  // 取消真正的删除 action
    setUndoQueue(q => q.filter(u => u.id !== id))
    // 恢复乐观层：清空整个队列最简单（todos 数据仍在 serverState）
    clearOptimistic()
  }
  return (
    <ul>
      {optimistic.map(t => <li key={t.id}>{t.title}<button onClick={() => remove(t.id)}>删</button></li>)}
      {undoQueue.map(u => <li key={u.id}><button onClick={() => undo(u.id)}>撤销删除</button> 5s</li>)}
    </ul>
  )
}
```
**评分点**：
- 乐观移除与真实删除分离（倒计时后 action）。
- Undo 清 timer 取消 action（不白删）。
- clearOptimistic 恢复展示。
- 软删除兜底（超时后 undo 走恢复接口）说明。
