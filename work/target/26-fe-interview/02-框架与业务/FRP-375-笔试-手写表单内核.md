## FRP-375：笔试：手写表单内核

**答**：实现 `createForm`：① values/errors/touched 三态 + status（idle/submitting/submitted）；② setField 触发「依赖该字段的其他字段重校验」（validateDeps 声明）；③ 异步校验竞态守卫（seq）；④ submit：全量校验 → focus 首错（返回 field 名）→ 提交锁；⑤ reset(initial) 与 dirty 判定（values 与 initial 深比较）。写出实现。

```javascript
export function createForm({ initial, validators = {}, validateDeps = {} } = {}) {
  let values = { ...initial };
  let errors = {}, touched = {}, status = 'idle';
  let seq = 0;
  const subs = new Set();

  async function validateField(name) {
    const mySeq = ++seq;
    const fn = validators[name];
    if (!fn) { delete errors[name]; return true; }
    const msg = await fn(values[name], values);
    if (mySeq !== seq) return null;                    // 旧校验丢弃
    if (msg) errors[name] = msg; else delete errors[name];
    notify();
    return !msg;
  }
  async function setField(name, v) {
    values = { ...values, [name]: v };
    touched[name] = true;
    const deps = validateDeps[name] ?? [];             // 联动重校验
    await Promise.all([validateField(name), ...deps.map((d) => validateField(d))]);
  }
  async function submit(onSubmit) {
    status = 'submitting'; notify();
    const results = await Promise.all(Object.keys(validators).map((n) => validateField(n)));
    const bad = Object.keys(validators).find((n, i) => results[i] === false);
    if (bad) { status = 'idle'; notify(); return { ok: false, field: bad }; }
    try { await onSubmit(values); status = 'submitted'; return { ok: true }; }
    finally { notify(); }
  }
  const isDirty = () => JSON.stringify(values) !== JSON.stringify(initial);
  function notify() { subs.forEach((f) => f(snapshot())); }
  function snapshot() { return { values, errors, touched, status, dirty: isDirty() }; }
  return { setField, submit, sub(fn) { subs.add(fn); return () => subs.delete(fn); }, snapshot,
    reset: () => { values = { ...initial }; errors = {}; touched = {}; status = 'idle'; notify(); } };
}
```

**评分点**：
- validateDeps 联动（字段 A 变更触发依赖字段重校验）
- seq 竞态守卫（异步校验旧结果丢弃）
- submit 全量校验 → 首错 field 返回 + 提交锁
- dirty 判定与 reset 的快照语义
