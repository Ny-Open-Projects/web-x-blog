## RBL-472：手写值班响应机器人
**需求**：实现 `createOncallBot({ roster })`：1) 告警接收：分级路由（P0 全员广播 + 电话；P3 仅工单）；2) 自动诊断附随：告警触发时自动跑诊断脚本（结果附在告警里——值班 3 分钟内可行动）；3) 升级链：P0 未确认 10 分钟 → 升级到平台 owner；4) 静默窗口管理：维护窗口内告警抑制（不 false-alarm 打电话）；5) 值班表轮转与节假日覆盖；6) 事后：告警 → 工单自动创建（复盘不丢）。

```javascript
export function createOncallBot({ roster, channels, diagnostics }) {
  const silenced = []          // 维护窗口 [{ start, end }]
  let currentOncall = 0

  function oncallNow() {
    return roster[currentOncall % roster.length]               // 轮转
  }
  function rotate() { currentOncall++ }

  function inSilence() {
    const now = Date.now()
    return silenced.some((w) => now >= w.start && now <= w.end)
  }

  async function alert({ level, title, detail }) {
    if (inSilence()) return { suppressed: true }               // 维护窗口抑制
    const oncall = oncallNow()
    // 自动诊断附随
    const diag = diagnostics[title] ? await diagnostics[title]().catch((e) => ({ error: String(e) })) : null
    const message = { level, title, detail, diag, oncall, at: Date.now() }
    if (level === 'P0') {
      channels.phone(message)                                  // 电话
      channels.broadcast(message)                              // 全员广播
      const ack = await waitForAck(10 * 60_000)                // 升级链
      if (!ack) { channels.phone({ ...message, escalate: 'platform-owner' }); rotate() }
    } else if (level === 'P1') channels.im(message)
    else channels.ticket(message)                              // P3 工单
    await createTicket(message)                                // 事后工单
    return { delivered: true, diag }
  }

  function waitForAck(ms) { return new Promise((r) => { /* ack 监听简化 */ setTimeout(() => r(false), ms) }) }
  async function createTicket(m) { /* 工单系统 API */ }

  return { alert, rotate, oncallNow,
    silence: (start, end) => silenced.push({ start, end }),
    resume: () => silenced.length = 0 }
}
```
**评分点**：
- 分级路由（P0 电话+广播 / P3 工单——响应成本与严重度匹配）
- 自动诊断附随（告警带诊断结果——值班 3 分钟可行动）
- 升级链（P0 未确认升级 owner——告警不能沉没）
- 维护窗口抑制（防 false-alarm 疲劳——告警疲劳是真实问题）
- 事后工单（告警 → 复盘不丢）
