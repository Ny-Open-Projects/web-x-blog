## RBL-496：手写产物回放验证
**答**：实现 `smokeTestBundle(bundleDir, { scenarios })`：1) 无头浏览器加载产物（本地 server 起产物）——真实运行环境验证；2) 场景断言：页面加载无 console error / 关键节点渲染（selector 存在）/ 路由跳转可达；3) 性能预算冒烟：LCP 预算内（超大产物回退检测）；4) 与构建联动：smoke 失败 → 构建门禁阻断（产物不可发布）；5) 场景录制回放（用户行为脚本 → 自动化回放——冒烟场景可维护）。

```javascript
import puppeteer from 'puppeteer'
import http from 'node:http'
import handler from 'serve-handler'

export async function smokeTestBundle(bundleDir, { scenarios = defaultScenarios(), budgetMs = 3000 } = {}) {
  const server = http.createServer((req, res) => handler(req, res, { public: bundleDir }))
  await new Promise((r) => server.listen(0, r))
  const port = server.address().port

  const browser = await puppeteer.launch({ headless: 'new' })
  const results = []
  try {
    for (const s of scenarios) {
      const page = await browser.newPage()
      const errors = []
      page.on('pageerror', (e) => errors.push(e.message))
      page.on('console', (m) => m.type() === 'error' && errors.push(m.text()))

      const t0 = Date.now()
      await page.goto(`http://localhost:${port}${s.route}`, { waitUntil: 'networkidle2' })
      const lcp = await page.evaluate(() => new Promise((r) =>
        new PerformanceObserver((l) => r(l.entries.at(-1)?.startTime ?? 0)).observe({ type: 'largest-contentful-paint', buffered: true })))

      const ok = errors.length === 0
        && s.assertSelectors.every((sel) => page.$(sel) !== null)
        && lcp < budgetMs
      results.push({ route: s.route, ok, errors: errors.slice(0, 3), lcpMs: Math.round(lcp) })
      await page.close()
    }
  } finally {
    await browser.close(); server.close()
  }
  const pass = results.every((r) => r.ok)
  return { pass, results }                                      // pass=false → 构建门禁阻断
}

function defaultScenarios() {
  return [
    { route: '/', assertSelectors: ['#root', '[data-testid=header]'] },
    { route: '/products', assertSelectors: ['[data-testid=product-list]'] },
  ]
}
```
**评分点**：
- 真实环境验证（无头浏览器 + 本地 server——不是静态检查）
- 三类断言（console error/selector/LCP——功能+性能双冒烟）
- LCP 预算联动（超大产物回退检测——产物结构与运行时挂钩）
- 构建门禁联动（smoke 失败阻断发布——冒烟是有牙齿的）
