## RBL-487：手写发布窗口管理
**需求**：实现 `createReleaseWindow(calendar)`：1) 窗口定义：允许发布时段（如工作日 10:00-18:00，避开大促封网期）——日历驱动；2) 判定：发布请求时校验当前是否在窗口内（时区正确——多 region 各自窗口）；3) 紧急通道：封网期紧急发布走审批（双人 + 值班确认——不是硬拦截而是有门槛的例外）；4) 窗口冲突检测：同应用并发发布排队（锁）；5) 提前预告：窗口即将关闭提醒（「还有 30 分钟，是否发布」）；6) 审计：窗口外发布记录（谁走的紧急通道）。

```javascript
export function createReleaseWindow(calendar /* [{ region, days: [1-5], start: '10:00', end: '18:00', tz }] */, freezePeriods) {
  function isInWindow(region = 'default', now = new Date()) {
    if (freezePeriods.some((f) => now >= new Date(f.start) && now <= new Date(f.end)))
      return { ok: false, reason: 'freeze period (封网)' }
    const rules = calendar.filter((c) => c.region === region)
    for (const rule of rules) {
      const local = new Date(now.toLocaleString('en-US', { timeZone: rule.tz }))
      if (!rule.days.includes(local.getDay())) continue
      const [sh, sm] = rule.start.split(':').map(Number)
      const [eh, em] = rule.end.split(':').map(Number)
      const mins = local.getHours() * 60 + local.getMinutes()
      if (mins >= sh * 60 + sm && mins <= eh * 60 + em)
        return { ok: true, remainingMin: eh * 60 + em - mins }
    }
    return { ok: false, reason: `outside window (region ${region})` }
  }

  const locks = new Map()                                        // 并发发布锁
  async function tryRelease({ app, region, urgent = false, approvals = [] }) {
    const win = isInWindow(region)
    if (!win.ok) {
      if (!urgent || approvals.length < 2)                       // 紧急通道双人
        return { released: false, reason: win.reason,
          hint: urgent ? '紧急发布需要 2 人审批（值班 + owner）' : '等待窗口或走紧急通道' }
    }
    if (locks.has(app)) return { released: false, reason: 'concurrent release in progress' }   // 排队
    locks.set(app, Date.now())
    try {
      return { released: true, remainingMin: win.remainingMin }
    } finally { setTimeout(() => locks.delete(app), 60_000) }    // 释放防死锁
  }

  return { isInWindow, tryRelease }
}
```
**评分点**：
- 时区正确（多 region 各自窗口——toLocaleString tz 处理）
- 封网期独立于时段窗口（业务日历维度——两套日历叠加）
- 紧急通道设计（双人审批——不是硬拦截而是有门槛例外）
- 并发发布锁 + 超时释放（防死锁）
