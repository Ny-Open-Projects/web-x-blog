## RBL-483：手写依赖解析缓存
**答**：实现 `createResolveCache(resolver)`：1) 缓存键：`{ importer, spec, conditions }` → 解析结果——同 importer+spec 复用；2) 失效粒度：文件级（importer 所在目录变更失效该目录相关条目）、包级（node_modules/xxx 变更失效该包全部）；3) 缓存穿透保护：解析失败的缓存（negative cache——不存在的路径反复 stat 浪费）但 TTL 短（新文件创建后尽快生效）；4) 缓存一致性：目录 mtime 快照校验（缓存命中前校验目录未变——防误命中）；5) stats 命中率与 miss 原因。

```javascript
export function createResolveCache(resolver) {
  const cache = new Map()          // key → { result, dirMtimes }
  const negCache = new Map()       // key → { at, ttlMs: 30_000 }
  let hits = 0, misses = 0
  const missReasons = { cold: 0, invalidated: 0, negative: 0 }

  async function resolve(importer, spec, conditions) {
    const key = JSON.stringify([importer, spec, conditions])

    const neg = negCache.get(key)
    if (neg && Date.now() - neg.at < neg.ttlMs) {
      missReasons.negative++; misses++
      return null                                                  // negative cache 短 TTL
    }

    const hit = cache.get(key)
    if (hit && await dirsUnchanged(hit.dirMtimes)) { hits++; return hit.result }
    if (hit) missReasons.invalidated++; else missReasons.cold++
    misses++

    const result = await resolver(importer, spec, conditions)
    if (result === null) { negCache.set(key, { at: Date.now() }); return null }   // 失败缓存
    cache.set(key, { result, dirMtimes: await snapshotDirs([importer, pathDir(result)]) })
    return result
  }

  async function snapshotDirs(dirs) {
    const out = {}
    for (const d of dirs) {
      try { out[d] = fs.statSync(d).mtimeMs } catch { out[d] = null }
    }
    return out
  }
  async function dirsUnchanged(mtimes) {
    for (const [d, m] of Object.entries(mtimes)) {
      let cur = null
      try { cur = fs.statSync(d).mtimeMs } catch {}
      if (cur !== m) return false                                  // 目录变 → 失效
    }
    return true
  }

  function invalidatePackage(pkgName) {                            // 包级失效
    for (const [key, hit] of cache)
      if (key.includes(`node_modules/${pkgName}`)) cache.delete(key)
  }

  return { resolve, invalidatePackage, invalidateAll: () => cache.clear(),
    stats: () => ({ hits, misses, rate: hits + misses ? hits / (hits + misses) : 0, missReasons }) }
}
const pathDir = (p) => p.replace(/\/[^/]+$/, '')
const fs = require('node:fs')
```
**评分点**：
- 缓存键含 conditions（resolve 条件影响结果——键不完整即错误命中）
- 目录 mtime 校验（命中前校验未变——防止误命中的低成本防线）
- negative cache 短 TTL（解析失败也缓存但不永久——新文件创建生效延迟可控）
- invalidatePackage 包级失效（node_modules 变更联动）+ miss 原因分类
