## RBL-636：笔试：手写增量编译最小失效集计算器

**答**：实现 `invalidate({ changed, oldMeta, newMeta, importersOf, isEntry })`。oldMeta/newMeta: `Map<file, { exports: string[], imports: string[] }>`（newMeta 只需含 changed 与其新增依赖）。返回 `{ retransform: string[], rechunck: boolean, reason }`：1) exports 与 imports 均无变化 → 只重 transform changed，rechunck=false；2) imports 新增/删除 → rechunck=true（chunk 结构可能变），传播 changed 的 importers；3) 仅 exports 变化 → 沿 importers 反向传播（BFS，环去重），rechunck=true；4) changed 是 entry 本身 → 全量失效（retransform 含所有模块，reason='entry'）；5) 传播只到「accept 边界」可选跳过（isEntry(file) 为 true 时停止继续向上传播但仍计入）。

```javascript
function invalidate({ changed, oldMeta = new Map(), newMeta, importersOf, isEntry = () => false }) {
  const old = oldMeta.get(changed) ?? { exports: [], imports: [] };
  const cur = newMeta.get(changed) ?? { exports: [], imports: [] };

  const expDiff = diff(old.exports, cur.exports);
  const impDiff = diff(old.imports, cur.imports);

  if (isEntry(changed))
    return { retransform: 'ALL', rechunck: true, reason: 'entry' };

  if (expDiff.size === 0 && impDiff.size === 0)
    return { retransform: [changed], rechunck: false, reason: 'content-only' };

  const stale = new Set([changed]);
  const queue = [changed];
  while (queue.length) {
    const id = queue.shift();
    for (const imp of importersOf(id) || []) {
      if (stale.has(imp) || isEntry(imp)) {              // entry 计入但停止向上传播
        if (isEntry(imp)) stale.add(imp);
        continue;
      }
      stale.add(imp);
      queue.push(imp);
    }
  }
  return { retransform: [...stale], rechunck: impDiff.size > 0 || expDiff.size > 0,
           reason: expDiff.size ? 'exports' : 'imports' };
}
const diff = (a, b) => {
  const s = new Set([...a.filter(x => !b.includes(x)), ...b.filter(x => !a.includes(x))]);
  return s;
};

// 自测
// 图：entry -> page -> Comp；lib 被 page/Comp 引用；util 被 lib 引用
const importersOf = (id) => ({
  page: ['entry'], Comp: ['page'], lib: ['page', 'Comp'],
  util: ['lib'],
}[id] || []);

// 1) 内容变化（接口不变）
let r = invalidate({
  changed: 'Comp',
  oldMeta: new Map([['Comp', { exports: ['C'], imports: ['lib'] }]]),
  newMeta: new Map([['Comp', { exports: ['C'], imports: ['lib'] }]]),
  importersOf, isEntry: (f) => f === 'entry',
});
console.log(r.reason === 'content-only' && r.rechunck === false);

// 2) exports 变化：传播 page -> entry
r = invalidate({
  changed: 'lib',
  oldMeta: new Map([['lib', { exports: ['a'], imports: [] }]]),
  newMeta: new Map([['lib', { exports: ['a', 'b'], imports: [] }]]),
  importersOf, isEntry: (f) => f === 'entry',
});
console.log(r.reason === 'exports' && r.retransform.includes('page') &&
            r.retransform.includes('entry'));

// 3) entry 变化：全量
r = invalidate({ changed: 'entry', oldMeta: new Map(), newMeta: new Map(), importersOf, isEntry: f => f === 'entry' });
console.log(r.reason === 'entry' && r.retransform === 'ALL');

// 4) 环收敛
r = invalidate({
  changed: 'lib',
  oldMeta: new Map([['lib', { exports: [], imports: [] }]]),
  newMeta: new Map([['lib', { exports: [], imports: ['Comp'] }]]),   // lib -> Comp（环）
  importersOf, isEntry: () => false,
});
console.log(r.rechunck === true);                        // imports 变化触发 rechunck，无死循环
```

**评分点**：
- 三级粒度判定正确：content-only 不 rechunck；exports/imports 变化触发反向传播与 rechunck
- BFS 反向传播带访问去重（环收敛）；entry 双语义正确（变更即全量、传播到 entry 停止但计入）
- reason 可追溯（content-only/exports/imports/entry），便于失效指标观测
- 能说清「漏失效是静默错误、存疑向保守降级」属于加分
