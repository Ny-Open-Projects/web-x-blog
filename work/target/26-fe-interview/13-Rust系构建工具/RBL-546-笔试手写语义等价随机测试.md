## RBL-546：手写语义等价随机测试（源码 vs 产物输出比对）
**答**：545 L3 落地——实现 `randomEquivalenceTest(srcFn, prodFn, gen, { samples = 100 })`：1) 输入生成器 `gen()` 产出随机输入；2) 采样执行：srcFn(input) 与 prodFn(input) 输出比对（深比较）；3) 失败最小化（缩小输入到最小复现）；4) 附用例：检出压缩引入的边界 bug 并最小化。

```javascript
function deepEqual(a, b) { return JSON.stringify(a) === JSON.stringify(b) }

function randomEquivalenceTest(srcFn, prodFn, gen, { samples = 100, minimize = true } = {}) {
  const failures = []
  for (let i = 0; i < samples; i++) {
    const input = gen()
    let srcOut, prodOut
    try { srcOut = srcFn(input) } catch (e) { srcOut = { error: e.message } }
    try { prodOut = prodFn(input) } catch (e) { prodOut = { error: e.message } }
    if (!deepEqual(srcOut, prodOut)) {
      failures.push({ input, srcOut, prodOut })
    }
  }
  // 失败最小化：随机删减输入字段直到仍复现
  const minimized = minimize && failures.length
    ? failures.slice(0, 3).map((f) => minimizeInput(f, srcFn, prodFn))
    : []
  return { failures: failures.length, samples, minimized, pass: failures.length === 0 }
}

function minimizeInput(failure, srcFn, prodFn) {
  let input = failure.input
  if (typeof input !== 'object' || input === null) return input
  let keys = Object.keys(input)
  let changed = true
  while (changed) {
    changed = false
    for (const k of [...keys]) {
      const smaller = {}
      keys.filter((x) => x !== k).forEach((x) => (smaller[x] = input[x]))
      let s, p
      try { s = srcFn(smaller) } catch (e) { s = { error: e.message } }
      try { p = prodFn(smaller) } catch (e) { p = { error: e.message } }
      if (!deepEqual(s, p)) { input = smaller; keys = Object.keys(smaller); changed = true }   // 仍复现：保留删减
    }
  }
  return input
}

// 用例：压缩 bug 模拟（prod 版对 NaN 输入行为不同）
const srcFn = (x) => ({ r: x.value === undefined ? 'u' : String(x.value) })
const prodFn = (x) => ({ r: x.value == null ? 'u' : String(x.value) })   // == null 把 undefined 和 null 混同
const gen = () => ({
  value: [undefined, null, 0, 's', false, NaN][Math.floor(Math.random() * 6)],
})
const r = randomEquivalenceTest(srcFn, prodFn, gen, { samples: 50 })
console.log(r.failures > 0)                          // true：检出（null 情形行为不一致）
console.log(typeof r.minimized[0])                   // 最小化输入
console.log(r.pass)                                  // false
```
**评分点**：
- 随机输入采样执行（源码 vs 产物双跑比对）
- 失败最小化（随机删减字段直到仍复现——delta debugging 思想）
- 错误形态归一（catch 后统一 { error } 结构比对）
- 采样数成本控制（samples 可配）
