## RBL-921：笔试：手写游标分页模拟器（游标 + 插入去重）

**答**：实现 `createFeed(items)`，返回 `fetch({ cursor, limit })`。items 可中途插入新数据（`insert(item)`）。规则：1) 按 sortKey 降序 + id 升序稳定排序；2) cursor 为 `{ sort, id }`：返回 (sort, id) 之后（更小/更旧）的 limit 条 + nextCursor + hasMore；3) cursor 为 null → 从头取；4) 到尾：nextCursor=null, hasMore=false；5) **客户端合并器** `createMerger()`：`merge(page)` 按已存 id 去重（中间插入导致重复拉取）；返回新增条数。边界：limit 0；items 空；cursor 指向已删除的项（按 sort/id 仍可定位——用 (sort,id) 比较而非引用）。

```javascript
function createFeed(items = [], { cmp = (a, b) => b.sort - a.sort || (a.id < b.id ? -1 : 1) } = {}) {
  const all = [...items];
  return {
    insert(item) { all.push(item); },
    fetch({ cursor = null, limit = 10 } = {}) {
      const sorted = [...all].sort(cmp);
      let start = 0;
      if (cursor) {
        const c = sorted.findIndex(x => x.id === cursor.id && x.sort === cursor.sort);
        if (c >= 0) start = c + 1;
        else start = sorted.findIndex(x =>
          cmp(x, { id: cursor.id, sort: cursor.sort }) > 0);   // 原项已删：按比较定位
        if (start < 0) start = sorted.length;
      }
      const page = sorted.slice(start, start + limit);
      const next = page.length ? page[page.length - 1] : null;
      const hasMore = start + limit < sorted.length;
      return { items: page,
        nextCursor: hasMore && next ? { sort: next.sort, id: next.id } : null,
        hasMore };
    },
  };
}
function createMerger() {
  const seen = new Set();
  return {
    merge(page) {
      let added = 0;
      for (const it of page.items ?? []) {
        if (!seen.has(it.id)) { seen.add(it.id); added++; }
      }
      return added;
    },
    size: () => seen.size,
  };
}

// 自测
const feed = createFeed([
  { id: 'a', sort: 100 }, { id: 'b', sort: 90 }, { id: 'c', sort: 80 },
  { id: 'd', sort: 70 }, { id: 'e', sort: 60 },
]);
let p1 = feed.fetch({ limit: 2 });
console.log(p1.items.map(x => x.id).join(','));          // a,b
console.log(JSON.stringify(p1.nextCursor));              // { sort: 90, id: 'b' }
// 中间插入（旧游标之前的新数据）
feed.insert({ id: 'n', sort: 95 });
let p2 = feed.fetch({ cursor: p1.nextCursor, limit: 2 });
console.log(p2.items.map(x => x.id).join(','));          // c,d（n 在游标前不重复出现）

// 合并器：重复拉取去重
const m = createMerger();
m.merge(p1);
const added = m.merge(p2);
console.log(added === 2 && m.size() === 4);              // c,d 新增（a,b 已见）

// 原项被删（cursor 定位失效 → 比较定位）
p2 = feed.fetch({ cursor: { sort: 80, id: 'ghost' }, limit: 2 });
console.log(p2.items.map(x => x.id).join(','));          // c,d（80 之后）

// 到尾
const p3 = feed.fetch({ cursor: { sort: 60, id: 'e' }, limit: 10 });
console.log(p3.hasMore === false && p3.nextCursor === null);
```

**评分点**：
- 排序 + 游标定位正确（存在匹配 → 后一条；已删除 → 比较定位——游标锚定实体语义）
- 中间插入不产生重复（新数据在游标之前自然被跳过——游标语义的稳定性）；合并器去重兜底
- hasMore/nextCursor 到尾语义；边界
- 能说清「游标锚定实体 vs 页码锚定位置——数据变更场景游标保证增量语义稳定」属于加分
