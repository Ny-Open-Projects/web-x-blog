## RBL-433：手写构建配置热重载
**答**：实现 `createConfigWatcher(loadConfig, applyConfig)`：1) watch 配置文件变更 → 防抖 300ms → 重新加载 → 校验（schema）→ 通过才应用（失败保持旧配置 + 报错提示）；2) 部分可热更（如 resolve.alias）与需重启（如 plugins——模块图级变更）分类；3) 可热更项增量应用（diff 新旧配置只更新变化部分）；4) 应用事件通知（哪些配置生效了）；5) 快速连续编辑合并（防抖）；6) 回滚：新配置应用后异常 → 自动回滚旧配置。

```javascript
import chokidar from 'chokidar'

export function createConfigWatcher(loadConfig, applyConfig, { validate, hotKeys = ['resolve'] } = {}) {
  let current = null
  let timer = null

  const watcher = chokidar.watch(['rsbuild.config.*', 'vite.config.*'])
  watcher.on('change', () => {
    clearTimeout(timer)
    timer = setTimeout(reload, 300)                              // 防抖
  })

  async function reload() {
    let next
    try { next = await loadConfig() } catch (e) { return warn(`config parse fail: ${e.message}`) }
    const errors = validate?.(next)
    if (errors?.length) return warn(`config invalid: ${errors[0].message}`)   // 校验不过保持旧配置
    const prev = current
    try {
      const hotDiff = diffConfigs(prev, next, hotKeys)           // 可热更部分
      applyConfig(hotDiff)
      current = next
      notify({ applied: Object.keys(hotDiff) })                  // 生效通知
    } catch (e) {
      warn(`apply fail, rollback: ${e.message}`)
      applyConfig(diffConfigs(next, prev, hotKeys))              // 自动回滚旧配置
    }
  }

  function diffConfigs(prev, next, hotKeys) {
    const diff = {}
    for (const key of hotKeys)
      if (JSON.stringify(prev?.[key]) !== JSON.stringify(next?.[key])) diff[key] = next?.[key]
    return diff
  }
  function warn(m) { notify({ error: m }) }
  function notify(msg) { /* WS 推送到浏览器 overlay */ }
  return { reload }
}
```
**评分点**：
- 防抖 + 校验前置（失败保持旧配置——热重载不引入坏状态）
- 冷热分类（alias 可热更 / plugins 需重启——诚实标注边界）
- diff 增量应用 + 应用事件通知（用户知道什么生效了）
- 异常自动回滚（应用失败恢复旧配置）
