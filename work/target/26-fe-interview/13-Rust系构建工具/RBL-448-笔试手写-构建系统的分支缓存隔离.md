## RBL-448：手写分支缓存隔离与合并策略
**需求**：实现 `createBranchCache(remoteCache, { baseBranch })`：1) 缓存命名空间：`branch:feature-x` 独立于 `branch:main`——特性分支缓存不污染基线；2) 读取链：本分支 → base 分支（main 缓存可被特性分支读——同 commit 前提下产物一致）；3) 写策略：特性分支写本命名空间；main 写全局命名空间（CI main 才写）——本地分支永不写共享层；4) 缓存迁移：分支合并后（PR merged）把分支缓存标记晋升到 main 命名空间（避免 main 重建）；5) 过期：分支删除 7 天后命名空间清理。

```javascript
export function createBranchCache(remoteCache, { baseBranch = 'main' } = {}) {
  function nsOf(branch, isCI) {
    if (branch === baseBranch && isCI) return 'shared:main'   // CI main 全局共享
    if (branch === baseBranch) return 'local:main'            // 本地 main 独立
    return `branch:${branch}`                                  // 特性分支独立
  }

  async function fetch(key, branch, isCI) {
    // 读链：本分支 → base 分支
    const own = await remoteCache.fetch(`${nsOf(branch, isCI)}/${key}`)
    if (own) return { data: own, source: nsOf(branch, isCI) }
    if (branch !== baseBranch) {
      const base = await remoteCache.fetch(`shared:main/${key}`)
      if (base) return { data: base, source: 'shared:main', borrowed: true }   // 借用基线
    }
    return null
  }

  async function put(key, files, branch, isCI) {
    return remoteCache.put(`${nsOf(branch, isCI)}/${key}`, files)   // 本地分支永不写 shared
  }

  async function promote(branch) {                             // PR merged 晋升
    const entries = await remoteCache.list(`branch:${branch}`)
    for (const { key, files } of entries)
      await remoteCache.putIfAbsent(`shared:main/${key}`, files)   // 不覆盖已有
  }

  return { fetch, put, promote,
    cleanup: (branch, deletedAt) => Date.now() - deletedAt > 7 * 86400_000 }
}
```
**评分点**：
- 三级命名空间（CI main 共享 / 本地 main 独立 / 特性分支独立——本地不污染共享层）
- 读链借用基线（main 缓存对特性分支只读共享——跨分支复用核心）
- 晋升 putIfAbsent（合并后不重建、不覆盖——竞态安全）
- 过期清理（分支删除后命名空间回收）
