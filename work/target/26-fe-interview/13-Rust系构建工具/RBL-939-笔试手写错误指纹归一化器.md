## RBL-939：笔试：手写错误指纹归一化器（归一 + 分组）

**答**：实现两部分。1) `fingerprint(err)`：err: `{ name, message, stack }`——stack 解析 `at 函数名 (file:line:col)` 帧：归一化 = 类型 + 前 5 帧的「文件名（剥 hash 与 query）」序列（忽略函数名/行号/列号——跨构建稳定）；无堆栈 → 类型 + message 前 80 字符。返回 8 位 hash（djb2）。2) `group(errors)`：按指纹分组 `{ fingerprint, count, sample: 首个错误, buildIds: Set }`——按 count 降序。边界：无 stack；file 含 hash/query；同错误不同构建的堆栈文件路径不同（hash 化产物名——剥 `[name].[hash].js` → `[name].js`）。

```javascript
const djb2 = (s) => { let h = 5381; for (const c of s) h = ((h << 5) + h + c.charCodeAt(0)) >>> 0;
  return h.toString(16).padStart(8, '0'); };
const normalizeFile = (f) => f.split(/[?#]/)[0]
  .replace(/([.\w-]+)\.[a-f0-9]{6,}\.js$/, '$1.js')      // main.abc123.js -> main.js
  .replace(/^https?:\/\/[^/]+/, '');

function fingerprint(err = {}) {
  const frames = [];
  for (const line of (err.stack ?? '').split('\n')) {
    const m = /at\s+(?:[\w$.]+\s+\()?([^):]+):(\d+):(\d+)\)?/.exec(line)
      ?? /at\s+([^(:]+):(\d+):(\d+)/.exec(line);
    if (m) {
      frames.push(normalizeFile(m[1]));
      if (frames.length >= 5) break;
    }
  }
  const key = frames.length
    ? [err.name ?? 'Error', ...frames].join('|')
    : `${err.name ?? 'Error'}|${(err.message ?? '').slice(0, 80)}`;
  return djb2(key);
}

function group(errors = []) {
  const by = new Map();
  for (const e of errors) {
    const fp = fingerprint(e);
    (by.get(fp) || by.set(fp, { fingerprint: fp, count: 0, sample: e, buildIds: new Set() }).get(fp))
      .count++;
    if (e.buildId) by.get(fp).buildIds.add(e.buildId);
  }
  return [...by.values()].sort((a, b) => b.count - a.count);
}

// 自测
const stackA = [
  'TypeError: x is not a function',
  '    at handleClick (/js/main.abc123.js:1:234)',
  '    at dispatch (/js/vendor.def456.js:9:100)',
  '    at /js/main.abc123.js:2:100',
].join('\n');
const stackB = [                                        // 同 bug 新构建：hash 变了
  'TypeError: x is not a function',
  '    at handleClick (/js/main.987xyz.js:1:888)',
  '    at dispatch (/js/vendor.111aaa.js:9:55)',
  '    at /js/main.987xyz.js:2:77',
].join('\n');
const errA = { name: 'TypeError', message: 'x is not a function', stack: stackA, buildId: 'r1' };
const errB = { name: 'TypeError', message: 'x is not a function', stack: stackB, buildId: 'r2' };
console.log(fingerprint(errA) === fingerprint(errB));   // 跨构建同指纹

const errC = { name: 'TypeError', message: 'y undefined' };   // 无堆栈
console.log(typeof fingerprint(errC) === 'string');
console.log(fingerprint(errC) !== fingerprint(errA));

const g = group([errA, errB, errC, { ...errA, buildId: 'r3' }]);
console.log(g[0].count === 3);                          // 同 bug 3 条
console.log(g[0].buildIds.size === 3);                  // r1/r2/r3
console.log(g[1].count === 1);
console.log(group([]).length === 0);
```

**评分点**：
- 归一化剥除 hash 与 query（产物名跨构建稳定——指纹跨版本有效的关键）；函数名/行列忽略
- 无堆栈回退（类型+消息前 80）；5 帧截断
- 分组聚合正确（count/buildIds 集合/降序）；边界
- 能说清「指纹跨构建稳定性依赖反解与文件名归一——聚合质量决定监控价值（3 个 bug vs 10 万噪声）」属于加分
