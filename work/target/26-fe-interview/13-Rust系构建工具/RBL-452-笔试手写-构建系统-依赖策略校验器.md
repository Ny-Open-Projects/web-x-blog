## RBL-452：手写依赖策略校验器
**答**：实现 `validateDeps(pkgJson, policy)`：1) policy：`{ banned: [{ name, reason }], ranges: { name: '>=1.2.0 <2' }, locked: { name: '1.5.2' } }`；2) semver 范围校验（简化实现：>= < 精确三种比较——真实用 semver 库）；3) banned 命中报错带 reason 与替代建议；4) monorepo 扩展：包间依赖也受策略管（内部包版本对齐）；5) 输出 { ok, violations: [{ pkg, rule, detail }] }；6) CI 集成：PR 改 package.json/lockfile 才触发（增量检查）。

```javascript
function compareVersion(a, b) {
  const pa = a.split('.').map(Number), pb = b.split('.').map(Number)
  for (let i = 0; i < 3; i++) {
    if ((pa[i] ?? 0) > (pb[i] ?? 0)) return 1
    if ((pa[i] ?? 0) < (pb[i] ?? 0)) return -1
  }
  return 0
}

function satisfiesRange(version, range) {
  // 支持 '>=1.2.0 <2' 与 '1.5.2'
  for (const part of range.split(/\s+(?=<)/)) {
    if (part.startsWith('>=')) { if (compareVersion(version, part.slice(2)) < 0) return false }
    else if (part.startsWith('<')) { if (compareVersion(version, part.slice(1)) >= 0) return false }
    else if (compareVersion(version, part) !== 0) return false
  }
  return true
}

export function validateDeps(pkgJson, policy) {
  const violations = []
  const all = { ...pkgJson.dependencies, ...pkgJson.devDependencies }
  for (const [name] of Object.entries(all)) {
    const banned = policy.banned.find((b) => b.name === name)
    if (banned) violations.push({ pkg: name, rule: 'banned',
      detail: `${banned.reason}${banned.alternative ? `，替代: ${banned.alternative}` : ''}` })
    const range = policy.ranges?.[name]
    if (range) {
      const v = all[name].replace(/[^0-9.]/g, '')                // 取实际锁定版本需 lockfile，简化用声明
      if (!satisfiesRange(v, range)) violations.push({ pkg: name, rule: 'range', detail: `${v} 不满足 ${range}` })
    }
    const locked = policy.locked?.[name]
    if (locked && all[name] !== locked)
      violations.push({ pkg: name, rule: 'locked', detail: `必须锁定 ${locked}，当前 ${all[name]}` })
  }
  return { ok: violations.length === 0, violations }
}
```
**评分点**：
- 三种策略语义（banned 带替代建议 / range / locked 精确锁定）
- semver 比较正确（逐段数值比较；范围解析 >= < 空格分段）
- 简化点标注（真实实现 semver 库 + lockfile 实际解析版本——声明版本不等于锁定版本）
- CI 增量触发（只查变更文件——策略检查不拖慢全部 PR）
