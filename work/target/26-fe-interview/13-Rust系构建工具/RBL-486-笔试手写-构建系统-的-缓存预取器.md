## RBL-486：手写缓存预取器
**答**：实现 `createPrefetcher(remoteCache)`：1) 预测任务：PR 创建时预测将构建的任务集（affected 计算 + 历史任务序列）→ 提前预取缓存（构建开始时缓存已在本地）；2) 预取并发控制（不挤占正常流量——低优先级带宽）；3) 预取收益度量：命中率提升 vs 带宽成本（预取的 ROI 追踪）；4) 失效保护：预取后构建 key 变了（预取的 key 与实际不符）→ 记录预测偏差（改进预测）；5) 预测准确性：预测集 vs 实际构建集的 Jaccard 相似度。

```javascript
export function createPrefetcher(remoteCache, { bandwidthLimitMBs = 50 } = {}) {
  let predictions = [], actuals = []

  async function prefetchFor(pr) {
    const predictedTasks = predictTasks(pr)                      // affected + 历史序列
    predictions.push({ pr: pr.id, tasks: predictedTasks, at: Date.now() })
    let budget = bandwidthLimitMBs
    for (const task of predictedTasks) {
      const size = await remoteCache.head(task.cacheKey)
      if (!size) continue
      if (budget - size / 1024 ** 2 < 0) break                   // 带宽预算
      budget -= size / 1024 ** 2
      await remoteCache.prefetch(task.cacheKey, { priority: 'low' }).catch(() => {})
    }
  }

  function predictTasks(pr) {
    const affected = computeAffected(pr.changedFiles)            // affected 闭包
    const historical = pr.historyTasks?.slice(-10) ?? []         // 该 PR 作者常跑的任务
    return [...new Set([...affected, ...historical])]
  }

  function calibrate() {
    let hits = 0, total = 0
    for (const p of predictions) {
      const act = actuals.find((a) => a.pr === p.pr)
      if (!act) continue
      total++
      const inter = p.tasks.filter((t) => act.tasks.includes(t)).length
      const union = new Set([...p.tasks, ...act.tasks]).size
      if (inter / union > 0.5) hits++                            // Jaccard > 0.5 算命中
    }
    return { predictHitRate: total ? hits / total : 0, samples: total,
      note: 'Jaccard 相似度 > 0.5 算有效预测' }
  }

  return { prefetchFor, recordActual: (pr, tasks) => actuals.push({ pr: pr.id, tasks }),
    calibrate }
}
```
**评分点**：
- 预测任务集（affected + 历史行为——预测不是瞎猜）
- 带宽预算（预取不挤占正常流量——低优先级）
- Jaccard 校准（预测准确性自证——预取器的可信度）
- ROI 意识（命中率提升 vs 带宽成本——预取不是无脑拉）
