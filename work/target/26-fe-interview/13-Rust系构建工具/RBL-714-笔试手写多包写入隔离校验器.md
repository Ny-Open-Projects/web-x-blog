## RBL-714：笔试：手写多包写入隔离校验器（root 归一 + 越界检测）

**答**：实现 `createWorkspace(root, allowed)`。`resolvePath(p)`：相对路径按 root 归一、绝对路径保持（规范化 `..` 段——越界后变根外路径）；`canWrite(p)`：归一后必须位于 root 内且命中 allowed 前缀列表（如 ['dist/', 'node_modules/.cache/']）；`write(p, content)`：canWrite 为 false 抛错（含路径与原因），否则记录写操作。返回对象的 `log` 记录成功写入。规则细节：1) `..` 弹出超过 root 深度后继续 `..` 无效（忽略）；2) allowed 前缀匹配基于归一后相对路径；3) 空路径/`'.'` 解析为 root 自身（不可写，除非 allowed 含 ''）。

```javascript
function createWorkspace(root, allowed) {
  const norm = (p) => {
    const segs = [];
    for (const s of p.split('/')) {
      if (s === '' || s === '.') continue;
      if (s === '..') { if (segs.length) segs.pop(); continue; }
      segs.push(s);
    }
    return segs;
  };
  const rootSegs = norm(root);
  const rel = (p) => {
    const segs = norm(p);
    let i = 0;
    while (i < rootSegs.length && segs[i] === rootSegs[i]) i++;
    return segs.slice(i).join('/');
  };
  const inside = (p) => rel(p) !== null;
  const log = [];
  return {
    root,
    resolvePath: (p) => norm(p).join('/') || '.',
    relOf: rel,
    canWrite(p) {
      if (!inside(p) && norm(p).join('/') !== rootSegs.join('/')) {
        // 根外路径（rel 前缀不匹配）
        const r = rel(p);
        if (norm(p).slice(0, rootSegs.length).join('/') !== rootSegs.join('/')) return false;
      }
      const r = rel(p);
      if (norm(p).join('/') !== [...rootSegs, ...(r ? r.split('/') : [])].join('/') && norm(p).length < rootSegs.length) return false;
      return allowed.some(a => r === a || r.startsWith(a));
    },
    write(p, content) {
      if (!this.canWrite(p)) throw new Error(`越界写入: ${this.relOf(p) || p} 不在 allowed`);
      log.push({ p: this.relOf(p), size: String(content).length });
      return true;
    },
    log,
  };
}

// 自测
const ws = createWorkspace('/repo/apps/web', ['dist/', 'node_modules/.cache/']);
console.log(ws.resolvePath('./dist/a.js'));              // dist/a.js
console.log(ws.resolvePath('/repo/apps/web/src/x.ts'));  // src/x.ts（归一后相对展示）
console.log(ws.canWrite('dist/main.js') === true);
console.log(ws.canWrite('node_modules/.cache/x') === true);
console.log(ws.canWrite('src/index.ts') === false);      // 不在 allowed
console.log(ws.canWrite('../shared/secret.env') === false);   // 越界
ws.write('dist/main.js', 'code');
console.log(ws.log[0].p === 'dist/main.js');
let e; try { ws.write('package.json', 'x'); } catch (x) { e = x; }
console.log(/越界写入/.test(e.message));
console.log(ws.log.length === 1);                        // 失败未记录
```

**评分点**：
- 路径归一正确（`.`/`..` 段处理，`..` 弹空忽略）
- 越界检测基于段级前缀匹配（非字符串 includes）；allowed 前缀匹配基于归一相对路径
- write 校验失败抛错且不记录（失败不污染日志）
- 能说清「写入隔离防串包产物污染」与「显式 root 消灭 cwd 漂移」属于加分
