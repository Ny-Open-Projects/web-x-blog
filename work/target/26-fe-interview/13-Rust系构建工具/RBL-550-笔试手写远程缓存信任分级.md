## RBL-550：手写远程缓存信任分级（分支隔离 + 签名验签 + 抽检）
**答**：549 落地——实现 `createRemoteCache()`：1) `put(key, value, { branch, sign })`：写入——签名（简单 HMAC）+ 分支标记（PR 缓存 key 带 branch 前缀隔离）；2) `get(key, { branch, verify })`：读取——PR 分支只读本分支 + 主分支缓存验签通过才可用；3) 抽检：`audit(sampleRate)` 随机采样重算比对；4) 附用例：PR 隔离与验签拒绝。

```javascript
import { createHmac, createHash } from 'crypto'
const SECRET = 'team-secret'

function sign(value) { return createHmac('sha256', SECRET).update(value).digest('hex').slice(0, 16) }

function createRemoteCache({ sampleRate = 0.2, recompute } = {}) {
  const store = new Map()       // key -> { value, sig, branch }
  const auditLog = []

  function scopedKey(key, branch) {
    return branch.startsWith('pr-') ? `pr/${branch}/${key}` : `main/${key}`   // PR 隔离
  }

  function put(key, value, { branch = 'main' } = {}) {
    const sk = scopedKey(key, branch)
    store.set(sk, { value, sig: sign(value), branch })
    return sk
  }

  function get(key, { branch = 'main', verify = true, recomputeFn } = {}) {
    const sk = scopedKey(key, branch)
    const entry = store.get(sk)
    if (!entry) return null
    if (verify && entry.sig !== sign(entry.value)) {
      return { corrupted: true, reason: '签名校验失败——缓存可能被篡改' }     // 完整性
    }
    // 抽检：采样重算比对（LRU 比例验证）
    if (recomputeFn && Math.random() < sampleRate) {
      const fresh = recomputeFn(key)
      const match = fresh === entry.value
      auditLog.push({ key: sk, match })
      if (!match) return { corrupted: true, reason: '抽检重算不一致——缓存脏' }
    }
    return { value: entry.value, branch: entry.branch }
  }

  function audit() { return { samples: auditLog.length, allMatch: auditLog.every((a) => a.match) } }

  return { put, get, audit, size: () => store.size }
}

// 用例
const cache = createRemoteCache({ sampleRate: 1, recompute: (key) => key === 'k1' ? 'artifact-k1' : 'artifact-tampered' })
cache.put('k1', 'artifact-k1', { branch: 'main' })
cache.put('k1', 'artifact-k1', { branch: 'pr-123' })          // PR 隔离存储

// PR 分支读不到主分支缓存（隔离）
console.log(cache.get('k1', { branch: 'pr-456' }))            // null
// PR 分支读本分支缓存
console.log(cache.get('k1', { branch: 'pr-123' }).value)      // 'artifact-k1'
// 主分支 + 验签通过
console.log(cache.get('k1', { branch: 'main' }).value)        // 'artifact-k1'

// 篡改检测
cache.put('k2', 'artifact-k2', { branch: 'main' })
cache.get('k2', { branch: 'main' })
const tampered = createRemoteCache({ recompute: () => 'different' })
tampered.put('k1', 'artifact-k1', { branch: 'main' })
console.log(tampered.get('k1', { branch: 'main', recomputeFn: tampered.put && (() => 'different') }))
// { corrupted: true, reason: '抽检重算不一致——缓存脏' }
```
**评分点**：
- 分支隔离 key（pr/ 前缀——PR 缓存不污染主缓存）
- HMAC 签名验签（完整性——篡改检出）
- 抽检重算（LRU 采样验证缓存正确性——auditLog）
- 主分支缓存全员可复用（信任分级语义）
