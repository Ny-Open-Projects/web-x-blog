## RBL-935：笔试：手写自顶向下预算分配器（TTI 反推）

**答**：实现 `allocateBudget({ ttiTargetSec, networkSec, renderSec, jsExecSec, bandwidthKbps, thirdPartyShare })`。规则：1) jsTransferSec = TTI − 网络 − 渲染 − 执行；≤ 0 → 抛错（含诊断：哪个参数组合不可行）；2) totalJsKB = jsTransferSec × bandwidthKbps / 8（KB gzip）取整；3) 三方预留 thirdPartyShare（0~1）；4) entry 占 firstParty 的 60%（可配 entryShare）；5) 返回 `{ derivation, totalJsKB, thirdPartyKB, firstPartyKB, entryKB, routeChunksKB }`。边界：thirdPartyShare=0；参数导致 jsTransferSec 恰为 0。附 `validate(params)`：参数合理性预检（TTI > 各项和；带宽 > 0；share ∈ [0,1]）。

```javascript
function validate({ ttiTargetSec, networkSec, renderSec, jsExecSec, bandwidthKbps, thirdPartyShare = 0.2 }) {
  const problems = [];
  if (ttiTargetSec <= networkSec + renderSec + jsExecSec)
    problems.push(`TTI ${ttiTargetSec}s ≤ 固定开销 ${(networkSec + renderSec + jsExecSec).toFixed(1)}s`);
  if (bandwidthKbps <= 0) problems.push('带宽非法');
  if (thirdPartyShare < 0 || thirdPartyShare > 1) problems.push('三方份额越界');
  return { ok: problems.length === 0, problems };
}

function allocateBudget(params) {
  const v = validate(params);
  if (!v.ok) throw new Error('参数不可行: ' + v.problems.join('; '));
  const { ttiTargetSec, networkSec, renderSec, jsExecSec, bandwidthKbps } = params;
  const share = params.thirdPartyShare ?? 0.2;
  const entryShare = params.entryShare ?? 0.6;
  const jsTransferSec = +(ttiTargetSec - networkSec - renderSec - jsExecSec).toFixed(3);
  const totalJsKB = Math.floor(jsTransferSec * bandwidthKbps / 8);
  const thirdPartyKB = Math.floor(totalJsKB * share);
  const firstPartyKB = totalJsKB - thirdPartyKB;
  const entryKB = Math.floor(firstPartyKB * entryShare);
  return {
    derivation: `TTI ${ttiTargetSec}s - 网络${networkSec}s - 渲染${renderSec}s - 执行${jsExecSec}s = JS ${jsTransferSec}s`,
    totalJsKB, thirdPartyKB, firstPartyKB, entryKB, routeChunksKB: firstPartyKB - entryKB,
  };
}

// 自测
let r = allocateBudget({ ttiTargetSec: 3, networkSec: 0.8, renderSec: 0.7, jsExecSec: 0.8, bandwidthKbps: 1000 });
console.log(r.derivation.includes('JS 0.7s'));
console.log(r.totalJsKB === 87);                         // 0.7×1000/8 = 87.5 取整
console.log(r.thirdPartyKB === 17 && r.firstPartyKB === 70);
console.log(r.entryKB === 42 && r.routeChunksKB === 28);

// 三方 0
r = allocateBudget({ ttiTargetSec: 3, networkSec: 0.8, renderSec: 0.7, jsExecSec: 0.8, bandwidthKbps: 1000, thirdPartyShare: 0 });
console.log(r.thirdPartyKB === 0 && r.firstPartyKB === 87);

// 不可行
let e; try { allocateBudget({ ttiTargetSec: 1, networkSec: 0.8, renderSec: 0.7, jsExecSec: 0.8, bandwidthKbps: 1000 }); }
catch (x) { e = x; }
console.log(/不可行/.test(e.message));
e = null; try { allocateBudget({ ttiTargetSec: 3, networkSec: 0.8, renderSec: 0.7, jsExecSec: 0.8, bandwidthKbps: 1000, thirdPartyShare: 2 }); }
catch (x) { e = x; }
console.log(/三方份额/.test(e.message));

console.log(validate({ ttiTargetSec: 3, networkSec: 0.8, renderSec: 0.7, jsExecSec: 0.8, bandwidthKbps: 1000 }).ok === true);
```

**评分点**：
- 反推公式正确（TTI − 固定开销 = JS 传输窗口）；KB 换算（×带宽/8）
- 不可行参数的预检与诊断（validate 分离——先验后算）；三方份额/入口占比可配
- 边界：share=0、share 越界、TTI 恰等于固定开销
- 能说清「自顶向下预算用目标倒逼优化，分配推导表可审计（每个数字有来源）」属于加分
