## RBL-516：手写协作式取消构建句柄（检查点 + 事务清理）
**答**：模拟 RBL-515——实现 `createBuildHandle(modules, transformFn)`：1) 返回 `{ promise, cancel }`：promise resolve `{ cancelled } | { done, artifacts }`（取消不 reject）；2) 协作检查点：每模块转换后检查取消标志；3) 事务清理：取消/失败时清除半成品（已转换的 artifacts 丢弃）；4) 统计：取消时已完成模块数；5) 附用例：中途取消与完整完成对比。

```javascript
function createBuildHandle(modules, transformFn, { checkpointEvery = 1 } = {}) {
  let cancelled = false
  const artifacts = new Map()
  let processed = 0
  let settle = null

  const promise = new Promise((resolve) => { settle = resolve })

  function run() {
    for (let i = 0; i < modules.length; i++) {
      if (cancelled) {                                        // 协作检查点
        settle({ cancelled: true, processed })                // 取消不 reject
        return
      }
      artifacts.set(modules[i], transformFn(modules[i]))
      processed++
      if ((i + 1) % checkpointEvery === 0 && cancelled) {     // 粗粒度检查（每 N 个）
        settle({ cancelled: true, processed })
        return
      }
    }
    settle({ cancelled: false, done: true, artifacts: [...artifacts.entries()] })   // 完整提交
  }

  function cancel() { cancelled = true }                      // 设置标志：检查点生效

  return { promise, cancel, run }
}

// 事务清理：取消/失败丢弃半成品
function withTransaction(handle, tmpDir) {
  return {
    promise: handle.promise.then((r) => {
      if (r.cancelled) {
        tmpDir.clear()                                       // 清除半成品（ZSM-524 事务语义同构）
        return { cancelled: true }
      }
      tmpDir.commit()                                        // 完整才 rename 提交
      return r
    }),
    cancel: handle.cancel,
    run: handle.run,
  }
}

// 用例
const tmp = { files: [], clear() { this.files = [] }, commit() { this.committed = true } }
const h = withTransaction(createBuildHandle(['a', 'b', 'c', 'd'], (m) => `code:${m}`), tmp)
h.run()
setTimeout(() => h.cancel(), 5)                              // 中途取消
const r = await h.promise
console.log(r.cancelled, r.processed, tmp.files.length)      // true <4 0：事务清理
console.log(h.promise.then(() => {}))                        // promise resolved 非 reject

// 完整完成
const h2 = withTransaction(createBuildHandle(['a', 'b'], (m) => `code:${m}`), tmp)
h2.run()
const r2 = await h2.promise
console.log(r2.done, r2.artifacts.length, tmp.committed)     // true 2 true
```
**评分点**：
- 协作检查点（每模块/每 N 模块粒度可配）
- 取消不 reject（resolve { cancelled: true }——语义区分取消与失败）
- 事务清理（取消丢弃半成品、完整才 commit）
- processed 统计（取消点可观测）
