## RBL-464：手写门禁豁免治理
**需求**：实现 `createExemptionManager(gates)`：1) 豁免申请：`request({ gate, reason, issue, until })`——必须关联 issue 与过期日期（无限豁免禁止）；2) 生效判定：gate 未过时查豁免——有效则放行但标记 EXEMPT（可观测不是静默绿）；3) 过期处理：豁免过期后自动恢复阻断（CI 每次运行校验 until）；4) 豁免审批：需要 gate owner 批准（申请人 ≠ 批准人）；5) 审计：豁免清单周报（数量/即将过期/过期未修复——债务可见）。

```javascript
export function createExemptionManager(gates) {
  const requests = []

  function request({ gate, reason, issue, until, requester }) {
    const g = gates.find((x) => x.name === gate)
    if (!g) throw new Error(`unknown gate ${gate}`)
    if (!issue || !until) throw new Error('exemption requires issue + until')
    if (new Date(until) < Date.now()) throw new Error('until must be future')
    const req = { gate, reason, issue, until, requester, status: 'pending', at: Date.now() }
    requests.push(req)
    return req                                                  // gate owner 批准后生效
  }
  function approve(req, approver) {
    if (req.requester === approver)
      throw new Error('申请人不能自批（separation of duties）')
    req.status = 'approved'; req.approver = approver
    return req
  }

  function evaluate(gateName, gateResult) {
    if (gateResult.ok) return { pass: true, status: 'PASS' }
    const exempt = requests.find((r) => r.gate === gateName && r.status === 'approved'
      && new Date(r.until) > Date.now())
    if (exempt) return { pass: true, status: 'EXEMPT', issue: exempt.issue,
      until: exempt.until }                                     // 放行但可观测
    return { pass: false, status: 'FAIL' }
  }

  function audit() {
    const now = Date.now()
    return {
      active: requests.filter((r) => r.status === 'approved' && new Date(r.until) > now),
      expired: requests.filter((r) => r.status === 'approved' && new Date(r.until) <= now),   // 过期未修复
      pending: requests.filter((r) => r.status === 'pending'),
    }
  }

  return { request, approve, evaluate, audit }
}
```
**评分点**：
- 豁免三要素强制（reason/issue/until——无限豁免禁止）
- 职责分离（申请人 ≠ 批准人——治理红线）
- EXEMPT 可观测（放行但不静默——豁免是债务不是绿灯）
- audit 三分类（active/expired/pending——过期未修复是治理重点）
