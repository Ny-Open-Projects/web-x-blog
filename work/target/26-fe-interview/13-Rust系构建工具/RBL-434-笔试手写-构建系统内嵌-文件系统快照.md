## RBL-434：手写文件系统快照
**需求**：实现 `createFsSnapshot(dir)`：1) `snapshot()`：递归遍历（含 mtime/size）生成快照 Map<path, {mtimeMs, size, hash?}>——跳过忽略目录；2) `diff(prev, next)`：变更集 { added, modified, removed }（mtime+size 快判，可疑再 hash 深判）；3) 增量刷新：只 stat 变更目录（基于目录 mtime——目录 mtime 变即子项可能有变）；4) 快照持久化（JSON 落盘，下次构建跳过全量扫描）；5) 忽略规则（glob 排除 node_modules 等）；6) 边界：mtime 精度问题（同秒修改）→ hash 兜底。

```javascript
import fs from 'node:fs'
import path from 'node:path'
import crypto from 'node:crypto'

export function createFsSnapshot(dir, { ignore = [/node_modules/, /\.git/] } = {}) {
  function walk(p, out) {
    for (const name of fs.readdirSync(p)) {
      if (ignore.some((r) => r.test(name))) continue
      const full = path.join(p, name)
      const st = fs.statSync(full)
      if (st.isDirectory()) walk(full, out)
      else out.set(full, { mtimeMs: st.mtimeMs, size: st.size })
    }
    return out
  }

  function snapshot() {
    const snap = walk(dir, new Map())
    fs.writeFileSync('.cache/fs-snapshot.json', JSON.stringify({ dir, snap: [...snap] }))
    return snap
  }

  function diff(prev, next) {
    const result = { added: [], modified: [], removed: [] }
    for (const [p, meta] of next) {
      const old = prev.get(p)
      if (!old) result.added.push(p)
      else if (old.mtimeMs !== meta.mtimeMs || old.size !== meta.size) {
        // mtime 精度兜底：hash 深判
        const h = crypto.createHash('md5').update(fs.readFileSync(p)).digest('hex')
        if (h !== old.hash) { result.modified.push(p); meta.hash = h }
      }
    }
    for (const p of prev.keys()) if (!next.has(p)) result.removed.push(p)
    return result
  }

  function loadPersisted() {
    try {
      const { snap } = JSON.parse(fs.readFileSync('.cache/fs-snapshot.json', 'utf8'))
      return new Map(snap)
    } catch { return null }
  }

  return { snapshot, diff, loadPersisted }
}
```
**评分点**：
- 快判（mtime+size）+ 深判（hash）两级——mtime 同秒精度问题的兜底
- 增量目录 mtime 优化意识（目录 mtime 变才深入——大仓库扫描提速）
- 快照持久化（跨进程跳过全量扫描——冷启动优化）
- 忽略规则前置（node_modules 不进快照）
