## RBL-565：笔试：手写带优先级与并发上限的构建任务调度器

**答**：实现 `createScheduler(concurrency)`，返回 `schedule(fn, prio)` 与 `run()`。优先级数字越小越先执行；并发上限 `concurrency`；同优先级按提交顺序 FIFO；`run()` 返回 Promise，全部任务完成后 resolve（结果按提交顺序）。`schedule` 需支持在 `run()` 之后追加任务。边界：空任务、fn 抛错（不能让整个 run reject，错误放进对应结果位）、并发槽位回收。

```javascript
function createScheduler(concurrency) {
  const tasks = [];            // { fn, prio, seq }
  const results = [];          // 按提交顺序
  let seq = 0, settled = 0, running = 0;
  let release; const done = new Promise(r => (release = r));
  const total = () => tasks.length;

  function pump() {
    while (running < concurrency) {
      // 取最小优先级、同优先级最早提交的任务
      let best = -1;
      for (let i = 0; i < tasks.length; i++) {
        if (tasks[i].done) continue;
        if (best === -1 || tasks[i].prio < tasks[best].prio ||
            (tasks[i].prio === tasks[best].prio && tasks[i].seq < tasks[best].seq)) best = i;
      }
      if (best === -1) break;
      const t = tasks[best]; t.done = true; running++;
      Promise.resolve()
        .then(t.fn)
        .then(v => (results[t.seq] = { ok: true, value: v }),
              e => (results[t.seq] = { ok: false, error: e }))
        .finally(() => {
          running--; settled++;
          if (settled === total() && running === 0) release();
          pump();
        });
    }
  }

  return {
    schedule(fn, prio = 5) { tasks.push({ fn, prio, seq: seq++, done: false }); pump(); },
    run() {
      return done.then(() => {
        if (settled < total()) return new Promise(r => { release = r; pump(); })
          .then(() => results);
        return results;
      });
    },
  };
}

// 自测：优先级生效 + 槽位回收 + 追加任务 + 错误隔离
const s = createScheduler(2); const order = [];
s.schedule(() => sleep(30).then(() => order.push('b')), 5);
s.schedule(() => sleep(10).then(() => order.push('a')), 1);   // 高优先级先抢槽
s.schedule(() => Promise.reject(new Error('x')), 9);           // 错误不冒泡
const out = await s.run();
s.schedule(() => order.push('c'), 0);                          // run 后追加
await s.run();
// 断言：order[0]==='a'；out[2].ok===false；order 含 'c'
```

**评分点**：
- 优先级选择逻辑正确（最小优先级 + 同级 seq 先进先出），且支持 run 后追加任务（监听总数动态变化）
- 并发槽位在任务 finally 中正确回收，防止并发数泄漏或超发
- 单任务失败不影响整体 run 完成，错误落位到对应下标（{ ok, error } 结构）
- 空任务/0 并发等边界有定义（至少不抛未处理 rejection）
