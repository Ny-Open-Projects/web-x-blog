## RBL-857：笔试：手写构建任务重试判定器（白名单 + 排除 + 历史）

**答**：实现 `shouldRetry(error, opts)`。规则：1) attempts ≥ maxAttempts → false；2) 确定性失败模式（assert/expect/SyntaxError/Cannot find module/config invalid）→ false；3) 瞬态白名单（ETIMEDOUT/ECONNRESET/502/503/registry timeout/runner evicted）→ true；4) 未知错误：历史成功率（注入 history.successRate(fp)）> 0.5 → true 否则 false；5) fingerprint：错误消息取首个非空行前 60 字符的 djb2 hash。边界：error 为 null（false）；maxAttempts=0。

```javascript
const RETRY_WHITELIST = [/ETIMEDOUT|ECONNRESET|\b50[23]\b|registry.*timeout/i, /runner.*evicted/i];
const DETERMINISTIC = [/assert|expect|SyntaxError|Cannot find module|config.*invalid/i];
const djb2 = (s) => { let h = 5381; for (const c of s) h = ((h << 5) + h + c.charCodeAt(0)) >>> 0; return String(h); };

function fingerprint(error) {
  const first = String(error?.message ?? error ?? '').split('\n').find(Boolean) ?? '';
  return djb2(first.slice(0, 60));
}
function shouldRetry(error, { attempts = 0, maxAttempts = 2, history } = {}) {
  if (!error) return false;
  if (attempts >= maxAttempts) return false;
  const msg = String(error.message ?? error);
  if (DETERMINISTIC.some(re => re.test(msg))) return false;
  if (RETRY_WHITELIST.some(re => re.test(msg))) return true;
  return (history?.successRate?.(fingerprint(error)) ?? 0) > 0.5;
}

// 自测
const netErr = new Error('ETIMEDOUT registry timeout after 30s');
console.log(shouldRetry(netErr, { attempts: 0 }) === true);
console.log(shouldRetry(netErr, { attempts: 2 }) === false);   // 上限
console.log(shouldRetry(new Error('expect(received).toBe(3)'), { attempts: 0 }) === false);
console.log(shouldRetry(new Error('Cannot find module ./x'), { attempts: 0 }) === false);

// 未知错误历史驱动
const unknown = new Error('weird failure mode');
console.log(shouldRetry(unknown, { history: { successRate: () => 0.8 } }) === true);
console.log(shouldRetry(unknown, { history: { successRate: () => 0.3 } }) === false);
console.log(shouldRetry(unknown, {}) === false);         // 无历史默认不重试（保守）
console.log(shouldRetry(null, {}) === false);
console.log(shouldRetry(new Error('runner evicted by scheduler'), { attempts: 1 }) === true);

// fingerprint 稳定性
console.log(fingerprint(new Error('a\nb')) === fingerprint(new Error('a\nc')));   // 只看首行
console.log(typeof fingerprint(new Error('x')) === 'string');
```

**评分点**：
- 三层判定顺序正确：上限 → 确定性排除 → 白名单 → 历史驱动；确定性优先于白名单（防两种模式同时命中时误重试）
- fingerprint 只取首行前 60 字符（压缩日志不稳定尾部的鲁棒性）
- 无历史默认 false（未知错误保守不重试）；null error 安全
- 能说清「确定性失败重试浪费分钟数并掩盖代码问题，重试语义是对随机性的正确建模」属于加分
