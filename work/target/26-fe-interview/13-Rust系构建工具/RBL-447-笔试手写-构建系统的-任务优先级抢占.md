## RBL-447：手写任务优先级抢占调度
**需求**：实现 `createPriorityScheduler({ slots })`：1) 任务带优先级（P0 紧急修复 / P2 日常）；2) 插队语义：高优先级任务到达时——不杀运行中任务（杀的成本高），但插到队列最前 + 拒绝同优先级以下新任务入队（让位）；3) 全部 slots 被 P2 占满且 P0 到达 → 可配置「抢占模式」：杀最低优先级运行任务（带检查点则可恢复）；4) 队列防饿死：P2 任务等待超 10 分钟自动升级优先级（aging）；5) stats 队列/抢占/升级计数。

```javascript
export function createPriorityScheduler({ slots = 4 } = {}) {
  const queue = [], running = new Map()
  let preemptCount = 0, agedCount = 0
  const PRIORITY = { P0: 0, P1: 1, P2: 2 }

  function enqueue(task) {
    const lowestRunning = [...running.entries()].sort((a, b) => PRIORITY[b[1].priority] - PRIORITY[a[1].priority])[0]
    // P0 且 slots 全被低优先级占满 → 抢占模式
    if (task.priority === 'P0' && running.size >= slots &&
        lowestRunning && PRIORITY[lowestRunning[1].priority] > 0 && task.preemptive) {
      const [id, victim] = lowestRunning
      victim.cancel?.()                                        // 带检查点可恢复
      running.delete(id)
      preemptCount++
      enqueue({ ...task })                                     // P0 入队
      if (victim.checkpoint) enqueue({ ...victim, priority: 'P2', resumed: true })  // 受害者重排队
      return { preempted: id }
    }
    queue.push(task)
    queue.sort((a, b) => PRIORITY[a.priority] - PRIORITY[b.priority] || a.enqueuedAt - b.enqueuedAt)
    pump()
  }

  function pump() {
    while (running.size < slots && queue.length) {
      const task = queue.shift()
      running.set(task.id, task)
      task.start?.().finally(() => { running.delete(task.id); pump() })
    }
  }

  // aging：防饿死
  setInterval(() => {
    for (const t of queue) {
      if (Date.now() - t.enqueuedAt > 600_000 && t.priority === 'P2') {
        t.priority = 'P1'; agedCount++
      }
    }
    queue.sort((a, b) => PRIORITY[a.priority] - PRIORITY[b.priority])
  }, 60_000)

  return { enqueue, stats: () => ({ running: running.size, queued: queue.length,
    preemptCount, agedCount }) }
}
```
**评分点**：
- 不杀运行中任务的默认策略（插队 + 拒绝低优先级入队——抢占代价认知）
- 可选抢占模式（受害者带检查点恢复——不是简单杀掉）
- aging 防饿死（P2 超时自动升级——公平性机制）
- 排序稳定性（同优先级 FIFO——enqueuedAt 次序保留）
