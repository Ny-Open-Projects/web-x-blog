## RBL-422：手写错误码注册表与查找
**需求**：实现 `createErrorRegistry()`：1) `register(code, { msg, doc, fix })`：编号唯一（重复注册抛错）；2) `throw(code, detail)`：抛 BuildError（message 模板渲染 + detail 插值 + doc 链接）；3) 错误模板插值：`{got} {expected}` 占位替换；4) deprecation：错误码废弃映射到新码（旧码使用打警告并重定向）；5) `exportDocs()` 生成错误码文档（markdown 表格）；6) 未注册码 throw 直接报「内部错误」并标注（帮助发现漏注册）。

```javascript
export function createErrorRegistry() {
  const table = new Map()      // code → { msg, doc, fix, deprecatedTo }
  const aliases = new Map()    // oldCode → newCode

  function register(code, def) {
    if (table.has(code)) throw new Error(`duplicate error code: ${code}`)
    table.set(code, def)
  }
  function deprecate(oldCode, newCode) {
    if (!table.has(newCode)) throw new Error(`target code missing: ${newCode}`)
    aliases.set(oldCode, newCode)
  }

  function render(tpl, detail = {}) {
    return tpl.replace(/\{(\w+)\}/g, (_m, k) => detail[k] ?? `{${k}}`)
  }

  function throw(code, detail = {}) {
    const target = aliases.get(code)
    if (target) {
      console.warn(`[deprecated] error code ${code} 已迁移到 ${target}`)
      code = target
    }
    const def = table.get(code)
    if (!def) {
      const e = new Error(`INTERNAL: unregistered error code ${code}`)
      e.unregistered = true
      throw e
    }
    const e = new Error(`[${code}] ${render(def.msg, detail)}\n  fix: ${def.fix ?? '见文档'}\n  doc: ${def.doc}`)
    e.code = code
    e.detail = detail
    throw e
  }

  function exportDocs() {
    const rows = [...table.entries()].map(([code, d]) => `| ${code} | ${d.msg} | ${d.fix ?? '-'} | ${d.doc} |`)
    return ['# 错误码表', '| 码 | 消息 | 修复 | 文档 |', '|---|---|---|---|', ...rows].join('\n')
  }

  return { register, deprecate, throw, exportDocs }
}
```
**评分点**：
- 编号唯一性（重复注册抛错——注册表纪律）
- 模板插值渲染（{got} 占位 + detail 插值）
- deprecation 重定向（旧码警告 + 转发——错误码演进不破坏用户搜索）
- 未注册码标注 INTERNAL（漏注册在运行时暴露）；exportDocs 单一事实源生成文档
