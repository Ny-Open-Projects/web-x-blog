## RBL-595：笔试：手写 re-export 链的导出绑定解析器

**答**：给定模块声明表 `modules: Map<id, { exports: { local, exported }[], reexports: { exported, from }[], stars: string[] }>`（相对导入 id 已 resolve 成绝对 id），实现 `resolveBinding(modId, name)`：返回 `{ source, local }`——穿透任意层 re-export 与 `export *`，最终落到**真实声明**的模块。规则：1) re-export 优先级高于 `export *`（显式 > 通配，冲突时显式赢）；2) `export *` 多源同名冲突时该名字解析失败（返回 null，对应 ESM 的 ambiguous star export）；3) 自身的 local 导出直接命中；4) 环形 re-export 抛错（检测环路）。要求 memoization 避免指数级重复解析。

```javascript
function createResolver(modules) {
  const memo = new Map();
  function resolveBinding(modId, name, seen = new Set()) {
    if (seen.has(modId)) throw new Error(`环形 re-export: ${[...seen, modId].join(' -> ')}`);
    const key = modId + ':' + name;
    if (memo.has(key)) return memo.get(key);            // memo：防指数
    memo.set(key, undefined);                            // 占位防同帧重复（环检在 seen）
    const m = modules.get(modId);
    if (!m) return undefined;

    // 1) 显式 re-export 优先
    for (const r of m.reexports) {
      if (r.exported === name) {
        const out = resolveBinding(r.from, name, new Set([...seen, modId]));
        return out;
      }
    }
    // 2) 自身 local 导出
    const local = m.exports.find(e => e.exported === name);
    if (local) return { source: modId, local: local.local };

    // 3) export *：收集所有源的同名绑定
    const found = [];
    for (const s of m.stars) {
      const out = resolveBinding(s, name, new Set([...seen, modId]));
      if (out) found.push(out);
    }
    let result;
    if (found.length === 1) result = found[0];
    else if (found.length > 1) result = null;            // ambiguous star export
    else result = undefined;                             // 未找到
    memo.set(key, result);
    return result;
  }
  return { resolveBinding };
}

// 自测：转发链 + 显式覆盖通配 + 歧义 + 环
const mods = new Map([
  ['app', { exports: [], reexports: [{ exported: 'x', from: 'lib' }], stars: ['theme'] }],
  ['lib', { exports: [], reexports: [{ exported: 'x', from: 'core' }], stars: [] }],
  ['core', { exports: [{ local: 'realX', exported: 'x' }], reexports: [], stars: [] }],
  ['theme-dark', { exports: [{ local: 'darkX', exported: 'x' }], reexports: [], stars: [] }],
  ['theme-light', { exports: [{ local: 'lightX', exported: 'x' }], reexports: [], stars: [] }],
  ['bad', { exports: [], reexports: [{ exported: 'y', from: 'bad2' }], stars: [] }],
  ['bad2', { exports: [], reexports: [{ exported: 'y', from: 'bad' }], stars: [] }],
]);
const { resolveBinding } = createResolver(mods);
console.log(resolveBinding('app', 'x'));               // { source: 'core', local: 'realX' }——两层穿透
console.log(resolveBinding('app', 'x').local === 'realX');

// 歧义 star：theme 的 x 来自两个模块
const mods2 = new Map([...mods]);
mods2.set('theme', { exports: [], reexports: [], stars: ['theme-dark', 'theme-light'] });
console.log(resolveBinding('app', 'x') === null);      // ambiguous -> null（若显式 reexport 仍在则优先显式）
const mods3 = new Map([...mods]);
mods3.get('app').stars = [];                            // 去掉通配后显式链生效
console.log(resolveBinding('app', 'x').local === 'realX');

let err; try { resolveBinding('bad', 'y'); } catch (e) { err = e; }
console.log(/环形/.test(err.message));                  // 环检测
```

**评分点**：
- 优先级正确：显式 re-export > local 导出 > `export *`（ESM 规范语义）
- 歧义 star export 返回 null 而非任取一个（与打包器「报 ambiguous 警告」对齐）
- 环形 re-export 显式抛错且路径可读；memoization 完整（同键不重复下钻）
- 穿透多层（≥2 层）转发仍正确，能说清「绑定解析是 tree shaking 的前提」属于加分
