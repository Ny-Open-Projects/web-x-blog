## RBL-406：手写体积预算门禁器
**答**：实现 `createBudgetGuard(budget, baseline)`：1) budget：`{ total: { gzip: 200_000 }, perChunk: { vendor: 120_000 }, delta: { ratio: 0.1 } }`；2) `check(manifest, prevManifest)`：总量/分项/环比三级检查；3) 超预算 → 违规列表 { type, chunk, value, limit, delta }；4) 白名单：`{ chunk, until, issue }`——过期的白名单视为无效（仍违规）；5) 结果 { pass, violations, warnings }，blocking 违规 → pass false。

```javascript
function gzipSizeOf(manifest, chunk) {
  return Object.entries(manifest.files)
    .filter(([p]) => chunk === 'total' || p.includes(chunk))
    .reduce((s, [, f]) => s + (f.gzip ?? f.size), 0)
}

function inWhitelist(chunk, whitelist, now) {
  const w = whitelist?.find((x) => x.chunk === chunk)
  return w && new Date(w.until) > now ? w : null
}

export function createBudgetGuard(budget, { whitelist = [] } = {}) {
  function check(manifest, prevManifest) {
    const now = Date.now()
    const violations = [], warnings = []
    const checkOne = (type, chunk, limit, value) => {
      const value_ = value
      if (value_ <= limit) return
      const item = { type, chunk, value: value_, limit }
      if (inWhitelist(chunk, whitelist, now)) warnings.push(item)
      else violations.push(item)
    }
    checkOne('total', 'total', budget.total.gzip, gzipSizeOf(manifest, 'total'))
    for (const [chunk, limit] of Object.entries(budget.perChunk ?? {}))
      checkOne('chunk', chunk, limit, gzipSizeOf(manifest, chunk))
    if (prevManifest && budget.delta) {
      for (const [chunk] of Object.entries(budget.perChunk ?? {})) {
        const prev = gzipSizeOf(prevManifest, chunk), cur = gzipSizeOf(manifest, chunk)
        if (prev > 0 && (cur - prev) / prev > budget.delta.ratio && !inWhitelist(chunk, whitelist, now))
          violations.push({ type: 'delta', chunk, value: cur, baseline: prev,
            ratio: +((cur - prev) / prev).toFixed(3) })
      }
    }
    return { pass: violations.length === 0, violations, warnings }
  }
  return { check }
}
```
**评分点**：
- 三级预算（total/perChunk/delta）全覆盖；环比分母防零
- 白名单过期判定（until > now 才有效——过期自动收紧）
- 违规/警告分离（白名单内是 warning 不是 pass——债务可见）
- gzip 口径统一（缺失时退回 size 并应标注口径）
