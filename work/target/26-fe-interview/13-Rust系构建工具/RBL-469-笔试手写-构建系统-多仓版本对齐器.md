## RBL-469：手写多仓库工具版本对齐器
**需求**：实现 `alignToolVersions(repos, target)`：1) 扫描：各仓 package.json 中工具依赖版本（rspack/typescript/eslint 等）；2) 对齐报告：与 target 版本 diff（落后 N 个 minor / 落后 major——风险分级）；3) 批量升级 PR 生成：每仓一个升级 PR（codemod + changelog），带验证状态（CI 通过才合并）；4) 对齐门禁：新仓库接入时版本必须在 target 范围内（防新增漂移）；5) 兼容缓冲：允许 ±1 minor（对齐不等于完全一致——迁移缓冲期）。

```javascript
import semver from 'semver'

export function createVersionAligner(repos, target) {
  function scan() {
    return repos.map((r) => ({
      repo: r.name,
      tools: Object.fromEntries(
        Object.entries(target.tools).map(([tool, targetV]) => {
          const current = r.deps[tool] ?? 'missing'
          const diffLevel = semver.valid(semver.coerce(current)) &&
            semver.diff(semver.coerce(current), semver.coerce(targetV))
          return [tool, { current, target: targetV, diffLevel: diffLevel ?? 'aligned' }]
        })),
    }))
  }

  function report() {
    const rows = scan().flatMap((r) =>
      Object.entries(r.tools).filter(([_, v]) => v.diffLevel !== 'aligned')
        .map(([tool, v]) => ({ repo: r.repo, tool, ...v,
          risk: v.diffLevel === 'major' ? 'high' : v.diffLevel === 'minor' ? 'medium' : 'low' })))
    return { rows: rows.sort((a) => (a.risk === 'high' ? -1 : 1)), missing: rows.filter((r) => r.current === 'missing') }
  }

  function gateNewRepo(repo) {
    const violations = []
    for (const [tool, targetV] of Object.entries(target.tools)) {
      const current = repo.deps[tool]
      if (!current) { violations.push(`${tool} missing`); continue }
      const c = semver.coerce(current), t = semver.coerce(targetV)
      if (!semver.satisfies(c, `~${t.major}.${t.minor}`))             // ±1 minor 缓冲
        violations.push(`${tool} ${current} 不在 target ${targetV} ±1 minor 范围`)
    }
    return { ok: violations.length === 0, violations }
  }

  return { scan, report, gateNewRepo }
}
```
**评分点**：
- semver diff 分级（major/high — minor/medium——风险分级驱动升级顺序）
- ±1 minor 缓冲（对齐不等于完全一致——务实的治理窗口）
- 新仓门禁（防新增漂移——治理是双向的：收敛旧 + 阻止新）
- 批量 PR 流程（codemod + CI 验证状态——规模化升级的执行方式）
