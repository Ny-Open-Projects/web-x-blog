## RBL-694：笔试：手写发布编排器（预推分级 + 失效最小化 + 就绪验证）

**答**：实现 `async publish(manifest, cdn)`。manifest: `{ files: Array<{ name, role, hash }>, indexHtml }`；cdn 是注入适配器：`pushBatch(files, { priority }), invalidate(name), probe(name) -> { status: 'fresh'|'stale' }, commitHtml(html) -> Promise`。编排顺序：1) role ∈ chunk/entry-js 预推 priority 1，asset 预推 priority 5（两批）；2) 无 hash 文件逐个 invalidate；3) 抽查前 5 个 P0 产物 probe，任一非 fresh 抛错且不 commitHtml；4) 全部 fresh 后 commitHtml；5) 返回 `{ pushed: n, invalidated: n }`。边界：无 mutable 文件时跳过失效；probe 异常按非 fresh 处理。

```javascript
async function publish(manifest, cdn) {
  const p0 = manifest.files.filter(f => f.role === 'chunk' || f.role === 'entry-js');
  const assets = manifest.files.filter(f => f.role === 'asset');
  await cdn.pushBatch(p0, { priority: 1 });
  await cdn.pushBatch(assets, { priority: 5 });

  const mutable = manifest.files.filter(f => !f.hash);
  for (const f of mutable) await cdn.invalidate(f.name);

  const probes = await Promise.all(p0.slice(0, 5).map(async f => {
    try { return (await cdn.probe(f.name)).status; }
    catch { return 'stale'; }                            // probe 异常按非 fresh
  }));
  if (probes.some(s => s !== 'fresh'))
    throw new Error('CDN 预推未生效，中止 html 切换');

  await cdn.commitHtml(manifest.indexHtml);
  return { pushed: p0.length + assets.length, invalidated: mutable.length };
}

// 自测
function fakeCdn() {
  const log = []; let fresh = true;
  return {
    log, setFresh: (v) => (fresh = v),
    pushBatch: async (files, opts) => log.push(['push', opts.priority, files.length]),
    invalidate: async (name) => log.push(['invalidate', name]),
    probe: async () => ({ status: fresh ? 'fresh' : 'stale' }),
    commitHtml: async (h) => log.push(['commit', h.length]),
  };
}
const manifest = {
  files: [
    { name: 'main.js', role: 'entry-js', hash: 'h1' },
    { name: 'chunk-a.js', role: 'chunk', hash: 'h2' },
    { name: 'logo.png', role: 'asset', hash: 'h3' },
    { name: 'index.html', role: 'entry-html', hash: null },
    { name: 'manifest.json', role: 'asset', hash: null },
  ],
  indexHtml: '<html>...</html>',
};
const cdn = fakeCdn();
let r = await publish(manifest, cdn);
console.log(JSON.stringify(r));                          // pushed 3, invalidated 2
console.log(cdn.log[0].join(',') === 'push,1,2');
console.log(cdn.log[1].join(',') === 'push,5,1');
console.log(cdn.log.at(-1)[0] === 'commit');

cdn.setFresh(false);
let err; try { await publish(manifest, cdn); } catch (e) { err = e; }
console.log(/中止/.test(err.message) && !cdn.log.some(x => x[0] === 'commit' && x[1] === cdn.log.at(-1)[1] + 1) || !cdn.log.at(-1)[0]);
// 最后一条不再是 commit（第二次发布未 commit）

// 无 mutable
const m2 = { files: manifest.files.filter(f => f.hash), indexHtml: 'x' };
const c2 = fakeCdn();
r = await publish(m2, c2);
console.log(r.invalidated === 0 && !c2.log.some(x => x[0] === 'invalidate'));
```

**评分点**：
- 编排顺序正确：hash 产物预推（两级优先级）→ mutable 失效 → probe 就绪 → 最后 commitHtml
- probe 异常按非 fresh 处理（防御）；未就绪不 commit（顺序契约：先产物后 html）
- 返回统计准确；无 mutable 跳过失效（失效最小化）
- 能说清「顺序颠倒会新 html 引旧 chunk 404」与「hash 产物免失效」属于加分
