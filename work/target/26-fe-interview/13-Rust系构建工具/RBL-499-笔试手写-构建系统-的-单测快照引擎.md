## RBL-499：手写构建产物快照测试引擎
**答**：实现 `createSnapshotEngine(store)`：1) 快照声明式定义：`{ name, capture: (buildResult) => data, mask: (data) => data }`——采集点可插拔（chunk 结构/体积/依赖数）；2) mask 层：不稳定字段掩码（hash/时间戳——快照确定性）；3) diff 报告：与基线逐字段对比（首个差异定位 + 语义 diff——「多了 3 个 chunk」而非字节 diff）；4) 基线更新流程：显式更新（CI 不允许自动更新——快照变更是 review 项）；5) 分支基线：feature 分支继承 main 基线（合并后对齐）。

```javascript
export function createSnapshotEngine(store) {
  const CAPTURES = {
    chunkStructure: { capture: (r) => Object.keys(r.files).sort(),
      mask: (d) => d.map((f) => f.replace(/\.[a-f0-9]{8}\./, '.HASH.')) },
    sizes: { capture: (r) => Object.fromEntries(Object.entries(r.files)
      .map(([k, v]) => [k.replace(/\.[a-f0-9]{8}\./, '.HASH.'), v.size])) },
    depCount: { capture: (r) => r.stats.externalDeps.length },
  }

  async function snapshot(name, buildResult, captures) {
    const out = {}
    for (const [key, cap] of Object.entries(captures ?? CAPTURES))
      out[key] = cap.mask(cap.capture(buildResult))
    const baseline = store.get(name)
    if (!baseline) return { status: 'new', data: out }           // 首次：需显式接受
    return { status: 'compare', diff: semanticDiff(baseline.data, out) }
  }

  function semanticDiff(prev, next) {
    const diffs = []
    for (const key of new Set([...Object.keys(prev), ...Object.keys(next)])) {
      if (Array.isArray(prev[key]) && Array.isArray(next[key])) {
        const added = next[key].filter((x) => !prev[key].includes(x))
        const removed = prev[key].filter((x) => !next[key].includes(x))
        if (added.length || removed.length)
          diffs.push({ key, summary: `+${added.length} -${removed.length}`,
            detail: { added, removed } })                        // 语义 diff：「多了 3 个 chunk」
      } else if (prev[key] !== next[key]) diffs.push({ key, from: prev[key], to: next[key] })
    }
    return diffs
  }

  function accept(name, data) { return store.set(name, { data, at: Date.now() }) }   // 显式更新
  return { snapshot, accept }
}

// CI 纪律：diff 非空 → fail；更新快照需要单独 commit（review 可见）
```
**评分点**：
- 采集点插拔（chunk/体积/依赖数——快照维度可扩展）
- mask 层确定性（hash 掩码——快照不被构建噪音污染）
- 语义 diff（「多了 3 个 chunk」——不是字节 diff 的可读性）
- 显式更新纪律（CI 禁自动更新——快照变更必须过 review）
