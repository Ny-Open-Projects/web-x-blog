## RBL-497：手写多仓统一构建入口
**需求**：实现 `createMultiRepoBuild(repos)`：1) 统一命令：一个 CLI 入口跨仓执行（build/test/lint——参数透传）；2) fanout 策略：串行（顺序保证）vs 并行（资源池上限）按任务类型；3) 聚合结果：每仓状态汇总表（成功/失败/耗时）+ 失败仓自动重试策略；4) 快速失败 vs 尽力而为模式：CI 门禁模式（首败即停）vs 全量诊断模式（跑完收集全部失败）；5) 进度上报（跨仓构建的实时进度——总时长预估）。

```javascript
import pLimit from 'p-limit'

export function createMultiRepoBuild(repos, { concurrency = 4 } = {}) {
  async function run(task, { mode = 'fail-fast' } = {}) {
    const results = []
    const limit = pLimit(concurrency)
    const tasks = repos.map((r) => ({ repo: r, promise: limit(() => r.run(task)) }))

    let failed = null
    for (const { repo, promise } of tasks) {                    // 按顺序检查（fail-fast 语义）
      try {
        const r = await promise
        results.push({ repo: repo.name, ok: true, ms: r.ms })
      } catch (e) {
        results.push({ repo: repo.name, ok: false, error: e.message })
        if (mode === 'fail-fast') { failed = repo.name; break }  // 首败即停（后续取消）
      }
    }
    // 进度上报
    const done = results.length
    console.log(`[multi-repo] ${done}/${repos.length} done, ETA ~${estimateRemaining(repos, done)}min`)

    const failedRepos = results.filter((r) => !r.ok)
    if (failedRepos.length && mode === 'collect-all')
      for (const r of failedRepos) await retryOnce(r)            // 尽力而为：失败重试一次
    return {
      ok: failedRepos.length === 0,
      results,
      summary: results.map((r) => `${r.ok ? '✓' : '✗'} ${r.repo} ${(r.ms / 1000).toFixed(1)}s`).join('\n'),
    }
  }

  function estimateRemaining(repos, done) {
    const avg = repos.reduce((s, r) => s + (r.avgBuildMs ?? 300_000), 0) / repos.length
    return Math.ceil((repos.length - done) * avg / concurrency / 60_000)
  }
  async function retryOnce(r) { /* 单仓重试 */ }

  return { run }
}
```
**评分点**：
- 双模式语义（fail-fast 门禁 / collect-all 诊断——CI 与排障的不同需求）
- 并发池 + 按序检查（pLimit 并行 + 顺序等待——fail-fast 语义正确）
- ETA 预估（跨仓构建的进度可预期）
- 失败重试策略（collect-all 才重试——门禁模式不掩盖失败）
