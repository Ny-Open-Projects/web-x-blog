## RBL-440：手写发布回滚器
**需求**：实现 `createRollback(cdn, history)`：1) history：发布历史 [{ version, manifestVersion, at, pointer }]；2) `rollback(targetVersion?)`：默认回滚到上一版本——切入口指针到目标 manifest；3) 回滚验证：目标版本产物仍存在（head 校验全部文件——产物被清理则回滚失败并提示）→ 指针切换 → 拨测确认（多节点拉 manifest hash 一致）；4) 回滚审计（触发人/原因/前后版本）；5) 级联回滚保护：连续回滚超 2 次需要人工确认（系统性问题不该靠回滚解决）。

```javascript
export function createRollback(cdn, historyStore) {
  async function rollback({ targetVersion, operator, reason, confirmCascade = false } = {}) {
    const history = historyStore.list()                        // 按时间降序
    const current = history[0]
    const target = targetVersion ? history.find((h) => h.version === targetVersion) : history[1]
    if (!target) throw new Error(`version ${targetVersion ?? 'previous'} not found`)

    // 级联保护
    const recentRollbacks = history.filter((h) => h.type === 'rollback').length
    if (recentRollbacks >= 2 && !confirmCascade)
      throw new Error('cascade rollback needs manual confirm (systemic issue?)')

    // 产物存在性校验
    for (const [path, hash] of Object.entries(target.hashes)) {
      const actual = await cdn.head(path)
      if (actual !== hash) throw new Error(`rollback blocked: ${path} missing/changed (cleaned up?)`)
    }
    await cdn.switchPointer(target.manifestVersion)
    // 拨测确认
    const probes = await cdn.probeAllRegions(target.manifestVersion)
    if (!probes.every((p) => p.match))
      throw new Error(`rollback inconsistent across regions: ${JSON.stringify(probes)}`)

    historyStore.push({ type: 'rollback', version: `rollback-to-${target.version}`,
      operator, reason, from: current.version, at: Date.now() })
    return { ok: true, from: current.version, to: target.version }
  }

  return { rollback }
}
```
**评分点**：
- 回滚前提：目标产物存在性校验（产物保留策略联动——被清理就回滚失败）
- 拨测确认（回滚后全网一致才算完成——半回滚状态危险）
- 级联保护（连续回滚 ≥2 需人工确认——系统性问题回滚不解决）
- 审计完整（operator/reason/from/to）
