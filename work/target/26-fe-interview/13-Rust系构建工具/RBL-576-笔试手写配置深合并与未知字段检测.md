## RBL-576：笔试：手写配置深合并器（数组替换语义 + 未知字段收集）

**答**：实现 `mergeConfig(defaults, ...overrides)`：对象深合并（后者覆盖前者）、**数组整体替换**、undefined 跳过、null 直接覆盖。同时提供 `collectUnknown(obj, schema)`：schema 是嵌套白名单对象（值可以是 true 或嵌套对象），返回所有不在白名单内的字段点路径数组（如 `output.fileName`）。两者都要纯函数（不改入参）。边界：override 非对象（数字/字符串）直接替换；循环引用对象可安全处理（可选加分）。

```javascript
function isPlainObj(v) { return v !== null && typeof v === 'object' && !Array.isArray(v); }

function mergeConfig(base, ...overs) {
  let out = base;
  for (const o of overs) {
    if (o === undefined) continue;                       // undefined 跳过
    if (!isPlainObj(out) || !isPlainObj(o)) { out = o; continue; }  // 数组/标量/null 整体替换
    const merged = { ...out };
    for (const k of Object.keys(o)) merged[k] = mergeConfig(merged[k], o[k]);
    out = merged;
  }
  return out;
}

function collectUnknown(obj, schema, path = '', acc = []) {
  if (!isPlainObj(obj) || !isPlainObj(schema)) return acc;
  for (const k of Object.keys(obj)) {
    if (!(k in schema)) { acc.push(path + k); continue; }
    const s = schema[k], v = obj[k];
    if (isPlainObj(s)) collectUnknown(v, s, path + k + '.', acc);   // 嵌套递归
  }
  return acc;
}

const SCHEMA = { output: { format: true, fileName: true }, resolve: { alias: true }, minify: true };

// 自测
const cfg = mergeConfig(
  { output: { format: 'es', map: true }, resolve: { alias: {} }, extra: 1 },
  { output: { format: 'iife', map: [1, 2, 3] } },        // 数组整体替换
  { minify: undefined, output: { fileName: 'x' } }
);
// 断言：
// cfg.output.format === 'iife'
// cfg.output.map 深度相等 [1,2,3]（替换而非合并）
// cfg.minify === undefined（跳过）且 'minify' in cfg === false? ——注意 undefined 值本身会被写入，
//   本实现 mergeConfig(1, undefined) 返回 1，故 key 不产生 —— 用 Object.keys(cfg) 不含 minify
console.log(JSON.stringify(collectUnknown(cfg, SCHEMA)));   // ['extra', 'output.map']

// 纯函数验证
const src = { a: { b: 1 } }; mergeConfig(src, { a: { c: 2 } });
console.log(JSON.stringify(src) === '{"a":{"b":1}}');       // true：入参未改
```

**评分点**：
- 数组整体替换语义正确实现（不是 concat），undefined 跳过、null 与标量覆盖符合规格
- collectUnknown 递归收集点路径正确，白名单嵌套与叶字段区分清楚
- 纯函数性保持（不修改 defaults 与 overrides），递归无栈溢出风险（浅层配置树）
- 能指出「undefined 写入 vs 跳过」的语义差异并按规格实现属于加分
