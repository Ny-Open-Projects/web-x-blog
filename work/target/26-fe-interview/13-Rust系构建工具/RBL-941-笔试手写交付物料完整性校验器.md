## RBL-941：笔试：手写交付物料完整性校验器（清单 + 消费方）

**答**：实现 `verifyDelivery(dir, fs, spec)`。spec: `Record<material, { file?: string, files?: (m) => string[], consumer }>`；fs 注入：`exists(path) -> bool`、`glob(dir, pattern) -> string[]`、`readJson(path)`。规则：1) 单文件物料：exists 校验（且 JSON 类物料可 parse——解析失败也算缺失）；2) 集合物料：glob 计数 > 0；3) manifest 存在时额外校验「buildId 与 attestation 一致」（物料间绑定——RBL-825）；4) 返回 `{ problems: Array<{ material, consumer, detail }>, complete }`。边界：spec 空（complete）；manifest 缺失时跳过绑定校验。

```javascript
async function verifyDelivery(dir, fs, spec = DELIVERABLES_DEFAULT) {
  const problems = [];
  let manifest = null;
  for (const [name, s] of Object.entries(spec)) {
    if (s.file) {
      const path = `${dir}/${s.file}`;
      const exists = await fs.exists(path);
      if (!exists) { problems.push({ material: name, consumer: s.consumer, detail: '缺失' }); continue; }
      if (s.file.endsWith('.json')) {
        try {
          const json = await fs.readJson(path);
          if (name === 'manifest') manifest = json;
        } catch (e) { problems.push({ material: name, consumer: s.consumer, detail: `JSON 解析失败: ${e.message}` }); }
      }
    } else if (s.files) {
      const found = await fs.glob(dir, s.files);
      if (!found.length) problems.push({ material: name, consumer: s.consumer, detail: '集合为空' });
    }
  }
  // 物料间绑定：manifest.buildId 与 attestation 一致
  if (manifest) {
    try {
      const att = await fs.readJson(`${dir}/.attestation.json`);
      if (att.buildId && att.buildId !== manifest.buildId)
        problems.push({ material: 'binding', consumer: '部署编排',
          detail: `buildId 不一致: manifest=${manifest.buildId} attestation=${att.buildId}` });
    } catch { /* attestation 已在 spec 校验 */ }
  }
  return { problems, complete: problems.length === 0 };
}

const DELIVERABLES_DEFAULT = {
  manifest: { file: 'manifest.json', consumer: '部署编排' },
  attestation: { file: '.attestation.json', consumer: '合规' },
  sbom: { file: 'sbom.json', consumer: '安全扫描' },
  sourcemaps: { files: () => '**/*.map', consumer: '错误监控' },
};

// 自测
function mkFs(store) {
  return {
    exists: async (p) => store.has(p),
    glob: async (dir, pattern) => [...store.keys()].filter(k => k.startsWith(dir) && k.endsWith('.map')),
    readJson: async (p) => { const v = store.get(p); if (!v) throw new Error('missing'); return v; },
  };
}
const store = new Map([
  ['/d/manifest.json', { buildId: 'b1' }],
  ['/d/.attestation.json', { buildId: 'b1' }],
  ['/d/sbom.json', { components: [] }],
  ['/d/main.js.map', {}],
]);
let r = await verifyDelivery('/d', mkFs(store));
console.log(r.complete === true && r.problems.length === 0);

// buildId 绑定不一致
store.set('/d/.attestation.json', { buildId: 'b2' });
r = await verifyDelivery('/d', mkFs(store));
console.log(r.problems.some(p => p.material === 'binding' && p.detail.includes('b2')));
console.log(r.complete === false);

// 缺 sbom
store.set('/d/.attestation.json', { buildId: 'b1' });
store.delete('/d/sbom.json');
r = await verifyDelivery('/d', mkFs(store));
console.log(r.problems.find(p => p.material === 'sbom').consumer === '安全扫描');

// JSON 解析失败
store.set('/d/sbom.json', 'not-json{{{');
r = await verifyDelivery('/d', mkFs(store));
console.log(r.problems.some(p => p.detail.includes('JSON 解析失败')));

// 无 map 文件
r = await verifyDelivery('/empty', mkFs(new Map()));
console.log(r.problems.length >= 3);
```

**评分点**：
- 双形态物料校验（单文件 exists + JSON 可解析 / 集合 glob 非空）；JSON 解析失败独立诊断
- 物料间绑定校验（manifest.buildId ↔ attestation——混版本物料的防线 RBL-825）；manifest 缺失跳过
- 消费方标注（物料价值由消费方定义）；边界
- 能说清「构建输出是一个系统而非一个目录——混版本物料是最常见的发布事故」属于加分
