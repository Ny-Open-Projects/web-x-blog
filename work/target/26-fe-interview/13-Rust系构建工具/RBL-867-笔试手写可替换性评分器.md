## RBL-867：笔试：手写可替换性评分器（耦合 + 基建 + 成本）

**答**：实现 `score(repo)`。repo: `{ plugins: Array<{ name, viaAbstraction: boolean }>, presetCoverage: number(0~1), deepCoupling: Array<{ file, symbol }>, semanticTests: number }`. 规则：1) 综合分（0-100）= presetCoverage×30 + pluginAbstractionRate×30 + max(0, 1 - deepCoupling.length/20)×20 + min(1, semanticTests/50)×20；2) 分级：≥75 'replaceable'、≥50 'moderate'、<50 'locked-in'；3) 成本预估（hours）：config = (1-presetCoverage)×8、plugin = 非抽象插件数×2、verify = semanticTests ≥ 50 ? 4 : 20；返回 `{ score, grade, cost: { config, plugin, verify, total } }`（total 四舍五入）。边界：plugins 空；presetCoverage 越界 clamp。

```javascript
function score(repo) {
  const pc = Math.max(0, Math.min(1, repo.presetCoverage ?? 0));
  const plugins = repo.plugins ?? [];
  const par = plugins.length ? plugins.filter(p => p.viaAbstraction).length / plugins.length : 1;
  const dc = Math.max(0, 1 - (repo.deepCoupling?.length ?? 0) / 20);
  const st = Math.min(1, (repo.semanticTests ?? 0) / 50);
  const total = +(pc * 30 + par * 30 + dc * 20 + st * 20).toFixed(1);
  const grade = total >= 75 ? 'replaceable' : total >= 50 ? 'moderate' : 'locked-in';
  const config = +((1 - pc) * 8).toFixed(1);
  const plugin = plugins.filter(p => !p.viaAbstraction).length * 2;
  const verify = (repo.semanticTests ?? 0) >= 50 ? 4 : 20;
  return { score: total, grade,
    cost: { config, plugin, verify, total: +(config + plugin + verify).toFixed(1) } };
}

// 自测
let r = score({ presetCoverage: 1, plugins: [
  { name: 'a', viaAbstraction: true }, { name: 'b', viaAbstraction: true }],
  deepCoupling: [], semanticTests: 80 });
console.log(r.score === 100 && r.grade === 'replaceable');
console.log(r.cost.total === 4);                        // config 0 + plugin 0 + verify 4

r = score({ presetCoverage: 0.3, plugins: [
  { name: 'w1', viaAbstraction: false }, { name: 'w2', viaAbstraction: false },
  { name: 'w3', viaAbstraction: false }, { name: 'w4', viaAbstraction: false }],
  deepCoupling: Array.from({ length: 25 }, (_, i) => ({ file: `f${i}`, symbol: 'ctx' })),
  semanticTests: 5 });
// 0.3*30=9 + 0*30=0 + 0*20=0 + 0.1*20=2 = 11
console.log(r.score === 11 && r.grade === 'locked-in');
console.log(r.cost.config === 5.6 && r.cost.plugin === 8 && r.cost.verify === 20 && r.cost.total === 33.6);

r = score({ presetCoverage: 0.7, plugins: [{ name: 'x', viaAbstraction: true }],
  deepCoupling: [{ file: 'f', symbol: 's' }], semanticTests: 30 });
// 21+30+19+12 = 82
console.log(r.grade === 'replaceable');

r = score({ presetCoverage: 1.5, plugins: [], deepCoupling: [], semanticTests: 0 });
console.log(r.score === 80);                            // clamp 1: 30+30+20+0
r = score({ presetCoverage: -1, plugins: [], deepCoupling: [], semanticTests: 0 });
console.log(r.grade === 'moderate');                    // clamp 0: 0+30+20+0=50
```

**评分点**：
- 四因子加权正确（preset/插件抽象/深耦合/测试基建各占 30/30/20/20）；clamp 与空插件防御（空视为全抽象 par=1）
- 分级三档；成本三段（config/plugin/verify）与测试基建的门禁效应（≥50 tests → verify 便宜）
- 边界：越界 coverage、空 plugins、0 tests
- 能说清「语义测试基建是可替换性最大资产——换工具从赌博变验证流程」属于加分
