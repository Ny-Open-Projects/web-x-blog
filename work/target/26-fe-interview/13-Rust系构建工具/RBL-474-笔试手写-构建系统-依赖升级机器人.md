## RBL-474：手写依赖升级机器人
**答**：实现 `createUpgradeBot({ repos, policy })`：1) 升级机会发现：weekly 扫描 outdated（semver 分级：patch/minor/major——风险分级决定自动化程度）；2) 自动 PR：patch/minor 自动升级 PR（带 changelog 与 CI 验证——低风险全自动）；major 生成升级提案（人工评估 breaking）；3) 升级 PR 质量：CI 全绿 + 产物 diff（体积/行为快照）才可合并；4) 升级频率控制：每仓每周最多 5 个升级 PR（PR 风暴防护）；5) 回归检测：升级后生产事故关联（升级 commit ↔ 事故率——升级器自我校准）。

```javascript
import semver from 'semver'

export function createUpgradeBot({ repos, policy }) {
  const prLog = []

  async function scan(repo) {
    const outdated = []
    for (const [name, current] of Object.entries(repo.deps)) {
      const latest = await fetchLatest(name)
      if (semver.gt(latest, semver.coerce(current))) {
        const level = semver.diff(semver.coerce(current), latest)   // major/minor/patch
        outdated.push({ name, from: current, to: latest, level })
      }
    }
    return outdated
  }

  async function proposeUpgrades(repo) {
    const outdated = (await scan(repo)).sort((a) => (a.level === 'major' ? 1 : -1))
    const recent = prLog.filter((p) => p.repo === repo.name && Date.now() - p.at < 7 * 86400_000)
    const quota = Math.max(0, 5 - recent.length)                    // PR 风暴防护
    const prs = []
    for (const u of outdated.slice(0, quota)) {
      if (u.level === 'major') {
        prs.push({ type: 'proposal', name: u.name, from: u.from, to: u.to,
          note: 'major 需人工评估 breaking' })                      // major 提案不自动
        continue
      }
      const pr = await repo.createUpgradePR(u, {
        checks: ['ci', 'size-diff', 'behavior-snapshot'],           // PR 质量三件
      })
      prLog.push({ repo: repo.name, name: u.name, pr, at: Date.now(), level: u.level })
      prs.push(pr)
    }
    return prs
  }

  async function calibrate(incidents) {                             // 升级 ↔ 事故关联
    const correlated = incidents.filter((i) => {
      const pr = prLog.find((p) => p.repo === i.repo && i.commit?.includes(p.pr.sha))
      return pr
    })
    return { upgradeRelatedIncidents: correlated.length,
      rate: incidents.length ? correlated.length / incidents.length : 0,
      note: '升级相关事故率 → 调整 policy（如 major 灰度更长）' }
  }

  return { scan, proposeUpgrades, calibrate }
}
```
**评分点**：
- semver 分级自动化程度（patch/minor 自动 PR / major 提案——风险与自动化匹配）
- PR 质量三件（CI + 体积 diff + 行为快照——升级 PR 不是改版本号就完）
- 频率配额（每周 5 个——PR 风暴防护）
- 自我校准（升级相关事故率回溯——策略调整的数据依据）
