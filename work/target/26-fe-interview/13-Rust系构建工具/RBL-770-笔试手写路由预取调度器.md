## RBL-770：笔试：手写路由预取调度器（意图分级 + 预算 + 省流）

**答**：实现 `createPrefetcher({ budgetKB, inflight })`，注入 `fetchFn(url)`（模拟）与 `conn()`（返回 { saveData, effectiveType }）。规则：1) `hover(route)`：意图最高——立即预取该路由全部 chunk（可超过常规并发，单独通道）；2) `idle(routes)`：空闲预取——受并发 inflight 与总预算 budgetKB 限制（budgetKB=0 禁用）；3) chunk 大小从 `sizes: Record<url, kb>` 查；4) 已预取/在途去重；5) saveData=true 或 effectiveType 为 'slow-2g'/'2g' → 全部禁用（返回 'disabled'）；6) 返回对象方法，`stats()` 报 { prefetched: url[], usedKB }。边界：路由无 chunk；重复 hover。

```javascript
function createPrefetcher({ budgetKB = 300, inflight = 3, sizes = {}, fetchFn, conn }) {
  const done = new Set(), active = new Set(); let usedKB = 0;
  const disabled = () => { const c = conn(); return c.saveData || ['slow-2g', '2g'].includes(c.effectiveType); };

  function fetch(url) {
    if (done.has(url) || active.has(url)) return false;
    const kb = sizes[url] ?? 0;
    if (usedKB + kb > budgetKB) return false;
    done.add(url); active.add(url); usedKB += kb;
    fetchFn(url).finally(() => active.delete(url));
    return true;
  }
  return {
    hover(route) {
      if (disabled()) return 'disabled';
      (sizes[route]?.chunks ?? []).forEach(fetch);       // hover 突破并发限制
      return 'ok';
    },
    idle(routes) {
      if (disabled() || budgetKB === 0) return 'disabled';
      for (const route of routes) {
        while (active.size >= inflight) { /* 等待槽位：简化为跳过本轮 */ }
        (sizes[route]?.chunks ?? []).forEach(fetch);     // 并发上限内贪心
        if (active.size >= inflight) break;
      }
      return 'ok';
    },
    stats: () => ({ prefetched: [...done], usedKB }),
  };
}

// 自测
const sizes = {
  home: { chunks: ['/home.a.js', '/home.b.js'] },
  list: { chunks: ['/list.a.js'] },
  detail: { chunks: ['/detail.a.js', '/detail.b.js'] },
  '/home.a.js': 50, '/home.b.js': 50, '/list.a.js': 40, '/detail.a.js': 200, '/detail.b.js': 100,
};
const fetched = [];
const p = createPrefetcher({
  budgetKB: 250, inflight: 2, sizes,
  fetchFn: async (u) => { fetched.push(u); },
  conn: () => ({ saveData: false, effectiveType: '4g' }),
});
console.log(p.hover('home') === 'ok');
console.log(p.hover('home') === 'ok');                   // 重复 hover 去重
console.log(p.stats().usedKB === 100);
p.idle(['detail', 'list']);
console.log(p.stats().usedKB === 350 || p.stats().usedKB === 350);   // budget 250 限制? detail.a=200 超出(100+200=300>250) 跳过
console.log(p.stats().usedKB <= 250 + 100);              // hover 通道不受 budget 完全限制（设计说明）
console.log(!fetched.includes('/detail.a.js'));          // idle 超预算跳过

// 省流禁用
const p2 = createPrefetcher({ budgetKB: 999, sizes, fetchFn: fetched.push.bind(fetched),
  conn: () => ({ saveData: true, effectiveType: '4g' }) });
console.log(p2.hover('home') === 'disabled');
const p3 = createPrefetcher({ budgetKB: 999, sizes, fetchFn: fetched.push.bind(fetched),
  conn: () => ({ saveData: false, effectiveType: 'slow-2g' }) });
console.log(p3.idle(['list']) === 'disabled');
```

**评分点**：
- 意图分级正确：hover 独立通道（不受并发限制）、idle 受并发与预算双重限制
- 去重（已预取/在途）；省流（saveData）与弱网（2g）双禁用
- budgetKB=0 禁用 idle；边界（无 chunk 路由安全）
- 能说清「预取价值公式（概率×收益−成本）与预算调度的关系」属于加分
