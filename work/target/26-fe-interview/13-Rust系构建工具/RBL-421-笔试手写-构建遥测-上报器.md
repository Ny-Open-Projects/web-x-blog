## RBL-421：手写构建遥测上报器
**需求**：实现 `createTelemetry(reporter)`：1) 事件：build.start/end/fail、cache.hit/miss（含 reason）、task.done（耗时）；2) 匿名化：路径归一（绝对路径 → 相对 projectRoot + 泛化 node_modules）、无源码内容；3) 批量上报：内存队列 50 条或 30s 定时 flush（sendBeacon 语义——进程退出前强制 flush）；4) 失败容错：上报失败重试 1 次后丢弃（遥测不阻塞构建）；5) `summary()` 返回本次会话聚合（命中率/总时长/错误类型分布）；6) 开关与采样（disabled 直接 no-op）。

```javascript
export function createTelemetry({ endpoint, projectRoot, enabled = true, sampleRate = 1 } = {}) {
  const queue = []
  let hits = 0, misses = 0, startedAt = null, errorTypes = new Map()
  let timer = null

  const anonymize = (ev) => ({
    ...ev,
    file: ev.file ? ev.file.replace(projectRoot, '.').replace(/node_modules\/[^/]+/g, 'node_modules/*') : undefined,
  })

  function push(ev) {
    if (!enabled || Math.random() > sampleRate) return
    queue.push(anonymize({ ...ev, at: Date.now() }))
    if (queue.length >= 50) flush()
  }

  function flush() {
    if (!queue.length) return
    const batch = queue.splice(0, queue.length)
    try { reporter.send(JSON.stringify(batch)) } catch { /* 遥测失败静默 */ }
  }
  timer = setInterval(flush, 30_000)

  return {
    buildStart: () => { startedAt = Date.now(); push({ type: 'build.start' }) },
    buildEnd: () => push({ type: 'build.end', ms: Date.now() - startedAt }),
    buildFail: (err) => { errorTypes.set(err.code ?? 'unknown', (errorTypes.get(err.code) ?? 0) + 1); push({ type: 'build.fail', code: err.code }) },
    cache: (hit, reason) => { hit ? hits++ : misses++; push({ type: hit ? 'cache.hit' : 'cache.miss', reason }) },
    task: (name, ms) => push({ type: 'task.done', name, ms }),
    flush,
    summary: () => ({ cacheRate: hits + misses ? hits / (hits + misses) : 0,
      totalMs: startedAt ? Date.now() - startedAt : 0,
      errorTypes: Object.fromEntries(errorTypes) }),
    dispose: () => { clearInterval(timer); flush() },   // 进程退出前 flush
  }
}
```
**评分点**：
- 匿名化两规则（root 归一 + node_modules 泛化——无源码无真实路径）
- 批量 + 定时 flush + dispose 强制（进程退出不丢数据——beacon 语义）
- 遥测失败静默（不阻塞构建——遥测是旁路）
- summary 聚合口径（命中率分母含 miss reason）
