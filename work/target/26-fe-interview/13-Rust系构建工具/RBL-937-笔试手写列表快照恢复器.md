## RBL-937：笔试：手写列表快照恢复器（过期 + 审计）

**答**：实现两部分。1) `restoreSnapshot(snap, { maxAgeMs, now })`：snap 为 null → null；过期（now − snap.ts > maxAgeMs）→ null（不恢复——重新拉数据）；有效 → 返回 `{ scrollY, filters, cursor, stale: now − snap.ts }`。2) `auditStateFields(code)`：提取 `history.(push|replace)State({...}` 的第一层键，敏感模式（token/secret/password/credit/auth，大小写不敏感）→ findings `{ field }`；嵌套对象不深入（第一层即可）；非 state 调用忽略。返回 `{ findings, ok }`。边界：snap 无 ts（视为过期——完整性存疑）；code 无 state 调用。

```javascript
function restoreSnapshot(snap, { maxAgeMs = 30 * 60000, now = Date.now() } = {}) {
  if (!snap) return null;
  if (typeof snap.ts !== 'number' || now - snap.ts > maxAgeMs) return null;
  return { scrollY: snap.scrollY ?? 0, filters: snap.filters ?? {},
    cursor: snap.cursor ?? null, stale: now - snap.ts };
}

const SENSITIVE = /token|secret|password|credit|auth/i;
function auditStateFields(code = '') {
  const findings = [];
  for (const m of code.matchAll(/history\.(?:push|replace)State\(\s*\{([^}]*)\}/g)) {
    for (const k of m[1].matchAll(/(\w+)\s*:/g)) {
      if (SENSITIVE.test(k[1]))
        findings.push({ field: k[1], detail: `敏感字段 ${k[1]} 不得入 history.state` });
    }
  }
  return { findings, ok: findings.length === 0 };
}

// 自测
const now = 1000000;
// 有效快照
let r = restoreSnapshot({ scrollY: 800, filters: { q: 'x' }, cursor: { sort: 90, id: 'b' }, ts: now - 1000 }, { now });
console.log(r.scrollY === 800 && r.stale === 1000);

// 过期
r = restoreSnapshot({ scrollY: 1, ts: now - 60000 }, { now, maxAgeMs: 30000 });
console.log(r === null);
// 无 ts
console.log(restoreSnapshot({ scrollY: 1 }, { now }) === null);
console.log(restoreSnapshot(null, { now }) === null);

// 敏感字段审计
const code = [
  "history.replaceState({ listSnapshot: s, token: t }, '');",
  "history.pushState({ page: 2, authToken: a }, '');",
  "history.replaceState({ q: 1 }, '');",                 // 干净
  "localStorage.setItem('token', t);",                   // 非 state 忽略
].join('\n');
let a = auditStateFields(code);
console.log(a.findings.map(f => f.field).sort().join(','));   // authToken,token
console.log(a.ok === false);
console.log(auditStateFields("history.replaceState({ q: 1 }, '')").ok === true);
console.log(auditStateFields('').ok === true);
```

**评分点**：
- 过期判定（含无 ts 视为过期的保守语义——完整性存疑不恢复）；stale 时长返回供 UI 提示
- 敏感字段提取限第一层（浅扫描声明）；非 state 调用不误报
- 边界：null 快照、无调用
- 能说清「history.state 进浏览器历史与同步——敏感数据不入 state 是序列化边界的安全审计」属于加分
