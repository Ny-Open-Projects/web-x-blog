## RBL-725：笔试：手写分层产物预算与依赖规则门禁

**答**：实现 `auditLayers(layers, imports, budgets)`。layers: `Record<name, { modules: string[], target? }>`；imports: `Array<{ from, to, kind }>`（from/to 为模块 id）；budgets: `Record<name, budgetKB>`。规则：1) 每层体积 = 层内模块体积和，超 budgetKB → blocker；2) 依赖方向规则：core 层模块不得 import enhanced/pwa 层模块（增强层可以 import 核心层——方向向下）；违规 → blocker `layer-violation`；3) core 层依赖外部包（node_modules）→ blocker `core-vendor`（零依赖承诺）；4) 返回 `{ blockers, sizeOf }`。边界：模块跨层归属重复取首次；imports 中 from/to 未分层忽略。

```javascript
function auditLayers(layers, imports, budgets, { moduleSize } = {}) {
  const layerOf = new Map();
  for (const [name, l] of Object.entries(layers))
    for (const m of l.modules)
      if (!layerOf.has(m)) layerOf.set(m, name);
  const LAYER_RANK = { core: 0, enhanced: 1, pwa: 2 };

  const sizeOf = {};
  const blockers = [];
  for (const [name, l] of Object.entries(layers)) {
    const size = l.modules.reduce((s, m) => s + (moduleSize?.[m] ?? 0), 0);
    sizeOf[name] = size;
    if (budgets[name] != null && size > budgets[name])
      blockers.push(`${name} 超预算 ${size}KB > ${budgets[name]}KB`);
  }
  const rank = (m) => LAYER_RANK[layerOf.get(m)] ?? null;
  for (const { from, to } of imports) {
    const rf = rank(from), rt = rank(to);
    if (rf === null || rt === null) continue;
    if (rf < rt) blockers.push(`layer-violation: ${layerOf.get(from)} 不能依赖 ${layerOf.get(to)}（${from} -> ${to}）`);
    if (layerOf.get(from) === 'core' && to.includes('node_modules/'))
      blockers.push(`core-vendor: ${from} 依赖外部包 ${to}（核心层零依赖）`);
  }
  return { blockers, sizeOf };
}

// 自测
const layers = {
  core: { modules: ['core/form.ts', 'core/nav.ts'] },
  enhanced: { modules: ['enh/dashboard.ts'] },
  pwa: { modules: ['sw.ts'] },
};
const size = { 'core/form.ts': 20, 'core/nav.ts': 15, 'enh/dashboard.ts': 60, 'sw.ts': 10 };
const imports = [
  { from: 'enh/dashboard.ts', to: 'core/nav.ts' },       // 合法：向下
  { from: 'core/form.ts', to: 'enh/dashboard.ts' },      // 违规
  { from: 'enh/dashboard.ts', to: 'node_modules/lodash' },  // 合法
];
let r = auditLayers(layers, imports, { core: 40, enhanced: 80 }, { moduleSize: size });
console.log(r.blockers.some(b => b.startsWith('layer-violation')));
console.log(r.blockers.some(b => b.startsWith('core 超预算')));   // core 35KB 未超？20+15=35 < 40 —— 不超
// 修正断言
console.log(r.sizeOf.core === 35 && r.sizeOf.enhanced === 60);

r = auditLayers(layers, [
  { from: 'core/nav.ts', to: 'node_modules/axios' },     // core-vendor
], { core: 100 }, { moduleSize: size });
console.log(r.blockers.some(b => b.startsWith('core-vendor')));

r = auditLayers(layers, [], { core: 40, enhanced: 50 }, { moduleSize: size });
console.log(r.blockers.some(b => b.startsWith('enhanced 超预算')));
console.log(auditLayers(layers, [], {}, { moduleSize: size }).blockers.length === 0);
```

**评分点**：
- 分层依赖方向判定正确（rank 向下合法、向上违规）；跨层归属首次优先
- core 零依赖承诺的检查（node_modules 路径判定）；体积预算独立门禁
- 边界：未分层模块忽略、空 imports 通过
- 能说清「分层是承诺，门禁把承诺变成 CI 约束」属于加分
