## RBL-176：笔试：手写任务级本地缓存（hash 存储 + 命中回放）

**题目**：写 task-cache.mjs：实现 runCached(task, inputs, outputsDir, cmd)：1) 对 inputs（文件列表）与 envKeys 计算任务指纹 hash；2) 缓存目录 .cache/tasks/{hash}/ 存 outputs（产物文件）与 meta.json（stdout 日志）；3) 命中：把产物复制回输出位置并原样输出日志，返回 {cached: true}；4) miss：执行命令，成功后把 outputs 收进缓存，失败不缓存；5) 写入用"临时目录 + rename"保证原子，避免并发写坏缓存。

```javascript
// task-cache.mjs
import { createHash } from 'node:crypto';
import { execSync } from 'node:child_process';
import { existsSync, mkdirSync, readFileSync, writeFileSync, cpSync, renameSync, rmSync } from 'node:fs';
import { join } from 'node:path';

function fingerprint(inputs, envKeys) {
  const h = createHash('sha256');
  for (const f of inputs.sort()) { h.update(f); h.update(readFileSync(f)); }
  h.update(envKeys.sort().map(k => `${k}=${process.env[k] ?? ''}`).join('\n'));
  return h.digest('hex');
}

export function runCached(task, { inputs, outputs, envKeys = [] }, cmd) {
  const hash = fingerprint(inputs, envKeys);
  const entry = join('.cache', 'tasks', hash);

  if (existsSync(join(entry, 'meta.json'))) {          // 命中：回放
    const meta = JSON.parse(readFileSync(join(entry, 'meta.json'), 'utf8'));
    for (const o of outputs) cpSync(join(entry, o), o, { recursive: true }); // 恢复产物
    process.stdout.write(meta.log);
    return { cached: true };
  }
  const log = execSync(cmd, { encoding: 'utf8', stdio: ['ignore', 'pipe', 'inherit'] });
  const tmp = entry + '.tmp-' + process.pid;           // 原子写入：先写临时目录
  mkdirSync(tmp, { recursive: true });
  for (const o of outputs) if (existsSync(o)) cpSync(o, join(tmp, o), { recursive: true });
  writeFileSync(join(tmp, 'meta.json'), JSON.stringify({ task, log }));
  mkdirSync(join('.cache', 'tasks'), { recursive: true });
  rmSync(entry, { recursive: true, force: true });
  renameSync(tmp, entry);                              // rename 原子可见
  return { cached: false };
}
// 用法：runCached('build', { inputs: ['src/**', 'package.json'], outputs: ['dist'] }, 'npm run build')
```
**评分点**：
- 指纹含输入文件内容 + 显式 env。
- 命中回放日志与产物（不是只跳过）。
- 失败不写缓存（错误结果复用是灾难）。
- 临时目录 + rename 原子写，防并发污染。
- 能说出与 turbo 的差距：无远程层、outputs 收集要自己声明、glob 展开略。
