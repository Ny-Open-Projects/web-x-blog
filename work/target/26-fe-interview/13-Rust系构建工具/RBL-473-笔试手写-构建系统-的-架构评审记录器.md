## RBL-473：手写架构评审记录器
**答**：实现 `createReviewRecorder()`：1) 评审提案：`propose({ title, options, recommendation, reviewers })`——选项必须 ≥2（单选项不算评审）；2) 评审结论记录：每个 reviewer 的意见（approve/block + 理由）——意见留痕；3) 决策产出：结论 → 自动生成 ADR 草稿（RBL-390 结构）；4) 行动项跟踪：评审提出的 must-fix → 跟踪状态（未完成阻塞合入）；5) 定期回顾：老评审的决策是否被后续推翻（superseded 链）。

```javascript
export function createReviewRecorder() {
  const reviews = new Map()

  function propose({ title, options, recommendation, reviewers }) {
    if (!Array.isArray(options) || options.length < 2)
      throw new Error('评审必须提供 ≥2 个选项（单选项不是评审）')
    const id = `rev-${Date.now()}`
    reviews.set(id, { id, title, options, recommendation, reviewers,
      opinions: [], actionItems: [], status: 'in-review', at: Date.now() })
    return id
  }

  function opine(id, reviewer, { verdict, reason }) {
    const r = reviews.get(id)
    if (!r.reviewers.includes(reviewer)) throw new Error(`${reviewer} 不是指定评审人`)
    if (!['approve', 'block', 'needs-work'].includes(verdict)) throw new Error('verdict 无效')
    r.opinions.push({ reviewer, verdict, reason, at: Date.now() })   // 意见留痕
  }

  function conclude(id, decision, { actionItems = [] } = {}) {
    const r = reviews.get(id)
    if (r.opinions.length < r.reviewers.length) throw new Error('尚有评审人未发表意见')
    r.status = 'concluded'; r.decision = decision
    r.actionItems = actionItems.map((a) => ({ ...a, done: false })) // must-fix 跟踪
    r.adr = { title: `ADR: ${r.title}`, background: r.options.map((o) => `- ${o.name}: ${o.pros} / ${o.cons}`).join('\n'),
      decision, consequences: r.actionItems.map((a) => a.description) }
    return r.adr                                                    // 自动生成 ADR 草稿
  }

  function unblockActions(id) {
    const r = reviews.get(id)
    return r.actionItems.filter((a) => !a.done).map((a) => a.description)   // 未完成 must-fix
  }

  return { propose, opine, conclude, unblockActions,
    supersede: (oldId, newId) => { reviews.get(oldId).supersededBy = newId } }
}
```
**评分点**：
- ≥2 选项强制（单选项不是评审——决策质量的流程保障）
- 意见留痕（verdict + reason——评审可追溯）
- conclude 前置校验（全员发表意见——流程完整性）
- ADR 自动产出 + must-fix 跟踪（评审结论落地不丢）+ supersede 链
