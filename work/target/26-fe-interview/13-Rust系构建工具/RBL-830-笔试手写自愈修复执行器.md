## RBL-830：笔试：手写自愈修复执行器（分级 + 回滚 + 闭环）

**答**：实现 `selfHeal(diag, fixes, { exec, confirm, rebuild, autoZero })`。fixes: `Array<{ code, risk: 'zero'|'low'|'review', match(d), plan(d) -> { cmd, undo? }[] }>`。规则：1) 按 code 优先匹配（diag.code 相等），否则按 match；2) risk=review：不执行，返回 `{ healed: false, mode: 'review', steps }`；low：必须 confirm 为 true 才执行；zero：autoZero=true 时免确认；3) 步骤顺序执行，某步失败 → 执行已完成步骤的 undo（逆序）→ 返回 `{ healed: false, rolledBack: true }`；4) 全部步骤成功后 rebuild()：失败也回滚；成功 → `{ healed: true }`。边界：无匹配；plan 空（直接 rebuild）。

```javascript
async function selfHeal(diag, fixes = [], { exec, confirm, rebuild, autoZero = true }) {
  const fix = fixes.find(f => f.code === diag.code) ?? fixes.find(f => f.match?.(diag));
  if (!fix) return { healed: false, mode: 'no-fix' };
  const steps = fix.plan?.(diag) ?? [];

  if (fix.risk === 'review') return { healed: false, mode: 'review', steps };
  if (fix.risk === 'low' && !(await confirm(steps)))
    return { healed: false, mode: 'declined' };
  if (fix.risk === 'zero' && !autoZero)
    return { healed: false, mode: 'declined' };

  const done = [];
  for (const s of steps) {
    try { await exec(s.cmd); done.push(s); }
    catch (e) {
      for (const d of done.reverse())
        if (d.undo) await exec(d.undo).catch(() => {});   // 逆序回滚，undo 失败不吞主错
      return { healed: false, mode: 'rolled-back', error: e.message };
    }
  }
  const ok = await rebuild();
  if (!ok) {
    for (const d of done.reverse())
      if (d.undo) await exec(d.undo).catch(() => {});
    return { healed: false, mode: 'rolled-back', note: '修复执行但构建仍失败' };
  }
  return { healed: true, mode: 'auto', steps: steps.length };
}

// 自测
const execLog = [];
const exec = async (cmd) => {
  if (cmd.includes('fail-cmd')) throw new Error('boom');
  execLog.push(cmd);
};
const rebuildStates = [false];                            // 第一次 rebuild 失败
const rebuild = async () => rebuildStates.shift() ?? true;

const fixes = [
  { code: 'E1', risk: 'low', plan: () => [{ cmd: 'npm i x', undo: 'npm rm x' }] },
  { code: 'E2', risk: 'review', plan: () => [{ cmd: 'refactor' }] },
  { code: 'E3', risk: 'zero', plan: () => [{ cmd: 'rm -rf cache' }] },
  { code: 'E4', risk: 'low', plan: () => [{ cmd: 'step1', undo: 'undo1' }, { cmd: 'fail-cmd', undo: null }, { cmd: 'step2' }] },
];
let r = await selfHeal({ code: 'E1' }, fixes, { exec, confirm: async () => true, rebuild });
console.log(r.healed === false && r.mode === 'rolled-back');   // rebuild 失败回滚
console.log(execLog.includes('npm rm x'));                // undo 执行

r = await selfHeal({ code: 'E2' }, fixes, { exec, confirm: async () => true, rebuild });
console.log(r.mode === 'review' && r.steps[0].cmd === 'refactor');

r = await selfHeal({ code: 'E3' }, fixes, { exec, confirm: async () => false, rebuild });
console.log(r.healed === true && r.mode === 'auto');      // zero 免确认

r = await selfHeal({ code: 'E4' }, fixes, { exec, confirm: async () => true, rebuild });
console.log(r.mode === 'rolled-back' && execLog.includes('undo1'));   // 中途失败逆序回滚

r = await selfHeal({ code: 'E9' }, fixes, { exec, confirm: async () => true, rebuild });
console.log(r.mode === 'no-fix');

r = await selfHeal({ code: 'E1' }, fixes, { exec, confirm: async () => false, rebuild });
console.log(r.mode === 'declined');
```

**评分点**：
- 风险三级语义正确：review 只给步骤、low 需确认、zero 免确认（可关）
- 失败逆序回滚（undo 失败不吞主错）；rebuild 失败也回滚（验证闭环）
- code 优先匹配、plan 空直接 rebuild；无匹配 no-fix
- 能说清「自愈的价值是把确认降到一键而非消除确认，自动改代码必须排除」属于加分
