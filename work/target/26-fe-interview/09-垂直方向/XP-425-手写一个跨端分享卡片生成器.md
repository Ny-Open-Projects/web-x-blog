## XP-425：手写一个跨端分享卡片生成器

**答**：
- 需求：implement ShareCardBuilder(canvas, { w=500, h=400 })。render({ title, desc, img, qr }) 合成：背景/标题换行（自动截断 2 行）/图片 cover 裁剪/二维码；输出 blob；标题超 2 行加省略号。
- 边界：无图时纯色背景；qr 缺失跳过；文本测量截断用 measureText；导出 jpeg 0.9。
- 评分点：cover 裁剪；换行截断；二维码；导出参数

```javascript
class ShareCardBuilder {
  constructor(canvas, { w = 500, h = 400 } = {}) {
    this.c = canvas; this.w = w; this.h = h;
    this.ctx = canvas.getContext('2d');
  }
  async render({ title, desc, img, qr }) {
    const { ctx, w, h } = this;
    ctx.fillStyle = '#fff'; ctx.fillRect(0, 0, w, h);
    if (img) this.drawCover(img, 0, 0, w, h * 0.6);
    ctx.fillStyle = '#333'; ctx.font = 'bold 20px sans-serif';
    this.drawText(title ?? '', 20, h * 0.6 + 30, w - 40, 2);
    ctx.fillStyle = '#666'; ctx.font = '14px sans-serif';
    this.drawText(desc ?? '', 20, h * 0.6 + 70, w - 120, 1);
    if (qr) ctx.drawImage(qr, w - 90, h - 100, 70, 70);
    return this.c.convertToBlob({ type: 'image/jpeg', quality: 0.9 });
  }
  drawCover(img, x, y, cw, ch) {
    const scale = Math.max(cw / img.width, ch / img.height);
    const sw = cw / scale, sh = ch / scale;
    this.ctx.drawImage(img, (img.width - sw) / 2, (img.height - sh) / 2,
      sw, sh, x, y, cw, ch);                           // cover 裁剪
  }
  drawText(text, x, y, maxW, maxLines) {
    this.ctx.textBaseline = 'top';
    let line = '', lines = 0;
    for (const ch of text) {
      if (this.ctx.measureText(line + ch).width > maxW) {
        if (++lines >= maxLines)
          return this.ctx.fillText(line.slice(0, -1) + '…', x, y + (lines - 1) * 24);
        this.ctx.fillText(line, x, y + (lines - 1) * 24);
        line = ch;
      } else line += ch;
    }
    this.ctx.fillText(line, x, y + lines * 24);
  }
}
```
