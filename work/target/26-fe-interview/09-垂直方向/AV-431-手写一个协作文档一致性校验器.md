## AV-431：手写一个协作文档一致性校验器

**答**：
- 需求：implement consistencyCheck(docs)。docs 为各端文档对象（可序列化）；规则：JSON 相同即一致；不一致时输出各端 hash 与差异字段（浅层 diff）；输出 { ok, hashes, diffs }。
- 边界：嵌套对象递归 diff（最多 3 层）；数组按索引 diff；docs 为空 ok。
- 评分点：稳定 hash；递归 diff；深度限制；差异截断

```javascript
function consistencyCheck(docs) {
  const hashes = docs.map(d => stableHash(JSON.stringify(sortKeys(d))));
  const ok = new Set(hashes).size === 1;
  const diffs = [];
  if (!ok && docs.length >= 2) {
    const [a, b] = docs;
    deepDiff(a, b, '', diffs, 0);
  }
  return { ok, hashes, diffs: diffs.slice(0, 10) };
}
function deepDiff(a, b, path, out, depth) {
  if (depth > 3 || out.length >= 10) return;
  if (typeof a !== typeof b) return out.push({ path, a, b });
  if (typeof a === 'object' && a) {
    const keys = new Set([...Object.keys(a), ...Object.keys(b)]);
    for (const k of keys)
      deepDiff(a[k], b[k], path ? `${path}.${k}` : k, out, depth + 1);
  } else if (a !== b) out.push({ path, a, b });
}
function sortKeys(v) {
  if (Array.isArray(v)) return v.map(sortKeys);
  if (v && typeof v === 'object')
    return Object.fromEntries(Object.keys(v).sort()
      .map(k => [k, sortKeys(v[k])]));
  return v;
}
function stableHash(s) { let h = 0; for (const c of s) h = (h * 31 + c.charCodeAt(0)) >>> 0; return h.toString(16); }
```
