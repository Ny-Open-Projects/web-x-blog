## SCN-599：手写验证结果缓存

**需求与边界**：
- 需求：验证通过后按「场景+用户」缓存结果，低危场景短 TTL 内复用（免重复验证）；高危场景永不缓存；账号切换/风控事件触发失效。
- 边界：① TTL 按场景配置；② 缓存 key 必须含用户隔离；③ 风控事件（异地登录/IP 变化）立即失效全部；④ 内存上限淘汰。

**答**：
```javascript
function createVerifyCache({ maxEntries = 50 } = {}) {
  const store = new Map();                     // LRU：Map 插入序
  const TTL = { coupon: 300_000, comment: 60_000, pay: 0 };   // pay 永不缓存
  function key(user, scene) { return `${user}:${scene}`; }
  function put(user, scene, token) {
    const ttl = TTL[scene];
    if (!ttl) return false;                    // 高危场景不缓存
    if (store.size >= maxEntries) {
      const oldest = store.keys().next().value;  // LRU 淘汰最旧
      store.delete(oldest);
    }
    store.set(key(user, scene), { token, exp: Date.now() + ttl });
    return true;
  }
  function get(user, scene) {
    const e = store.get(key(user, scene));
    if (!e) return null;
    if (Date.now() > e.exp) { store.delete(key(user, scene)); return null; }
    store.delete(key(user, scene));            // 取出后重插 = LRU touch
    store.set(key(user, scene), e);
    return e.token;
  }
  function invalidateUser(user) {              // 账号切换/登出
    for (const k of store.keys()) if (k.startsWith(user + ':')) store.delete(k);
  }
  function invalidateAll() {                   // 风控事件：异地登录/IP 变化
    store.clear();
  }
  return { put, get, invalidateUser, invalidateAll, get size() { return store.size; } };
}
// 触发：authStore.subscribe(s => { if (s.userChanged) cache.invalidateUser(prev); });

```

**评分点**:
- 场景级 TTL 配置表 + 高危场景（pay）永不缓存。
- 缓存 key 含用户隔离（跨用户不可复用）。
- LRU 实现正确（重插 touch + 最旧淘汰）。
- 用户级/全局两级失效触发点明确。
