## SCN-541：手写防重复提交锁：幂等键复用、超时「不确定态」、结果查询恢复

**需求与边界**：
- 需求：submit(fn) 返回状态机对象；请求超时（网络错误）不自动重试，进入 UNKNOWN 态并启动结果查询轮询；成功/失败终态后解锁。
- 边界：① 幂等键在 UNKNOWN 恢复期间不可变；② 轮询最多 N 次后转人工确认提示；③ 组件卸载停止轮询；④ 重复调用 submit 在 BUSY/UNKNOWN 态直接拒绝。

**答**：
```javascript
function createSubmitLock(doRequest, { maxPoll = 6, pollEvery = 3000 } = {}) {
  let state = 'IDLE', key = null, timer = null;
  async function submit(payload) {
    if (state === 'BUSY' || state === 'UNKNOWN') return { state };       // 拒绝重复
    key ??= crypto.randomUUID();                  // 幂等键贯穿整个生命周期
    state = 'BUSY';
    try {
      const res = await doRequest({ ...payload, idempotencyKey: key });
      state = 'DONE'; return { state, res };
    } catch (e) {
      if (e.type === 'TIMEOUT' || e.type === 'NETWORK') return enterUnknown();
      state = 'FAILED'; return { state, err: e };  // 明确业务失败可换 key 重来
    }
  }
  function enterUnknown() {
    state = 'UNKNOWN';
    let polls = 0;
    const query = async () => {
      const r = await doRequest.query(key).catch(() => null);
      if (r?.status === 'SUCCESS' || r?.status === 'FAIL') {
        state = 'DONE';
        return { state, res: r };
      }
      if (++polls >= maxPoll) { stop(); return { state: 'MANUAL_CHECK', key }; }
      timer = setTimeout(query, pollEvery);
    };
    timer = setTimeout(query, pollEvery);
    return { state };
  }
  function stop() { clearTimeout(timer); state = 'DONE'; }   // 卸载时调用
  function reset() { key = null; state = 'IDLE'; }           // 新交易才允许换键
  return { submit, stop, reset, get state() { return state; } };
}
```

**评分点**:
- 幂等键生命周期管理（UNKNOWN/重试复用，reset 才换）。
- 超时不自动重试 POST，进入 UNKNOWN 并以查询替代——这是资损防线核心。
- 轮询上限 + MANUAL_CHECK 人工兜底。
- stop 防卸载泄漏，BUSY/UNKNOWN 态拒绝并发提交。
