## SCN-1054：物流批量导入（商家批量发货单）的解析与校验

**答**：
- 导入：Excel/CSV 批量发货单，列映射（模板校验）、逐行校验（地址/重量/禁运品）。
- 反馈：校验结果分页展示（错误行高亮+原因），修正后重传仅传错误行。
- 体验：模板下载/示例数据/进度反馈（大文件分块解析）。

```javascript
// 批量导入
async function importWaybills(file) {
  const rows = await parseCsv(file);
  const errors = [];
  const valid = [];
  rows.forEach((r, i) => {
    const e = validateRow(r);                          // 地址/重量/禁运
    e ? errors.push({ line: i + 1, ...e }) : valid.push(r);
  });
  return { valid, errors, retryOnly: () => reimport(errors) };
}
function validateRow(r) {
  if (!/^[\u4e00-\u9fa5]{2,}[省市区]/.test(r.address)) return { reason: '地址不完整' };
  if (r.weight <= 0 || r.weight > 50) return { reason: '重量超范围' };
  if (FORBIDDEN.some(f => r.desc.includes(f))) return { reason: '疑似禁运品' };
  return null;
}
```
