## SCN-546：手写余额推送处理器

**需求与边界**：
- 需求：WS 推送 `{ version, balanceFen, ts }` 高频到达；200ms 合并窗口内只渲染最新；version 乱序/重复丢弃；合并期间有 pending 交易时显示值 = 最新 + pending。
- 边界：① 窗口内最终必渲染一次（不丢帧）；② 页面隐藏时跳过渲染只记账，可见时立即渲染；③ pending 在推送 version 覆盖后清除。

**答**：
```javascript
function createPushHandler({ window = 200, render } = {}) {
  let latest = null, timer = null, version = 0;
  const pending = new Map();
  function handle(evt) {
    if (evt.version <= version) return;            // 旧/重复事件丢弃
    version = evt.version;
    // 清除已被推送确认的 pending（evt.confirmedIds）
    for (const id of evt.confirmedIds || []) pending.delete(id);
    latest = evt;
    if (document.hidden) return;                   // 后台只记账
    if (!timer) timer = setTimeout(flush, window); // 合并窗口
  }
  function flush() {
    timer = null;
    if (!latest) return;
    const pendingSum = [...pending.values()].reduce((s, v) => s + v, 0n);
    render({ confirmed: latest.balanceFen,
             display: latest.balanceFen + pendingSum,
             pendingCount: pending.size,
             ts: latest.ts });
    latest = null;
  }
  return {
    handle,
    addPending(id, fen) { pending.set(id, fen); flush(); },  // 本地意图立即渲染
    onVisible() { flush(); },                                // 切回前台立即渲染
    destroy() { clearTimeout(timer); },
  };
}
```

**评分点**:
- version 单调守卫（乱序/重复丢弃）。
- 合并窗口保证最终一致渲染（timer 一次性 flush）。
- hidden 时不渲染但保留 latest，onVisible flush。
- pending 意图叠加与 confirmedIds 对账清除。
