## SCN-644：手写审批草稿自动保存器

**需求与边界**：
- 需求：输入停顿 3s 自动保存；内容未变化不保存；保存冲突（服务端 version 不一致）提示并以服务端为准提供 diff；页面卸载前 flush。
- 边界：① 保存失败重试 1 次后静默降级（不打断输入）；② 恢复草稿时校验 schema 版本；③ in-flight 保护（上一笔未完成不叠发）。

**答**：
```javascript
function createDraftSaver(draftId, saveApi) {
  let lastSaved = null, inFlight = false, timer = null, pendingData = null;
  function onChange(data) {
    pendingData = data;
    if (JSON.stringify(data) === JSON.stringify(lastSaved)) return;   // 变脏检测
    clearTimeout(timer.current ?? timer);
    timer = setTimeout(flush, 3000);
  }
  async function flush() {
    if (inFlight || !pendingData) return;
    inFlight = true;
    const payload = pendingData;
    try {
      const res = await saveApi(draftId, payload, { version: currentVersion }).catch(() => null);
      if (!res) return retryOnce();                  // 网络失败重试一次
      if (res.conflict) {
        showConflict({ mine: payload, theirs: res.serverDraft });   // 冲突 diff
        currentVersion = res.version;
      } else {
        lastSaved = payload; currentVersion = res.version;
        setSavedAt(res.savedAt);
      }
    } finally { inFlight = false; }
  }
  function retryOnce() {
    setTimeout(async () => {
      try {
        await saveApi(draftId, pendingData, { version: currentVersion });
        lastSaved = pendingData;
      } catch { localStash(pendingData); }           // 再失败：本地暂存，静默
    }, 2000);
  }
  window.addEventListener('pagehide', flush);        // 卸载 flush
  return { onChange, flush };
}
```

**评分点**:
- 变脏检测（未变化不保存，省接口）。
- in-flight 保护 + 防抖定时器管理。
- 冲突以服务端为准并 diff 展示。
- 失败重试一次后本地暂存静默降级（不打断输入）。
