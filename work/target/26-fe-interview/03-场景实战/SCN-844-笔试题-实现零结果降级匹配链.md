## SCN-844：手写零结果降级链

**需求与边界**：
- 需求：联想无结果时依次降级：编辑距离 ≤2 纠错 → 子串匹配 → 品类热词 → 全局兜底；每级降级标记来源（展示「相似结果」）；零结果输入记录进词库缺口。
- 边界：① 每级最多取 5 条；② 纠错需词库长度过滤（编辑距离对短词噪声大）；③ 缺口记录去重（同输入 24h 一次）。

**答**：

function suggestFallback(q, dict, categories, { maxEach = 5 } = {}) {
  const steps = [
    { name: 'fuzzy', run: () =>
        dict.filter(w => w.length >= 3 && levenshtein(q, w.slice(0, q.length + 2)) <= 2) },
    { name: 'substring', run: () =>
        dict.filter(w => w.includes(q) || q.includes(w)) },
    { name: 'category', run: () =>
        categories[inferCategory(q)]?.hot ?? [] },
    { name: 'global', run: () => globalTopQueries },
  ];
  for (const step of steps) {
    const items = step.run().slice(0, maxEach);
    if (items.length) {
      logGapOnce(q, { filledBy: step.name });          // 缺口记录（24h 去重）
      return { items, source: step.name, degraded: step.name !== 'fuzzy' || true };
    }
  }
  logGapOnce(q, { filledBy: null });
  return { items: [], source: 'none' };
}
