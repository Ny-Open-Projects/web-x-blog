## SCN-750：手写面板渲染调度器

**需求与边界**：
- 需求：首屏核心面板立即渲染（最多 4 个/帧批次），其余按视口接近度（rootMargin 200px）懒初始化；数据到达但面板未可见时缓存数据，可见后立即渲染。
- 边界：① 初始化批次间隔 rAF（防主线程长任务）；② 卸载时销毁图表实例；③ 同一面板防重复初始化。

**答**：
```javascript
function createPanelScheduler({ coreCount = 4 } = {}) {
  const states = new Map();                            // el -> { chart, data, inited }
  const io = new IntersectionObserver(entries => {
    for (const e of entries) {
      if (e.isIntersecting) initPanel(e.target);
    }
  }, { rootMargin: '200px' });
  function initPanel(el) {
    const s = states.get(el);
    if (!s || s.inited) return;                        // 防重复
    s.inited = true;
    io.unobserve(el);
    s.chart = echarts.init(el);
    if (s.data) s.chart.setOption(buildOption(s.data));   // 数据已到立即渲染
  }
  function feed(el, data) {                            // 数据到达
    const s = states.get(el);
    if (!s) return;
    s.data = data;
    if (s.inited) s.chart.setOption(buildOption(data)); // 已可见直接更新
  }
  function mount(panels) {                             // 分批初始化核心面板
    panels.slice(0, coreCount).forEach((el, i) =>
      requestAnimationFrame(() => initPanel(el)));     // 每帧一个防长任务
    panels.slice(coreCount).forEach(el => io.observe(el));
  }
  function destroy(el) {
    const s = states.get(el);
    if (s?.chart) { s.chart.dispose(); s.inited = false; }   // 卸载销毁
  }
  return { mount, feed, initPanel, destroy };
}
```

**评分点**:
- 核心面板 rAF 分批初始化（防主线程长任务）。
- 视口懒渲染 + 数据早到缓存（可见即渲染）。
- inited 防重复 + 卸载 dispose。
- rootMargin 预渲染临近面板。
