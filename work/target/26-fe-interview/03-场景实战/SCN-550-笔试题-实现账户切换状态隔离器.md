## SCN-550：手写多账户切换隔离器

**需求与边界**：
- 需求：switchTo(id) 完成缓存隔离、旧请求取消、响应竞态丢弃、权限重载；get(key) 永远只返回当前账户数据。
- 边界：① 切换中再次切换（快速连点）以后一次为准；② 旧账户响应即使后到也必须丢弃；③ 隔离期间读取返回 undefined 而非旧账户数据。

**答**：
```javascript
function createAccountIsolator(loader) {
  const cache = new Map();                 // `${acctId}:${key}` -> data
  let current = null, switchSeq = 0;
  const inflight = new Map();              // acctId -> [AbortController]
  async function switchTo(nextId) {
    const seq = ++switchSeq;               // 连点：最后一次为准
    for (const ac of inflight.get(current) || []) ac.abort();
    current = nextId;
    return reload(nextId, seq);
  }
  async function reload(acctId, seq) {
    const needed = await listKeys(acctId);             // 权限与必需数据清单
    const results = {};
    await Promise.all(needed.map(async key => {
      const k = `${acctId}:${key}`;
      const ac = new AbortController();
      (inflight.get(acctId) || inflight.set(acctId, []).get(acctId)).push(ac);
      try {
        const data = await loader(key, ac.signal);
        if (seq !== switchSeq || current !== acctId) return;    // 竞态丢弃
        cache.set(k, data); results[key] = data;
      } catch (e) { if (e.name !== 'AbortError') throw e; }
    }));
    return results;
  }
  function get(key) {
    if (current === null) return undefined;
    return cache.get(`${current}:${key}`);             // 强命名空间：拿不到旧账户数据
  }
  function invalidate(key) { cache.delete(`${current}:${key}`); }
  return { switchTo, get, invalidate };
}
```

**评分点**:
- 缓存 key 强制带账户前缀（get 不可能串账户）。
- 切换时 abort 旧请求 + seq 双保险（连点以后者为准）。
- 响应落地前校验 seq/账户（旧响应丢弃）。
- 隔离期读取返回 undefined（宁可空态不串数据）。
