## SCN-645：手写待办红点聚合器

**需求与边界**：
- 需求：按维度（待办/抄送/催办/升级）维护红点计数；事件到达增减；聚合总数 = 各维之和；打开某维清零；持久化恢复。
- 边界：① 事件可能乱序/重复（带 seq 去重）；② 批量事件合并一次渲染；③ 计数不能为负。

**答**：
```javascript
function createBadgeStore() {
  const counts = { todo: 0, cc: 0, urge: 0, escalate: 0 };
  let seq = 0, timer = null;
  const listeners = new Set();
  function handle(events) {                          // 批量事件
    const fresh = events.filter(e => e.seq > seq);   // seq 去重（单调过滤乱序）
    if (!fresh.length) return;
    seq = Math.max(seq, ...fresh.map(e => e.seq));
    for (const e of fresh) {
      switch (e.type) {
        case 'ASSIGNED': counts.todo++; break;
        case 'DONE':    counts.todo = Math.max(0, counts.todo - 1); break;   // 防负
        case 'CC':      counts.cc++; break;
        case 'URGE':    counts.urge++; break;
        case 'ESCALATE': counts.escalate++; break;
      }
    }
    scheduleRender();
  }
  function scheduleRender() {                        // 100ms 合并渲染
    if (timer) return;
    timer = setTimeout(() => {
      timer = null;
      const total = Object.values(counts).reduce((s, v) => s + v, 0);
      listeners.forEach(fn => fn({ counts, total }));
    }, 100);
  }
  function clear(dim) { counts[dim] = 0; scheduleRender(); }   // 打开即清零
  function persist() { localStorage.setItem('badges', JSON.stringify({ counts, seq })); }
  function restore() {
    const s = JSON.parse(localStorage.getItem('badges') || '{}');
    if (s.counts) Object.assign(counts, s.counts), seq = s.seq || 0;
  }
  return { handle, clear, subscribe: fn => (listeners.add(fn), () => listeners.delete(fn)), persist, restore };
}
```

**评分点**:
- seq 单调去重（乱序/重复事件安全）。
- 计数 Math.max(0,…) 防负数。
- 批量事件合并渲染（100ms 窗口）。
- persist/restore 持久化 + 打开清零语义。
