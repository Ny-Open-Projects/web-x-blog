## SCN-893：手写快照版本校验

**需求与边界**：
- 需求：客户端缓存商品快照（含版本号）；渲染时与服务端最新版本比对；不一致拉增量 patch 应用（不全量重拉）；价格快照与下单接口价差异时立即刷新并提示。
- 边界：① patch 应用幂等（按字段覆盖）；② 版本乱序（旧版本晚到）丢弃；③ 校验时机：渲染前 + 下单前双检。

**答**：
```javascript
function snapshotManager(localSnapshot) {
  let ver = localSnapshot?.version ?? 0;
  function check(serverVer) {
    if (serverVer <= ver) return { fresh: true };      // 乱序/相同：无需更新
    return { fresh: false, needPatch: serverVer > ver };
  }
  async function syncPatch(serverVer) {
    const { baseVer, patch } = await api.get('/snapshot/patch', { params: { from: ver } });
    if (baseVer !== ver) return fullRefresh();         // 基线不匹配（版本跳跃过大）全量
    applyPatch(localSnapshot, patch);                  // 幂等字段覆盖
    ver = serverVer;
  }
  function priceGuard(orderPrice) {                    // 下单前双检
    if (orderPrice !== localSnapshot.price) {
      toast('价格已更新，请确认最新价格');
      report({ type: 'PRICE_DIFF', snapshot: localSnapshot.price, server: orderPrice });
      return syncPatch(serverVersion).then(() => ({ refreshed: true }));
    }
    return { refreshed: false };
  }
  return { check, syncPatch, priceGuard, get version() { return ver; } };
}
const applyPatch = (snap, patch) =>
  Object.entries(patch).forEach(([k, v]) => (snap[k] = v));   // 字段覆盖幂等
```

**评分点**:
- 版本乱序守卫（serverVer <= ver 忽略）。
- 增量 patch + 基线不匹配全量兜底（版本跳跃场景）。
- 下单前价格双检（快照价 ≠ 下单价 = 资损防线）。
- patch 幂等字段覆盖。
