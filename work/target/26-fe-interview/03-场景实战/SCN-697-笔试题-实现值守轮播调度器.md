## SCN-697：手写值守轮播调度器

**需求与边界**：
- 需求：按页面列表定时轮播（每页 dwell 时长可不同）；用户交互暂停轮播，闲置 N 分钟恢复（从暂停处继续）；切换前预取下一页数据；页面不可见时暂停计时。
- 边界：① dwell 按页面配置（undefined 用默认）；② 恢复后从下一页继续而非当前页重放；③ stop 后所有定时器清理。

**答**：
```javascript
function createPatrol(pages, { defaultDwell = 5 * 60_000, idleResume = 10 * 60_000 } = {}) {
  let idx = 0, timer = null, idleTimer = null, paused = false, visible = true;
  function schedule() {
    clearTimeout(timer);
    if (paused || !visible) return;
    const dwell = pages[idx].dwell ?? defaultDwell;
    timer = setTimeout(next, dwell);
  }
  async function next() {
    idx = (idx + 1) % pages.length;
    await prefetch(pages[idx]);                        // 切换前预取
    router.push(pages[idx].path);
    schedule();
  }
  function onUserActivity() {                          // 交互暂停
    paused = true;
    clearTimeout(timer); clearTimeout(idleTimer);
    idleTimer = setTimeout(() => { paused = false; schedule(); }, idleResume);   // 闲置恢复
  }
  document.addEventListener('visibilitychange', () => {
    visible = !document.hidden;
    visible ? schedule() : clearTimeout(timer);        // 隐藏暂停计时
  });
  ['pointerdown', 'keydown'].forEach(ev =>
    document.addEventListener(ev, onUserActivity, { passive: true }));
  function stop() {
    clearTimeout(timer); clearTimeout(idleTimer); paused = true;
  }
  schedule();
  return { next, stop, get current() { return idx; } };
}
```

**评分点**:
- 页面级 dwell 配置 + 默认值兜底。
- 交互暂停/闲置恢复从暂停处继续（idx 不重置）。
- visibilitychange 暂停计时（后台不切页）。
- 预取在切换前完成；stop 全清理。
