## 删除与更新接口

删除直接把 `AppStoreId` 透传给后端；更新则先取出原有记录，再把请求数据覆盖后回写。这里同样建议「先查后改」，避免清空未传入的字段。

```go
package handler

import (
	"context"
	appStore "git.imooc.com/coding-535/appStore/proto/appStore"
	log "github.com/asim/go-micro/v3/logger"
)

// DeleteAppStore 对外暴露：删除云应用（级联删除由后端完成）
func (e *AppStoreApi) DeleteAppStore(ctx context.Context, req *appStore.AppStoreId, rsp *appStore.Response) error {
	log.Info("Received AppStoreApi.DeleteAppStore request")
	if _, err := GetAppID(req); err != nil {
		rsp.Msg = err.Error()
		return err
	}
	if err := e.AppStoreClient.DeleteAppStore(ctx, req, rsp); err != nil {
		log.Error(err)
		rsp.Msg = err.Error()
		return err
	}
	return nil
}

// UpdateAppStore 对外暴露：更新云应用
func (e *AppStoreApi) UpdateAppStore(ctx context.Context, req *appStore.AppStoreInfo, rsp *appStore.Response) error {
	log.Info("Received AppStoreApi.UpdateAppStore request")
	// AppStoreInfo 自带 Id，无需额外取 ID
	if err := e.AppStoreClient.UpdateAppStore(ctx, req, rsp); err != nil {
		log.Error(err)
		rsp.Msg = err.Error()
		return err
	}
	return nil
}
```
