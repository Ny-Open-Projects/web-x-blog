### Q6532 · PostgreSQL 逻辑复制：publication/subscription 与 Python 侧消费

> 难度:★★★★ | 标签:PostgreSQL, 逻辑复制, 数据集成

**问**：PG 逻辑复制和物理复制、MySQL binlog 外挂解析（Canal 思路）比有什么差别？Python 侧怎么消费逻辑复制流？初始快照和增量怎么衔接才能不丢不重？为什么有的表订阅后 UPDATE 事件直接报错？

**答**：结论：逻辑复制按"发布/订阅"复制**行级变更事件**（INSERT/UPDATE/DELETE），复制的是解码后的逻辑变更而不是 WAL 页块，因此可跨大版本、可只订阅部分表、目标端不必是另一套同版本 PG。Python 侧用 psycopg3 的 replication 连接 + `pgoutput`（或 `wal2json`）解码消费。不丢不重的关键是"**先建槽拿一致点位，再导快照，增量从槽位 LSN 起消费，衔接点之前的事件跳过**"。

机理与实战：
- **三种路径**：物理复制=块级、同版本、整实例、从库只读；逻辑复制=表级、跨版本、目标可为异构系统；MySQL binlog 外挂=需要外部组件解析并维护位点，PG 从 10 起内核自带（logical decoding + 复制槽）。
- **初始衔接**：`CREATE_REPLICATION_SLOT ... LOGICAL` 返回 consistent_point（LSN）与导出的快照标识；在快照事务里读全量，之后从该 LSN 消费增量——LSN ≤ 衔接点的变更已体现在快照里，必须跳过，否则重复。订阅端 `CREATE SUBSCRIPTION` 的初始表同步就是这套机制的自动化版。
- **复制身份**：表必须有 PRIMARY KEY（或 `REPLICA IDENTITY FULL`）才能解码 UPDATE/DELETE 的"旧行定位"；无身份表的 UPDATE/DELETE 会直接报错中断复制。
- **会丢/不传的东西**：DDL 不复制（两端各自执行，顺序要"先订阅端后发布端"保持向后兼容）；序列状态、大对象不复制；TRUNCATE 会作为事件传播（PG 11+）。
- **Python 消费骨架**：replication 游标 `start_replication` → 逐条 `consume_stream`，处理完 `send_feedback(flush_lsn)` 汇报位点——**位点汇报时机决定重放边界**，宕机重启后从上次确认位点重放，消费端必须幂等。

```python
# 基线:Python 3.10+;纯标准库模拟逻辑复制的初始快照与增量衔接
# 真实场景:psycopg3 LogicalReplicationConnection + pgoutput/wal2json 解码消费
from __future__ import annotations
from dataclasses import dataclass


@dataclass
class Event:
    lsn: int
    op: str            # INSERT / UPDATE / DELETE
    table: str
    row_id: str
    payload: dict


class Primary:
    """发布端:维护 WAL/LSN 与复制身份(有无主键)。"""

    def __init__(self) -> None:
        self.lsn = 100
        self.tables: dict[str, dict[str, dict]] = {}       # table -> (pk -> row)
        self.has_pk: set[str] = set()
        self.log: list[Event] = []

    def _bump(self) -> int:
        self.lsn += 10
        return self.lsn

    def insert(self, table: str, row_id: str, payload: dict) -> Event:
        self.tables.setdefault(table, {})[row_id] = payload
        ev = Event(self._bump(), "INSERT", table, row_id, payload)
        self.log.append(ev)
        return ev

    def update(self, table: str, row_id: str, payload: dict,
               new_id: str | None = None) -> list[Event]:
        if table not in self.has_pk:
            raise ValueError(f"{table} 无复制身份(主键):UPDATE 事件不可解码,复制报错中断")
        events: list[Event] = []
        if new_id is None:
            self.tables[table][row_id] = payload
            events.append(Event(self._bump(), "UPDATE", table, row_id, payload))
        else:                                             # 改主键:拆 DELETE+INSERT
            old = self.tables[table].pop(row_id)
            events.append(Event(self._bump(), "DELETE", table, row_id, old))
            self.tables[table][new_id] = payload
            events.append(Event(self._bump(), "INSERT", table, new_id, payload))
        self.log.extend(events)
        return events

    def create_slot(self) -> tuple[int, list[tuple[str, dict]]]:
        """建槽返回 (consistent_lsn, 初始快照);快照内容 = 该 LSN 时刻的表状态。"""
        return self.lsn, [(t, dict(rows)) for t, rows in self.tables.items()]


class Subscriber:
    """订阅端:先快照后增量,衔接点之后的事件不丢不重。"""

    def __init__(self) -> None:
        self.state: dict[str, dict[str, dict]] = {}
        self.applied_lsn = 0

    def apply_snapshot(self, snapshot: list[tuple[str, dict]]) -> None:
        for t, rows in snapshot:
            self.state.setdefault(t, dict(rows))

    def apply_increment(self, events: list[Event], from_lsn: int) -> int:
        n = 0
        for ev in events:
            if ev.lsn <= from_lsn:                        # 快照已含其效果 -> 跳过
                continue
            rows = self.state.setdefault(ev.table, {})
            if ev.op == "DELETE":
                rows.pop(ev.row_id, None)
            else:
                rows[ev.row_id] = ev.payload
            self.applied_lsn = ev.lsn
            n += 1
        return n


def demo() -> None:
    pub = Primary()
    pub.has_pk |= {"orders"}
    pub.insert("orders", "o1", {"amt": 100})
    pub.insert("orders", "o2", {"amt": 200})

    consistent_lsn, snap = pub.create_slot()              # 建槽:得到一致点位+快照
    sub = Subscriber()
    sub.apply_snapshot(snap)
    assert sub.state["orders"]["o2"]["amt"] == 200
    print(f"初始快照对齐 consistent_lsn={consistent_lsn} ✅")

    # 快照之后写入的增量:新插入 + 普通更新 + 改主键拆分
    pub.insert("orders", "o3", {"amt": 300})
    pub.update("orders", "o1", {"amt": 150})
    split = pub.update("orders", "o2", {"amt": 200}, new_id="o9")
    assert [e.op for e in split] == ["DELETE", "INSERT"], "PK 变更拆 DELETE+INSERT"
    print("改主键拆为 DELETE+INSERT 事件 ✅")

    consumed = sub.apply_increment(pub.log, from_lsn=consistent_lsn)
    assert consumed == 4 and sub.state["orders"]["o3"]["amt"] == 300
    assert sub.state["orders"]["o1"]["amt"] == 150
    assert "o2" not in sub.state["orders"] and sub.state["orders"]["o9"]["amt"] == 200
    print("增量消费不丢不重,衔接点之前的事件被跳过 ✅")

    # 复制身份校验:无主键表的 UPDATE 必须被拒
    pub.insert("logs", "x", {"msg": "hi"})                # logs 未登记 has_pk
    try:
        pub.update("logs", "x", {"msg": "changed"})
    except ValueError as e:
        print("无主键表 UPDATE 被拒 ✅:", e)


if __name__ == "__main__":
    demo()
```

**追问一**：UPDATE 修改了主键，订阅端会看到什么？
答：PG 会拆成 DELETE(旧行)+INSERT(新行)两个事件，消费端按序处理即等价；若 `REPLICA IDENTITY FULL`，事件里携带旧值全列，下行体积变大但消费端可以做"变更前后 diff"。这也解释了为什么把 UPDATE 写成"删旧插新"的场景，逻辑复制流量会翻倍。

**追问二**：两端 schema 漂移时复制会怎样？加列的安全顺序是什么？
答：逻辑复制只传行数据、不传 DDL，订阅端列缺失/类型不兼容会让事件应用失败并卡住复制进度。安全顺序：先订阅端加列（旧事件列少，可空/有默认值兼容）→ 确认订阅端就绪 → 发布端加列；删除列则完全相反。NOT NULL 列要借"先加可空列、回填、再挂 NOT NULL 约束"三步走。

## 背诵要点
- 逻辑复制=行级变更事件（跨版本/表级/异构目标）；物理复制=块级（同版本/整实例）；Python 消费用 psycopg3 + pgoutput。
- 不丢不重：先建槽拿 consistent_point + 快照 → 快照读全量 → 增量从槽位 LSN 起、衔接点前跳过；位点汇报决定重放边界，消费端必须幂等。
- 无 PRIMARY KEY/REPLICA IDENTITY 的表 UPDATE/DELETE 直接报错中断复制；DDL 不复制，加列顺序"先订阅端后发布端"。
