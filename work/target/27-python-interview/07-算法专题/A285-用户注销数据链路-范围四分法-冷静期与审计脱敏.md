### A285 · 用户注销的数据链路：范围四分法、冷静期与 PII 不落审计

> 难度:★★★★ | 标签:数据合规, 注销, 删除范围, 冷静期, 审计, PII

**问**（笔试·场景题）：用户提交注销申请，法务要求「删除一切可关联的个人数据 + 出具删除证明 + 全程可审计」——面试官要的不是「DELETE FROM users」，而是**注销的完整数据链路设计**：① 删除范围四分法：直接标识数据（主库行/缓存/向量索引：即时物理删）、引用痕迹（他人会话中的引用：替换「已注销用户」占位符脱敏）、派生数据（聚合统计/推荐特征：不可定点清除，声明性披露）、法定留存（交易流水：保留并引用法律依据）——每一类的处置动作与理由都要说清；② **冷静期**：申请到执行留 7 天宽限，期内可撤销恢复账户——撤销与执行的互斥语义（撤销后不可执行、执行后重复申请幂等拒绝）；③ 备份残留的处置口径：备份不重写，靠保留期滚动过期 + 按用户派生密钥销毁实现「密文永久不可读」（当约束引用，不展开）；④ **审计悖论**：删除全程要审计，但审计里不能再存这个用户的 PII——审计与删除证明只落 user_id 的哈希，断言全链审计 JSON 中明文 PII 零出现；⑤ 复核：执行后抽验直接/引用存储中明文消失、幂等。请说清与 Q7766（会话数据合规删除的资产清单与扩散面清点）、A112（审计哈希链与销毁凭证）、A264（token 注销与黑名单）、Q2226（缓存 PII 治理）、A192（租户级生命周期）的分工——本题只回答「**账户级注销的范围裁决、冷静期编排与审计脱敏**」。

**答**：① 分工：Q7766 回答「会话数据散落在 8 类存储里的扩散面怎么逐项清点与出证明」（资产清单驱动，删除的执行层），A112 回答「审计记录本身怎么防篡改」，A264 回答「注销时 token/会话态怎么失效」，A192 是租户级生命周期——没有一张卡回答「**一个账户注销时，哪些数据删、哪些脱敏、哪些声明、哪些保留**」的范围裁决，以及裁决之上的冷静期编排与审计脱敏。② 四分法是范围裁决的骨架：直接标识数据**删**（可定位、无共享语义）；引用痕迹**脱敏不删**——他人会话里「user_B 的评价」删掉会破坏他人数据的完整性，替换占位符是唯一同时满足「被删者不可关联」与「保留者数据完整」的动作；派生数据**声明**——聚合统计里的一票在数学上无法定点减除（周活是计数器不是集合），合规姿态是「随口径滚动稀释 + 隐私政策披露」；法定留存**保留**——删除权与留存义务冲突时留存优先，但要在证明里引用法律依据，不是默默不删。③ 冷静期是误删保险也是合规要求（不少辖区要求注销有撤销窗口）；编排上的铁律是**状态机互斥**——pending 可撤销可执行、cancelled 回不去 pending（重新走申请）、executed 幂等拒绝——每个状态转移都留审计。④ 审计悖论的解法是**最小化再入账**：审计 subject 只落 `sha256(user_id)` 截断哈希——哈希不可逆推回明文，但同一 user_id 再来请求能对上账（可关联性只保留在哈希层）；审计记「删了什么类别、多少条、何时、凭何依据」，不记数据本体——A112 的哈希链防篡改是审计的完整性层，本题的哈希化是审计的隐私层，两者正交可叠加。⑤ 复核与幂等是收尾两道闸：复核断言「直接/引用两类存储明文 0 残留」，幂等断言「重复申请 already_executed 不二次执行」——注销编排最常见的线上事故就是重试把已脱敏的引用又写坏。

```python
# 全部标准库；用户注销数据链路：范围四分法 + 冷静期状态机 + 备份残留声明 + PII 不落审计
import datetime as dt
import hashlib
import json


SCOPE = {
    "direct": "主库行/缓存/向量索引：即时物理删除",
    "referenced": "他人可见的引用痕迹：替换为「已注销用户」占位符",
    "derived": "聚合统计/推荐特征：不可定点清除，随口径滚动稀释并披露",
    "legal_hold": "交易与支付流水：法定留存 3 年，声明依据不删",
}


def uhash(user_id):
    """审计与证明只落哈希：删除操作合规，但不能把 PII 换个地方存。"""
    return hashlib.sha256(user_id.encode()).hexdigest()[:12]


class DeletionPipeline:
    def __init__(self, today):
        self.today = today
        self.audit = []
        self.requests = {}
        self.destroyed_keys = []

    def _audit(self, user_id, action, note):
        self.audit.append({"date": self.today.isoformat(),
                           "subject": uhash(user_id),
                           "action": action, "note": note})

    def request(self, user_id, counts):
        st = self.requests.get(user_id)
        if st and st["state"] == "executed":
            return "already_executed"                  # 幂等：已注销不重复执行
        deadline = self.today + dt.timedelta(days=7)
        self.requests[user_id] = {"state": "pending", "counts": counts,
                                  "deadline": deadline}
        self._audit(user_id, "request", "注销申请，冷静期至 %s（可撤销）"
                     % deadline.isoformat())
        return "pending"

    def cancel(self, user_id):
        st = self.requests.get(user_id)
        if st and st["state"] == "pending":
            st["state"] = "cancelled"
            self._audit(user_id, "cancel", "冷静期内撤销，账户保留")
            return True
        return False

    def execute(self, user_id, stores):
        st = self.requests.get(user_id)
        if st is None or st["state"] != "pending":
            return None
        if self.today < st["deadline"]:
            return "in_grace"                          # 冷静期未到：不可执行
        n = st["counts"]
        stores["direct"] = [r for r in stores["direct"] if user_id not in r]
        stores["referenced"] = [r.replace(user_id, "已注销用户")
                                for r in stores["referenced"]]
        self.destroyed_keys.append(uhash(user_id))     # 派生密钥销毁：密文永久不可读
        report = {
            "subject": uhash(user_id),
            "executed_on": self.today.isoformat(),
            "direct": "删除 %d 条（主库/缓存/向量）" % n["direct"],
            "referenced": "脱敏 %d 条他人可见引用" % n["referenced"],
            "derived": "聚合 %d 项：不可定点清除，随口径滚动稀释（已披露）" % n["derived"],
            "legal_hold": "交易流水 %d 条：法定留存 3 年（声明依据）" % n["legal_hold"],
            "backup": "备份不重写：保留期滚动过期 + 派生密钥已销毁，残留密文永久不可读",
        }
        st["state"] = "executed"
        self._audit(user_id, "execute", "四类范围执行完毕，备份密钥销毁，证明已出")
        return report


def verify(stores, user_id):
    """复核：直接与引用存储中明文标识必须消失。"""
    return (not [r for r in stores["direct"] if user_id in r]
            and not [r for r in stores["referenced"] if user_id in r])


if __name__ == "__main__":
    T0 = dt.date(2026, 11, 24)
    stores = {
        "direct": ["user_A 的资料行", "user_B 的资料行", "user_B 的向量记忆"],
        "referenced": ["user_B 在 user_C 订单下的评价", "user_A 的评价"],
        "derived": ["周活聚合（含 user_B 贡献 1 次）", "推荐特征（含 user_B 行为）"],
        "legal_hold": ["user_B 的交易流水 3 条"],
    }
    pipe = DeletionPipeline(T0)

    # ① 冷静期撤销：user_A 申请后反悔，账户保留、删除不执行
    assert pipe.request("user_A", {"direct": 1, "referenced": 1,
                                   "derived": 1, "legal_hold": 0}) == "pending"
    assert pipe.cancel("user_A") is True
    assert pipe.execute("user_A", stores) is None          # 撤销后不可执行
    print("冷静期: user_A 撤销成功，账户保留，删除未执行 ✅")

    # ② user_B：冷静期未到拒绝执行；到期执行
    counts = {"direct": 2, "referenced": 1, "derived": 2, "legal_hold": 1}
    assert pipe.request("user_B", counts) == "pending"
    assert pipe.execute("user_B", stores) == "in_grace"    # 11-24 < 12-01
    pipe.today = dt.date(2026, 12, 1)
    report = pipe.execute("user_B", stores)
    assert report is not None and report["subject"] == uhash("user_B")
    print("注销执行: 四类范围各就各位，证明只含 subject 哈希 ✅")

    # ③ 复核：明文消失、引用脱敏、聚合声明留存、法定留存保留
    assert verify(stores, "user_B")
    assert stores["referenced"][0] == "已注销用户 在 user_C 订单下的评价"
    assert any("user_B" in r for r in stores["derived"])   # 聚合：声明性留存
    assert any("user_B" in r for r in stores["legal_hold"])  # 法定留存
    assert uhash("user_B") in pipe.destroyed_keys
    print("复核: 直接/引用存储明文 0 残留，聚合/法定按声明保留，密钥已销毁 ✅")

    # ④ 幂等：重复申请不重复执行
    assert pipe.request("user_B", counts) == "already_executed"
    assert pipe.execute("user_B", stores) is None
    print("幂等: 重复申请 already_executed，不二次执行 ✅")

    # ⑤ 审计合规：全链审计不含明文 PII，只含哈希
    audit_dump = json.dumps(pipe.audit, ensure_ascii=False)
    assert "user_B" not in audit_dump and "user_A" not in audit_dump
    assert uhash("user_B") in audit_dump and uhash("user_A") in audit_dump
    print("审计: %d 条记录只落 subject 哈希，明文 PII 零泄漏 ✅" % len(pipe.audit))
    print("备份残留: %s" % report["backup"])
```

**追问**：1) 派生数据「声明性披露」合规审查一般怎么挑战？怎么加固？—— 审查问的是「不可定点清除」是不是技术偷懒：加固路径三层——聚合口径按季度滚动重算（旧口径里该用户的贡献随重算自然消失）、差分隐私噪声让单用户贡献不可辨、k-匿名保证聚合单元 ≥k 人才发布；三层做到两层，声明就是可辩护的。反过来，如果系统能力明明能删（特征向量按用户存储）却归入派生类，就是违规——四分法的边界是「技术上的不可定点性」，不是「不想删」。2) 冷静期 7 天里用户数据还在被业务读写，执行删除时怎么处理「窗口期新写入」？—— 编排必须是**快照 + 追删两段式**：执行时刻按当时快照删一遍，再对「窗口期新写入且带该用户标识」的增量追删；更稳的是服务层先封禁注销用户的写路径（注销状态拦截写入），追删只兜漏网——只删快照不追删，冷静期的每一天都在制造新残留。3) 为什么审计保留哈希而不是完全不记标识？—— 完全不记会让审计失去对账能力：同一 user_id 重新注册后再次注销、或监管抽查「这个删除是否真执行了」，都需要哈希层的可关联性；sha256 截断 12 hex 在该规模下碰撞可忽略——审计的合规底线是「不含可还原的 PII」，不是「不含任何标识」，这也是 A112 追问里「审计存数据哈希与操作、不存个人数据本体」原则在注销场景的具体化。
